Quanto costa davvero ogni operazione? Da 0 a 0,5% su quasi un secolo di S&P 500

Quanto pesano davvero i costi per operazione, facendoli variare da 0 a 0,5% del valore scambiato? Pesano in proporzione ai cambi di posizione? Cambiano la classifica delle regole? Quanto costa ribilanciare un portafoglio? E che cosa succede a chi cambia posizione a caso e spesso?

L'articolo esce a breve sul blog⬇ Scarica il codice (.zip)

Il disegno

  • Otto regole fisse (le stesse dell'articolo 61), nessuna ottimizzata: buy & hold; prezzo sopra la media a 50, 100, 200 giorni; incrocio 50/200; momentum a 12 mesi; investito solo a mercato calmo; ritorno alla media nel trend. Il segnale della chiusura di ieri decide la posizione di oggi. Un cambio è un acquisto o una vendita dell'intera posizione; l'ingresso iniziale dalla cassa si paga.
  • Sei livelli di costo: 0, 0,05, 0,10, 0,20, 0,30, 0,50% del valore scambiato (commissione, spread e slippage insieme). Nessuna tassa, cassa a rendimento zero.
  • Ribilanciamento: portafoglio 50% S&P 500 e 50% cassa, riportato al 50/50 ogni seduta, ogni 21 sedute, ogni 252 sedute.
  • Scimmie: 300 posizioni casuali per ognuna di cinque probabilità giornaliere di cambiare stato (1, 5, 10, 20, 50%).
  • Sei ipotesi scritte in config.py prima di guardare i dati.

"""Passo 1 - Scarica da Yahoo Finance la serie giornaliera dell'S&P 500.

Salva:
  dati/gspc_giornaliero.csv   la serie (Date, Open, High, Low, Close, Volume)
  dati/gspc_metadati.json     da dove arrivano i dati, quando, con quale
                              versione di yfinance, e l'impronta SHA-256 del file

Gli script successivi leggono SOLO il file salvato: i risultati sono quindi
riproducibili anche se Yahoo corregge i dati in futuro.
"""
import hashlib
import json
from datetime import datetime, timezone
from zoneinfo import ZoneInfo

import pandas as pd
import yfinance as yf

import config


def main() -> None:
    config.DATI.mkdir(parents=True, exist_ok=True)

    ticker = yf.Ticker(config.TICKER)
    grezzo = ticker.history(start=config.INIZIO_DOWNLOAD, auto_adjust=False)
    if grezzo.empty:
        raise SystemExit("Yahoo Finance non ha restituito dati.")

    df = grezzo[["Open", "High", "Low", "Close", "Volume"]].copy()
    df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
    df.index.name = "Date"
    righe_scaricate = len(df)

    # Controlli di qualità di base
    df = df[~df.index.duplicated(keep="last")].sort_index()
    df = df.dropna(subset=["Close"])
    if (df["Close"] <= 0).any():
        raise SystemExit("Trovati prezzi di chiusura non positivi.")

    # Se lo script gira durante la seduta, l'ultima barra è incompleta: la scartiamo.
    ora_borsa = datetime.now(ZoneInfo(config.FUSO_BORSA))
    barra_scartata = None
    if df.index[-1].date() == ora_borsa.date() and ora_borsa.hour < 16:
        barra_scartata = str(df.index[-1].date())
        df = df.iloc[:-1]

    df.to_csv(config.FILE_DATI, float_format="%.6f")

    sha = hashlib.sha256(config.FILE_DATI.read_bytes()).hexdigest()
    metadati = {
        "ticker": config.TICKER,
        "descrizione": "S&P 500, indice di prezzo (esclude i dividendi)",
        "fonte": "Yahoo Finance",
        "url_fonte": "https://finance.yahoo.com/quote/%5EGSPC/history/",
        "libreria": f"yfinance {yf.__version__}",
        "scaricato_il_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"),
        "righe_scaricate": righe_scaricate,
        "righe_salvate": len(df),
        "prima_data": str(df.index[0].date()),
        "ultima_data": str(df.index[-1].date()),
        "barra_incompleta_scartata": barra_scartata,
        "colonna_usata": "Close (chiusura dell'indice, non corretta: l'indice non ha dividendi)",
        "sha256_file": sha,
    }
    config.FILE_METADATI.write_text(
        json.dumps(metadati, indent=2, ensure_ascii=False), encoding="utf-8"
    )
    print(json.dumps(metadati, indent=2, ensure_ascii=False))


if __name__ == "__main__":
    main()
flusso/02_backtest_e_test.pycosti, ribilanciamento, scimmie, verdetti
"""Passo 2 - Il costo nascosto: otto regole, sei livelli di costo, ribilanciamento e scimmie.

Legge SOLO dati/gspc_giornaliero.csv e produce in risultati/:
  metriche.json              ogni numero dell'articolo, con descrizione e origine di calcolo
  costi_regole.csv           per ogni regola e livello di costo: rendimento annuo, Sharpe, cambi all'anno, posto per Sharpe
  ribilanciamento.csv        per ogni frequenza di ribilanciamento e livello di costo: rendimento annuo e Sharpe del 50/50
  scimmie.csv                per ogni probabilità di cambio e livello di costo: rendimento annuo mediano e quota di scimmie in perdita
"""
import json
import platform
from datetime import datetime, timezone

import numpy as np
import pandas as pd

import config
import funzioni as fx

RIS: dict = {}
SORGENTE = "02_backtest_e_test.py con funzioni.py"


def reg(chiave, valore, formato, descrizione, origine=SORGENTE):
    if isinstance(valore, (np.floating, np.integer)):
        valore = valore.item()
    if isinstance(valore, float) and not np.isfinite(valore):
        raise SystemExit(f"Numero non finito per {chiave}")
    RIS[chiave] = {"valore": valore, "formato": formato, "descrizione": descrizione, "origine": origine}


def main() -> None:
    config.RISULTATI.mkdir(parents=True, exist_ok=True)
    df = pd.read_csv(config.FILE_DATI, parse_dates=["Date"], index_col="Date").sort_index()
    close = df["Close"].astype(float)
    if (close <= 0).any() or close.isna().any():
        raise SystemExit("Chiusure mancanti o non positive.")
    date = close.index
    S0 = config.INIZIO_TEST_SEDUTE
    ids = list(config.STRATEGIE)
    K, C = len(ids), len(config.COSTI)
    rng = np.random.default_rng(config.SEME)
    ret = close.pct_change().fillna(0.0)
    T = len(close) - S0
    anni = (date[-1] - date[S0 - 1]).days / 365.25
    nome = {i: config.STRATEGIE[i][0] for i in ids}

    # --- Posizioni (segnale di ieri decide oggi) e rendimenti netti a ogni livello di costo ------------------------------------
    pos = {i: fx.posizione_regola(close, config.STRATEGIE[i][1]).iloc[S0:] for i in ids}
    ret_s = ret.iloc[S0:]
    cambi = {i: float((pos[i].diff().abs().fillna(0.0).sum() + abs(pos[i].iloc[0]))) for i in ids}
    cambi_anno = {i: cambi[i] / anni for i in ids}
    CG = np.zeros((K, C))
    SH = np.zeros((K, C))
    for k, i in enumerate(ids):
        for c, costo in enumerate(config.COSTI):
            R = fx.rendimenti_netti(ret_s, pos[i], costo).to_numpy()
            CG[k, c] = fx.cagr(R, anni)
            SH[k, c] = fx.sharpe(R, config.SEDUTE_ANNO)
    PS = np.vstack([fx.posti(SH[:, c]) for c in range(C)]).T       # posti per Sharpe, per ogni costo
    righe = []
    for k, i in enumerate(ids):
        for c, costo in enumerate(config.COSTI):
            righe.append({"id": i, "nome": nome[i], "costo": costo, "cagr": CG[k, c], "sharpe": SH[k, c],
                          "posto_sharpe": PS[k, c], "cambi_anno": cambi_anno[i]})
    pd.DataFrame(righe).to_csv(config.RISULTATI / "costi_regole.csv", index=False, float_format="%.6f")

    ib, isf, isv = ids.index(config.ID_BH), ids.index(config.ID_FILTRO), ids.index(config.ID_VELOCE)
    c0, cm = 0, C - 1
    cp = list(config.COSTI).index(config.COSTO_PROVA)

    # --- Parametri e dati ---------------------------------------------------------------------------------------------------------
    reg("prima_data", str(date[0].date()), "data_lunga", "Prima seduta dei dati", "dati/")
    reg("data_inizio", str(date[S0].date()), "data_lunga", "Prima seduta in cui si misurano tutte le regole", "dati/ e config.py")
    reg("data_ultima", str(date[-1].date()), "data_lunga", "Ultima seduta dei dati", "dati/")
    reg("anni_campione", anni, "num1", "Anni dell'intero periodo misurato", "dati/")
    reg("w_breve", config.STRATEGIE["sma50"][1][1], "annonum", "Giorni della media mobile più corta", "config.py, STRATEGIE")
    reg("w_media", config.STRATEGIE["sma100"][1][1], "annonum", "Giorni della media mobile intermedia", "config.py, STRATEGIE")
    reg("w_lunga", config.STRATEGIE["sma200"][1][1], "annonum", "Giorni della media mobile più lunga", "config.py, STRATEGIE")
    reg("mom_mesi", config.STRATEGIE["mom"][1][1] / 21, "annonum", "Mesi di storia usati dal momentum (252 sedute, 21 per mese)", "config.py, STRATEGIE")
    reg("n_regole", K, "int", "Regole confrontate", "config.py, STRATEGIE")
    reg("n_costi", C, "int", "Livelli di costo provati", "config.py, COSTI")
    for c, costo in enumerate(config.COSTI):
        reg(f"costo_{c}", costo, "pct2", f"Livello di costo numero {c + 1} della griglia, quota del valore scambiato", "config.py, COSTI")
    reg("costo_max", config.COSTO_MAX, "pct1", "Costo più alto della griglia", "config.py, COSTI")
    reg("costo_prova", config.COSTO_PROVA, "pct1", "Costo tipico citato nel testo", "config.py, COSTO_PROVA")
    reg("sedute_anno", config.SEDUTE_ANNO, "int", "Sedute di borsa per anno", "config.py")
    reg("quota_azioni", config.QUOTA_AZIONI, "pct0", "Quota in azioni del portafoglio ribilanciato", "config.py, QUOTA_AZIONI")
    reg("n_scimmie", config.N_SCIMMIE, "int", "Scimmie estratte per ogni probabilità di cambio", "config.py")
    reg("cambi_min_anno", config.CAMBI_MIN_ANNO, "annonum", "Cambi all'anno minimi per entrare nella prova di proporzionalità", "config.py")
    reg("banda_lo", config.BANDA_PROPORZIONALITA[0], "num1", "Estremo basso della banda di proporzionalità", "config.py")
    reg("banda_hi", config.BANDA_PROPORZIONALITA[1], "num1", "Estremo alto della banda di proporzionalità", "config.py")
    reg("soglia_bh_pp", config.SOGLIA_BH, "pp2", "Perdita massima ammessa per il buy & hold, in punti percentuali di rendimento annuo", "config.py")
    reg("soglia_prob_scimmie", config.SOGLIA_PROB_SCIMMIE, "pct0", "Probabilità giornaliera di cambio da cui si valuta l'ipotesi sulle scimmie", "config.py")
    reg("soglia_ribil_pp", config.SOGLIA_RIBIL, "pp2", "Perdita massima ammessa con ribilanciamento annuale, in punti percentuali", "config.py, ipotesi 5")

    # --- Rendimenti, Sharpe e cambi per regola ---------------------------------------------------------------------------------------
    for k, i in enumerate(ids):
        reg(f"cambi_anno_{i}", cambi_anno[i], "num1", f"Cambi di posizione all'anno: {nome[i]}", "posizioni sul periodo misurato")
        reg(f"cagr_{i}_lordo", CG[k, c0], "pct1", f"Rendimento annuo composto senza costi: {nome[i]}", "costi_regole.csv")
        reg(f"cagr_{i}_prova", CG[k, cp], "pct1", f"Rendimento annuo composto con costo {config.COSTO_PROVA:.1%}: {nome[i]}", "costi_regole.csv")
        reg(f"cagr_{i}_max", CG[k, cm], "pct1", f"Rendimento annuo composto con costo {config.COSTO_MAX:.1%}: {nome[i]}", "costi_regole.csv")
        reg(f"perdita_{i}", CG[k, c0] - CG[k, cm], "pp2", f"Perdita di rendimento annuo (punti percentuali) tra costo zero e costo massimo: {nome[i]}", "costi_regole.csv")
        reg(f"sharpe_{i}_lordo", SH[k, c0], "num2", f"Sharpe senza costi: {nome[i]}", "costi_regole.csv")
        reg(f"sharpe_{i}_prova", SH[k, cp], "num2", f"Sharpe con costo {config.COSTO_PROVA:.1%}: {nome[i]}", "costi_regole.csv")
        reg(f"sharpe_{i}_max", SH[k, cm], "num2", f"Sharpe con costo {config.COSTO_MAX:.1%}: {nome[i]}", "costi_regole.csv")
        reg(f"posto_{i}_lordo", PS[k, c0], "int", f"Posto per Sharpe senza costi: {nome[i]}", "costi_regole.csv")
        reg(f"posto_{i}_max", PS[k, cm], "int", f"Posto per Sharpe con costo massimo: {nome[i]}", "costi_regole.csv")

    # --- Ipotesi 1: proporzionalità ------------------------------------------------------------------------------------------------------
    elig = [i for i in ids if cambi_anno[i] >= config.CAMBI_MIN_ANNO]
    rapp = {i: (CG[ids.index(i), c0] - CG[ids.index(i), cm]) / (cambi_anno[i] * config.COSTO_MAX) for i in elig}
    for i in elig:
        reg(f"rapporto_{i}", rapp[i], "num2", f"Perdita osservata divisa per (cambi all'anno x costo massimo): {nome[i]}", "costi_regole.csv")
    in_banda = [i for i in elig if config.BANDA_PROPORZIONALITA[0] <= rapp[i] <= config.BANDA_PROPORZIONALITA[1]]
    reg("n_eleggibili", len(elig), "int", "Regole con almeno un cambio all'anno", "posizioni")
    reg("n_in_banda", len(in_banda), "int", "Regole eleggibili con rapporto dentro la banda di proporzionalità", "costi_regole.csv")
    reg("rapporto_min", min(rapp.values()), "num2", "Rapporto più basso tra le regole eleggibili", "costi_regole.csv")
    reg("rapporto_max", max(rapp.values()), "num2", "Rapporto più alto tra le regole eleggibili", "costi_regole.csv")
    k_max_cambi = max(ids, key=lambda i: cambi_anno[i])
    reg("regola_piu_attiva", nome[k_max_cambi], "testo", "Regola con più cambi all'anno", "posizioni")
    reg("perdita_piu_attiva", CG[ids.index(k_max_cambi), c0] - CG[ids.index(k_max_cambi), cm], "pp2",
        "Perdita di rendimento annuo (punti percentuali) della regola più attiva, costo zero contro costo massimo", "costi_regole.csv")

    # --- Ipotesi 2: buy & hold ---------------------------------------------------------------------------------------------------------
    perdita_bh = CG[ib, c0] - CG[ib, cm]
    reg("perdita_bh_fine", perdita_bh, "pp3", "Perdita di rendimento annuo del buy & hold (punti percentuali, tre decimali) tra costo zero e costo massimo", "costi_regole.csv")

    # --- Ipotesi 3 e 4: Sharpe contro buy & hold --------------------------------------------------------------------------------------------
    altre = [k for k in range(K) if k != ib]
    n_sh = {c: int(sum(SH[k, c] > SH[ib, c] for k in altre)) for c in range(C)}
    n_cg = {c: int(sum(CG[k, c] > CG[ib, c] for k in altre)) for c in range(C)}
    for c, tag in ((c0, "lordo"), (cp, "prova"), (cm, "max")):
        reg(f"n_sharpe_sopra_bh_{tag}", n_sh[c], "int", f"Regole (escluso il buy & hold) con Sharpe più alto del buy & hold, costo livello {tag}", "costi_regole.csv")
        reg(f"n_cagr_sopra_bh_{tag}", n_cg[c], "int", f"Regole (escluso il buy & hold) con rendimento annuo più alto del buy & hold, costo livello {tag}", "costi_regole.csv")
    reg("n_altre_regole", len(altre), "int", "Regole diverse dal buy & hold", "config.py")
    n_neg = int((CG[:, cm] < 0).sum())
    reg("n_cagr_negativo_max", n_neg, "int", "Regole con rendimento annuo negativo al costo massimo", "costi_regole.csv")
    reg("tau_classifica", fx.kendall(SH[:, c0], SH[:, cm]), "num2", "Tau di Kendall tra la classifica per Sharpe senza costi e quella con costo massimo", "costi_regole.csv")
    reg("n_cambiano_posto", int((PS[:, c0] != PS[:, cm]).sum()), "int", "Regole che cambiano posto nella classifica per Sharpe tra costo zero e costo massimo", "costi_regole.csv")
    vinc0, vincm = ids[int(np.nanargmax(SH[:, c0]))], ids[int(np.nanargmax(SH[:, cm]))]
    reg("vinc_lordo", nome[vinc0], "testo", "Regola prima per Sharpe senza costi", "costi_regole.csv")
    reg("vinc_max", nome[vincm], "testo", "Regola prima per Sharpe con costo massimo", "costi_regole.csv")
    reg("filtro_meno_bh_max", SH[isf, cm] - SH[ib, cm], "num2", "Sharpe del filtro a 200 giorni meno Sharpe del buy & hold, costo massimo", "costi_regole.csv")

    # --- Ipotesi 5: ribilanciamento -----------------------------------------------------------------------------------------------------------
    righe_r = []
    RB = {}
    ret_a = ret_s.to_numpy()
    for nome_f, ogni in config.RIBILANCIAMENTI.items():
        for c, costo in enumerate(config.COSTI):
            R = fx.mix_ribilanciato(ret_a, config.QUOTA_AZIONI, ogni, costo)
            RB[(nome_f, c)] = (float(fx.cagr(R, anni)), float(fx.sharpe(R, config.SEDUTE_ANNO)))
            righe_r.append({"frequenza": nome_f, "ogni_sedute": ogni, "costo": costo, "cagr": RB[(nome_f, c)][0], "sharpe": RB[(nome_f, c)][1]})
    pd.DataFrame(righe_r).to_csv(config.RISULTATI / "ribilanciamento.csv", index=False, float_format="%.6f")
    perdite_rb = {}
    for nome_f, ogni in config.RIBILANCIAMENTI.items():
        perdite_rb[nome_f] = RB[(nome_f, c0)][0] - RB[(nome_f, cm)][0]
        reg(f"ribil_ogni_{nome_f}", ogni, "int", f"Sedute tra un ribilanciamento e l'altro: {nome_f}", "config.py, RIBILANCIAMENTI")
        reg(f"ribil_n_{nome_f}", len(ret_a) // ogni, "int", f"Ribilanciamenti eseguiti nel periodo: {nome_f}", "ribilanciamento.csv")
        reg(f"ribil_lordo_{nome_f}", RB[(nome_f, c0)][0], "pct2", f"Rendimento annuo del 50/50 senza costi: {nome_f}", "ribilanciamento.csv")
        reg(f"ribil_prova_{nome_f}", RB[(nome_f, cp)][0], "pct2", f"Rendimento annuo del 50/50 con costo {config.COSTO_PROVA:.1%}: {nome_f}", "ribilanciamento.csv")
        reg(f"ribil_max_{nome_f}", RB[(nome_f, cm)][0], "pct2", f"Rendimento annuo del 50/50 con costo massimo: {nome_f}", "ribilanciamento.csv")
        reg(f"ribil_perdita_{nome_f}", perdite_rb[nome_f], "pp2", f"Perdita di rendimento annuo (punti percentuali) tra costo zero e costo massimo: 50/50 {nome_f}", "ribilanciamento.csv")
        reg(f"ribil_perdita_prova_{nome_f}", RB[(nome_f, c0)][0] - RB[(nome_f, cp)][0], "pp2", f"Perdita di rendimento annuo (punti percentuali) tra costo zero e costo tipico: 50/50 {nome_f}", "ribilanciamento.csv")
    gnames = list(config.RIBILANCIAMENTI)
    reg("ribil_vantaggio_lordo_giorn_ann", RB[("giornaliero", c0)][0] - RB[("annuale", c0)][0], "pp2",
        "Rendimento annuo senza costi del 50/50 giornaliero meno quello annuale, in punti percentuali (positivo = il ribilanciamento frequente rende di più prima dei costi)", "ribilanciamento.csv")

    # --- Ipotesi 6: scimmie ---------------------------------------------------------------------------------------------------------------------
    righe_s = []
    SC = {}
    for j, p in enumerate(config.PROB_CAMBIO):
        pos_s = fx.posizioni_scimmie(config.N_SCIMMIE, T, p, rng)
        cambi_s = float((np.abs(np.diff(pos_s, axis=1, prepend=np.zeros((config.N_SCIMMIE, 1)))).sum(axis=1)).mean() / anni)
        reg(f"scim_prob_{j}", p, "pct0", f"Probabilità giornaliera di cambiare stato, gruppo di scimmie numero {j + 1}", "config.py, PROB_CAMBIO")
        reg(f"scim_cambi_anno_{j}", cambi_s, "num1", f"Cambi all'anno medi delle scimmie del gruppo {j + 1}", "scimmie.csv")
        for c, costo in enumerate(config.COSTI):
            R = fx.rendimenti_scimmie(ret_a, pos_s, costo)
            cg = fx.cagr(R, anni)
            SC[(j, c)] = (float(np.median(cg)), float((cg < 0).mean()))
            righe_s.append({"prob_cambio": p, "costo": costo, "cagr_mediano": SC[(j, c)][0], "quota_in_perdita": SC[(j, c)][1], "cambi_anno": cambi_s})
        reg(f"scim_med_lordo_{j}", SC[(j, c0)][0], "pct1", f"Rendimento annuo mediano senza costi, scimmie del gruppo {j + 1}", "scimmie.csv")

"""Funzioni dell'articolo 101: regole, rendimenti netti dei costi, ribilanciamento, scimmie.
Separate dal flusso perché si possano controllare su dati inventati (verifica_motore.py)."""
import numpy as np
import pandas as pd

import motore_vettoriale as m


# --- Regole (le stesse dell'articolo 61) --------------------------------------------------------------------------
def segnale(close: pd.Series, spec: tuple) -> pd.Series:
    """True se la regola è 'investito', calcolato con le chiusure fino a t incluse."""
    tipo = spec[0]
    if tipo == "bh":
        return pd.Series(True, index=close.index)
    if tipo == "sopra_media":
        return m.segnale_sopra_media(close, spec[1])
    if tipo == "incrocio":
        return m.segnale_medie(close, spec[1], spec[2])
    if tipo == "momentum":
        passato = close.shift(spec[1])
        return (close > passato) & passato.notna()
    if tipo == "calma":
        vol = m.rendimenti(close).rolling(spec[1]).std() * np.sqrt(252)
        return (vol < spec[2]) & vol.notna()
    if tipo == "ritorno_media":
        lenta = close.rolling(spec[2]).mean()
        return (close < close.rolling(spec[1]).mean()) & (close > lenta) & lenta.notna()
    raise ValueError(f"regola sconosciuta: {tipo}")


def posizione_regola(close: pd.Series, spec: tuple) -> pd.Series:
    """Quota investita (0 o 1) nella seduta; il segnale di ieri decide la posizione di oggi."""
    return m.posizione(segnale(close, spec), 1).astype(float)


def rendimenti_netti(rend: pd.Series, pos: pd.Series, costo: float) -> pd.Series:
    """Rendimento giornaliero della regola (rend = rendimenti giornalieri dell'indice), meno il costo su ogni variazione della quota investita."""
    cambi = pos.diff().abs()
    cambi.iloc[0] = abs(pos.iloc[0])          # l'ingresso iniziale dalla cassa è un cambio e si paga
    return rend * pos - costo * cambi


# --- Metriche ----------------------------------------------------------------------------------------------------------
def sharpe(R: np.ndarray, sedute_anno: int = 252) -> np.ndarray:
    R = np.asarray(R, dtype=float)
    sd = R.std(axis=-1, ddof=1)
    with np.errstate(divide="ignore", invalid="ignore"):
        v = R.mean(axis=-1) / sd * np.sqrt(sedute_anno)
    return np.where(sd > 0, v, np.nan)


def cagr(R: np.ndarray, anni: float) -> np.ndarray:
    R = np.asarray(R, dtype=float)
    return np.exp(np.log1p(R).sum(axis=-1) / anni) - 1.0


def perdita_massima(R: np.ndarray) -> np.ndarray:
    R = np.asarray(R, dtype=float)
    v = np.cumprod(1.0 + R, axis=-1)
    v = np.concatenate([np.ones(v.shape[:-1] + (1,)), v], axis=-1)
    return (v / np.maximum.accumulate(v, axis=-1) - 1.0).min(axis=-1)


# --- Ribilanciamento ---------------------------------------------------------------------------------------------------
def mix_ribilanciato(rend: np.ndarray, quota: float, ogni: int, costo: float) -> np.ndarray:
    """Rendimenti giornalieri netti di un portafoglio con una quota fissa nell'indice e il resto in cassa (rendimento zero).

    Ogni `ogni` sedute (alla chiusura) la quota derivata viene riportata a `quota`; si paga `costo` sul valore scambiato,
    cioè |quota derivata - quota| del valore del portafoglio. All'inizio si parte tutto in cassa e si compra la quota obiettivo
    il primo giorno (anche questo scambio si paga: `quota` del valore)."""
    rend = np.asarray(rend, dtype=float)
    out = np.empty(len(rend))
    w = quota
    for t, r in enumerate(rend):
        p = w * r
        w = w * (1.0 + r) / (1.0 + p)
        netto = p
        if t == 0:
            netto = (1.0 + p) * (1.0 - costo * quota) - 1.0   # ingresso iniziale dalla cassa
        if (t + 1) % ogni == 0:
            scambio = abs(w - quota)
            netto = (1.0 + netto) * (1.0 - costo * scambio) - 1.0
            w = quota
        out[t] = netto
    return out


# --- Scimmie -----------------------------------------------------------------------------------------------------------
def posizioni_scimmie(n: int, T: int, prob: float, rng) -> np.ndarray:
    """n posizioni casuali (1 investito, 0 cassa) lunghe T: partono investite e a ogni seduta cambiano stato con probabilità `prob`."""
    cambia = rng.random((n, T)) < prob
    cambia[:, 0] = False
    return (np.cumsum(cambia, axis=1) % 2 == 0).astype(float)


def rendimenti_scimmie(rend: np.ndarray, pos: np.ndarray, costo: float) -> np.ndarray:
    """Rendimenti netti di più posizioni (una per riga); il costo si paga a ogni cambio di stato."""
    cambi = np.abs(np.diff(pos, axis=1, prepend=np.zeros((pos.shape[0], 1))))   # l'ingresso iniziale dalla cassa si paga
    return rend[None, :] * pos - costo * cambi


# --- Correlazione di rango tra due classifiche ---------------------------------------------------------------------------
def posti(valori) -> np.ndarray:
    """Posto in classifica, 1 = valore più alto; parità = posto medio; valori non definiti = ultimi."""
    v = np.where(np.isnan(np.asarray(valori, dtype=float)), -np.inf, valori)
    return pd.Series(v).rank(ascending=False, method="average").to_numpy()


def kendall(a, b) -> float:
    """Tau-b di Kendall tra due vettori."""
    a, b = np.asarray(a, dtype=float), np.asarray(b, dtype=float)
    da, db = np.sign(a[:, None] - a[None, :]), np.sign(b[:, None] - b[None, :])
    iu = np.triu_indices(len(a), 1)
    x, y = da[iu], db[iu]
    den = np.sqrt((x != 0).sum() * (y != 0).sum())
    return float((x * y).sum() / den) if den > 0 else float("nan")

"""Motore vettoriale minimo: segnale, posizione, rendimenti. Senza commissioni né tasse, per isolare l'effetto del look-ahead.

Convenzioni:
  - rend[t] è il rendimento della seduta t: chiusura di t su chiusura di t-1;
  - il segnale[t] si calcola con le chiusure fino a t incluse;
  - posizione(segnale, ritardo): ritardo = 0 usa il segnale di t per il rendimento di t (LOOK-AHEAD: si compra alla
    chiusura di ieri sapendo la chiusura di oggi); ritardo = 1 usa il segnale di ieri (onesto).
"""
import numpy as np
import pandas as pd


def rendimenti(close: pd.Series) -> pd.Series:
    return close.pct_change().fillna(0.0)


def segnale_un_giorno(close: pd.Series) -> pd.Series:
    return close > close.shift(1)


def segnale_sopra_media(close: pd.Series, n: int) -> pd.Series:
    return (close > close.rolling(n).mean()) & close.rolling(n).mean().notna()


def segnale_medie(close: pd.Series, veloce: int, lenta: int) -> pd.Series:
    sl = close.rolling(lenta).mean()
    return (close.rolling(veloce).mean() > sl) & sl.notna()


def posizione(segnale: pd.Series, ritardo: int) -> pd.Series:
    """True se si è investiti nella seduta, dato il ritardo con cui il segnale è noto."""
    if ritardo < 0:
        raise ValueError("ritardo negativo: sarebbe usare il futuro")
    return segnale.shift(ritardo, fill_value=False).astype(bool)


def valore(rend: pd.Series, pos: pd.Series) -> pd.Series:
    """Valore di 1 investito, composto giorno per giorno: nei giorni in cassa il rendimento è zero."""
    return (1.0 + rend * pos.astype(float)).cumprod()


def cagr(v: pd.Series, anni: float) -> float:
    return float(v.iloc[-1] ** (1.0 / anni) - 1.0)


def perdita_massima(v: pd.Series) -> float:
    return float((v / v.cummax() - 1.0).min())

"""Controlli delle funzioni dell'articolo 101 su dati inventati. Se uno fallisce, il flusso si ferma."""
import numpy as np
import pandas as pd

import config
import funzioni as fx

n_ok = 0


def controlla(nome: str, condizione: bool) -> None:
    global n_ok
    print(("OK   " if condizione else "FAIL ") + nome)
    if not condizione:
        raise SystemExit(f"Controllo fallito: {nome}")
    n_ok += 1


def serie(valori, inizio="2000-01-03") -> pd.Series:
    return pd.Series(np.asarray(valori, float), index=pd.bdate_range(inizio, periods=len(valori)))


rng = np.random.default_rng(1)
close = serie(100 * np.exp(np.cumsum(rng.normal(0.0003, 0.012, 1200))))

# --- Regole: nessun look-ahead ---
for i, (nome, spec) in config.STRATEGIE.items():
    completa = fx.posizione_regola(close, spec)
    tronca = fx.posizione_regola(close.iloc[:800], spec)
    controlla(f"{i}: cambiare il futuro non cambia la posizione passata", np.array_equal(completa.iloc[:800].to_numpy(), tronca.to_numpy()))
    alt = close.copy()
    alt.iloc[900:] *= 3.0
    controlla(f"{i}: la posizione fino al giorno 900 non dipende dai prezzi dopo", np.array_equal(completa.iloc[:901].to_numpy(), fx.posizione_regola(alt, spec).iloc[:901].to_numpy()))
controlla("buy & hold è sempre investito dal secondo giorno", bool((fx.posizione_regola(close, config.STRATEGIE["bh"][1]).iloc[1:] == 1.0).all()))

# --- Rendimenti netti: costo a ogni cambio, ingresso iniziale compreso ---
pos = pd.Series([0, 0, 1, 1, 0, 0], index=close.index[:6], dtype=float)
rend = serie([0, 0, 0.10, 0.10, 0, 0])
rn = fx.rendimenti_netti(rend, pos, 0.01)
controlla("due cambi (entrata e uscita): lordo 0,20 meno 0,02", abs(float(rn.sum()) - 0.18) < 1e-12)
pos1 = pd.Series([1, 1, 1], index=close.index[:3], dtype=float)
controlla("partendo già investiti si paga l'ingresso iniziale", abs(float(fx.rendimenti_netti(serie([0, 0, 0]), pos1, 0.01).sum()) + 0.01) < 1e-12)
controlla("a costo zero il rendimento netto è quello lordo", np.allclose(fx.rendimenti_netti(serie([0.01, -0.02, 0.03]), pos1, 0.0), [0.01, -0.02, 0.03]))
controlla("il costo è proporzionale: il doppio del costo, il doppio della perdita", abs(float(fx.rendimenti_netti(rend, pos, 0.02).sum()) - 0.16) < 1e-12)

# --- Ribilanciamento ---
r = np.array([0.10, 0.0, 0.0])
sen = fx.mix_ribilanciato(r, 0.5, 1, 0.0)
controlla("50/50 senza costi: primo giorno rende metà", abs(sen[0] - 0.05) < 1e-12)
# giorno 1, costo 1%: ingresso 0,5% (costo*quota), poi quota derivata 0,5*1,1/1,05 = 0,5238..., scambio 0,0238...
w1 = 0.5 * 1.1 / 1.05
atteso = (1.05 * (1 - 0.01 * 0.5)) * (1 - 0.01 * abs(w1 - 0.5)) - 1
controlla("50/50 giornaliero con costo, giorno 1 a mano", abs(fx.mix_ribilanciato(r, 0.5, 1, 0.01)[0] - atteso) < 1e-12)
controlla("senza ribilanciamenti nel periodo (ogni 10 sedute su 3) il costo è solo l'ingresso", abs(fx.mix_ribilanciato(r, 0.5, 10, 0.01)[0] - (1.05 * 0.995 - 1)) < 1e-12)
rr = rng.normal(0.0004, 0.01, 2000)
cg = {k: np.sum(np.log1p(fx.mix_ribilanciato(rr, 0.5, k, 0.005))) for k in (1, 21, 252)}
cg0 = {k: np.sum(np.log1p(fx.mix_ribilanciato(rr, 0.5, k, 0.0))) for k in (1, 21, 252)}
controlla("il costo del ribilanciamento cala con la frequenza (giornaliero > mensile > annuale)", (cg0[1] - cg[1]) > (cg0[21] - cg[21]) > (cg0[252] - cg[252]) > 0)
controlla("il costo del ribilanciamento annuale è positivo e sotto lo 0,5% del valore all'anno in questi dati", 0 < (cg0[252] - cg[252]) < 0.005 * 2000 / 252)

# --- Scimmie ---
ps = fx.posizioni_scimmie(400, 5000, 0.10, np.random.default_rng(3))
controlla("scimmie: valori 0/1, partono investite", set(np.unique(ps)) <= {0.0, 1.0} and bool((ps[:, 0] == 1.0).all()))
cambi_medi = np.abs(np.diff(ps, axis=1)).sum(axis=1).mean() / 5000
controlla("scimmie: la frequenza dei cambi è vicina alla probabilità", abs(cambi_medi - 0.10) < 0.01)
rs = fx.rendimenti_scimmie(np.full(5000, 0.0), ps, 0.01)
controlla("scimmie: con rendimenti nulli si perde esattamente il costo dei cambi", abs(rs.sum(axis=1).mean() + 0.01 * (np.abs(np.diff(ps, axis=1, prepend=0)).sum(axis=1).mean())) < 1e-9)

# --- Metriche ---
controlla("rendimento annuo composto a mano", abs(float(fx.cagr(np.array([0.1, -0.1, 0.1]), 2.0)) - (1.089 ** 0.5 - 1)) < 1e-12)
controlla("Sharpe a mano su una serie piccola", abs(float(fx.sharpe(np.array([0.01, -0.01, 0.02, 0.0]))) - np.array([0.01, -0.01, 0.02, 0.0]).mean() / np.array([0.01, -0.01, 0.02, 0.0]).std(ddof=1) * np.sqrt(252)) < 1e-9)
controlla("Kendall di due classifiche uguali = 1 e opposte = -1", abs(fx.kendall([1, 2, 3], [4, 5, 9]) - 1) < 1e-12 and abs(fx.kendall([1, 2, 3], [9, 5, 4]) + 1) < 1e-12)
controlla("posti: il valore più alto è primo", list(fx.posti(np.array([3.0, 1.0, 2.0, 5.0]))) == [2.0, 4.0, 3.0, 1.0])

print(f"{n_ok} controlli superati")
