Volatility targeting: ridurre le azioni quando il mercato trema migliora il portafoglio?

Ridurre l'esposizione quando la volatilità sale rende un portafoglio azionario migliore, o solo più tranquillo?

L'articolo esce a breve sul blog⬇ Scarica il codice (.zip)

Il disegno

  • Strategia: a fine mese si misura la volatilità degli ultimi 21 giorni di borsa; il peso in azioni del mese successivo è obiettivo (10% annuo) diviso volatilità, con tetto a 1 (nessuna leva); il resto è in cassa, che non rende.
  • Niente look-ahead: il peso calcolato alla chiusura dell'ultima seduta del mese si scambia alla chiusura della seduta seguente e lavora dal rendimento della seduta ancora dopo.
  • Indici: S&P 500, Nasdaq Composite, DAX, FTSE 100, CAC 40, Nikkei 225, scelti prima di guardare i dati. Indici di prezzo, valuta locale.
  • Confronti: buy & hold; buy & hold con la stessa volatilità della strategia (peso fisso); scimmie con gli stessi pesi in mesi casuali.
  • Test: differenza di Sharpe con bootstrap a blocchi (21 giorni), p-value delle scimmie con correzione di Holm, cinque varianti di robustezza riportate tutte, due finestre di crisi fissate in anticipo.
  • Costi: 5 punti base su ogni scambio. Le tasse non sono modellate (dichiarato nei limiti).

"""Passo 1 - Scarica da Yahoo Finance la serie giornaliera di sei indici azionari.

Salva, per ogni indice, dati/<ticker>.csv (Date, Open, High, Low, Close, Volume) e dati/metadati.json con la fonte,
la data del download, la versione di yfinance e l'impronta SHA-256 di ogni file. Gli script successivi leggono SOLO
i file salvati: i risultati restano riproducibili anche se Yahoo corregge i dati in futuro.
"""
import hashlib
import json
import time
from datetime import datetime, timezone
from zoneinfo import ZoneInfo

import pandas as pd
import yfinance as yf

import config


def scarica(ticker: str) -> pd.DataFrame:
    for tentativo in range(3):
        grezzo = yf.Ticker(ticker).history(start=config.INIZIO_DOWNLOAD, auto_adjust=False)
        if not grezzo.empty:
            return grezzo
        print(f"{ticker}: nessun dato (tentativo {tentativo + 1})")
        time.sleep(10)
    return pd.DataFrame()


def main() -> None:
    config.DATI.mkdir(parents=True, exist_ok=True)
    ora_borsa = datetime.now(ZoneInfo(config.FUSO_BORSA))
    meta = {"fonte": "Yahoo Finance", "url_fonte": config.URL_FONTE, "libreria": f"yfinance {yf.__version__}",
            "scaricato_il_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"), "indici": []}
    for ticker, nome, valuta in config.INDICI:
        riga = {"ticker": ticker, "nome": nome, "valuta": valuta}
        grezzo = scarica(ticker)
        if grezzo.empty:
            riga["stato"] = "scartato: nessun dato restituito da Yahoo Finance"
            meta["indici"].append(riga)
            continue
        df = grezzo[["Open", "High", "Low", "Close", "Volume"]].copy()
        df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
        df.index.name = "Date"
        df = df[~df.index.duplicated(keep="last")].sort_index().dropna(subset=["Close"])
        df = df[df["Close"] > 0]
        # seduta ancora in corso: barra incompleta, da scartare
        if df.index[-1].date() == ora_borsa.date() and ora_borsa.hour < 18:
            df = df.iloc[:-1]
        file = config.file_dati(ticker)
        df.to_csv(file, float_format="%.6f")
        riga.update({"stato": "ok", "file": file.name, "righe": len(df), "prima_data": str(df.index[0].date()),
                     "ultima_data": str(df.index[-1].date()), "sha256": hashlib.sha256(file.read_bytes()).hexdigest()})
        meta["indici"].append(riga)
        print(f"{ticker}: {len(df)} righe, dal {riga['prima_data']} al {riga['ultima_data']}")
    config.FILE_METADATI.write_text(json.dumps(meta, indent=2, ensure_ascii=False), encoding="utf-8")
    if sum(1 for r in meta["indici"] if r["stato"] == "ok") < 4:
        raise SystemExit("Troppi pochi indici scaricati.")


if __name__ == "__main__":
    main()

"""Passo 2 - Volatility targeting: strategia contro buy & hold, contro il buy & hold con la stessa volatilità e contro le scimmie.

Legge SOLO i file in dati/ e produce in risultati/:
  metriche.json       ogni numero dell'articolo, con descrizione e origine di calcolo
  indici_usati.json   indici effettivamente usati
  varianti.csv        le varianti di robustezza, tutte, indice per indice
  scimmie_sharpe.csv  Sharpe lordo di ogni scimmia, per indice
  bootstrap.csv       differenze di Sharpe ricampionate (strategia - buy & hold), per indice
  curve_<ticker>.csv  curve, pesi e volatilità mobile dell'indice principale (una seduta su cinque)
Scimmia: stessi pesi mensili della strategia, ma assegnati ai mesi in ordine casuale (confronto lordo di costi).
"""
import json
import math
from datetime import datetime, timezone

import numpy as np
import pandas as pd

import config
import motore as m

RIS: dict = {}
SORGENTE = "02_backtest_e_test.py con motore.py"
GA = config.GIORNI_ANNO


def reg(chiave, valore, formato, descrizione, origine=SORGENTE):
    if isinstance(valore, (np.floating, np.integer)):
        valore = valore.item()
    if isinstance(valore, float) and not np.isfinite(valore):
        raise SystemExit(f"Numero non finito per {chiave}")
    RIS[chiave] = {"valore": valore, "formato": formato, "descrizione": descrizione, "origine": origine}


def slug(ticker: str) -> str:
    return ticker.replace("^", "").replace(".", "_").lower()


def holm(p: dict) -> dict:
    ordine = sorted(p, key=p.get)
    mm = len(ordine)
    out, massimo = {}, 0.0
    for k, nome in enumerate(ordine):
        massimo = max(massimo, min(1.0, (mm - k) * p[nome]))
        out[nome] = massimo
    return out


def carica(ticker: str) -> pd.Series:
    df = pd.read_csv(config.file_dati(ticker), parse_dates=["Date"], index_col="Date").sort_index()
    df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
    df = df[~df.index.duplicated(keep="last")]
    c = df["Close"].astype(float)
    if (c <= 0).any() or c.isna().any():
        raise SystemExit(f"{ticker}: chiusure mancanti o non positive.")
    return c


def metriche_base(d: pd.DataFrame, anni: float) -> dict:
    rs, rb = d["strategia"], d["buyhold"]
    vs, vb = m.vol_annua(rs, GA), m.vol_annua(rb, GA)
    k = vs / vb                                   # buy & hold con la stessa volatilità: peso fisso k, ribilanciato ogni giorno
    req = k * d["r"]
    req.iloc[0] -= config.COSTO_BP / 1e4 * k      # paga l'acquisto iniziale
    return dict(rs=rs, rb=rb, req=req, k=k, vs=vs, vb=vb, veq=m.vol_annua(req, GA),
                cagr_s=m.cagr(rs, anni), cagr_b=m.cagr(rb, anni), cagr_eq=m.cagr(req, anni),
                dd_s=m.perdita_massima(rs), dd_b=m.perdita_massima(rb), dd_eq=m.perdita_massima(req),
                sh_s=m.sharpe(rs, GA), sh_b=m.sharpe(rb, GA),
                vv_s=m.volvol(rs, GA), vv_b=m.volvol(rb, GA))


def analizza(tk, nome, valuta, k_rng):
    meta = [r for r in META["indici"] if r["ticker"] == tk][0]
    if meta["stato"] != "ok":
        return None
    c = carica(tk)
    d = m.serie_strategia(c, config.OBIETTIVO_VOL, config.FINESTRA, config.TETTO_PESO, GA, config.COSTO_BP)
    anni = (d.index[-1] - d.index[0]).days / 365.25
    if anni < config.ANNI_MINIMI:
        print(f"{tk}: scartato, solo {anni:.1f} anni")
        return None
    b = metriche_base(d, anni)
    rs, rb = b["rs"].to_numpy(), b["rb"].to_numpy()
    rng = np.random.default_rng([config.SEME, k_rng])

    # --- bootstrap a blocchi sulla differenza di Sharpe --------------------------------------------------
    boot = m.bootstrap_dsharpe(rs, rb, config.N_BOOTSTRAP, config.BLOCCO_BOOTSTRAP, GA, rng)
    alfa = (1 - config.LIVELLO_CONFIDENZA) / 2
    lo, hi = np.quantile(boot, [alfa, 1 - alfa])

    # --- scimmie: stessi pesi, mesi a caso (lordo di costi) --------------------------------------------
    g = d.groupby("periodo")
    w = g["peso"].first().to_numpy()
    s1 = g["r"].sum().to_numpy()
    s2 = g["r"].apply(lambda x: (x ** 2).sum()).to_numpy()
    n = len(d)
    sh_lordo = float(m.sharpe_da_somme(w, s1, s2, n, GA))
    perm = rng.permuted(np.tile(w, (config.N_SCIMMIE, 1)), axis=1)
    sh_sc = m.sharpe_da_somme(perm, s1, s2, n, GA)
    p_sc = (int((sh_sc >= sh_lordo).sum()) + 1) / (config.N_SCIMMIE + 1)

    # --- finestre di crisi --------------------------------------------------------------------------------
    crisi = {}
    for k, (etichetta, ini, fin) in enumerate(config.CRISI):
        sel = (d.index >= pd.Timestamp(ini)) & (d.index <= pd.Timestamp(fin))
        if d.index[0] <= pd.Timestamp(ini) and sel.sum() > 5:
            crisi[k] = dict(s=float(np.prod(1 + rs[sel]) - 1), b=float(np.prod(1 + rb[sel]) - 1),
                            eq=float(np.prod(1 + b["req"].to_numpy()[sel]) - 1),
                            peso_medio=float(d["peso"].to_numpy()[sel].mean()))
        else:
            crisi[k] = None

    # --- varianti di robustezza (tutte riportate) ------------------------------------------------------------
    var = []
    for et, obiettivo, finestra, tetto in config.VARIANTI:
        dv = m.serie_strategia(c, obiettivo, finestra, tetto, GA, config.COSTO_BP)
        av = (dv.index[-1] - dv.index[0]).days / 365.25
        bv = metriche_base(dv, av)
        var.append(dict(indice=tk, variante=et, obiettivo=obiettivo, finestra=finestra, tetto=tetto,
                        sharpe_s=bv["sh_s"], sharpe_b=bv["sh_b"], cagr_s=bv["cagr_s"], cagr_b=bv["cagr_b"],
                        dd_s=bv["dd_s"], dd_b=bv["dd_b"], vol_s=bv["vs"], vol_b=bv["vb"],
                        peso_medio=float(dv["peso"].mean())))

    # --- curve dell'indice principale ------------------------------------------------------------------------
    curva = None
    if tk == config.INDICE_PRINCIPALE:
        es, eb, eq = m.equity(rs), m.equity(rb), m.equity(b["req"])
        vol_s = pd.Series(rs, index=d.index).rolling(GA).std(ddof=0) * np.sqrt(GA)
        vol_b = pd.Series(rb, index=d.index).rolling(GA).std(ddof=0) * np.sqrt(GA)
        e1 = np.r_[1.0, es]
        e2 = np.r_[1.0, eb]
        dd_s = (es / np.maximum.accumulate(np.r_[1.0, es])[1:] - 1)
        dd_b = (eb / np.maximum.accumulate(np.r_[1.0, eb])[1:] - 1)
        camp = np.arange(0, n, 5)
        curva = pd.DataFrame({"data": d.index[camp], "strategia": es[camp], "buyhold": eb[camp], "stessa_vol": eq[camp],
                              "dd_strategia": dd_s[camp], "dd_buyhold": dd_b[camp], "peso": d["peso"].to_numpy()[camp],
                              "vol_strategia": vol_s.to_numpy()[camp], "vol_buyhold": vol_b.to_numpy()[camp]})

    return dict(tk=tk, sl=slug(tk), nome=nome, d=d, anni=anni, b=b, lo=lo, hi=hi, boot=boot, sh_lordo=sh_lordo, sh_sc=sh_sc,
                p_sc=p_sc, crisi=crisi, var=var, curva=curva, w=w)


def main() -> None:
    global META
    config.RISULTATI.mkdir(parents=True, exist_ok=True)
    META = json.loads(config.FILE_METADATI.read_text(encoding="utf-8"))
    res = []
    for k, (tk, nome, valuta) in enumerate(config.INDICI):
        r = analizza(tk, nome, valuta, k)
        if r:
            res.append(r)
            b = r["b"]
            print(f"{tk}: {r['anni']:.1f} anni, Sharpe {b['sh_s']:.2f} contro {b['sh_b']:.2f}, volatilità {b['vs']:.3f} contro {b['vb']:.3f}, "
                  f"perdita massima {b['dd_s']:.2f} contro {b['dd_b']:.2f}, p scimmie {r['p_sc']:.3f}")
    if len(res) < 4:
        raise SystemExit("Troppi pochi indici utilizzabili.")

    ph = holm({r["tk"]: r["p_sc"] for r in res})
    n_ind = len(res)
    for r in res:
        sl, b, d = r["sl"], r["b"], r["d"]
        pesi = d["peso"]
        reg(f"{sl}_nome", r["nome"], "testo", "Nome dell'indice", "config.py, INDICI")
        reg(f"{sl}_prima_data", str(d.index[0].date()), "data", f"Prima seduta in cui la strategia ha una posizione ({r['nome']})")
        reg(f"{sl}_anni", r["anni"], "num1", f"Anni di storia usati ({r['nome']})")
        reg(f"{sl}_cagr_s", b["cagr_s"], "pct1", f"Rendimento annuo composto netto del volatility targeting ({r['nome']})")
        reg(f"{sl}_cagr_b", b["cagr_b"], "pct1", f"Rendimento annuo composto del buy & hold ({r['nome']})")
        reg(f"{sl}_cagr_eq", b["cagr_eq"], "pct1", f"Rendimento annuo composto del buy & hold con la stessa volatilità della strategia ({r['nome']})")
        reg(f"{sl}_vol_s", b["vs"], "pct1", f"Volatilità annua realizzata della strategia ({r['nome']})")
        reg(f"{sl}_vol_b", b["vb"], "pct1", f"Volatilità annua realizzata del buy & hold ({r['nome']})")
        reg(f"{sl}_vv_s", b["vv_s"], "pct1", f"Deviazione standard della volatilità a un anno mobile, strategia ({r['nome']})")
        reg(f"{sl}_vv_b", b["vv_b"], "pct1", f"Deviazione standard della volatilità a un anno mobile, buy & hold ({r['nome']})")
        reg(f"{sl}_dd_s", b["dd_s"], "pct1", f"Perdita massima della strategia ({r['nome']})")
        reg(f"{sl}_dd_b", b["dd_b"], "pct1", f"Perdita massima del buy & hold ({r['nome']})")
        reg(f"{sl}_dd_eq", b["dd_eq"], "pct1", f"Perdita massima del buy & hold con la stessa volatilità ({r['nome']})")
        reg(f"{sl}_sh_s", b["sh_s"], "num2", f"Indice di Sharpe della strategia, cassa a rendimento zero ({r['nome']})")
        reg(f"{sl}_sh_b", b["sh_b"], "num2", f"Indice di Sharpe del buy & hold, cassa a rendimento zero ({r['nome']})")
        reg(f"{sl}_dsh", b["sh_s"] - b["sh_b"], "num2", f"Differenza di Sharpe strategia meno buy & hold ({r['nome']})")
        reg(f"{sl}_dsh_lo", r["lo"], "num2", f"Estremo inferiore dell'intervallo al 95% della differenza di Sharpe ({r['nome']})",
            "bootstrap a blocchi circolari di 21 giorni su coppie di giorni")
        reg(f"{sl}_dsh_hi", r["hi"], "num2", f"Estremo superiore dell'intervallo al 95% della differenza di Sharpe ({r['nome']})",
            "bootstrap a blocchi circolari di 21 giorni su coppie di giorni")
        reg(f"{sl}_finale_s", config.CAPITALE * m.equity(b["rs"])[-1], "int", f"Valore finale di {config.CAPITALE:.0f} euro con la strategia, al lordo delle tasse ({r['nome']})")
        reg(f"{sl}_finale_b", config.CAPITALE * m.equity(b["rb"])[-1], "int", f"Valore finale di {config.CAPITALE:.0f} euro con il buy & hold, al lordo delle tasse ({r['nome']})")
        reg(f"{sl}_peso_medio", float(pesi.mean()), "pct0", f"Peso medio in azioni della strategia ({r['nome']})")
        reg(f"{sl}_quota_tetto", float((pesi >= config.TETTO_PESO - 1e-12).mean()), "pct0", f"Quota di sedute con peso pieno (tetto) ({r['nome']})")
        reg(f"{sl}_peso_min", float(pesi.min()), "pct0", f"Peso minimo raggiunto dalla strategia ({r['nome']})")
        reg(f"{sl}_scambi_anno", float(d["scambio"].sum() / r["anni"]), "num1",
            f"Scambio annuo, in multipli del capitale (somma delle variazioni di peso per anno) ({r['nome']})")
        reg(f"{sl}_sh_lordo", r["sh_lordo"], "num2", f"Sharpe lordo di costi della strategia, base del confronto con le scimmie ({r['nome']})")
        reg(f"{sl}_sh_sc_med", float(np.mean(r["sh_sc"])), "num2", f"Sharpe lordo medio delle scimmie ({r['nome']})")
        reg(f"{sl}_p_sc", r["p_sc"], "pval", f"Quota di scimmie con Sharpe uguale o maggiore della strategia (p-value) ({r['nome']})")
        reg(f"{sl}_p_sc_holm", ph[r["tk"]], "pval", f"p-value delle scimmie dopo la correzione di Holm ({r['nome']})")
        for kk, (et, _, _) in enumerate(config.CRISI):
            cr = r["crisi"][kk]
            if cr:
                reg(f"{sl}_crisi{kk}_s", cr["s"], "pct1", f"Rendimento della strategia nella finestra «{et}» ({r['nome']})")
                reg(f"{sl}_crisi{kk}_b", cr["b"], "pct1", f"Rendimento del buy & hold nella finestra «{et}» ({r['nome']})")
                reg(f"{sl}_crisi{kk}_peso", cr["peso_medio"], "pct0", f"Peso medio della strategia nella finestra «{et}» ({r['nome']})")

    # --- aggregati -----------------------------------------------------------------------------------------
    B = [r["b"] for r in res]
    reg("n_indici", n_ind, "int", "Numero di indici utilizzabili", "02_backtest_e_test.py")
    reg("n_scartati", len(config.INDICI) - n_ind, "int", "Indici scartati per dati assenti o storia troppo corta", "02_backtest_e_test.py")
    reg("n_vol_minore", sum(b["vs"] < b["vb"] for b in B), "int", "Indici in cui la volatilità della strategia è più bassa di quella del buy & hold")
    reg("n_vv_minore", sum(b["vv_s"] < b["vv_b"] for b in B), "int", "Indici in cui la volatilità a un anno varia meno nella strategia")
    reg("n_dd_minore", sum(b["dd_s"] > b["dd_b"] for b in B), "int", "Indici in cui la perdita massima della strategia è più piccola")
    reg("n_cagr_minore", sum(b["cagr_s"] < b["cagr_b"] for b in B), "int", "Indici in cui il rendimento annuo della strategia è più basso del buy & hold")
    reg("n_sh_sopra", sum(b["sh_s"] > b["sh_b"] for b in B), "int", "Indici in cui lo Sharpe della strategia è più alto di quello del buy & hold")
    reg("n_ic_zero", sum(r["lo"] <= 0 <= r["hi"] for r in res), "int", "Indici in cui l'intervallo di confidenza della differenza di Sharpe contiene lo zero")
    reg("n_ic_sopra", sum(r["lo"] > 0 for r in res), "int", "Indici in cui l'intervallo della differenza di Sharpe sta tutto sopra lo zero")
    reg("n_ic_sotto", sum(r["hi"] < 0 for r in res), "int", "Indici in cui l'intervallo della differenza di Sharpe sta tutto sotto lo zero")
    reg("n_scimmie_sign", sum(r["p_sc"] < 0.05 for r in res), "int", "Indici in cui meno del 5% delle scimmie ha uno Sharpe uguale o maggiore (senza correzione)")
    reg("n_scimmie_holm", sum(ph[r["tk"]] < 0.05 for r in res), "int", "Indici in cui il p-value delle scimmie resta sotto il 5% dopo la correzione di Holm")
    reg("n_eq_cagr_sopra", sum(b["cagr_s"] > b["cagr_eq"] for b in B), "int", "Indici in cui la strategia batte in rendimento annuo il buy & hold con la stessa volatilità")
    reg("n_eq_dd_meglio", sum(b["dd_s"] > b["dd_eq"] for b in B), "int", "Indici in cui la strategia ha una perdita massima più piccola del buy & hold con la stessa volatilità")
    reg("sh_diff_media", float(np.mean([b["sh_s"] - b["sh_b"] for b in B])), "num2", "Differenza media di Sharpe (strategia meno buy & hold) sugli indici")
    reg("peso_medio_tutti", float(np.mean([r["d"]["peso"].mean() for r in res])), "pct0", "Peso medio in azioni, media sugli indici")

    # --- parametri e fonti (giustificano le cifre del testo) ----------------------------------------------
    reg("obiettivo", config.OBIETTIVO_VOL, "pct0", "Volatilità annua obiettivo", "config.py, OBIETTIVO_VOL")

flusso/motore.pyvolatilità, pesi mensili, posizione ritardata, costi, metriche, bootstrap
"""Motore del volatility targeting. Funzioni pure su serie di prezzi: nessun accesso a file, nessuna rete.

Regole (da dichiarare nell'articolo):
  - volatilità: deviazione standard dei rendimenti giornalieri semplici delle ultime `finestra` sedute, annualizzata;
  - a fine mese (ultima seduta del mese) il peso obiettivo è min(tetto, obiettivo / volatilità);
  - il peso calcolato alla chiusura t viene scambiato alla chiusura t+1 e lavora sul rendimento della seduta t+2
    (segnale di ieri, posizione di oggi): la posizione che guadagna il rendimento di t è quindi `segnale[t-2]`;
  - la quota non investita sta in cassa, che non rende; i dividendi non sono considerati;
  - ogni scambio costa `costo_bp` punti base sul valore scambiato (|variazione del peso|); anche l'acquisto iniziale.
"""
import numpy as np
import pandas as pd


def rendimenti(close: pd.Series) -> pd.Series:
    return close.pct_change()


def fine_mese(indice: pd.DatetimeIndex) -> np.ndarray:
    """True nell'ultima seduta di ogni mese; l'ultima riga dei dati non conta (il mese può non essere finito)."""
    mese = indice.year * 12 + indice.month
    ultimo = np.r_[mese[1:] != mese[:-1], False]
    return np.asarray(ultimo)


def volatilita(r: pd.Series, finestra: int, giorni_anno: int) -> pd.Series:
    return r.rolling(finestra).std(ddof=1) * np.sqrt(giorni_anno)


def segnale_peso(close: pd.Series, obiettivo: float, finestra: int, tetto: float, giorni_anno: int) -> pd.Series:
    """Peso obiettivo noto alla chiusura di ogni seduta (cambia solo a fine mese, poi resta fermo)."""
    r = rendimenti(close)
    vol = volatilita(r, finestra, giorni_anno)
    peso = np.minimum(tetto, obiettivo / vol)
    peso = peso.where(pd.Series(fine_mese(close.index), index=close.index))
    return peso.ffill()


def posizione(segnale: pd.Series) -> pd.Series:
    """Peso con cui si guadagna il rendimento della seduta t: segnale della seduta t-2."""
    return segnale.shift(2)


def serie_strategia(close: pd.Series, obiettivo: float, finestra: int, tetto: float, giorni_anno: int,
                    costo_bp: float) -> pd.DataFrame:
    """Rendimenti giornalieri netti di strategia e buy & hold, dalla prima seduta con posizione definita."""
    r = rendimenti(close)
    sig = segnale_peso(close, obiettivo, finestra, tetto, giorni_anno)
    pos = posizione(sig)
    valido = pos.notna() & r.notna()
    primo = np.flatnonzero(valido.to_numpy())
    if len(primo) == 0:
        raise ValueError("nessuna seduta con posizione definita")
    i0 = int(primo[0])
    r = r.iloc[i0:]
    pos = pos.iloc[i0:]
    sig = sig.iloc[i0:]
    prec = pos.shift(1).fillna(0.0)                     # prima seduta: si parte da zero, l'acquisto iniziale costa
    scambio = (pos - prec).abs()
    c = costo_bp / 1e4
    rs = pos * r - c * scambio
    rb = r.copy()
    rb.iloc[0] -= c * 1.0                                # buy & hold: anche lui paga l'acquisto iniziale
    fm = fine_mese(close.index)
    effettivo = np.r_[False, False, fm[:-2]][i0:]          # True nella seduta in cui il nuovo peso comincia a lavorare
    periodo = np.cumsum(effettivo)
    return pd.DataFrame({"r": r, "peso": pos, "scambio": scambio, "strategia": rs, "buyhold": rb,
                         "periodo": periodo})


def equity(rend: pd.Series | np.ndarray) -> np.ndarray:
    return np.cumprod(1.0 + np.asarray(rend, dtype=float))


def sharpe(rend, giorni_anno: int) -> float:
    x = np.asarray(rend, dtype=float)
    sd = x.std(ddof=0)
    return float(x.mean() / sd * np.sqrt(giorni_anno)) if sd > 0 else float("nan")


def vol_annua(rend, giorni_anno: int) -> float:
    return float(np.asarray(rend, dtype=float).std(ddof=0) * np.sqrt(giorni_anno))


def perdita_massima(rend) -> float:
    e = np.r_[1.0, equity(rend)]
    return float((e / np.maximum.accumulate(e) - 1).min())


def cagr(rend, anni: float) -> float:
    return float(equity(rend)[-1] ** (1 / anni) - 1)


def volvol(rend: pd.Series, giorni_anno: int) -> float:
    """Deviazione standard della volatilità annualizzata misurata su 252 sedute mobili."""
    v = rend.rolling(giorni_anno).std(ddof=0) * np.sqrt(giorni_anno)
    return float(v.dropna().std(ddof=0))


def sharpe_da_somme(w: np.ndarray, s1: np.ndarray, s2: np.ndarray, n: int, giorni_anno: int) -> np.ndarray:
    """Sharpe lordo di un portafoglio a pesi costanti per periodo: w, s1 (somma r), s2 (somma r^2) per periodo.
    Accetta w con più righe (una per scimmia)."""
    media = (w * s1).sum(axis=-1) / n
    m2 = (w ** 2 * s2).sum(axis=-1) / n
    sd = np.sqrt(np.maximum(m2 - media ** 2, 1e-30))
    return media / sd * np.sqrt(giorni_anno)


def bootstrap_dsharpe(rs: np.ndarray, rb: np.ndarray, n_boot: int, blocco: int, giorni_anno: int,
                      rng, a_pezzi: int = 100) -> np.ndarray:
    """Differenza di Sharpe (strategia - buy & hold) su ricampionamenti a blocchi circolari, coppie di giorni insieme."""
    n = len(rs)
    nb = int(np.ceil(n / blocco))
    out = np.empty(n_boot)
    off = np.arange(blocco)
    fatti = 0
    while fatti < n_boot:
        k = min(a_pezzi, n_boot - fatti)
        st = rng.integers(0, n, size=(k, nb))
        idx = ((st[:, :, None] + off) % n).reshape(k, nb * blocco)[:, :n]
        xs, xb = rs[idx], rb[idx]
        ss = xs.mean(axis=1) / np.maximum(xs.std(axis=1), 1e-30)
        sb = xb.mean(axis=1) / np.maximum(xb.std(axis=1), 1e-30)
        out[fatti:fatti + k] = (ss - sb) * np.sqrt(giorni_anno)
        fatti += k
    return out
flusso/verifica_motore.pycontrolli su dati inventati (conti a mano, assenza di look-ahead)
"""Controlli del motore del volatility targeting. Usa SOLO dati inventati: nessun numero qui è un risultato.

Si esegue prima del backtest (lo fa anche il flusso su GitHub): se un controllo fallisce, il flusso si ferma.
  1. conti fatti a mano su serie costruite apposta;
  2. assenza di look-ahead: cambiare un prezzo futuro non cambia mai le posizioni passate;
  3. coerenza tra formule: Sharpe da somme contro Sharpe diretto, periodi, bootstrap.
"""
import numpy as np
import pandas as pd

import motore as m

ok = 0
GA = 252


def verifica(nome, condizione, dettaglio=""):
    global ok
    if not condizione:
        raise SystemExit(f"FALLITO: {nome} {dettaglio}")
    ok += 1
    print(f"  ok  {nome}")


def serie(rend, inizio="2020-01-01"):
    idx = pd.bdate_range(inizio, periods=len(rend))
    return pd.Series(100.0 * np.cumprod(1 + np.asarray(rend, dtype=float)), index=idx)


rng = np.random.default_rng(1)
print("1. Conti a mano")
# Rendimenti alternati +1% / -1%: deviazione standard campionaria nota.
alt = np.tile([0.01, -0.01], 200)
c = serie(alt)
v = m.volatilita(m.rendimenti(c), 20, GA).dropna()
atteso = np.std(alt[1:21] if False else np.array([0.01, -0.01] * 10), ddof=1) * np.sqrt(GA)
verifica("volatilità di una serie +1%/-1%", abs(v.iloc[0] - atteso) < 1e-3 * atteso, f"{v.iloc[0]} vs {atteso}")

# Volatilità costante del 20% annuo: il peso obiettivo con obiettivo 10% deve essere 0,5.
sd_g = 0.20 / np.sqrt(GA)
base = np.tile([sd_g, -sd_g], 300)
c = serie(base)
sig = m.segnale_peso(c, 0.10, 20, 1.0, GA).dropna()
verifica("peso = obiettivo / volatilità (0,5, con la correzione n-1)", np.allclose(sig.to_numpy(), 0.5 * np.sqrt(19 / 20), atol=0.003), sig.iloc[:3].tolist())
sig2 = m.segnale_peso(c, 0.30, 20, 1.0, GA).dropna()
verifica("il tetto limita il peso a 1", np.allclose(sig2.to_numpy(), 1.0))
sig3 = m.segnale_peso(c, 0.30, 20, 1.5, GA).dropna()
verifica("con tetto 1,5 il peso è 1,46 (0,30 / 0,20 con la correzione n-1)", np.allclose(sig3.to_numpy(), 0.30 / 0.20 * np.sqrt(19 / 20), atol=0.01))

# Costi: peso costante 0,5 e costo zero => rendimento della strategia = 0,5 * rendimento dell'indice
d0 = m.serie_strategia(c, 0.10, 20, 1.0, GA, 0.0)
verifica("senza costi: rendimento = peso * indice", np.allclose(d0["strategia"], d0["peso"] * d0["r"]))
verifica("la prima seduta ha scambio pari al peso", abs(d0["scambio"].iloc[0] - d0["peso"].iloc[0]) < 1e-12)
d1 = m.serie_strategia(c, 0.10, 20, 1.0, GA, 10.0)
costo_tot = (d0["strategia"] - d1["strategia"]).sum()
verifica("costo totale = 10 punti base * scambi totali", abs(costo_tot - 0.001 * d0["scambio"].sum()) < 1e-12)
verifica("buy & hold paga solo l'acquisto iniziale", abs((d0["buyhold"] - d1["buyhold"]).sum() - 0.001) < 1e-12)

print("2. Nessun look-ahead")
rend = rng.normal(0.0004, 0.01, 1500)
rend[600:700] *= 3
c = serie(rend)
a = m.serie_strategia(c, 0.10, 21, 1.0, GA, 5.0)
for taglio in (900, 1200, 1400):
    r2 = rend.copy()
    r2[taglio:] = rng.normal(0, 0.05, len(r2) - taglio)         # futuro completamente diverso
    b = m.serie_strategia(serie(r2), 0.10, 21, 1.0, GA, 5.0)
    comune = a.index[a.index < c.index[taglio]]
    verifica(f"posizioni identiche fino al giorno {taglio} (futuro cambiato)",
             np.allclose(a.loc[comune, "peso"], b.loc[comune, "peso"], atol=0, rtol=0, equal_nan=True))
# Ritardo: il peso calcolato a fine mese non è mai usato prima di due sedute dopo.
fm = m.fine_mese(c.index)
sig = m.segnale_peso(c, 0.10, 21, 1.0, GA)
pos = m.posizione(sig)
primo_fm = np.flatnonzero(fm & sig.notna().to_numpy())[0]
verifica("il peso nuovo non è usato nelle due sedute dopo la fine del mese",
         pos.iloc[primo_fm + 1] != sig.iloc[primo_fm] or np.isnan(pos.iloc[primo_fm + 1]))
verifica("la posizione compare dopo due sedute", abs(pos.iloc[primo_fm + 2] - sig.iloc[primo_fm]) < 1e-15)
verifica("l'ultima riga non è mai una fine mese", not m.fine_mese(c.index)[-1])

print("3. Coerenza tra formule")
d = m.serie_strategia(c, 0.10, 21, 1.0, GA, 5.0)
verifica("equity finale = prodotto di (1+r)", abs(m.equity(d["strategia"])[-1] - np.prod(1 + d["strategia"])) < 1e-12)
verifica("perdita massima di una serie sempre crescente è zero", m.perdita_massima(np.full(50, 0.01)) == 0.0)
e = m.perdita_massima(np.array([0.10, -0.50, 0.20]))
verifica("perdita massima a mano (-50%)", abs(e - (-0.5)) < 1e-12, str(e))
# Sharpe da somme contro Sharpe diretto
per = d["periodo"].to_numpy()
w = d.groupby("periodo")["peso"].first().to_numpy()
s1 = d.groupby("periodo")["r"].sum().to_numpy()
s2 = d.groupby("periodo")["r"].apply(lambda x: (x ** 2).sum()).to_numpy()
lordo = d["peso"] * d["r"]
verifica("Sharpe da somme = Sharpe diretto", abs(m.sharpe_da_somme(w, s1, s2, len(d), GA) - m.sharpe(lordo, GA)) < 1e-9)
verifica("i periodi sono mensili (circa 21 sedute)", 15 <= len(d) / len(w) <= 25, str(len(d) / len(w)))
perm = np.stack([rng.permutation(w) for _ in range(3)])
verifica("Sharpe da somme accetta più permutazioni", m.sharpe_da_somme(perm, s1, s2, len(d), GA).shape == (3,))
# Bootstrap: due serie identiche => differenza di Sharpe nulla
x = rng.normal(0.0005, 0.01, 800)
bs = m.bootstrap_dsharpe(x, x, 50, 21, GA, np.random.default_rng(3))
verifica("bootstrap su serie identiche dà differenza nulla", np.allclose(bs, 0))
# volatilità e rolling
v2 = m.volvol(pd.Series(rng.normal(0, 0.01, 800)), GA)
verifica("volvol positiva e finita", np.isfinite(v2) and v2 > 0)
print(f"\n{ok} controlli superati.")
