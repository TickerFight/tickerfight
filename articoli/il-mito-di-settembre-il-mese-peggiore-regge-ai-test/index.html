"""Passo 1 - Scarica da Yahoo Finance la serie giornaliera dell'S&P 500.

Salva:
  dati/gspc_giornaliero.csv   la serie (Date, Open, High, Low, Close, Volume)
  dati/gspc_metadati.json     da dove arrivano i dati, quando, con quale
                              versione di yfinance, e l'impronta SHA-256 del file

Gli script successivi leggono SOLO il file salvato: i risultati sono quindi
riproducibili anche se Yahoo corregge i dati in futuro.
"""
import hashlib
import json
from datetime import datetime, timezone
from zoneinfo import ZoneInfo

import pandas as pd
import yfinance as yf

import config


def main() -> None:
    config.DATI.mkdir(parents=True, exist_ok=True)

    ticker = yf.Ticker(config.TICKER)
    grezzo = ticker.history(start=config.INIZIO_DOWNLOAD, auto_adjust=False)
    if grezzo.empty:
        raise SystemExit("Yahoo Finance non ha restituito dati.")

    df = grezzo[["Open", "High", "Low", "Close", "Volume"]].copy()
    df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
    df.index.name = "Date"
    righe_scaricate = len(df)

    # Controlli di qualità di base
    df = df[~df.index.duplicated(keep="last")].sort_index()
    df = df.dropna(subset=["Close"])
    if (df["Close"] <= 0).any():
        raise SystemExit("Trovati prezzi di chiusura non positivi.")

    # Se lo script gira durante la seduta, l'ultima barra è incompleta: la scartiamo.
    ora_borsa = datetime.now(ZoneInfo(config.FUSO_BORSA))
    barra_scartata = None
    if df.index[-1].date() == ora_borsa.date() and ora_borsa.hour < 16:
        barra_scartata = str(df.index[-1].date())
        df = df.iloc[:-1]

    df.to_csv(config.FILE_DATI, float_format="%.6f")

    sha = hashlib.sha256(config.FILE_DATI.read_bytes()).hexdigest()
    metadati = {
        "ticker": config.TICKER,
        "descrizione": "S&P 500, indice di prezzo (esclude i dividendi)",
        "fonte": "Yahoo Finance",
        "url_fonte": "https://finance.yahoo.com/quote/%5EGSPC/history/",
        "libreria": f"yfinance {yf.__version__}",
        "scaricato_il_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"),
        "righe_scaricate": righe_scaricate,
        "righe_salvate": len(df),
        "prima_data": str(df.index[0].date()),
        "ultima_data": str(df.index[-1].date()),
        "barra_incompleta_scartata": barra_scartata,
        "colonna_usata": "Close (chiusura dell'indice, non corretta: l'indice non ha dividendi)",
        "sha256_file": sha,
    }
    config.FILE_METADATI.write_text(
        json.dumps(metadati, indent=2, ensure_ascii=False), encoding="utf-8"
    )
    print(json.dumps(metadati, indent=2, ensure_ascii=False))


if __name__ == "__main__":
    main()

"""Passo 2 - Il mito di settembre sull'S&P 500.

Legge SOLO dati/gspc_giornaliero.csv; produce in risultati/: metriche.json, medie_mensili.csv, settembri.csv.
"""
import json
import platform
from datetime import datetime, timezone

import numpy as np
import pandas as pd

import config
import funzioni as fx

RIS: dict = {}
SORGENTE = "02_backtest_e_test.py con funzioni.py"
MESI = ["gennaio", "febbraio", "marzo", "aprile", "maggio", "giugno", "luglio", "agosto", "settembre", "ottobre", "novembre", "dicembre"]


def reg(chiave, valore, formato, descrizione, origine=SORGENTE):
    if isinstance(valore, (np.floating, np.integer)):
        valore = valore.item()
    if isinstance(valore, float) and not np.isfinite(valore):
        raise SystemExit(f"Numero non finito per {chiave}")
    RIS[chiave] = {"valore": valore, "formato": formato, "descrizione": descrizione, "origine": origine}


def mese_peggiore(x: pd.Series) -> int:
    return int(x.groupby(x.index.month).mean().idxmin())


def main() -> None:
    config.RISULTATI.mkdir(parents=True, exist_ok=True)
    rng = np.random.default_rng(config.SEME)
    df = pd.read_csv(config.FILE_DATI, parse_dates=["Date"], index_col="Date").sort_index()
    close = df["Close"].astype(float)
    if (close <= 0).any() or close.isna().any():
        raise SystemExit("Chiusure mancanti o non positive.")
    rs = fx.rendimenti_mensili(close)
    mm = rs.index.month.to_numpy()
    sett = (mm == config.MESE)
    reg("prima_data", str(close.index[0].date()), "data_lunga", "Prima seduta dei dati", "dati/")
    reg("data_ultima", str(close.index[-1].date()), "data_lunga", "Ultima seduta dei dati", "dati/")
    reg("n_mesi", len(rs), "int", "Mesi completi", "dati/")
    reg("n_sett", int(sett.sum()), "int", "Settembri nel campione", "dati/")
    reg("n_perm", config.N_PERMUTAZIONI, "int", "Permutazioni", "config.py")
    reg("soglia_p", config.SOGLIA_P, "num2", "Soglia di significatività", "config.py")
    reg("n_peggiori", config.N_PEGGIORI, "int", "Settembri peggiori tolti nell'ipotesi 6", "config.py")

    medie = rs.groupby(rs.index.month).mean()
    pd.DataFrame({"mese": MESI, "media": medie.to_numpy()}).to_csv(config.RISULTATI / "medie_mensili.csv", index=False, float_format="%.6f")
    ordine = medie.sort_values().index.tolist()
    reg("rango_sett", ordine.index(config.MESE) + 1, "int", "Posizione di settembre dal peggiore (1 = il più basso)", "pandas")
    reg("sett_media", float(medie.loc[config.MESE]), "pct2", "Rendimento medio di settembre", "pandas")
    reg("altri_media", float(rs[~sett].mean()), "pct2", "Rendimento medio degli altri undici mesi", "pandas")
    reg("sett_mediana", float(rs[sett].median()), "pct2", "Rendimento mediano di settembre", "pandas")
    reg("sett_neg", float((rs[sett] < 0).mean()), "pct0", "Quota di settembri in calo", "pandas")
    reg("altri_neg", float((rs[~sett] < 0).mean()), "pct0", "Quota degli altri mesi in calo", "pandas")
    secondo = ordine[1]
    reg("secondo_nome", MESI[secondo - 1], "testo", "Secondo mese peggiore", "pandas")
    reg("secondo_media", float(medie.loc[secondo]), "pct2", "Rendimento medio del secondo mese peggiore", "pandas")
    reg("migliore_nome", MESI[ordine[-1] - 1], "testo", "Mese migliore", "pandas")
    reg("migliore_media", float(medie.loc[ordine[-1]]), "pct2", "Rendimento medio del mese migliore", "pandas")

    d, p = fx.permutazione_differenza(rs.to_numpy(), sett, config.N_PERMUTAZIONI, rng)
    reg("diff", d, "pp2", "Settembre meno gli altri mesi, punti percentuali", "funzioni")
    reg("pval", p, "pval", "P-value di permutazione di settembre contro gli altri mesi", "funzioni.permutazione_differenza")
    mn, pm = fx.minimo_permutazione(rs.to_numpy(), mm, config.N_PERMUTAZIONI, rng)
    reg("minimo", mn, "pct2", "Media più bassa tra i dodici mesi", "funzioni.minimo_permutazione")
    reg("pval_min", pm, "pval", "P-value della media più bassa dei dodici mesi, per permutazione dei rendimenti tra i mesi", "funzioni.minimo_permutazione")

    anni = sorted(set(rs.index.year))
    meta_anno = anni[len(anni) // 2]
    reg("anno_meta", meta_anno, "annonum", "Anno che divide il campione a metà", "dati/")
    for tag, sel in (("p", rs.index.year < meta_anno), ("d", rs.index.year >= meta_anno)):
        x = rs[sel]
        mp = mese_peggiore(x)
        reg(f"peggiore_{tag}", MESI[mp - 1], "testo", f"Mese con la media più bassa, metà {'prima' if tag == 'p' else 'seconda'}", "pandas")
        reg(f"sett_media_{tag}", float(x[x.index.month == config.MESE].mean()), "pct2", f"Rendimento medio di settembre, metà {'prima' if tag == 'p' else 'seconda'}", "pandas")
        reg(f"sett_rango_{tag}", int(x.groupby(x.index.month).mean().sort_values().index.tolist().index(config.MESE) + 1), "int", f"Posizione di settembre dal peggiore, metà {'prima' if tag == 'p' else 'seconda'}", "pandas")

    s = rs[sett].sort_values()
    pd.DataFrame({"anno": s.index.year, "rendimento": s.to_numpy()}).to_csv(config.RISULTATI / "settembri.csv", index=False, float_format="%.5f")
    reg("peggiore1", float(s.iloc[0]), "pct1", "Il settembre peggiore", "pandas")
    reg("anno_peggiore1", str(int(s.index[0].year)), "testo", "Anno del settembre peggiore", "pandas")
    senza = s.iloc[config.N_PEGGIORI:]
    reg("sett_media_senza", float(senza.mean()), "pct2", "Rendimento medio di settembre senza i tre peggiori", "pandas")

    h = [RIS["rango_sett"]["valore"] == 1, p < config.SOGLIA_P, pm >= config.SOGLIA_P,
         RIS["peggiore_p"]["valore"] == "settembre" and RIS["peggiore_d"]["valore"] == "settembre",
         RIS["sett_neg"]["valore"] > 0.5, float(senza.mean()) > 0]
    for i, v in enumerate(h, 1):
        reg(f"h{i}_esito", int(bool(v)), "int", f"Ipotesi {i}: 1 se confermata, 0 se smentita", "confronto con config.py")
    reg("n_conf", int(sum(bool(v) for v in h)), "int", "Ipotesi confermate (su 6)", "confronto con config.py")

    meta = {"generato": datetime.now(timezone.utc).isoformat(timespec="seconds"), "python": platform.python_version(),
            "numpy": np.__version__, "pandas": pd.__version__, "seme": config.SEME, "ipotesi": config.IPOTESI}
    (config.RISULTATI / "metriche.json").write_text(json.dumps({"meta": meta, "valori": RIS}, ensure_ascii=False, indent=1), encoding="utf-8")
    print(f"Scritte {len(RIS)} metriche.")


if __name__ == "__main__":
    main()

Tutti i parametri stanno qui; gli script numerati li leggono da questo file. Le ipotesi sono scritte prima di guardare i risultati di QUESTO articolo.
Attenzione: nell'articolo 162 (effetto gennaio) abbiamo già visto la classifica dei mesi dell'S&P 500 sull'intero periodo, e settembre era l'ultimo. Per questo la prima
ipotesi è dichiarata come già nota, e le altre riguardano aspetti che non avevamo guardato: significatività, correzione per la scelta del peggiore dei dodici mesi, stabilità
tra le due metà del campione.

«Settembre è il mese peggiore dell'anno». Si prova sull'S&P 500 (indice di prezzo, dal 1927, rendimenti mensili da fine mese a fine mese, senza dividendi). Il punto delicato:
tra dodici mesi uno è per forza il peggiore, quindi il confronto onesto non è «settembre contro gli altri» ma «il peggiore dei dodici contro il peggiore che si avrebbe se i

"""Funzioni di calcolo dell'articolo 162: rendimenti mensili e test di permutazione."""
import numpy as np
import pandas as pd


def rendimenti_mensili(close: pd.Series) -> pd.Series:
    """Rendimento semplice da fine mese a fine mese. Il primo mese (incompleto) è perso. Indice: Period mensile."""
    m = close.groupby(close.index.to_period("M")).last()
    return m.pct_change().dropna()


def permutazione_differenza(x: np.ndarray, etichetta: np.ndarray, n: int, rng: np.random.Generator) -> tuple[float, float]:
    """Differenza tra la media dei rendimenti con etichetta vera e quella degli altri; p-value bilaterale per permutazione delle etichette
    (con +1 al numeratore e al denominatore)."""
    x = np.asarray(x, float)
    e = np.asarray(etichetta, bool)
    oss = x[e].mean() - x[~e].mean()
    k = int(e.sum())
    cont = 0
    for _ in range(n):
        idx = rng.permutation(len(x))[:k]
        m = np.zeros(len(x), bool)
        m[idx] = True
        if abs(x[m].mean() - x[~m].mean()) >= abs(oss):
            cont += 1
    return float(oss), float((1 + cont) / (n + 1))


def resto_anno(rend_mensili: pd.Series) -> pd.DataFrame:
    """Per ogni anno completo: rendimento di gennaio e rendimento composto da febbraio a dicembre."""
    righe = []
    for anno, g in rend_mensili.groupby(rend_mensili.index.year):
        if len(g) == 12 and g.index[0].month == 1:
            righe.append({"anno": anno, "gennaio": g.iloc[0], "resto": float(np.prod(1 + g.iloc[1:].to_numpy()) - 1)})
    return pd.DataFrame(righe)


def minimo_permutazione(x: np.ndarray, mese: np.ndarray, n: int, rng: np.random.Generator) -> tuple[float, float]:
    """Media più bassa tra i dodici mesi di calendario: valore osservato e p-value per permutazione dei rendimenti tra i mesi
    (quota di permutazioni con un minimo almeno così basso, con +1). Misura quanto è raro il «peggiore dei dodici», non di un mese scelto prima."""
    x = np.asarray(x, float)
    mese = np.asarray(mese, int)
    conta = np.bincount(mese, minlength=13)[1:]
    oss = (np.bincount(mese, weights=x, minlength=13)[1:] / conta).min()
    cont = 0
    for _ in range(n):
        xp = rng.permutation(x)
        m = (np.bincount(mese, weights=xp, minlength=13)[1:] / conta).min()
        if m <= oss:
            cont += 1
    return float(oss), float((1 + cont) / (n + 1))

"""Controlli di funzioni.py su dati inventati. Se uno fallisce, l'Action si ferma."""
import numpy as np
import pandas as pd

import funzioni as f


def ok(nome, cond):
    assert cond, nome
    print(f"ok  {nome}")


idx = pd.bdate_range("2020-01-01", "2021-12-31")
c = pd.Series(np.arange(1.0, len(idx) + 1), index=idx)
r = f.rendimenti_mensili(c)
ok("il primo mese incompleto è perso", r.index[0] == pd.Period("2020-02"))
m = c.groupby(c.index.to_period("M")).last()
ok("rendimento da fine mese a fine mese", abs(r.iloc[0] - (m.iloc[1] / m.iloc[0] - 1)) < 1e-12)
rng = np.random.default_rng(0)
x = rng.normal(0, 1, 600)
e = np.zeros(600, bool)
e[:50] = True
x[e] += 3.0
d, p = f.permutazione_differenza(x, e, 500, rng)
ok("differenza grande: p-value piccolo", d > 2.5 and p < 0.01)
x2 = rng.normal(0, 1, 600)
d2, p2 = f.permutazione_differenza(x2, e, 500, rng)
ok("nessuna differenza: p-value non piccolo", p2 > 0.05)
idx2 = pd.period_range("2020-01", "2020-12", freq="M")
rm = pd.Series([0.10] + [0.01] * 11, index=idx2)
d3 = f.resto_anno(rm)
ok("resto dell'anno: undici mesi composti", len(d3) == 1 and abs(d3["resto"].iloc[0] - (1.01 ** 11 - 1)) < 1e-12 and d3["gennaio"].iloc[0] == 0.10)
# minimo per permutazione: un mese molto più basso degli altri -> p-value piccolo; nessuna differenza -> p-value grande
mese = np.tile(np.arange(1, 13), 60)
y = rng.normal(0, 1, len(mese))
y[mese == 9] -= 1.5
mn, pm = f.minimo_permutazione(y, mese, 400, rng)
ok("un mese molto più basso: p-value piccolo", pm < 0.01 and mn < -0.8)
y2 = rng.normal(0, 1, len(mese))
mn2, pm2 = f.minimo_permutazione(y2, mese, 400, rng)
ok("nessun mese diverso: il minimo non è raro", pm2 > 0.05)
