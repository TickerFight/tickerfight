"""Passo 1 - Scarica da Yahoo Finance aperture, massimi, minimi e chiusure NON corretti di sette ETF.

Salva dati/<ticker>.csv per ogni ETF e dati/metadati.json (fonte, data, versione, SHA-256).
Gli script successivi leggono SOLO i file salvati.
"""
import hashlib
import json
from datetime import datetime, timezone
from zoneinfo import ZoneInfo

import pandas as pd
import yfinance as yf

import config


def main() -> None:
    config.DATI.mkdir(parents=True, exist_ok=True)
    ora_borsa = datetime.now(ZoneInfo(config.FUSO_BORSA))
    meta = {"fonte": "Yahoo Finance", "libreria": f"yfinance {yf.__version__}",
            "scaricato_il_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"),
            "colonne": "Open, High, Low, Close non corretti (auto_adjust=False)", "serie": {}}
    for t in config.TICKERS:
        g = yf.Ticker(t).history(start=config.INIZIO_DOWNLOAD, auto_adjust=False)
        if g.empty:
            raise SystemExit(f"Nessun dato per {t}")
        df = g[["Open", "High", "Low", "Close", "Volume"]].copy()
        df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
        df.index.name = "Date"
        df = df[~df.index.duplicated(keep="last")].sort_index().dropna(subset=["Close", "High", "Low"])
        if (df[["High", "Low", "Close"]] <= 0).any().any():
            raise SystemExit(f"Prezzi non positivi in {t}")
        if df.index[-1].date() == ora_borsa.date() and ora_borsa.hour < 16:
            df = df.iloc[:-1]
        f = config.file_dati(t)
        df.to_csv(f, float_format="%.6f")
        meta["serie"][t] = {"righe": len(df), "prima": str(df.index[0].date()), "ultima": str(df.index[-1].date()),
                            "sha256": hashlib.sha256(f.read_bytes()).hexdigest()}
        print(t, meta["serie"][t])
    config.FILE_METADATI.write_text(json.dumps(meta, indent=2, ensure_ascii=False), encoding="utf-8")


if __name__ == "__main__":
    main()

"""Passo 2 - Stima dello spread dai prezzi (Corwin-Schultz e Roll) su sette ETF.

Legge SOLO dati/<ticker>.csv e produce in risultati/: metriche.json, per_etf.csv, mensile_spy.csv.
"""
import json
import platform
from datetime import datetime, timezone

import numpy as np
import pandas as pd

import config
import funzioni as fx

RIS: dict = {}
SORGENTE = "02_backtest_e_test.py con funzioni.py"


def reg(chiave, valore, formato, descrizione, origine=SORGENTE):
    if isinstance(valore, (np.floating, np.integer)):
        valore = valore.item()
    if isinstance(valore, float) and not np.isfinite(valore):
        raise SystemExit(f"Numero non finito per {chiave}")
    RIS[chiave] = {"valore": valore, "formato": formato, "descrizione": descrizione, "origine": origine}


def main() -> None:
    config.RISULTATI.mkdir(parents=True, exist_ok=True)
    righe, serie = [], {}
    for t in config.TICKERS:
        df = pd.read_csv(config.file_dati(t), parse_dates=["Date"], index_col="Date").sort_index()
        df = df[(df["High"] >= df["Low"]) & (df["Low"] > 0)]
        raw = pd.Series(fx.cs_grezzo(df["High"].to_numpy(), df["Low"].to_numpy()), index=df.index).dropna()
        cs = raw.clip(lower=0)
        tick = (config.TICK / df["Close"]).mean()
        roll = fx.roll_mensile(df["Close"])
        k = t.lower()
        serie[t] = (df, raw, cs)
        reg(f"{k}_prima", str(df.index[0].date()), "data_lunga", f"Prima seduta di {t}", "dati/")
        reg(f"{k}_n", len(raw), "int", f"Coppie di giorni usate per {t}", "dati/")
        reg(f"{k}_cs", cs.mean(), "pct3", f"Stima media di Corwin-Schultz per {t} (negativi posti a zero)", "funzioni.cs_grezzo")
        reg(f"{k}_cs_med", cs.median(), "pct3", f"Stima mediana di Corwin-Schultz per {t}", "funzioni.cs_grezzo")
        reg(f"{k}_neg", float((raw < 0).mean()), "pct0", f"Quota di stime grezze negative per {t}", "funzioni.cs_grezzo")
        reg(f"{k}_tick", tick, "pct3", f"Minimo teorico dello spread per {t}: un centesimo diviso la chiusura, in media", "config.TICK / Close")
        reg(f"{k}_mult", cs.mean() / tick, "num1", f"Stima media diviso minimo teorico per {t}", "rapporto")
        reg(f"{k}_roll_nan", float(roll.isna().mean()), "pct0", f"Quota di mesi in cui Roll non è definito per {t}", "funzioni.roll_mensile")
        reg(f"{k}_roll_med", float(roll.median()), "pct3", f"Stima mediana di Roll sui mesi in cui è definita, {t}", "funzioni.roll_mensile")
        righe.append({"etf": t, "cs": cs.mean(), "cs_mediana": cs.median(), "neg": (raw < 0).mean(), "tick": tick,
                      "roll_nan": roll.isna().mean(), "roll_mediana": roll.median()})
    pd.DataFrame(righe).to_csv(config.RISULTATI / "per_etf.csv", index=False, float_format="%.6f")

    reg("n_etf", len(config.TICKERS), "int", "ETF analizzati", "config.py")
    df, raw, cs = serie["SPY"]
    reg("data_ultima", str(df.index[-1].date()), "data_lunga", "Ultima seduta dei dati", "dati/")
    reg("tick_dollari", config.TICK, "num2", "Tick minimo in dollari", "config.py")

    # Stima mensile di SPY contro la volatilità del mese
    m_cs = fx.media_mensile(cs)
    m_vol = fx.vol_mensile(df["Close"])
    mm = pd.concat([m_cs.rename("cs"), m_vol.rename("vol")], axis=1).dropna()
    mm.to_csv(config.RISULTATI / "mensile_spy.csv", float_format="%.6f")
    corr = float(mm["cs"].corr(mm["vol"]))
    reg("corr_vol", corr, "num2", "Correlazione mensile tra la stima di SPY e la volatilità giornaliera del mese", "pandas corr")
    reg("n_mesi", len(mm), "int", "Mesi usati per la correlazione", "dati/")
    mediana_mesi = float(m_cs.median())
    mar = float(m_cs[pd.Period(config.MESE_CRISI)])
    reg("mese_crisi", config.MESE_CRISI + "-01", "data_lunga", "Mese del crollo del Covid", "config.py")
    reg("cs_crisi", mar, "pct3", "Stima media di SPY nel mese del crollo", "funzioni")
    reg("cs_med_mesi", mediana_mesi, "pct3", "Mediana delle stime mensili di SPY", "funzioni")
    reg("mult_crisi", mar / mediana_mesi, "num1", "Stima del mese del crollo diviso la mediana dei mesi, SPY", "rapporto")
    prima = cs[cs.index < config.SOGLIA_PRIMA_ANNO].mean()
    seconda = cs[cs.index >= config.SOGLIA_PRIMA_ANNO].mean()
    reg("cs_prima", prima, "pct3", "Stima media di SPY prima del 2013", "funzioni")
    reg("cs_seconda", seconda, "pct3", "Stima media di SPY dal 2013", "funzioni")
    reg("costo_giro", RIS["spy_cs"]["valore"], "pct3", "Costo di un acquisto e una vendita di SPY se si pagasse l'intera stima (un giro completo)", "funzioni")

    # Costanti di soglia
    reg("h1_mult", config.H1_MULTIPLO, "int", "Ipotesi 1: multiplo minimo", "config.py")
    reg("h2_quota", config.H2_QUOTA_NEG, "pct0", "Ipotesi 2: quota minima di stime negative", "config.py")
    reg("h3_corr", config.H3_CORR, "num1", "Ipotesi 3: correlazione minima", "config.py")
    reg("h5_mult", config.H5_MULTIPLO, "int", "Ipotesi 5: multiplo minimo", "config.py")
    reg("h6_quota", config.H6_QUOTA_ROLL, "pct0", "Ipotesi 6: quota minima di mesi con Roll non definito", "config.py")

    h = [RIS["spy_mult"]["valore"] >= config.H1_MULTIPLO,
         RIS["spy_neg"]["valore"] > config.H2_QUOTA_NEG,
         corr > config.H3_CORR,
         RIS["eem_cs"]["valore"] > RIS["spy_cs"]["valore"] and RIS["iwm_cs"]["valore"] > RIS["spy_cs"]["valore"],
         mar / mediana_mesi >= config.H5_MULTIPLO,
         RIS["spy_roll_nan"]["valore"] > config.H6_QUOTA_ROLL]
    for i, v in enumerate(h, 1):
        reg(f"h{i}_esito", int(bool(v)), "int", f"Ipotesi {i}: 1 se confermata, 0 se smentita", "confronto con config.py")
    reg("n_conf", int(sum(bool(v) for v in h)), "int", "Ipotesi confermate (su 6)", "confronto con config.py")

    meta = {"generato": datetime.now(timezone.utc).isoformat(timespec="seconds"), "python": platform.python_version(),
            "numpy": np.__version__, "pandas": pd.__version__, "ipotesi": config.IPOTESI}
    (config.RISULTATI / "metriche.json").write_text(json.dumps({"meta": meta, "valori": RIS}, ensure_ascii=False, indent=1), encoding="utf-8")
    print(f"Scritte {len(RIS)} metriche.")


if __name__ == "__main__":
    main()

"""Parametri dell'articolo n. 103 - "Stimare lo spread denaro-lettera dai prezzi".

Tutti i parametri stanno qui; gli script numerati li leggono da questo file. Le ipotesi sono scritte prima di guardare i risultati.

Lo spread denaro-lettera (la differenza tra il prezzo a cui si compra e quello a cui si vende) è il costo di operare più citato e meno osservato: nei dati
giornalieri gratuiti non ci sono le quotazioni, ci sono solo apertura, massimo, minimo e chiusura. Si provano due stimatori che ricavano lo spread dai soli prezzi:
quello di Corwin e Schultz (dal rapporto tra massimo e minimo su due giorni consecutivi: il massimo contiene lo spread, la varianza cresce con il tempo ma lo spread no)
e quello di Roll (dalla covarianza tra variazioni consecutive della chiusura). Si applicano a sette ETF dal 2003 (prezzi NON corretti, perché i rapporti
intragiornalieri non cambiano ma la chiusura grezza serve per il minimo imposto dal tick). Non abbiamo le quotazioni vere: l'unico riferimento esterno ai dati è il
minimo teorico dello spread, un centesimo di dollaro diviso il prezzo (il tick minimo). Si usa la versione base dello stimatore, senza la correzione per i salti
notturni, e si pongono a zero le stime negative (come fanno gli autori).
"""
from pathlib import Path

BASE = Path(__file__).resolve().parent.parent
DATI = BASE / "dati"
RISULTATI = BASE / "risultati"
FIGURE = BASE / "figure"

TICKERS = ("SPY", "QQQ", "IWM", "EFA", "EEM", "GLD", "TLT")
INIZIO_DOWNLOAD = "2003-01-01"
FUSO_BORSA = "America/New_York"
FILE_METADATI = DATI / "metadati.json"
SENTINELLA = DATI / "spy.csv"
DESCRIZIONE_DATI = "le aperture, i massimi, i minimi e le chiusure giornaliere (non corrette) di sette ETF"
NOMI_ESENTI = ("SPY", "QQQ", "IWM", "EFA", "EEM", "GLD", "TLT", "ETF", "Corwin", "Schultz", "Roll", "S&P 500", "OHLC")

    return DATI / f"{t.lower()}.csv"

TICK = 0.01                          # dollari: il minimo scostamento di prezzo
GIORNI_ANNO = 252
MESE_CRISI = "2020-03"               # il mese del crollo del Covid

"""Funzioni di calcolo dell'articolo 103: stimatori dello spread dai prezzi."""
import numpy as np
import pandas as pd

K = 3.0 - 2.0 * np.sqrt(2.0)


def cs_grezzo(h: np.ndarray, l: np.ndarray) -> np.ndarray:
    """Stima di Corwin-Schultz per ogni coppia di giorni consecutivi (t-1, t), assegnata al giorno t. Il primo elemento è nan.
    Può essere negativa. Spread in frazione del prezzo."""
    h, l = np.asarray(h, float), np.asarray(l, float)
    b1 = np.log(h / l) ** 2
    beta = b1[1:] + b1[:-1]
    hh = np.maximum(h[1:], h[:-1])
    ll = np.minimum(l[1:], l[:-1])
    gamma = np.log(hh / ll) ** 2
    alfa = (np.sqrt(2 * beta) - np.sqrt(beta)) / K - np.sqrt(gamma / K)
    s = 2 * (np.exp(alfa) - 1) / (1 + np.exp(alfa))
    return np.concatenate([[np.nan], s])


def roll_mensile(close: pd.Series) -> pd.Series:
    """Stimatore di Roll per mese: 2*sqrt(-cov(dp_t, dp_t-1)) / prezzo medio; nan se la covarianza è >= 0."""
    dp = close.diff()
    out = {}
    for mese, g in dp.groupby(close.index.to_period("M")):
        d = g.dropna().to_numpy()
        if len(d) < 10:
            continue
        cov = np.mean((d[1:] - d.mean()) * (d[:-1] - d.mean()))
        pm = close[close.index.to_period("M") == mese].mean()
        out[mese] = 2 * np.sqrt(-cov) / pm if cov < 0 else np.nan
    return pd.Series(out)


def media_mensile(serie: pd.Series) -> pd.Series:
    return serie.groupby(serie.index.to_period("M")).mean()


def vol_mensile(close: pd.Series) -> pd.Series:
    r = np.log(close).diff()
    return r.groupby(r.index.to_period("M")).std()
flusso/verifica_motore.py
"""Controlli di funzioni.py su dati inventati. Se uno fallisce, l'Action si ferma."""
import numpy as np
import pandas as pd

import funzioni as f


def ok(nome, cond):
    assert cond, nome
    print(f"ok  {nome}")


# 1. Corwin-Schultz: serie con spread vero noto. Prezzo efficiente = random walk; massimo = max(p)*(1+s/2), minimo = min(p)*(1-s/2).
rng = np.random.default_rng(3)
n, S = 4000, 0.004
cl = 100 * np.exp(np.cumsum(rng.normal(0, 0.004, n)))
amp = np.abs(rng.normal(0, 0.006, n)) + 0.0005
h = cl * (1 + amp) * (1 + S / 2)
l = cl * (1 - amp) * (1 - S / 2)
est = np.nanmean(f.cs_grezzo(h, l))
ok("spread noto 0,4%: la stima è dello stesso ordine", 0.001 < est < 0.012)
# 2. prezzo fermo, massimo uguale al minimo: la stima è zero
e0 = f.cs_grezzo(np.full(50, 100.0), np.full(50, 100.0))
ok("prezzo fermo: stima zero", abs(np.nanmean(e0)) < 1e-12)
# 3. il primo elemento è nan e la lunghezza è conservata
ok("lunghezza e primo elemento", len(est * np.ones(1)) == 1 and np.isnan(f.cs_grezzo(h, l)[0]) and len(f.cs_grezzo(h, l)) == n)
# 4. Roll: serie con rimbalzo denaro-lettera: p = m + s/2 * q, q = ±1 casuale; la covarianza è -(s/2)^2, quindi lo spread vale s
idx = pd.bdate_range("2020-01-01", periods=500)
q = rng.choice([-1.0, 1.0], 500)
m = np.full(500, 100.0)
s = 0.5
p = pd.Series(m + s / 2 * q, index=idx)
r = f.roll_mensile(p).dropna()
ok("Roll recupera lo spread del rimbalzo (0,5 su 100)", abs(r.mean() - 0.005) < 0.0015)
# 5. Roll su un trend: covarianza positiva -> nan
trend = pd.Series(np.cumsum(1 + 0.5 * np.sin(np.arange(300) / 8.0)) + 100, index=pd.bdate_range("2020-01-01", periods=300))
ok("Roll con covarianza positiva è nan", f.roll_mensile(trend).isna().all())
# 6. medie mensili
ser = pd.Series(np.arange(40.0), index=pd.bdate_range("2020-01-01", periods=40))
mm = f.media_mensile(ser)
ok("media mensile", abs(mm.iloc[0] - ser[:23].mean()) < 1e-9)
