Quante operazioni all'anno si possono permettere 1.000 euro?

Con 1.000 € di capitale, 2 € a ordine (o 0,1%) e la tassa del 26%, quanti ordini all'anno si possono fare prima che i costi si mangino il rendimento? Per misurare solo il peso della frequenza si usa una «regola a cicli» senza alcun segnale di mercato: si è investiti tutte le sedute tranne una ogni L; la si prova a sei frequenze e la si confronta con il buy & hold e con le scimmie.

L'articolo esce a breve sul blog⬇ Scarica il codice (.zip)

Il disegno

  • Regola a cicli: ciclo di L sedute (252, 126, 63, 21, 10, 5): L−1 sedute investite e una fuori; vendita e riacquisto alla chiusura. Nessun segnale di mercato, nessun look-ahead.
  • Strumenti: un ETF azionario mondiale e un ETF sull'S&P 500 (euro, Milano), più l'indice S&P 500 in dollari convertito in euro.
  • Condizioni: 1.000 €; commissione 0 €, 2 € a ordine e 0,1%; tassa 26% con minusvalenze riportate; dividendi, bollo e spread non considerati.
  • Soglia di sostenibilità (scritta prima): perdita di rendimento annuo netto, rispetto al buy & hold, non sopra 0,5 punti con 2 € a ordine.
  • Test: scimmie (stesso numero di periodi e stessa durata totale), 18 test con la correzione di Holm.

flusso/01_scarica_dati.py
"""Passo 1 - Scarica da Yahoo Finance le serie giornaliere degli strumenti e del cambio euro-dollaro.

Salva dati/<ticker>.csv (Date, Open, High, Low, Close, Volume) e dati/metadati.json con la fonte, la data del download,
la versione di yfinance e l'impronta SHA-256 di ogni file. Gli script successivi leggono SOLO i file salvati: i risultati
restano riproducibili anche se Yahoo corregge i dati in futuro.
"""
import hashlib
import json
import time
from datetime import datetime, timezone
from zoneinfo import ZoneInfo

import pandas as pd
import yfinance as yf

import config


def scarica(ticker: str) -> pd.DataFrame:
    for tentativo in range(3):
        grezzo = yf.Ticker(ticker).history(start=config.INIZIO_DOWNLOAD, auto_adjust=False)
        if not grezzo.empty:
            return grezzo
        print(f"{ticker}: nessun dato (tentativo {tentativo + 1})")
        time.sleep(10)
    return pd.DataFrame()


def main() -> None:
    config.DATI.mkdir(parents=True, exist_ok=True)
    ora_borsa = datetime.now(ZoneInfo(config.FUSO_BORSA))
    meta = {"fonte": "Yahoo Finance", "url_fonte": config.URL_FONTE, "libreria": f"yfinance {yf.__version__}",
            "scaricato_il_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"), "strumenti": []}
    elenco = [(t, n, v) for t, n, v in config.STRUMENTI] + [(config.TICKER_CAMBIO, "cambio euro-dollaro", "USD")]
    for ticker, nome, valuta in elenco:
        riga = {"ticker": ticker, "nome": nome, "valuta": valuta}
        grezzo = scarica(ticker)
        if grezzo.empty:
            riga["stato"] = "scartato: nessun dato restituito da Yahoo Finance"
            meta["strumenti"].append(riga)
            continue
        df = grezzo[["Open", "High", "Low", "Close", "Volume"]].copy()
        df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
        df.index.name = "Date"
        df = df[~df.index.duplicated(keep="last")].sort_index().dropna(subset=["Close"])
        df = df[df["Close"] > 0]
        if df.index[-1].date() == ora_borsa.date() and ora_borsa.hour < 18:       # seduta in corso: barra incompleta
            df = df.iloc[:-1]
        salti = (df["Close"].pct_change().abs() > 0.25).sum()
        file = config.file_dati(ticker)
        df.to_csv(file, float_format="%.6f")
        riga.update({"stato": "ok", "file": file.name, "righe": len(df), "prima_data": str(df.index[0].date()),
                     "ultima_data": str(df.index[-1].date()), "variazioni_giornaliere_oltre_25_per_cento": int(salti),
                     "sha256": hashlib.sha256(file.read_bytes()).hexdigest()})
        meta["strumenti"].append(riga)
        print(f"{ticker}: {len(df)} righe, dal {riga['prima_data']} al {riga['ultima_data']}, salti oltre il 25%: {salti}")
    config.FILE_METADATI.write_text(json.dumps(meta, indent=2, ensure_ascii=False), encoding="utf-8")
    ok = {r["ticker"] for r in meta["strumenti"] if r["stato"] == "ok"}
    if config.TICKER_CAMBIO not in ok and any(v == "USD" for _, _, v in config.STRUMENTI):
        print("Attenzione: cambio non scaricato, gli strumenti in dollari verranno scartati.")
    if sum(1 for t, _, _ in config.STRUMENTI if t in ok) < 1:
        raise SystemExit("Nessuno strumento scaricato.")


if __name__ == "__main__":
    main()
flusso/02_backtest_e_test.pybuy & hold, sei frequenze, scimmie, correzione di Holm
"""Passo 2 - La regola a cicli a sei frequenze, con le condizioni di chi parte con 1.000 euro.

Legge SOLO i file in dati/ e produce in risultati/:
  metriche.json     ogni numero dell'articolo, con descrizione e origine di calcolo
  prove.csv         per strumento, frequenza e scenario: valore finale, rendimento annuo, ordini, costi, tasse
  scimmie_finali.csv  valore finale netto di ogni scimmia (scenario 2 euro), per strumento e frequenza

Regole: buy & hold ("bh"); "cNN" = regola a cicli con ciclo di NN sedute (NN-1 dentro, 1 fuori).
Quattro scenari: lordo (nessun costo, nessuna tassa) e tre scenari di commissione, tutti con la tassa del 26%.
"""
import json
import platform
from datetime import datetime, timezone

import numpy as np
import pandas as pd

import config
import simulazione as s

RIS: dict = {}
SORGENTE = "02_backtest_e_test.py con simulazione.py"
SCEN = [("lordo", 0.0, 0.0, 0.0)] + [(f"c{k}", f, p, config.ALIQUOTA) for k, (_, f, p) in enumerate(config.SCENARI_COSTO)]
M = f"c{config.SCENARIO_PRINCIPALE}"
NOME_SCEN = {f"c{k}": nome for k, (nome, _, _) in enumerate(config.SCENARI_COSTO)}
NOME_SCEN["lordo"] = "lordo"
RID = [f"y{L}" for L in config.CICLI]          # y252, y126, ...


def reg(chiave, valore, formato, descrizione, origine=SORGENTE):
    if isinstance(valore, (np.floating, np.integer)):
        valore = valore.item()
    if isinstance(valore, float) and not np.isfinite(valore):
        raise SystemExit(f"Numero non finito per {chiave}")
    RIS[chiave] = {"valore": valore, "formato": formato, "descrizione": descrizione, "origine": origine}


def holm(p: dict) -> dict:
    ordine = sorted(p, key=p.get)
    m = len(ordine)
    out, massimo = {}, 0.0
    for k, nome in enumerate(ordine):
        massimo = max(massimo, min(1.0, (m - k) * p[nome]))
        out[nome] = massimo
    return out


def agenda(n: int, i0: int, L: int) -> np.ndarray:
    return np.array([(i - i0) % L != L - 1 for i in range(n)])


def analizza(k_str, tk, nome, valuta, meta, fx_df):
    riga = [r for r in meta["strumenti"] if r["ticker"] == tk][0]
    if riga["stato"] != "ok":
        print(f"{tk}: scartato, {riga['stato']}")
        return None
    t = s.carica_csv(config.file_dati(tk))
    if valuta == "USD":
        if fx_df is None:
            return None
        df = s.prepara(t, fx_df, 1)
    else:
        df = s.prepara(t, None, 1)
    i0 = s.primo_indice(df)
    anni_test = (df.index[-1] - df.index[i0]).days / 365.25
    if anni_test < config.STORIA_MINIMA_ANNI:
        print(f"{tk}: scartato, solo {anni_test:.1f} anni")
        return None
    n = len(df)
    C, FX = df["Close"].to_numpy(float), df["fx"].to_numpy(float)
    anni_idx = df.index.year.to_numpy()
    sim, prove = {}, []
    for rid in ["bh"] + RID:
        ob = None if rid == "bh" else agenda(n, i0, int(rid[1:]))
        for sid, f, p, al in SCEN:
            sim[(rid, sid)] = s.simula(df, i0, config.CAPITALE, f, p, al, rid != "bh", obiettivo=ob)
    for (rid, sid), r in sim.items():
        prove.append({"strumento": tk, "regola": rid, "scenario": sid, "finale": r["valore_finale_eur"], "cagr": r["cagr"],
                      "ordini": r["n_acquisti"] + r["n_vendite"], "commissioni": r["commissioni_eur"], "tasse": r["tasse_eur"],
                      "quota_investito": r["quota_giorni_investito"]})
    rng = np.random.default_rng([config.SEME, k_str, 1])
    N = n - i0
    scim = {}
    f1, p1, a1 = [(f, p, al) for sid, f, p, al in SCEN if sid == M][0]
    for rid in RID:
        a, b = s.blocchi(sim[(rid, "lordo")])
        k, D = len(a), int((b - a).sum())
        fin = np.empty(config.N_SCIMMIE)
        for j in range(config.N_SCIMMIE):
            ia, ib = s.scimmia(k, D, N, rng)
            fin[j] = s.valuta_blocchi(C, FX, anni_idx, ia + i0, ib + i0, config.CAPITALE, f1, p1, a1)[0]
        scim[rid] = {"k": k, "finali": fin}
    return dict(tk=tk, sl=config.slug(tk), nome=nome, df=df, i0=i0, sim=sim, prove=prove, scim=scim, anni=anni_test)


def main() -> None:
    config.RISULTATI.mkdir(parents=True, exist_ok=True)
    meta = json.loads(config.FILE_METADATI.read_text(encoding="utf-8"))
    cambio_ok = any(r["ticker"] == config.TICKER_CAMBIO and r["stato"] == "ok" for r in meta["strumenti"])
    fx_df = s.carica_csv(config.file_dati(config.TICKER_CAMBIO)) if cambio_ok else None
    res = []
    for k, (tk, nome, valuta) in enumerate(config.STRUMENTI):
        r = analizza(k, tk, nome, valuta, meta, fx_df)
        if r:
            res.append(r)
            print(f"{tk}: {r['anni']:.1f} anni; netto 2 euro: bh {r['sim'][('bh', M)]['valore_finale_eur']:.0f}, " +
                  ", ".join(f"{x} {r['sim'][(x, M)]['valore_finale_eur']:.0f}" for x in RID))
    if len(res) < 2:
        raise SystemExit("Troppo pochi strumenti utilizzabili.")

    pgrezzo = {}
    for r in res:
        for rid in RID:
            reale = r["sim"][(rid, M)]["valore_finale_eur"]
            pgrezzo[(r["tk"], rid)] = (int((r["scim"][rid]["finali"] >= reale).sum()) + 1) / (config.N_SCIMMIE + 1)
    ph = holm(pgrezzo)

    for r in res:
        sl, nome, sm = r["sl"], r["nome"], r["sim"]
        reg(f"{sl}_nome", nome, "testo", "Nome dello strumento", "config.py, STRUMENTI")
        reg(f"{sl}_data_inizio", str(r["df"].index[r["i0"]].date()), "data_lunga", f"Prima seduta del test su {nome}", "dati/")
        reg(f"{sl}_anni", r["anni"], "num1", f"Anni del test su {nome}")
        ordini_anno = {}
        for rid in ["bh"] + RID:
            for sid in NOME_SCEN:
                x = sm[(rid, sid)]
                reg(f"{sl}_{rid}_{sid}_finale", x["valore_finale_eur"], "int", f"Valore finale di {config.CAPITALE:.0f} euro ({rid}, scenario {NOME_SCEN[sid]}), {nome}")
                reg(f"{sl}_{rid}_{sid}_cagr", x["cagr"], "pct1", f"Rendimento annuo composto ({rid}, scenario {NOME_SCEN[sid]}), {nome}")
                if sid != "lordo":
                    reg(f"{sl}_{rid}_{sid}_costo", x["commissioni_eur"], "int", f"Commissioni pagate in tutto ({rid}, scenario {NOME_SCEN[sid]}), {nome}")
                    reg(f"{sl}_{rid}_{sid}_tasse", x["tasse_eur"], "int", f"Tasse pagate in tutto ({rid}, scenario {NOME_SCEN[sid]}), {nome}")
                    reg(f"{sl}_{rid}_{sid}_perdita", sm[("bh", sid)]["cagr"] - x["cagr"], "pp1", f"Rendimento annuo netto del buy & hold meno quello della frequenza {rid}, scenario {NOME_SCEN[sid]}, in punti percentuali, {nome}")
            x = sm[(rid, M)]
            oa = (x["n_acquisti"] + x["n_vendite"]) / r["anni"]
            nom = 2 * config.SEDUTE_ANNO / int(rid[1:]) if rid != "bh" else 2 / r["anni"]
            ordini_anno[rid] = nom
            eq = x["equity"]
            morto = np.flatnonzero(eq <= 1e-9)
            vita = (r["df"].index[r["i0"] + int(morto[0])] - r["df"].index[r["i0"]]).days / 365.25 if len(morto) else r["anni"]
            reg(f"{sl}_{rid}_ordini_nom", nom, "num1", f"Ordini all'anno previsti dalla regola se il capitale non si esaurisce ({rid}), {nome}", "config.py, CICLI")
            reg(f"{sl}_{rid}_anni_vita", vita, "num1", f"Anni dopo i quali il capitale è esaurito (scenario principale, {rid}); se non si esaurisce, gli anni del test, {nome}")
            reg(f"{sl}_{rid}_esaurito", int(len(morto) > 0), "int", f"1 se il capitale si esaurisce nello scenario principale ({rid}), {nome}")
            reg(f"{sl}_{rid}_ordini_anno", oa, "num1", f"Ordini all'anno ({rid}), {nome}")
            reg(f"{sl}_{rid}_ordini", x["n_acquisti"] + x["n_vendite"], "int", f"Ordini totali ({rid}), {nome}")
            reg(f"{sl}_{rid}_quota", x["quota_giorni_investito"], "pct1", f"Quota di sedute investite ({rid}), {nome}")
            reg(f"{sl}_{rid}_costo_anno_quota", x["commissioni_eur"] / r["anni"] / config.CAPITALE, "pct1", f"Commissioni pagate in media ogni anno, in quota del capitale iniziale ({rid}, scenario principale), {nome}")
            reg(f"{sl}_{rid}_tasse_anno", x["tasse_eur"] / r["anni"], "num1", f"Tasse pagate in media ogni anno in euro ({rid}, scenario principale), {nome}")
            reg(f"{sl}_{rid}_costo_anno", x["commissioni_eur"] / r["anni"], "num1", f"Commissioni pagate in media ogni anno in euro ({rid}, scenario principale), {nome}")
        for rid in RID:
            fin = r["scim"][rid]["finali"]
            reale = sm[(rid, M)]["valore_finale_eur"]
            reg(f"{sl}_{rid}_sc_mediana", float(np.median(fin)), "int", f"Valore finale mediano delle scimmie ({rid}, scenario principale), {nome}", "scimmie_finali.csv")
            reg(f"{sl}_{rid}_sc_p5", float(np.quantile(fin, 0.05)), "int", f"5° percentile del valore finale delle scimmie ({rid}), {nome}", "scimmie_finali.csv")
            reg(f"{sl}_{rid}_sc_p95", float(np.quantile(fin, 0.95)), "int", f"95° percentile del valore finale delle scimmie ({rid}), {nome}", "scimmie_finali.csv")
            reg(f"{sl}_{rid}_p_sc", pgrezzo[(r["tk"], rid)], "pval", f"p-value a una coda: quota di scimmie che fanno almeno quanto la regola ({rid}), {nome}", "scimmie_finali.csv")
            reg(f"{sl}_{rid}_p_holm", ph[(r["tk"], rid)], "pval", f"p-value delle scimmie corretto con Holm sui 18 test ({rid}), {nome}", "scimmie_finali.csv")
        # frequenza massima sostenibile (perdita netta con 2 euro non sopra la soglia)
        buone = [rid for rid in RID if sm[("bh", M)]["cagr"] - sm[(rid, M)]["cagr"] <= config.SOGLIA_SOSTENIBILE]
        sost = max((ordini_anno[x] for x in buone), default=0.0)
        r["sost"] = sost
        r["buone"] = buone
        reg(f"{sl}_sost_ordini", sost, "num1", f"Ordini all'anno della frequenza più alta provata con perdita netta (2 euro) sotto la soglia, {nome}; zero se nessuna lo è")
        reg(f"{sl}_pareggio_ordini", sm[("bh", "lordo")]["cagr"] * config.CAPITALE / config.SCENARI_COSTO[config.SCENARIO_PRINCIPALE][1], "num1", f"Ordini all'anno con cui le commissioni fisse (non composte) eguagliano il rendimento annuo lordo del buy & hold su {config.CAPITALE:.0f} euro, {nome}", "rendimento lordo del buy & hold diviso la commissione, per 1.000 euro")

    # --- Tabella teorica (nessun dato di mercato) --------------------------------------------------------------------
    fissa = config.SCENARI_COSTO[config.SCENARIO_PRINCIPALE][1]
    perc = config.SCENARI_COSTO[2][2]
    for n_ord in config.ORDINI_TABELLA:
        reg(f"ana_{n_ord}_ordini", n_ord, "int", "Ordini all'anno nella tabella teorica", "config.py, ORDINI_TABELLA")
        reg(f"ana_{n_ord}_fisso", n_ord * fissa / config.CAPITALE, "pct1", f"Costo annuo di {n_ord} ordini da {fissa:.0f} euro su {config.CAPITALE:.0f} euro, in quota del capitale", "ordini x commissione / capitale")
        reg(f"ana_{n_ord}_perc", n_ord * perc, "pct1", f"Costo annuo di {n_ord} ordini con commissione proporzionale su capitale invariato, in quota del capitale", "ordini x aliquota")
    reg("ana_ordini_soglia", config.SOGLIA_SOSTENIBILE * config.CAPITALE / fissa, "num1", "Ordini all'anno che consumano da soli la soglia di sostenibilità con la commissione fissa", "soglia x capitale / commissione")

    # --- Ipotesi ----------------------------------------------------------------------------------------------------
    nS = len(res)
    reg("n_strumenti", nS, "int", "Strumenti utilizzabili", "config.py e dati/")
    reg("n_scartati", len(config.STRUMENTI) - nS, "int", "Strumenti scartati", "metadati.json")
    reg("n_h1", sum(r["sim"][("y10", M)]["cagr"] < 0 for r in res), "int", "Strumenti in cui il rendimento annuo netto (2 euro) del ciclo da 10 sedute è negativo")
    reg("n_h2", sum(r["sim"][("bh", M)]["cagr"] - r["sim"][("y252", M)]["cagr"] <= config.SOGLIA_SOSTENIBILE for r in res), "int", "Strumenti in cui il ciclo annuale perde meno della soglia rispetto al buy & hold (2 euro)")
    reg("n_h3", sum(r["sost"] < 10 for r in res), "int", "Strumenti in cui gli ordini all'anno sostenibili sono meno di 10")
    reg("n_h4", sum(r["sim"][("y5", "c0")]["cagr"] < r["sim"][("bh", "c0")]["cagr"] for r in res), "int", "Strumenti in cui, con commissione zero e tassa, il ciclo da 5 sedute rende meno del buy & hold")
    reg("n_h5", sum(all(r["sim"][("bh", "c2")]["cagr"] - r["sim"][(x, "c2")]["cagr"] < r["sim"][("bh", "c1")]["cagr"] - r["sim"][(x, "c1")]["cagr"] for x in ("y21", "y10", "y5")) for r in res), "int", "Strumenti in cui la commissione 0,1% perde meno della commissione da 2 euro alle tre frequenze alte (21, 10, 5 sedute)")
    reg("n_h6_signif", sum(v < 0.05 for v in ph.values()), "int", "Test sulle scimmie significativi dopo Holm (su 18)", "scimmie_finali.csv")
    reg("n_h6_grezzo", sum(v < 0.05 for v in pgrezzo.values()), "int", "Test sulle scimmie con p-value sotto 0,05 senza correzione (su 18)", "scimmie_finali.csv")
    for x in ("y10", "y5"):
        reg(f"n_esaurito_{x}", sum(int(r["sim"][(x, M)]["equity"].min() <= 1e-9) for r in res), "int", f"Strumenti in cui il capitale si esaurisce con la frequenza {x}, scenario principale")
    reg("n_prove", nS * (1 + len(RID)) * len(NOME_SCEN), "int", "Simulazioni complete: strumenti x (buy & hold + frequenze) x scenari", "config.py")
    reg("n_test", nS * len(RID), "int", "Test di ipotesi con correzione di Holm", "config.py")
    reg("n_frequenze", len(RID), "int", "Frequenze provate", "config.py, CICLI")
    reg("n_scimmie", config.N_SCIMMIE, "int", "Scimmie per strumento e frequenza", "config.py")
    reg("seme", config.SEME, "annonum", "Seme dei sorteggi", "config.py")
    reg("capitale", config.CAPITALE, "int", "Capitale iniziale in euro", "config.py")
    reg("commissione_fissa", fissa, "int", "Commissione fissa per ordine, scenario principale, in euro", "config.py")
    reg("commissione_perc", perc, "pct1", "Commissione proporzionale, terzo scenario", "config.py")
    reg("aliquota", config.ALIQUOTA, "pct0", "Aliquota sulle plusvalenze", "config.py")
    reg("soglia", config.SOGLIA_SOSTENIBILE, "pp1", "Soglia di sostenibilità: perdita annua massima rispetto al buy & hold, in punti percentuali", "config.py")
    for L in config.CICLI:
        reg(f"ciclo_{L}", L, "int", "Sedute di un ciclo", "config.py, CICLI")
    reg("data_scarico", meta["scaricato_il_utc"][:10], "data", "Data del download dei dati", "dati/metadati.json")
    reg("data_ultima", max(str(r["df"].index[-1].date()) for r in res), "data_lunga", "Ultima seduta presente nei dati", "dati/")

flusso/simulazione.pymotore in euro: ordini, commissioni, tassa, minusvalenze, scimmie
"""Motore di simulazione in euro di una regola su un singolo strumento (trend con stop loss facoltativo, oppure un'agenda di
posizione stabilita in anticipo, `obiettivo`).

Regole (da dichiarare nell'articolo):
  - segnale: investito se la chiusura sta sopra la media mobile semplice (calcolata sulle chiusure);
  - il segnale a fine seduta t decide l'ordine della seduta t+1, eseguito alla chiusura di t+1 (nessun look-ahead),
    con il cambio della stessa seduta;
  - stop loss: se alla chiusura di t la chiusura è scesa dello `stop` o più sotto il prezzo di acquisto (nella valuta
    dello strumento) e il segnale di trend è ancora vero, si vende alla chiusura di t+1; si rientra solo dopo che il
    segnale è tornato falso e poi vero (se il segnale è già falso, l'uscita è quella del trend, senza blocco);
  - azioni frazionate, si investe tutta la cassa; commissione per ogni ordine = parte fissa + parte proporzionale
    all'importo investito o incassato;
  - la cassa non rende nulla; i dividendi non sono considerati;
  - tassa sulle plusvalenze calcolate in euro (cambio incluso), pagata a ogni vendita come nel regime amministrato;
    la commissione si somma al costo di acquisto e si sottrae dall'incasso; le minusvalenze compensano le plusvalenze
    successive fino al quarto anno seguente;
  - a fine periodo si vende tutto, con commissione e tassa, per confrontare cifre nette.
Non modellati: imposta di bollo, tassa sulle transazioni finanziarie, spread di cambio, scarto tra denaro e lettera.
"""
from pathlib import Path

import numpy as np
import pandas as pd


class Zaino:
    """Minusvalenze compensabili con le plusvalenze successive, fino al 4° anno seguente."""

    def __init__(self, aliquota: float):
        self.aliquota = aliquota
        self.voci: list[list[float]] = []   # [ultimo anno utilizzabile, importo residuo]

    def tassa(self, anno: int, guadagno: float) -> float:
        if guadagno < 0:
            self.voci.append([anno + 4, -guadagno])
            return 0.0
        residuo = guadagno
        for voce in sorted(self.voci, key=lambda v: v[0]):      # prima le più vecchie
            if voce[0] < anno or voce[1] <= 0:
                continue
            usa = min(voce[1], residuo)
            voce[1] -= usa
            residuo -= usa
            if residuo <= 0:
                break
        return self.aliquota * residuo


def carica_csv(percorso: Path) -> pd.DataFrame:
    df = pd.read_csv(percorso, parse_dates=["Date"], index_col="Date").sort_index()
    df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
    df = df[~df.index.duplicated(keep="last")]
    if (df["Close"] <= 0).any() or df["Close"].isna().any():
        raise SystemExit(f"{percorso}: chiusure mancanti o non positive.")
    return df


def prepara(titolo: pd.DataFrame, cambio: pd.DataFrame | None, media: int) -> pd.DataFrame:
    """Chiusure, cambio (dollari per euro; 1 per gli strumenti in euro), media mobile e segnale di trend."""
    df = titolo[["Close"]].astype(float).copy()
    if cambio is None:
        df["fx"] = 1.0
    else:
        fx = cambio["Close"].astype(float)
        df["fx"] = fx.reindex(df.index.union(fx.index)).ffill().reindex(df.index)
        df = df.dropna(subset=["fx"])
    df["sma"] = df["Close"].rolling(media).mean()
    df["segnale"] = (df["Close"] > df["sma"]) & df["sma"].notna()
    return df


def primo_indice(df: pd.DataFrame) -> int:
    """Prima seduta il cui segnale precedente è calcolabile con la media completa."""
    return int(np.flatnonzero(df["sma"].notna().to_numpy())[0]) + 1


def _compra_importo(cassa: float, fissa: float, perc: float) -> tuple[float, float]:
    """Dalla cassa all'importo investito e alla commissione pagata: cassa = investito + fissa + perc*investito."""
    investito = (cassa - fissa) / (1.0 + perc)
    return investito, fissa + perc * investito


def valuta_blocchi(C: np.ndarray, FX: np.ndarray, anni: np.ndarray, acquisti, vendite, capitale: float,
                   fissa: float, perc: float, aliquota: float) -> tuple[float, float, float]:
    """Valore finale, commissioni e tasse di una sequenza di blocchi (acquisto alla chiusura a[j], vendita alla chiusura
    b[j]). È la stessa contabilità di `simula`, senza il registro: serve alle scimmie."""
    cassa = capitale
    zaino = Zaino(aliquota)
    comm_tot = tasse_tot = 0.0
    for a, b in zip(acquisti, vendite):
        investito, comm = _compra_importo(cassa, fissa, perc)
        if investito <= 0:
            continue
        base = cassa
        azioni = investito / (C[a] / FX[a])
        lordo = azioni * C[b] / FX[b]
        comm_v = min(fissa + perc * lordo, lordo)
        netto = lordo - comm_v
        tassa = zaino.tassa(int(anni[b]), netto - base)
        comm_tot += comm + comm_v
        tasse_tot += tassa
        cassa = netto - tassa
    return cassa, comm_tot, tasse_tot


def simula(df: pd.DataFrame, i0: int, capitale: float, fissa: float, perc: float, aliquota: float,
           strategia: bool, stop: float | None = None, obiettivo=None) -> dict:
    """strategia=True: regola di trend (con stop loss se `stop` è un numero); False: buy & hold (compra a i0, vende alla fine)."""
    C, FX = (df[c].to_numpy(float) for c in ("Close", "fx"))
    SEG = df["segnale"].to_numpy(bool)
    date = df.index
    n = len(df)
    cassa, azioni, base = float(capitale), 0.0, 0.0
    entrata_locale = 0.0
    bloccato = False
    zaino = Zaino(aliquota)
    log: list[dict] = []
    tot = {"commissioni": 0.0, "tasse": 0.0}
    equity = np.empty(n - i0)
    in_posizione = np.zeros(n - i0, dtype=bool)

    def compra(i, motivo):
        nonlocal cassa, azioni, base, entrata_locale
        investito, comm = _compra_importo(cassa, fissa, perc)
        if investito <= 0:
            return
        px_eur = C[i] / FX[i]
        azioni = investito / px_eur
        base = cassa
        entrata_locale = C[i]
        tot["commissioni"] += comm
        log.append(dict(i=i, data=str(date[i].date()), lato="acquisto", motivo=motivo, prezzo=C[i], cambio=FX[i],
                        prezzo_eur=px_eur, azioni=azioni, commissione=comm, spesa_eur=cassa,
                        incasso_netto_eur=0.0, plusvalenza_eur=0.0, tassa_eur=0.0))
        cassa = 0.0

    def vendi(i, motivo):
        nonlocal cassa, azioni, base
        px_eur = C[i] / FX[i]
        lordo = azioni * px_eur
        comm = min(fissa + perc * lordo, lordo)       # la commissione non può superare l'incasso: il capitale non va sotto zero
        netto = lordo - comm
        guadagno = netto - base
        tassa = zaino.tassa(int(date[i].year), guadagno)
        tot["commissioni"] += comm
        tot["tasse"] += tassa
        log.append(dict(i=i, data=str(date[i].date()), lato="vendita", motivo=motivo, prezzo=C[i], cambio=FX[i],
                        prezzo_eur=px_eur, azioni=azioni, commissione=comm, spesa_eur=0.0,
                        incasso_netto_eur=netto, plusvalenza_eur=guadagno, tassa_eur=tassa))
        cassa = netto - tassa
        azioni, base = 0.0, 0.0

    for i in range(i0, n):
        if strategia:
            sig = bool(SEG[i - 1]) if obiettivo is None else bool(obiettivo[i])
            if bloccato and not sig:
                bloccato = False
            # uno stop che scatta nella stessa seduta in cui il trend chiede comunque di uscire è un'uscita di trend
            colpito = stop is not None and azioni > 0.0 and sig and C[i - 1] <= entrata_locale * (1.0 - stop)
            if colpito:
                vendi(i, "stop loss")
                bloccato = True
            elif sig and not bloccato and azioni == 0.0:
                compra(i, "segnale")
            elif (not sig) and azioni > 0.0:
                vendi(i, "segnale")
        elif i == i0:
            compra(i, "acquisto iniziale")
        equity[i - i0] = cassa + azioni * C[i] / FX[i]
        in_posizione[i - i0] = azioni > 0.0
    if azioni > 0.0:
        vendi(n - 1, "liquidazione finale")

    picco = np.maximum.accumulate(equity)
    anni = (date[-1] - date[i0]).days / 365.25
    finale = cassa
    return {
        "valore_finale_eur": finale,
        "rendimento_totale": finale / capitale - 1,
        "cagr": (finale / capitale) ** (1 / anni) - 1 if finale > 0 else -1.0,
        "anni": anni,
        "n_acquisti": sum(1 for r in log if r["lato"] == "acquisto"),
        "n_vendite": sum(1 for r in log if r["lato"] == "vendita"),
        "n_stop": sum(1 for r in log if r["motivo"] == "stop loss"),
        "commissioni_eur": tot["commissioni"],
        "tasse_eur": tot["tasse"],
        "quota_giorni_investito": float(in_posizione.mean()),
        "giorni_investito": int(in_posizione.sum()),
        "perdita_massima": float((equity / picco - 1).min()),
        "operazioni": log,
        "equity": equity,
        "in_posizione": in_posizione,
    }


def blocchi(res: dict) -> tuple[np.ndarray, np.ndarray]:
    """Indici di acquisto e di vendita (nelle sedute del dataframe) dei periodi investiti di una simulazione."""
    a = np.array([r["i"] for r in res["operazioni"] if r["lato"] == "acquisto"], dtype=int)
    b = np.array([r["i"] for r in res["operazioni"] if r["lato"] == "vendita"], dtype=int)
    return a, b


def scimmia(k: int, durata_totale: int, n_sedute: int, rng) -> tuple[np.ndarray, np.ndarray]:
    """Una scimmia: k periodi investiti, la cui durata totale (sedute tra acquisto e vendita) è `durata_totale`, piazzati
    a caso nelle `n_sedute` sedute disponibili, senza sovrapposizioni. Restituisce gli indici relativi di acquisto e vendita."""
    if k == 1:
        durate = np.array([durata_totale])
    else:
        tagli = np.sort(rng.choice(durata_totale - 1, size=k - 1, replace=False)) + 1
        durate = np.diff(np.r_[0, tagli, durata_totale])               # k durate, ciascuna almeno 1
    libere = n_sedute - durata_totale - k                              # sedute rimaste fuori dai blocchi (b - a + 1 sedute l'uno)
    if libere < 0:
        raise ValueError("blocchi troppo lunghi per le sedute disponibili")
    tagli_l = np.sort(rng.choice(libere + k, size=k, replace=False)) - np.arange(k)    # k+1 vuoti non negativi
    vuoti = np.diff(np.r_[0, tagli_l, libere])
    inizi = np.cumsum(np.r_[vuoti[0], vuoti[1:k] + durate[:-1] + 1])
    return inizi, inizi + durate

"""Controlli del motore sulla regola a cicli. Usa SOLO dati inventati: nessun numero qui è un risultato."""
import numpy as np
import pandas as pd

import simulazione as s

ok = 0


def verifica(nome, cond, det=""):
    global ok
    if not cond:
        raise SystemExit(f"FALLITO: {nome} {det}")
    ok += 1
    print(f"  ok  {nome}")


def uguali(a, b, tol=1e-9):
    return abs(a - b) <= tol * max(1.0, abs(a), abs(b))


def serie(chiusure, fx=1.0):
    return pd.DataFrame({"Close": np.array(chiusure, float), "fx": fx, "segnale": True},
                        index=pd.bdate_range("2022-03-01", periods=len(chiusure)))


def agenda(n, i0, L):
    return np.array([(i - i0) % L != L - 1 for i in range(n)])


print("Regola a cicli")
d = serie([100.0] * 12)
ob = agenda(12, 1, 4)           # sedute 1,2,3 dentro, 4 fuori, 5,6,7 dentro, 8 fuori, 9,10,11 dentro
r = s.simula(d, 1, 1000, 2.0, 0.0, 0.26, True, obiettivo=ob)
verifica("ordini: 3 acquisti e 3 vendite (compresa la liquidazione finale)", (r["n_acquisti"], r["n_vendite"]) == (3, 3), (r["n_acquisti"], r["n_vendite"]))
verifica("prezzi fermi: ogni ordine costa 2 euro e la cassa cala di 12 euro", uguali(r["valore_finale_eur"], 988.0) and uguali(r["commissioni_eur"], 12.0), r["valore_finale_eur"])
verifica("nessuna tassa se non ci sono plusvalenze", r["tasse_eur"] == 0.0)
verifica("quota di sedute investite", uguali(r["quota_giorni_investito"], 9 / 11), r["quota_giorni_investito"])

# Prezzo che sale del 10% tra l'acquisto e la prima vendita, poi fermo: 1.000 -> 998 investiti -> 1.097,8 -> 1.095,8 netti
d = serie([100, 100, 100, 110, 110, 110, 110, 110, 110])
ob = np.array([1, 1, 1, 0, 1, 1, 1, 1, 1], bool)
r = s.simula(d, 1, 1000, 2.0, 0.0, 0.26, True, obiettivo=ob)
az = 998 / 100
inc = az * 110 - 2
tassa = 0.26 * (inc - 1000)
verifica("prima plusvalenza tassata alla vendita del ciclo", uguali(r["operazioni"][1]["tassa_eur"], tassa), r["operazioni"][1]["tassa_eur"])

# Nessun look-ahead: l'agenda dipende solo dall'indice, cambiare i prezzi futuri non cambia gli ordini già dati
a = serie(list(np.linspace(100, 120, 30)))
b = a.copy(); b.loc[b.index[20:], "Close"] *= 0.5
ob = agenda(30, 1, 5)
ra = s.simula(a, 1, 1000, 2.0, 0.0, 0.26, True, obiettivo=ob)
rb = s.simula(b, 1, 1000, 2.0, 0.0, 0.26, True, obiettivo=ob)
ia = [(o["i"], o["lato"]) for o in ra["operazioni"] if o["i"] < 20]
ib = [(o["i"], o["lato"]) for o in rb["operazioni"] if o["i"] < 20]
verifica("nessun look-ahead", ia == ib)

# Coerenza con il contabile delle scimmie: stessi blocchi, stesso valore finale
rng = np.random.default_rng(1)
C = 100 * np.cumprod(1 + rng.normal(0.0004, 0.01, 400)); d = serie(C)
ob = agenda(400, 1, 10)
r = s.simula(d, 1, 1000, 2.0, 0.001, 0.26, True, obiettivo=ob)
ia, ib = s.blocchi(r)
fin = s.valuta_blocchi(C, np.ones(400), d.index.year.to_numpy(), ia, ib, 1000, 2.0, 0.001, 0.26)[0]
verifica("motore completo e contabile delle scimmie coincidono", uguali(fin, r["valore_finale_eur"]), (fin, r["valore_finale_eur"]))
for _ in range(50):
    k, D = 7, 120
    x, y = s.scimmia(k, D, 400, rng)
    assert len(x) == k and (y - x).sum() == D and (x[1:] > y[:-1]).all() and x[0] >= 0 and y[-1] < 400
