ETF covered call: rendita o rinuncia?

Gli ETF che vendono ogni mese opzioni call sull'indice e distribuiscono il premio (XYLD sull'S&P 500, QYLD sul Nasdaq-100) rendono di più, di meno o con meno rischio dell'indice? E uno stesso rischio si può ottenere più semplicemente con indice e liquidità?

L'articolo esce a breve sul blog⬇ Scarica il codice (.zip)

Il disegno

  • Dati: chiusure aggiustate (distribuzioni reinvestite) di XYLD, QYLD, SPY e QQQ dal giorno in cui esistono tutti e quattro.
  • Coppie: XYLD contro SPY, QYLD contro QQQ.
  • Misure: rendimento annuo composto, volatilità, Sharpe, perdita massima, cattura mensile dei rialzi e dei ribassi.
  • Test: bootstrap a blocchi della differenza di Sharpe; portafoglio di indice e liquidità con la stessa volatilità del covered call; crollo del 2020 e rialzo dei tassi del 2022.

Il primo passo controlla il motore di simulazione su dati inventati (se c'è); poi si scaricano i prezzi da Yahoo Finance in dati/, si calcolano i risultati in risultati/ e i grafici in figure/. Parametri, ipotesi e seme dei sorteggi sono in flusso/config.py. """Passo 1 - Scarica da Yahoo Finance la serie giornaliera di quattro ETF (SPY, QQQ, XYLD, QYLD) e del tasso a tre mesi (^IRX).

Salva, per ogni ETF, dati/<ticker>.csv (Date, Close, Adj Close, Volume) e dati/metadati.json con la fonte,
la data del download, la versione di yfinance e l'impronta SHA-256 di ogni file. Gli script successivi leggono SOLO
i file salvati: i risultati restano riproducibili anche se Yahoo corregge i dati in futuro.
"Adj Close" è il prezzo con i dividendi reinvestiti: è quello che si usa per i rendimenti.
"""
import hashlib
import json
import time
from datetime import datetime, timezone
from zoneinfo import ZoneInfo

import pandas as pd
import yfinance as yf

import config


def scarica(ticker: str) -> pd.DataFrame:
    for tentativo in range(3):
        grezzo = yf.Ticker(ticker).history(start=config.INIZIO_DOWNLOAD, auto_adjust=False)
        if not grezzo.empty:
            return grezzo
        print(f"{ticker}: nessun dato (tentativo {tentativo + 1})")
        time.sleep(10)
    return pd.DataFrame()


def main() -> None:
    config.DATI.mkdir(parents=True, exist_ok=True)
    ora_borsa = datetime.now(ZoneInfo(config.FUSO_BORSA))
    meta = {"fonte": "Yahoo Finance", "url_fonte": config.URL_FONTE, "libreria": f"yfinance {yf.__version__}",
            "scaricato_il_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"), "etf": []}
    nomi = {t: (n, a) for t, n, a in config.ETF}
    nomi[config.TICKER_LIQUIDITA] = ("Tasso a 3 mesi dei Treasury americani (percentuale annua)", "IRX")
    for ticker in [t for t, _, _ in config.ETF] + [config.TICKER_LIQUIDITA]:
        riga = {"ticker": ticker, "nome": nomi[ticker][0], "sigla": nomi[ticker][1]}
        grezzo = scarica(ticker)
        if grezzo.empty or "Adj Close" not in grezzo.columns:
            riga["stato"] = "scartato: nessun dato restituito da Yahoo Finance"
            meta["etf"].append(riga)
            continue
        df = grezzo[["Close", "Adj Close", "Volume"]].copy()
        df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
        df.index.name = "Date"
        df = df[~df.index.duplicated(keep="last")].sort_index().dropna(subset=["Adj Close"])
        df = df[df["Adj Close"] > 0]
        # seduta ancora in corso: barra incompleta, da scartare
        if df.index[-1].date() == ora_borsa.date() and ora_borsa.hour < 18:
            df = df.iloc[:-1]
        file = config.file_dati(ticker)
        df.to_csv(file, float_format="%.6f")
        riga.update({"stato": "ok", "file": file.name, "righe": len(df), "prima_data": str(df.index[0].date()),
                     "ultima_data": str(df.index[-1].date()), "sha256": hashlib.sha256(file.read_bytes()).hexdigest()})
        meta["etf"].append(riga)
        print(f"{ticker}: {len(df)} righe, dal {riga['prima_data']} al {riga['ultima_data']}")
    config.FILE_METADATI.write_text(json.dumps(meta, indent=2, ensure_ascii=False), encoding="utf-8")
    principali = {t for t, _, _ in config.ETF} | {config.TICKER_LIQUIDITA}
    ok = {r["ticker"] for r in meta["etf"] if r["stato"] == "ok"}
    if not principali <= ok:
        raise SystemExit(f"Mancano ETF principali: {sorted(principali - ok)}")


if __name__ == "__main__":
    main()
flusso/02_backtest_e_test.pymetriche, cattura, bootstrap, portafoglio con la stessa volatilità, due fasi di crisi
"""Passo 2 - ETF covered call: XYLD contro SPY e QYLD contro QQQ.

Legge SOLO i file in dati/ e produce in risultati/:
  metriche.json     ogni numero dell'articolo, con descrizione e origine
  curve.csv         valore di 1 dollaro nei quattro ETF (una seduta su cinque)
  mensili.csv       rendimenti mensili dei quattro ETF
  bootstrap.csv     differenze di Sharpe ricampionate (covered call meno indice) per le due coppie
  miste.csv         valore di 1 dollaro del portafoglio con la stessa volatilità, per le due coppie (una seduta su cinque)
"""
import json
from datetime import datetime, timezone

import numpy as np
import pandas as pd

import config
import funzioni as f
import motore as m

RIS: dict = {}
SORGENTE = "02_backtest_e_test.py con motore.py e funzioni.py"
GA = config.GIORNI_ANNO


def reg(chiave, valore, formato, descrizione, origine=SORGENTE):
    if isinstance(valore, (np.floating, np.integer)):
        valore = valore.item()
    if isinstance(valore, float) and not np.isfinite(valore):
        raise SystemExit(f"Numero non finito per {chiave}")
    RIS[chiave] = {"valore": valore, "formato": formato, "descrizione": descrizione, "origine": origine}


def carica(ticker: str, colonna: str) -> pd.Series:
    df = pd.read_csv(config.file_dati(ticker), parse_dates=["Date"], index_col="Date").sort_index()
    df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
    df = df[~df.index.duplicated(keep="last")]
    c = df[colonna].astype(float)
    if c.isna().any() or (c < 0).any():
        raise SystemExit(f"{ticker}: dati mancanti o negativi.")
    return c


def main() -> None:
    config.RISULTATI.mkdir(parents=True, exist_ok=True)
    sigle = [s for _, _, s in config.ETF]
    prezzi = pd.concat({s: carica(t, config.COLONNA_PREZZO) for t, _, s in config.ETF}, axis=1).dropna()
    anni = (prezzi.index[-1] - prezzi.index[0]).days / 365.25
    if anni < config.ANNI_MINIMI:
        raise SystemExit(f"Storia comune di {anni:.1f} anni: troppo corta.")
    tasso = carica(config.TICKER_LIQUIDITA, "Close").reindex(prezzi.index, method="ffill").bfill() / 100.0
    R = prezzi.pct_change().dropna()
    liq = (tasso.reindex(R.index) / GA).to_numpy()          # rendimento giornaliero della liquidità
    fini = m.fine_mese(R.index)
    mensili = m.rendimenti_mensili(R.to_numpy(), fini)
    ci = {s: i for i, s in enumerate(sigle)}
    rng = np.random.default_rng(config.SEME)
    a = (1 - config.LIVELLO_CONFIDENZA) / 2

    reg("data_inizio", str(prezzi.index[0].date()), "data_lunga", "Prima seduta in comune dei quattro ETF", "dati/")
    reg("data_ultima", str(prezzi.index[-1].date()), "data_lunga", "Ultima seduta nei dati", "dati/")
    reg("anni", anni, "num1", "Anni di storia comune", "dati/")
    reg("n_sedute", len(R), "int", "Rendimenti giornalieri in comune", "dati/")
    reg("n_mesi", len(mensili), "int", "Mesi usati per la cattura", "dati/")
    reg("capitale", config.CAPITALE, "int", "Capitale iniziale per tradurre i rendimenti in dollari", "config.py")
    reg("tasso_medio", float(tasso.mean()), "pct1", "Tasso a tre mesi medio nel periodo", "dati/IRX.csv")
    reg("h2_riduzione", config.H2_VOL_RIDUZIONE, "pct0", "Ipotesi 2: riduzione relativa minima della volatilità", "config.py")
    reg("h3_dd", config.H3_DD_MIN, "pct0", "Ipotesi 3: perdita massima minima del covered call", "config.py")
    reg("h4_up", config.H4_UP_MAX, "pct0", "Ipotesi 4: cattura massima dei mesi di rialzo", "config.py")
    reg("h4_down", config.H4_DOWN_MIN, "pct0", "Ipotesi 4: cattura minima dei mesi di ribasso", "config.py")
    reg("n_bootstrap", config.N_BOOTSTRAP, "int", "Ricampionamenti del bootstrap", "config.py")
    reg("blocco", config.BLOCCO_BOOTSTRAP, "int", "Giorni per blocco del bootstrap", "config.py")
    reg("livello_ic", config.LIVELLO_CONFIDENZA, "pct0", "Livello degli intervalli di confidenza", "config.py")

    stat = {}
    for s in sigle:
        r = R[s].to_numpy()
        e = m.equity(r)
        stat[s] = dict(cagr=m.cagr(r, GA), vol=m.vol(r, GA), sharpe=m.sharpe(r, GA), dd=-m.max_dd(r), finale=config.CAPITALE * float(e[-1]))
        k = s.lower()
        reg(f"{k}_cagr", stat[s]["cagr"], "pct1", f"Rendimento annuo composto di {s} (distribuzioni reinvestite)", "motore.cagr")
        reg(f"{k}_vol", stat[s]["vol"], "pct1", f"Volatilità annua di {s}", "motore.vol")
        reg(f"{k}_sharpe", stat[s]["sharpe"], "num2", f"Rapporto di Sharpe di {s} (tasso privo di rischio zero)", "motore.sharpe")
        reg(f"{k}_dd", stat[s]["dd"], "pct0", f"Perdita massima dal picco di {s}", "motore.max_dd")
        reg(f"{k}_finale", stat[s]["finale"], "int", f"Valore finale del capitale iniziale in {s}", "motore.equity")

    esiti = {i: [] for i in range(1, 8)}
    boots = {}
    miste = {}
    for cc, ix in config.COPPIE:
        k, kx = cc.lower(), ix.lower()
        rc, rx = R[cc].to_numpy(), R[ix].to_numpy()
        reg(f"{k}_diff_cagr", stat[cc]["cagr"] - stat[ix]["cagr"], "pp1", f"Rendimento annuo di {cc} meno quello di {ix}, in punti percentuali", "motore.cagr")
        rel = stat[cc]["vol"] / stat[ix]["vol"] - 1.0
        reg(f"{k}_vol_rel", -rel, "pct0", f"Riduzione relativa della volatilità di {cc} rispetto a {ix}", "motore.vol")
        reg(f"{k}_diff_dd", stat[ix]["dd"] - stat[cc]["dd"], "pp1", f"Perdita massima di {ix} meno quella di {cc}, in punti percentuali", "motore.max_dd")
        reg(f"{k}_rapporto_finale", stat[cc]["finale"] / stat[ix]["finale"], "num2", f"Valore finale di {cc} diviso quello di {ix}", "motore.equity")
        reg(f"{k}_corr", float(np.corrcoef(rc, rx)[0, 1]), "num2", f"Correlazione dei rendimenti giornalieri tra {cc} e {ix}", "dati/")
        su, giu = f.cattura(mensili[:, ci[cc]], mensili[:, ci[ix]])
        reg(f"{k}_cat_su", su, "pct0", f"Cattura dei mesi di rialzo di {ix} da parte di {cc}", "funzioni.cattura")
        reg(f"{k}_cat_giu", giu, "pct0", f"Cattura dei mesi di ribasso di {ix} da parte di {cc}", "funzioni.cattura")
        boot = m.bootstrap_diff_sharpe(rc, rx, config.N_BOOTSTRAP, config.BLOCCO_BOOTSTRAP, GA, rng)
        lo, hi = [float(x) for x in np.quantile(boot, [a, 1 - a])]
        boots[cc] = boot
        reg(f"{k}_sh_diff", stat[cc]["sharpe"] - stat[ix]["sharpe"], "num2", f"Sharpe di {cc} meno Sharpe di {ix}", "motore.sharpe")
        reg(f"{k}_sh_lo", lo, "num2", f"Estremo inferiore dell'intervallo bootstrap a blocchi della differenza di Sharpe ({cc} meno {ix})", "motore.bootstrap_diff_sharpe")
        reg(f"{k}_sh_hi", hi, "num2", f"Estremo superiore dell'intervallo bootstrap a blocchi della differenza di Sharpe ({cc} meno {ix})", "motore.bootstrap_diff_sharpe")
        w = f.peso_stessa_volatilita(rx, liq, stat[cc]["vol"], GA)
        rm = f.miscela(rx, liq, w)
        mix_cagr, mix_sh, mix_dd = m.cagr(rm, GA), m.sharpe(rm, GA), -m.max_dd(rm)
        miste[cc] = m.equity(rm)
        reg(f"{k}_w", w, "pct0", f"Peso di {ix} nel portafoglio con la stessa volatilità di {cc} (il resto in liquidità)", "funzioni.peso_stessa_volatilita")
        reg(f"{k}_mix_cagr", mix_cagr, "pct1", f"Rendimento annuo composto del portafoglio {ix} e liquidità con la volatilità di {cc}", "motore.cagr")
        reg(f"{k}_mix_dd", mix_dd, "pct0", f"Perdita massima del portafoglio {ix} e liquidità con la volatilità di {cc}", "motore.max_dd")
        reg(f"{k}_mix_diff", mix_cagr - stat[cc]["cagr"], "pp1", f"Rendimento annuo del portafoglio con la stessa volatilità meno quello di {cc}, in punti percentuali", "motore.cagr")
        n_crisi_meglio = 0
        crisi = []
        for j, (nome, ini, fin) in enumerate(config.CRISI):
            sel = (R.index >= pd.Timestamp(ini)) & (R.index <= pd.Timestamp(fin))
            if sel.sum() < 5:
                raise SystemExit(f"Fase «{nome}» fuori dai dati.")
            rcc, rix = float(np.prod(1 + rc[sel]) - 1), float(np.prod(1 + rx[sel]) - 1)
            crisi.append((rcc, rix))
            reg(f"{k}_crisi{j}_cc", rcc, "pct0", f"Rendimento di {cc} nella fase «{nome}»", "dati/")
            reg(f"{k}_crisi{j}_ix", rix, "pct0", f"Rendimento di {ix} nella fase «{nome}»", "dati/")
            reg(f"{k}_crisi{j}_meglio", int(rcc > rix), "int", f"1 se {cc} perde meno di {ix} nella fase «{nome}»", "dati/")
            n_crisi_meglio += int(rcc > rix)
        reg(f"{k}_n_crisi_meglio", n_crisi_meglio, "int", f"Fasi di crisi in cui {cc} perde meno di {ix}", "dati/")
        esiti[1].append(stat[cc]["cagr"] < stat[ix]["cagr"])
        esiti[2].append(rel <= -config.H2_VOL_RIDUZIONE)
        esiti[3].append(stat[cc]["dd"] > config.H3_DD_MIN)
        esiti[4].append(su < config.H4_UP_MAX and giu > config.H4_DOWN_MIN)
        esiti[5].append(lo <= 0 <= hi)
        esiti[6].append(mix_cagr > stat[cc]["cagr"])
        esiti[7].append(n_crisi_meglio == len(config.CRISI))
    for i in range(1, 8):
        reg(f"h{i}_esito", int(all(esiti[i])), "int", f"Ipotesi {i}: 1 se confermata in entrambe le coppie, 0 se smentita", "confronto con config.py")
        reg(f"h{i}_n_coppie", int(sum(esiti[i])), "int", f"Coppie su due in cui l'ipotesi {i} è confermata", "confronto con config.py")
    reg("n_conf", int(sum(all(esiti[i]) for i in range(1, 8))), "int", "Ipotesi confermate (su sette)", "confronto con config.py")
    reg("n_coppie", len(config.COPPIE), "int", "Coppie covered call e indice", "config.py")
    reg("run_utc", datetime.now(timezone.utc).isoformat(timespec="seconds"), "testo", "Data di esecuzione dell'analisi")
    json.dump({"valori": RIS}, open(config.RISULTATI / "metriche.json", "w", encoding="utf-8"), indent=1, ensure_ascii=False)

    cum = np.cumprod(1.0 + R.to_numpy(), axis=0)
    camp = np.arange(0, len(R), 5)
    pd.DataFrame({"data": R.index[camp].strftime("%Y-%m-%d"), **{s.lower(): cum[camp, ci[s]] for s in sigle}}).to_csv(
        config.RISULTATI / "curve.csv", index=False, float_format="%.5f")
    pd.DataFrame({"data": R.index[camp].strftime("%Y-%m-%d"), **{f"mix_{cc.lower()}": miste[cc][camp] for cc, _ in config.COPPIE}}).to_csv(
        config.RISULTATI / "miste.csv", index=False, float_format="%.5f")
    pd.DataFrame(mensili, columns=sigle).to_csv(config.RISULTATI / "mensili.csv", index=False, float_format="%.6f")
    pd.DataFrame({cc.lower(): boots[cc] for cc, _ in config.COPPIE}).to_csv(config.RISULTATI / "bootstrap.csv", index=False, float_format="%.5f")
    print(f"Scritti {len(RIS)} numeri in metriche.json")
    for i in range(1, 8):
        print(f"H{i}: {esiti[i]}")


if __name__ == "__main__":
    main()

"""Passo 3 - Grafici dell'articolo. Legge SOLO risultati/ e salva i PNG in figure/.

  fig1_valore.png      valore di un dollaro nei quattro ETF (scala logaritmica)
  fig2_cattura.png     cattura dei mesi di rialzo e di ribasso, per le due coppie
  fig3_stessa_vol.png  covered call contro portafoglio indice e liquidità con la stessa volatilità
"""
import json

import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt  # noqa: E402
import numpy as np  # noqa: E402
import pandas as pd  # noqa: E402

import config  # noqa: E402

BLU, ARANCIO, GRIGIO, VERDE = "#1f5fa8", "#c9610a", "#8a8a8a", "#2a7f4f"
plt.rcParams.update({"font.size": 12, "axes.spines.top": False, "axes.spines.right": False,
                     "axes.grid": True, "grid.alpha": 0.25, "figure.dpi": 120, "savefig.bbox": "tight",
                     "savefig.facecolor": "white", "legend.frameon": False})
R = config.RISULTATI

    ax.plot(c["data"], c["qyld"], color=ARANCIO, lw=2, ls="--", label="QYLD (covered call sul Nasdaq-100)")

    ax.set_ylabel("Valore di 1 dollaro (scala logaritmica)")
    ax.legend(loc="upper left", fontsize=10)
    fig.savefig(config.FIGURE / "fig1_valore.png")

    fig, ax = plt.subplots(figsize=(9, 5))
    etich = ["XYLD su SPY", "QYLD su QQQ"]
    x = np.arange(2)
    su = [V["xyld_cat_su"]["valore"] * 100, V["qyld_cat_su"]["valore"] * 100]
    giu = [V["xyld_cat_giu"]["valore"] * 100, V["qyld_cat_giu"]["valore"] * 100]
    ax.bar(x - 0.2, su, width=0.4, color=VERDE, label="Mesi di rialzo dell'indice")
    ax.bar(x + 0.2, giu, width=0.4, color=ARANCIO, label="Mesi di ribasso dell'indice")
    ax.axhline(100, color=GRIGIO, ls="--", lw=1)
    ax.set_xticks(x)
    ax.set_xticklabels(etich)
    ax.set_ylabel("Quota dell'indice catturata (%)")
    ax.legend(loc="upper left", fontsize=10)
    fig.savefig(config.FIGURE / "fig2_cattura.png")
    plt.close(fig)

    fig, axs = plt.subplots(1, 2, figsize=(11, 4.6), sharey=False)
    for ax, (cc, nome, colore) in zip(axs, (("xyld", "XYLD", BLU), ("qyld", "QYLD", ARANCIO))):
        ax.plot(c["data"], c[cc], color=colore, lw=2, label=nome)
        ax.plot(mx["data"], mx[f"mix_{cc}"], color=GRIGIO, lw=2, label="Indice e liquidità, stessa volatilità")
        ax.set_title(f"{nome}: peso dell'indice {V[cc + '_w']['valore'] * 100:.0f}%".replace(".", ","), loc="left", fontsize=12)
        ax.legend(loc="upper left", fontsize=9)
        ax.tick_params(axis="x", rotation=30)
    axs[0].set_ylabel("Valore di 1 dollaro")
    fig.savefig(config.FIGURE / "fig3_stessa_vol.png")
    plt.close(fig)
    print("Figure salvate.")


if __name__ == "__main__":
    main()
flusso/config.pyparametri e ipotesi, scritti prima dei risultati

"""Funzioni dell'articolo 483: cattura dei mesi di rialzo e di ribasso, portafoglio con la stessa volatilità."""
import numpy as np


def cattura(r_fondo: np.ndarray, r_indice: np.ndarray) -> tuple[float, float]:
    """Cattura dei mesi di rialzo e di ribasso: rendimento medio del fondo diviso quello dell'indice, nei mesi in cui l'indice sale e in cui scende."""
    r_fondo, r_indice = np.asarray(r_fondo, float), np.asarray(r_indice, float)
    su, giu = r_indice > 0, r_indice < 0
    return float(r_fondo[su].mean() / r_indice[su].mean()), float(r_fondo[giu].mean() / r_indice[giu].mean())


def miscela(r_indice: np.ndarray, r_liquidita: np.ndarray, w: float) -> np.ndarray:
    """Rendimenti giornalieri di un portafoglio con peso w nell'indice e 1 - w in liquidità, ribilanciato ogni giorno."""
    return w * np.asarray(r_indice, float) + (1.0 - w) * np.asarray(r_liquidita, float)


def peso_stessa_volatilita(r_indice: np.ndarray, r_liquidita: np.ndarray, vol_obiettivo: float, ga: int) -> float:
    """Peso dell'indice che dà alla miscela la volatilità annua `vol_obiettivo` (bisezione tra 0 e 1; se non raggiungibile, l'estremo)."""
    def vol(w):
        return float(np.std(miscela(r_indice, r_liquidita, w), ddof=1) * np.sqrt(ga))
    lo, hi = 0.0, 1.0
    if vol(hi) <= vol_obiettivo:
        return 1.0
    if vol(lo) >= vol_obiettivo:
        return 0.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if vol(mid) < vol_obiettivo:
            lo = mid
        else:
            hi = mid
    return (lo + hi) / 2
flusso/motore.pyrendimento, volatilità, Sharpe, perdita massima, bootstrap a blocchi
"""Funzioni di calcolo dell'articolo 202: metriche, finestre mobili, persistenza, bootstrap, scimmie.

Nessuna strategia di trading: sono confronti tra buy & hold. Le funzioni sono pure, così verifica_motore.py
le controlla su dati inventati con risultati calcolabili a mano.
"""
import numpy as np
import pandas as pd


def equity(r) -> np.ndarray:
    return np.cumprod(1.0 + np.asarray(r, dtype=float))


def cagr(r, ga: int) -> float:
    r = np.asarray(r, dtype=float)
    return float(np.prod(1.0 + r) ** (ga / len(r)) - 1.0)


def vol(r, ga: int) -> float:
    return float(np.std(np.asarray(r, dtype=float), ddof=1) * np.sqrt(ga))


def sharpe(r, ga: int) -> float:
    r = np.asarray(r, dtype=float)
    return float(np.mean(r) / np.std(r, ddof=1) * np.sqrt(ga))


def max_dd(r) -> float:
    e = equity(r)
    picco = np.maximum.accumulate(np.concatenate([[1.0], e]))[1:]
    return float(np.min(e / picco - 1.0))


def fine_mese(indice: pd.DatetimeIndex) -> np.ndarray:
    """Posizioni dell'ultima seduta di ogni mese (escluso l'ultimo mese se incompleto non è controllato: si usa quanto c'è)."""
    s = pd.Series(np.arange(len(indice)), index=indice)
    return s.groupby([indice.year, indice.month]).last().to_numpy()


def rendimento_finestre(cum: np.ndarray, fini: np.ndarray, lung: int) -> tuple[np.ndarray, np.ndarray]:
    """Rendimento totale delle ultime `lung` sedute alla fine di ogni mese utile. cum: (T, k). Restituisce (posizioni, matrice)."""
    ok = fini[fini >= lung]
    base = np.concatenate([np.ones((1, cum.shape[1])), cum], axis=0)   # base[i] = valore prima della seduta i
    return ok, cum[ok] / base[ok - lung + 1] - 1.0


def quote_vincitrici(rend: np.ndarray) -> np.ndarray:
    """Quota di finestre in cui ciascuna colonna ha il rendimento più alto."""
    v = np.argmax(rend, axis=1)
    return np.bincount(v, minlength=rend.shape[1]) / len(v)


def persistenza(cum: np.ndarray, fini: np.ndarray, lung: int) -> tuple[np.ndarray, np.ndarray]:
    """Per ogni mese t con `lung` sedute prima e dopo: area vincitrice nel passato (usa solo dati fino a t) e nel futuro."""
    base = np.concatenate([np.ones((1, cum.shape[1])), cum], axis=0)
    passato, futuro = [], []
    for e in fini:
        if e - lung + 1 < 0 or e + lung >= len(cum):
            continue
        passato.append(int(np.argmax(cum[e] / base[e - lung + 1])))
        futuro.append(int(np.argmax(cum[e + lung] / cum[e])))
    return np.array(passato), np.array(futuro)


def p_permutazione(passato: np.ndarray, futuro: np.ndarray, n: int, rng) -> tuple[float, float]:
    """Quota di coincidenze osservata e p-value: frazione di permutazioni del futuro con almeno altrettante coincidenze."""
    oss = float(np.mean(passato == futuro))
    sup = 0
    for _ in range(n):
        if np.mean(passato == rng.permutation(futuro)) >= oss - 1e-12:
            sup += 1
    return oss, (sup + 1) / (n + 1)


def volatilita_mobile(r: pd.Series, finestra: int, ga: int) -> pd.Series:
    return r.rolling(finestra).std(ddof=1) * np.sqrt(ga)


def correlazione_mobile(a: pd.Series, b: pd.Series, finestra: int) -> pd.Series:
    return a.rolling(finestra).corr(b)


def bootstrap_diff_sharpe(ra, rb, n: int, blocco: int, ga: int, rng) -> np.ndarray:
    """Differenza di Sharpe (a - b) ricampionando a blocchi le stesse date per entrambe le serie."""
    ra, rb = np.asarray(ra, dtype=float), np.asarray(rb, dtype=float)
    T = len(ra)
    nb = int(np.ceil(T / blocco))
    out = np.empty(n)
    for i in range(n):
        inizi = rng.integers(0, T - blocco + 1, size=nb)
        idx = (inizi[:, None] + np.arange(blocco)[None, :]).ravel()[:T]
        out[i] = sharpe(ra[idx], ga) - sharpe(rb[idx], ga)
    return out


def rendimenti_mensili(r: np.ndarray, fini: np.ndarray) -> np.ndarray:
    """Rendimenti composti per mese: r (T, k); i mesi sono i tratti che finiscono nelle posizioni `fini`."""
    e = np.vstack([np.ones((1, r.shape[1])), np.cumprod(1.0 + r, axis=0)])
    estremi = np.concatenate([[0], fini + 1])
    if estremi[-1] != len(r):            # coda: ultimo mese incompleto, si include
        estremi = np.concatenate([estremi, [len(r)]])
    return e[estremi[1:]] / e[estremi[:-1]] - 1.0


def scimmie_valore_finale(mensili: np.ndarray, n: int, rng) -> np.ndarray:
    """Valore finale di 1 per n scimmie che scelgono a caso l'area (colonna) ogni mese."""
    M, k = mensili.shape
    scelta = rng.integers(0, k, size=(n, M))
    return np.prod(1.0 + mensili[np.arange(M)[None, :], scelta], axis=1)

"""Controlli di motore.py su dati inventati, con risultati calcolabili a mano. Se uno fallisce, il flusso si ferma."""
import numpy as np
import pandas as pd

import funzioni as f
import motore as m

rng = np.random.default_rng(1)


def uguale(a, b, tol=1e-9, nome=""):
    assert abs(a - b) < tol, f"{nome}: {a} != {b}"
    print(f"ok  {nome}")


# 1. rendimento composto: +10% e -10% in due sedute
r = np.array([0.10, -0.10])
uguale(m.equity(r)[-1], 0.99, nome="equity")
uguale(m.cagr(np.full(252, 0.0), 252), 0.0, nome="cagr nullo")
uguale(m.cagr(np.full(252, 0.001), 252), 1.001 ** 252 - 1, nome="cagr costante")
# 2. perdita massima: sale a 1,2 poi scende a 0,6 -> -50%
uguale(m.max_dd(np.array([0.2, -0.5])), -0.5, nome="perdita massima")
# 3. volatilità e Sharpe di una serie alternata +1% / -1%
alt = np.tile([0.01, -0.01], 126)
uguale(m.vol(alt, 252), np.std(alt, ddof=1) * np.sqrt(252), nome="volatilità")
uguale(m.sharpe(alt + 0.0, 252), 0.0, tol=1e-9, nome="Sharpe nullo")
# 4. fine mese: tre mesi finti
idx = pd.bdate_range("2020-01-01", periods=70)
fini = m.fine_mese(idx)
assert idx[fini[0]].month == 1 and idx[fini[0] + 1].month == 2, "fine mese"
print("ok  fine mese")
# 5. finestre: A cresce 1% al giorno, B ferma, C scende 1%: A vince sempre, rendimento a 10 sedute noto
T = 600
R = np.column_stack([np.full(T, 0.01), np.zeros(T), np.full(T, -0.01)])
cum = np.cumprod(1.0 + R, axis=0)
pos, rend = m.rendimento_finestre(cum, np.array([50, 100, 599]), 10)
uguale(rend[0, 0], 1.01 ** 10 - 1, nome="rendimento finestra A")
uguale(rend[0, 1], 0.0, nome="rendimento finestra B")
uguale(rend[0, 2], 0.99 ** 10 - 1, nome="rendimento finestra C")
q = m.quote_vincitrici(rend)
assert q.tolist() == [1.0, 0.0, 0.0], "quote vincitrici"
print("ok  quote vincitrici")
# 6. persistenza: se A vince sempre, passato = futuro sempre
fm = m.fine_mese(pd.bdate_range("2000-01-03", periods=T))
pa, fu = m.persistenza(cum, fm, 60)
assert len(pa) > 5 and (pa == fu).all() and (pa == 0).all(), "persistenza"
print("ok  persistenza")
# 7. permutazione: su dati indipendenti il p-value è alto; su dati identici con due classi è basso
a = rng.integers(0, 3, 300)
b = rng.integers(0, 3, 300)
_, p = m.p_permutazione(a, b, 500, rng)
assert p > 0.05, f"permutazione su dati indipendenti: p={p}"
_, p2 = m.p_permutazione(a, a, 500, rng)
assert p2 < 0.01, "permutazione su dati identici"
print("ok  permutazione")
# 8. correlazione mobile: serie identiche -> 1; opposte -> -1
x = pd.Series(rng.normal(size=300))
uguale(m.correlazione_mobile(x, x, 50).dropna().min(), 1.0, tol=1e-9, nome="correlazione 1")
uguale(m.correlazione_mobile(x, -x, 50).dropna().max(), -1.0, tol=1e-9, nome="correlazione -1")
# 9. bootstrap: due serie identiche -> differenza di Sharpe zero
d = m.bootstrap_diff_sharpe(x.to_numpy() * 0.01, x.to_numpy() * 0.01, 50, 21, 252, rng)
assert np.abs(d).max() < 1e-9, "bootstrap identico"
print("ok  bootstrap")
# 10. mensili e scimmie: tre aree con rendimento mensile costante 1%, 0, -1% -> la scimmia ha valore tra i tre estremi
Rm = np.column_stack([np.full(T, 0.0005), np.zeros(T), np.full(T, -0.0005)])
mens = m.rendimenti_mensili(Rm, fm)
uguale(np.prod(1 + mens[:, 0]), np.prod(1 + Rm[:, 0]), tol=1e-9, nome="mesi ricompongono il totale")
sc = m.scimmie_valore_finale(mens, 2000, rng)
assert (sc <= np.prod(1 + mens[:, 0]) + 1e-9).all() and (sc >= np.prod(1 + mens[:, 2]) - 1e-9).all(), "scimmie nei limiti"
print("ok  scimmie")
# 11. cattura: fondo che prende metà dei rialzi e tutti i ribassi
idx_m = np.array([0.04, -0.02, 0.02, -0.04])
fondo = np.array([0.02, -0.02, 0.01, -0.04])
su, giu = f.cattura(fondo, idx_m)
uguale(su, 0.5, nome="cattura rialzi")
uguale(giu, 1.0, nome="cattura ribassi")
# 12. peso a stessa volatilità: con liquidità costante la volatilità è lineare nel peso
ri = rng.normal(0.0004, 0.01, 2000)
liq = np.full(2000, 0.0001)
w = f.peso_stessa_volatilita(ri, liq, 0.5 * m.vol(ri, 252), 252)
uguale(w, 0.5, tol=1e-6, nome="peso a stessa volatilità")
uguale(m.vol(f.miscela(ri, liq, w), 252), 0.5 * m.vol(ri, 252), tol=1e-6, nome="volatilità della miscela")
assert f.peso_stessa_volatilita(ri, liq, 2 * m.vol(ri, 252), 252) == 1.0, "obiettivo oltre l'indice"
print("ok  peso limite")
print("Tutti i controlli superati.")
