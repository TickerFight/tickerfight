50 titoli dell'S&P 500, medie mobili contro buy & hold: la classifica

Su 50 titoli dell'S&P 500 scelti a sorte, la regola a medie mobili 50/200, con 500 euro per titolo, una commissione fissa di 1 euro a ordine e il 26% di imposta sulle plusvalenze, ha fatto guadagnare più del semplice buy & hold negli ultimi cinque anni? Il risultato è una classifica titolo per titolo.

flusso/01_scarica_dati.py
"""Passo 1 - Sceglie 50 titoli dell'S&P 500 con un sorteggio a seme fisso e scarica i loro prezzi da Yahoo Finance.

Salva:
  dati/sp500_composizione.csv   elenco dei componenti dell'indice al momento del download
  dati/selezione.csv            ordine del sorteggio, esito per ogni titolo provato (preso o scartato, e perché)
  dati/titoli/<TICKER>.csv      Date, Open, High, Low, Close, Volume dei 50 titoli scelti
  dati/eurusd_giornaliero.csv   cambio, in dollari per un euro
  dati/metadati.json            fonte, data del download, versione di yfinance, impronta SHA-256 dei file

Gli script successivi leggono SOLO i file salvati: i risultati sono riproducibili anche se Yahoo
corregge i dati in futuro.
"""
import hashlib
import json
import time
from datetime import datetime, timezone
from zoneinfo import ZoneInfo

import numpy as np
import pandas as pd
import yfinance as yf

import config


def scarica(ticker: str, tentativi: int = 3) -> pd.DataFrame | None:
    for t in range(tentativi):
        try:
            grezzo = yf.Ticker(ticker).history(start=config.INIZIO_DOWNLOAD, auto_adjust=False)
        except Exception as e:  # noqa: BLE001 - errori di rete: si riprova
            print(f"  {ticker}: errore {type(e).__name__}, tentativo {t + 1}")
            grezzo = None
        if grezzo is not None and not grezzo.empty:
            break
        time.sleep(3)
    else:
        return None
    df = grezzo[["Open", "High", "Low", "Close", "Volume"]].copy()
    df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
    df.index.name = "Date"
    df = df[~df.index.duplicated(keep="last")].sort_index()
    df = df.dropna(subset=["Open", "Close"])
    if df.empty or (df[["Open", "Close"]] <= 0).any().any():
        return None
    return df


def leggi_composizione() -> pd.DataFrame:
    grezzo = pd.read_csv(config.URL_COMPOSIZIONE)
    colonne = {"Symbol": "simbolo", "Security": "nome", "Name": "nome", "GICS Sector": "settore"}
    df = grezzo.rename(columns=colonne)[["simbolo", "nome", "settore"]].copy()
    df["simbolo"] = df["simbolo"].astype(str).str.strip()
    df["ticker_yahoo"] = df["simbolo"].str.replace(".", "-", regex=False)   # BRK.B -> BRK-B
    df = df.drop_duplicates("ticker_yahoo").sort_values("ticker_yahoo").reset_index(drop=True)
    if len(df) < 400:
        raise SystemExit(f"Elenco dei componenti troppo corto ({len(df)} righe): scarico non valido.")
    return df


def main() -> None:
    config.DATI_TITOLI.mkdir(parents=True, exist_ok=True)
    comp = leggi_composizione()
    comp.to_csv(config.FILE_COMPOSIZIONE, index=False)

    cambio = scarica(config.TICKER_CAMBIO)
    if cambio is None:
        raise SystemExit("Yahoo Finance non ha restituito il cambio euro/dollaro.")
    cambio.to_csv(config.FILE_CAMBIO, float_format="%.6f")

    ordine = np.random.default_rng(config.SEME).permutation(len(comp))
    ora_borsa = datetime.now(ZoneInfo(config.FUSO_BORSA))
    prese, righe = [], []
    ultima_data = None
    for pos, k in enumerate(ordine, 1):
        if len(prese) >= config.N_TITOLI:
            break
        r = comp.iloc[k]
        t = r["ticker_yahoo"]
        df = scarica(t)
        esito = "preso"
        if df is None:
            esito = "scartato: dati non disponibili"
        else:
            # Se lo script gira durante la seduta l'ultima barra è incompleta: la scartiamo.
            if df.index[-1].date() == ora_borsa.date() and ora_borsa.hour < 16:
                df = df.iloc[:-1]
            fine = df.index[-1]
            i0 = int(df.index.searchsorted(fine - pd.DateOffset(years=config.ANNI_TEST)))
            if i0 < config.SEDUTE_MINIME_PRIMA:
                esito = f"scartato: storia troppo corta ({i0} sedute prima dell'inizio del test)"
            elif ultima_data is not None and (ultima_data - fine).days > 5:
                esito = "scartato: dati non aggiornati"
        if esito == "preso":
            ultima_data = df.index[-1] if ultima_data is None else max(ultima_data, df.index[-1])
            df.to_csv(config.DATI_TITOLI / f"{t}.csv", float_format="%.6f")
            prese.append(t)
        righe.append({"ordine_sorteggio": pos, "ticker": t, "nome": r["nome"], "settore": r["settore"], "esito": esito})
        print(f"{pos:3d} {t:8s} {esito}")
    if len(prese) < config.N_TITOLI:
        raise SystemExit(f"Solo {len(prese)} titoli validi su {config.N_TITOLI} richiesti.")
    pd.DataFrame(righe).to_csv(config.DATI / "selezione.csv", index=False)

    def impronta(p):
        return hashlib.sha256(p.read_bytes()).hexdigest()

    metadati = {
        "descrizione": config.DESCRIZIONE_DATI,
        "fonte": "Yahoo Finance",
        "fonte_composizione": config.URL_COMPOSIZIONE,
        "url_fonte": config.URL_FONTE,
        "libreria": f"yfinance {yf.__version__}",
        "scaricato_il_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"),
        "seme": config.SEME,
        "componenti_nell_elenco": len(comp),
        "titoli_provati": len(righe),
        "titoli_scartati": len(righe) - len(prese),
        "titoli": prese,
        "ultima_data_titoli": str(ultima_data.date()),
        "prima_data_cambio": str(cambio.index[0].date()),
        "ultima_data_cambio": str(cambio.index[-1].date()),
        "colonne_usate": config.NOTA_COLONNA,
        "sha256_composizione": impronta(config.FILE_COMPOSIZIONE),
        "sha256_cambio": impronta(config.FILE_CAMBIO),
        "sha256_titoli": {t: impronta(config.DATI_TITOLI / f"{t}.csv") for t in prese},
    }
    config.FILE_METADATI.write_text(json.dumps(metadati, indent=2, ensure_ascii=False), encoding="utf-8")
    print(f"\nScaricati {len(prese)} titoli, {len(righe) - len(prese)} scartati. Ultima seduta: {ultima_data.date()}")


if __name__ == "__main__":
    main()
flusso/02_backtest_e_test.pysimulazione su ogni titolo, classifica, portafoglio, controlli
"""Passo 2 - 500 euro su ciascuno di 50 titoli dell'S&P 500: regola 50/200 contro buy & hold, in euro.

Legge SOLO i file in dati/ (scaricati dal passo 1) e produce in risultati/:
  metriche.json              ogni numero dell'articolo, con descrizione e origine di calcolo
  classifica.csv             una riga per titolo: valori finali netti, differenza, perdita massima, operazioni
  operazioni_regola.csv      registro di tutte le operazioni della regola, per titolo
  curva_portafoglio.csv      valore giorno per giorno dei 50 x 500 euro con la regola e con il buy & hold

Regole del metodo:
  - l'ipotesi, la scelta dei titoli e i parametri sono dichiarati in config.py prima di guardare i risultati;
  - il segnale di fine seduta t decide l'ordine di t+1 (nessun look-ahead);
  - ogni titolo parte con 500 euro, commissione fissa per ordine e tassa del 26% in euro, come nell'articolo su Apple;
  - ogni numero dell'articolo passa da reg(): niente cifre scritte a mano nel testo.
"""
import json
import platform
from datetime import datetime, timezone

import numpy as np
import pandas as pd

import config
import simulazione as s

RIS: dict = {}
SORGENTE = "02_backtest_e_test.py con simulazione.py"
SCEN = {"lordo": (0.0, 0.0), "comm": (config.COMMISSIONE, 0.0), "netto": (config.COMMISSIONE, config.ALIQUOTA)}


def reg(chiave, valore, formato, descrizione, origine):
    """Registra un numero con la sua giustificazione (finisce nelle note dell'articolo)."""
    if isinstance(valore, (np.floating, np.integer)):
        valore = valore.item()
    if isinstance(valore, float) and not np.isfinite(valore):
        raise SystemExit(f"Numero non finito per {chiave}")
    RIS[chiave] = {"valore": valore, "formato": formato, "descrizione": descrizione, "origine": origine}


def main() -> None:
    config.RISULTATI.mkdir(parents=True, exist_ok=True)
    meta = json.loads(config.FILE_METADATI.read_text(encoding="utf-8"))
    sel = pd.read_csv(config.DATI / "selezione.csv")
    comp = pd.read_csv(config.FILE_COMPOSIZIONE)
    info = comp.set_index("ticker_yahoo")
    cambio = s.carica_csv(config.FILE_CAMBIO)
    tickers = meta["titoli"]

    righe, operazioni, curve_mm, curve_bh = [], [], {}, {}
    date_inizio, date_fine = set(), set()
    for t in tickers:
        df = s.prepara(s.carica_csv(config.DATI_TITOLI / f"{t}.csv"), cambio, config.SMA_VELOCE, config.SMA_LENTA)
        fine = df.index[-1]
        i0 = s.indice_data(df, fine - pd.DateOffset(years=config.ANNI_TEST))
        if i0 < config.SMA_LENTA + 1:
            raise SystemExit(f"{t}: storia insufficiente per la media lenta.")
        if df["fx"].iloc[i0 - 1:].isna().any():
            raise SystemExit(f"{t}: cambio mancante nel periodo del test.")
        date_inizio.add(df.index[i0])
        date_fine.add(fine)
        R = {}
        for modo in (config.MODO_PRINCIPALE, config.MODO_CONTROLLO):
            for sc, (c, a) in SCEN.items():
                for nome, strat in (("mm", True), ("bh", False)):
                    R[(modo, sc, nome)] = s.simula(df, i0, config.CAPITALE, c, a, modo, strat)
        P = config.MODO_PRINCIPALE
        mm, bh = R[(P, "netto", "mm")], R[(P, "netto", "bh")]
        righe.append({
            "ticker": t, "nome": info.loc[t, "nome"], "settore": info.loc[t, "settore"],
            "finale_mm_lordo": R[(P, "lordo", "mm")]["valore_finale_eur"], "finale_bh_lordo": R[(P, "lordo", "bh")]["valore_finale_eur"],
            "finale_mm_netto": mm["valore_finale_eur"], "finale_bh_netto": bh["valore_finale_eur"],
            "differenza": mm["valore_finale_eur"] - bh["valore_finale_eur"],
            "differenza_chiusura": R[(config.MODO_CONTROLLO, "netto", "mm")]["valore_finale_eur"]
            - R[(config.MODO_CONTROLLO, "netto", "bh")]["valore_finale_eur"],
            "rend_mm_netto": mm["rendimento_totale"], "rend_bh_netto": bh["rendimento_totale"],
            "cagr_mm_netto": mm["cagr"], "cagr_bh_netto": bh["cagr"],
            "maxdd_mm": -mm["perdita_massima"], "maxdd_bh": -bh["perdita_massima"],
            "quota_investito": mm["quota_giorni_investito"], "n_ordini_mm": mm["n_acquisti"] + mm["n_vendite"],
            "tasse_mm": mm["tasse_eur"], "tasse_bh": bh["tasse_eur"],
            "commissioni_mm": mm["commissioni_eur"], "commissioni_bh": bh["commissioni_eur"],
        })
        for o in mm["operazioni"]:
            operazioni.append({"ticker": t, **o})
        d = df.index[i0:]
        curve_mm[t] = pd.Series(mm["equity"], index=d)
        curve_bh[t] = pd.Series(bh["equity"], index=d)

    if max(date_fine) - min(date_fine) > pd.Timedelta(days=5) or max(date_inizio) - min(date_inizio) > pd.Timedelta(days=5):
        raise SystemExit("I titoli non coprono lo stesso periodo di test.")
    tab = pd.DataFrame(righe).sort_values("differenza", ascending=False).reset_index(drop=True)
    tab.insert(0, "posizione", np.arange(1, len(tab) + 1))
    tab.to_csv(config.RISULTATI / "classifica.csv", index=False, float_format="%.6f")
    pd.DataFrame(operazioni).to_csv(config.RISULTATI / "operazioni_regola.csv", index=False, float_format="%.6f")
    pf_mm = pd.DataFrame(curve_mm).sum(axis=1, min_count=len(tickers))
    pf_bh = pd.DataFrame(curve_bh).sum(axis=1, min_count=len(tickers))
    ok = pf_mm.notna() & pf_bh.notna()
    pf = pd.DataFrame({"data": pf_mm.index[ok], "valore_regola_netto": pf_mm[ok].to_numpy(), "valore_bh_netto": pf_bh[ok].to_numpy()})
    pf.to_csv(config.RISULTATI / "curva_portafoglio.csv", index=False, float_format="%.6f")

    n = len(tab)
    origine_dati = f"dati/ ({meta['fonte']}, {n} titoli e {config.TICKER_CAMBIO}, scaricati il {meta['scaricato_il_utc'][:10]})"
    cl = "classifica.csv"

    # --- Parametri e dati ---------------------------------------------------------------------------------
    reg("n_titoli", n, "int", "Numero di titoli del campione", "config.py, N_TITOLI (controllato su classifica.csv)")
    reg("capitale", config.CAPITALE, "int", "Capitale iniziale per ogni titolo, in euro", "config.py, CAPITALE")
    reg("capitale_totale", config.CAPITALE * n, "int", "Capitale iniziale complessivo dei 50 titoli, in euro", "config.py: CAPITALE per numero di titoli")
    reg("commissione", config.COMMISSIONE, "num2", "Commissione fissa per ogni ordine, in euro", "config.py, COMMISSIONE")
    reg("aliquota", config.ALIQUOTA, "pct0", "Aliquota sulle plusvalenze", "config.py, ALIQUOTA")
    reg("sma_veloce", config.SMA_VELOCE, "annonum", "Giorni della media veloce", "config.py, SMA_VELOCE")
    reg("sma_lenta", config.SMA_LENTA, "annonum", "Giorni della media lenta", "config.py, SMA_LENTA")
    reg("anni_test", config.ANNI_TEST, "annonum", "Anni del test", "config.py, ANNI_TEST")
    reg("seme", config.SEME, "annonum", "Seme del sorteggio dei titoli", "config.py, SEME")
    reg("data_inizio", str(min(date_inizio).date()), "data_lunga", "Prima seduta del test", origine_dati)
    reg("data_fine", str(max(date_fine).date()), "data_lunga", "Ultima seduta del test (ultimo dato scaricato)", origine_dati)
    reg("data_scarico", meta["scaricato_il_utc"][:10], "data", "Data del download dei dati", "dati/metadati.json")
    reg("n_componenti", meta["componenti_nell_elenco"], "int", "Componenti dell'S&P 500 nell'elenco usato per il sorteggio", "dati/metadati.json")
    reg("n_provati", meta["titoli_provati"], "int", "Titoli provati nel sorteggio per ottenerne 50 validi", "dati/metadati.json")
    reg("n_scartati", meta["titoli_scartati"], "int", "Titoli scartati per dati assenti o storia troppo corta", "dati/metadati.json")

    # --- Risultati sul campione ---------------------------------------------------------------------------
    vince = tab["differenza"] > 0
    reg("n_vince", int(vince.sum()), "int", "Titoli su cui la regola chiude sopra il buy & hold, dopo costi e tasse", f"{cl}: differenza > 0")
    reg("n_perde", int((~vince).sum()), "int", "Titoli su cui il buy & hold chiude sopra o alla pari", f"{cl}: differenza <= 0")
    reg("quota_vince", float(vince.mean()), "pct0", "Quota dei titoli su cui vince la regola", f"{cl}: differenza > 0, su 50")
    reg("mediana_diff", float(tab["differenza"].median()), "num2", "Differenza mediana in euro, regola meno buy & hold, netto", f"{cl}: mediana di differenza")
    reg("media_diff", float(tab["differenza"].mean()), "num2", "Differenza media in euro, regola meno buy & hold, netto", f"{cl}: media di differenza")
    reg("mediana_finale_mm", float(tab["finale_mm_netto"].median()), "num2", "Valore finale netto mediano della regola, in euro", f"{cl}: mediana di finale_mm_netto")
    reg("mediana_finale_bh", float(tab["finale_bh_netto"].median()), "num2", "Valore finale netto mediano del buy & hold, in euro", f"{cl}: mediana di finale_bh_netto")
    reg("mediana_rend_mm", float(tab["rend_mm_netto"].median()), "pct0", "Rendimento totale netto mediano della regola", f"{cl}: mediana di rend_mm_netto")
    reg("mediana_rend_bh", float(tab["rend_bh_netto"].median()), "pct0", "Rendimento totale netto mediano del buy & hold", f"{cl}: mediana di rend_bh_netto")
    reg("diff_min", float(tab["differenza"].min()), "num2", "Differenza peggiore per la regola, in euro", f"{cl}: minimo di differenza")
    reg("diff_max", float(tab["differenza"].max()), "num2", "Differenza migliore per la regola, in euro", f"{cl}: massimo di differenza")

    fin_mm, fin_bh = float(tab["finale_mm_netto"].sum()), float(tab["finale_bh_netto"].sum())
    reg("finale_pf_mm", fin_mm, "num2", "Valore finale in euro del portafoglio dei 50 titoli con la regola, netto", f"{cl}: somma di finale_mm_netto")
    reg("finale_pf_bh", fin_bh, "num2", "Valore finale in euro del portafoglio dei 50 titoli con il buy & hold, netto", f"{cl}: somma di finale_bh_netto")
    reg("diff_pf_abs", abs(fin_mm - fin_bh), "num2", "Distanza finale in euro tra i due portafogli, netto", "differenza assoluta tra le due somme")
    reg("rend_pf_mm", fin_mm / (config.CAPITALE * n) - 1, "pct0", "Rendimento totale netto del portafoglio con la regola", "finale_pf_mm / capitale_totale - 1")
    reg("rend_pf_bh", fin_bh / (config.CAPITALE * n) - 1, "pct0", "Rendimento totale netto del portafoglio con il buy & hold", "finale_pf_bh / capitale_totale - 1")
    for nome, col in (("mm", "valore_regola_netto"), ("bh", "valore_bh_netto")):
        v = pf[col].to_numpy()
        reg(f"maxdd_pf_{nome}", float(-(v / np.maximum.accumulate(v) - 1).min()), "pct0",
            f"Perdita massima dal picco del portafoglio dei 50 titoli: {'regola' if nome == 'mm' else 'buy & hold'}",
            "curva_portafoglio.csv: minimo di valore/picco - 1")

    # --- Chi vince e quando -------------------------------------------------------------------------------
    scesi = tab["finale_bh_netto"] < config.CAPITALE
    reg("n_bh_perdita", int(scesi.sum()), "int", "Titoli su cui il buy & hold chiude sotto i 500 euro iniziali", f"{cl}: finale_bh_netto < capitale")
    reg("n_vince_tra_perdita", int((vince & scesi).sum()), "int", "Di questi, quelli su cui la regola chiude sopra il buy & hold", f"{cl}: differenza > 0 tra i titoli in perdita")
    reg("n_bh_guadagno", int((~scesi).sum()), "int", "Titoli su cui il buy & hold chiude sopra i 500 euro iniziali", f"{cl}: finale_bh_netto >= capitale")
    reg("n_vince_tra_guadagno", int((vince & ~scesi).sum()), "int", "Di questi, quelli su cui la regola chiude sopra il buy & hold", f"{cl}: differenza > 0 tra i titoli in guadagno")
    reg("corr_rango", float(tab[["rend_bh_netto", "differenza"]].rank().corr().iloc[0, 1]), "num2",
        "Correlazione di rango (Spearman) tra rendimento netto del buy & hold e differenza regola meno buy & hold",
        f"{cl}: correlazione tra i ranghi di rend_bh_netto e differenza")
    meglio_dd = tab["maxdd_mm"] < tab["maxdd_bh"]
    reg("n_dd_meglio", int(meglio_dd.sum()), "int", "Titoli su cui la regola ha una perdita massima minore del buy & hold", f"{cl}: maxdd_mm < maxdd_bh")
    reg("mediana_maxdd_mm", float(tab["maxdd_mm"].median()), "pct0", "Perdita massima mediana della regola", f"{cl}: mediana di maxdd_mm")
    reg("mediana_maxdd_bh", float(tab["maxdd_bh"].median()), "pct0", "Perdita massima mediana del buy & hold", f"{cl}: mediana di maxdd_bh")
    reg("mediana_ordini", float(tab["n_ordini_mm"].median()), "num1", "Ordini mediani della regola per titolo (acquisti e vendite, compresa la vendita finale)", f"{cl}: mediana di n_ordini_mm")
    reg("mediana_quota_investito", float(tab["quota_investito"].median()), "pct0", "Quota mediana di sedute in cui la regola è investita", f"{cl}: mediana di quota_investito")
    reg("tasse_tot_mm", float(tab["tasse_mm"].sum()), "num2", "Tasse totali pagate in euro dalla regola sui 50 titoli", f"{cl}: somma di tasse_mm")
    reg("tasse_tot_bh", float(tab["tasse_bh"].sum()), "num2", "Tasse totali pagate in euro dal buy & hold sui 50 titoli", f"{cl}: somma di tasse_bh")
    reg("comm_tot_mm", float(tab["commissioni_mm"].sum()), "num2", "Commissioni totali pagate in euro dalla regola sui 50 titoli", f"{cl}: somma di commissioni_mm")
    reg("comm_tot_bh", float(tab["commissioni_bh"].sum()), "num2", "Commissioni totali pagate in euro dal buy & hold sui 50 titoli", f"{cl}: somma di commissioni_bh")
    reg("n_lordo_vince", int((tab["finale_mm_lordo"] > tab["finale_bh_lordo"]).sum()), "int",

flusso/simulazione.pymotore: segnale, ordini, commissioni, cambio, tassa e minusvalenze
"""Motore di simulazione in euro di una regola a medie mobili su un singolo titolo.

Regole (da dichiarare nell'articolo):
  - segnale: investito se la media mobile veloce è sopra la lenta, calcolate sulle chiusure;
  - il segnale a fine seduta t decide l'operazione della seduta t+1 (nessun look-ahead);
  - esecuzione "apertura": prezzo di apertura di t+1, cambio dell'ultima chiusura nota (t);
    esecuzione "chiusura": prezzo di chiusura di t+1, cambio di t+1 (controllo di robustezza);
  - azioni frazionate, si investe tutta la cassa; commissione fissa per ogni ordine, acquisto o vendita;
  - la cassa non rende nulla; i dividendi non sono considerati;
  - tassa sulle plusvalenze calcolate in euro (cambio incluso), pagata a ogni vendita come nel regime
    amministrato; la commissione si somma al costo di acquisto e si sottrae dall'incasso; le minusvalenze
    compensano le plusvalenze successive fino al quarto anno seguente;
  - a fine periodo si vende tutto, con commissione e tassa, per confrontare cifre nette.
Non modellati: imposta di bollo, ritenuta sui dividendi, spread di cambio.
"""
from pathlib import Path

import numpy as np
import pandas as pd


class Zaino:
    """Minusvalenze compensabili con le plusvalenze successive, fino al 4° anno seguente."""

    def __init__(self, aliquota: float):
        self.aliquota = aliquota
        self.voci: list[list[float]] = []   # [ultimo anno utilizzabile, importo residuo]

    def tassa(self, anno: int, guadagno: float) -> float:
        if guadagno < 0:
            self.voci.append([anno + 4, -guadagno])
            return 0.0
        residuo = guadagno
        for voce in sorted(self.voci, key=lambda v: v[0]):      # prima le più vecchie
            if voce[0] < anno or voce[1] <= 0:
                continue
            usa = min(voce[1], residuo)
            voce[1] -= usa
            residuo -= usa
            if residuo <= 0:
                break
        return self.aliquota * residuo


def carica_csv(percorso: Path) -> pd.DataFrame:
    df = pd.read_csv(percorso, parse_dates=["Date"], index_col="Date").sort_index()
    df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
    df = df[~df.index.duplicated(keep="last")]
    if (df["Close"] <= 0).any() or df["Close"].isna().any():
        raise SystemExit(f"{percorso}: chiusure mancanti o non positive.")
    return df


def prepara(titolo: pd.DataFrame, cambio: pd.DataFrame, veloce: int, lenta: int) -> pd.DataFrame:
    df = titolo[["Open", "Close"]].astype(float).copy()
    fx = cambio["Close"].astype(float)
    df["fx"] = fx.reindex(df.index.union(fx.index)).ffill().reindex(df.index)
    df["sma_v"] = df["Close"].rolling(veloce).mean()
    df["sma_l"] = df["Close"].rolling(lenta).mean()
    df["segnale"] = (df["sma_v"] > df["sma_l"]) & df["sma_l"].notna()
    return df


def indice_data(df: pd.DataFrame, data) -> int:
    """Posizione della prima seduta con data uguale o successiva a quella indicata."""
    return int(np.searchsorted(df.index.values, np.datetime64(data)))


def simula(df: pd.DataFrame, i0: int, capitale: float, commissione: float, aliquota: float,
           modo: str, strategia: bool) -> dict:
    """strategia=True: regola a medie mobili; False: buy & hold (compra a i0, vende alla fine)."""
    O, C, FX = (df[c].to_numpy(float) for c in ("Open", "Close", "fx"))
    SEG = df["segnale"].to_numpy(bool)
    date = df.index
    n = len(df)
    cassa, azioni, base = float(capitale), 0.0, 0.0
    zaino = Zaino(aliquota)
    log: list[dict] = []
    tot = {"commissioni": 0.0, "tasse": 0.0}
    equity = np.empty(n - i0)
    in_posizione = np.zeros(n - i0, dtype=bool)

    def compra(i, px_usd, fx, etichetta):
        nonlocal cassa, azioni, base
        investito = cassa - commissione
        if investito <= 0:
            return
        px_eur = px_usd / fx
        spesa = cassa
        azioni = investito / px_eur
        base = spesa
        tot["commissioni"] += commissione
        log.append(dict(data=str(date[i].date()), lato="acquisto", motivo=etichetta, prezzo_usd=px_usd, cambio=fx,
                        prezzo_eur=px_eur, azioni=azioni, commissione=commissione, spesa_eur=spesa,
                        incasso_netto_eur=0.0, plusvalenza_eur=0.0, tassa_eur=0.0))
        cassa = 0.0

    def vendi(i, px_usd, fx, etichetta):
        nonlocal cassa, azioni, base
        px_eur = px_usd / fx
        lordo = azioni * px_eur
        netto = lordo - commissione
        guadagno = netto - base
        tassa = zaino.tassa(int(date[i].year), guadagno)
        tot["commissioni"] += commissione
        tot["tasse"] += tassa
        log.append(dict(data=str(date[i].date()), lato="vendita", motivo=etichetta, prezzo_usd=px_usd, cambio=fx,
                        prezzo_eur=px_eur, azioni=azioni, commissione=commissione, spesa_eur=0.0,
                        incasso_netto_eur=netto, plusvalenza_eur=guadagno, tassa_eur=tassa))
        cassa = netto - tassa
        azioni, base = 0.0, 0.0

    for i in range(i0, n):
        voglio = bool(SEG[i - 1]) if strategia else True
        px, fx = (O[i], FX[i - 1]) if modo == "apertura" else (C[i], FX[i])
        if voglio and azioni == 0.0:
            compra(i, px, fx, "segnale" if strategia else "acquisto iniziale")
        elif (not voglio) and azioni > 0.0:
            vendi(i, px, fx, "segnale")
        equity[i - i0] = cassa + azioni * C[i] / FX[i]
        in_posizione[i - i0] = azioni > 0.0
    if azioni > 0.0:
        vendi(n - 1, C[n - 1], FX[n - 1], "liquidazione finale")

    picco = np.maximum.accumulate(equity)
    anni = (date[-1] - date[i0]).days / 365.25
    finale = cassa
    return {
        "valore_finale_eur": finale,
        "rendimento_totale": finale / capitale - 1,
        "cagr": (finale / capitale) ** (1 / anni) - 1 if finale > 0 else float("nan"),
        "anni": anni,
        "n_acquisti": sum(1 for r in log if r["lato"] == "acquisto"),
        "n_vendite": sum(1 for r in log if r["lato"] == "vendita"),
        "commissioni_eur": tot["commissioni"],
        "tasse_eur": tot["tasse"],
        "quota_giorni_investito": float(in_posizione.mean()),
        "perdita_massima": float((equity / picco - 1).min()),
        "operazioni": log,
        "equity": equity,
        "in_posizione": in_posizione,
    }

cambio_df = pd.DataFrame({"Close": fx_var}, index=pd.Index(date, name="Date"))
piatto = pd.DataFrame({"Close": 1.0}, index=cambio_df.index)
df = s.prepara(titolo, piatto, 50, 200)
i0 = s.indice_data(df, df.index[-1] - pd.DateOffset(years=5))
SEG = df["segnale"].to_numpy(bool)
C = df["Close"].to_numpy()
formula = 500 * np.prod([(C[i + 1] / C[i]) ** SEG[i - 1] for i in range(i0, len(df) - 1)])
r = s.simula(df, i0, 500, 0.0, 0.0, "chiusura", True)
verifica("regola, esecuzione alla chiusura, lordo = prodotto dei rendimenti", uguali(r["valore_finale_eur"], formula),
         f"{r['valore_finale_eur']} contro {formula}")
bh = s.simula(df, i0, 500, 0.0, 0.0, "chiusura", False)
verifica("buy & hold lordo = chiusura finale / chiusura iniziale", uguali(bh["valore_finale_eur"], 500 * C[-1] / C[i0]))
bh_a = s.simula(df, i0, 500, 0.0, 0.0, "apertura", False)
verifica("buy & hold, esecuzione all'apertura = ultima chiusura / prima apertura",
         uguali(bh_a["valore_finale_eur"], 500 * C[-1] / df["Open"].iloc[i0]))
con = s.simula(df, i0, 500, 1.0, 0.0, "apertura", True)
verifica("commissioni = 1 euro per ogni ordine", uguali(con["commissioni_eur"], len(con["operazioni"]) * 1.0))
netto = s.simula(df, i0, 500, 1.0, 0.26, "apertura", True)
verifica("con la tassa non si finisce mai sopra il caso senza tassa",
         netto["valore_finale_eur"] <= con["valore_finale_eur"] + 1e-9)
verifica("il registro delle operazioni quadra con la cassa finale",
         uguali(500 - sum(o["spesa_eur"] for o in netto["operazioni"]) +
                sum(o["incasso_netto_eur"] - o["tassa_eur"] for o in netto["operazioni"]), netto["valore_finale_eur"]))

print("3. Il registro delle operazioni contro i prezzi grezzi (ricerca per data, senza gli array del motore)")
dfv = s.prepara(titolo, cambio_df, 50, 200)
i0v = s.indice_data(dfv, dfv.index[-1] - pd.DateOffset(years=5))
rv = s.simula(dfv, i0v, 500, 1.0, 0.26, "apertura", True)
errori = 0
for o in rv["operazioni"]:
    d = pd.Timestamp(o["data"])
    if o["motivo"] == "liquidazione finale":
        px, fx = titolo.loc[d, "Close"], cambio_df.loc[d, "Close"]
    else:
        prec = cambio_df.index[cambio_df.index.get_loc(d) - 1]
        px, fx = titolo.loc[d, "Open"], cambio_df.loc[prec, "Close"]
    if not (uguali(o["prezzo_usd"], px) and uguali(o["cambio"], fx) and uguali(o["prezzo_eur"], px / fx)):
        errori += 1
verifica(f"{len(rv['operazioni'])} operazioni con prezzo e cambio corretti", errori == 0, f"{errori} errori")
