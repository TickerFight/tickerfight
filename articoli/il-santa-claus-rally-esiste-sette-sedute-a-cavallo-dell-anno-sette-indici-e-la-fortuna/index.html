Il Santa Claus rally esiste? Sette sedute a cavallo dell'anno, sette indici e la fortuna

Le ultime cinque sedute di dicembre e le prime due di gennaio rendono davvero più di sette sedute qualunque, o è fortuna?

L'articolo esce a breve sul blog⬇ Scarica il codice (.zip)Vedi su GitHub

Il disegno

  • Strategia: si compra alla chiusura della sesta-ultima seduta di dicembre e si vende alla chiusura della seconda seduta di gennaio (ultime cinque sedute dell'anno più prime due del nuovo: sette sedute); il resto dell'anno si sta in cassa, che non rende. Il calendario è noto in anticipo: nessun look-ahead.
  • Indici: S&P 500, Dow Jones, Nasdaq Composite, DAX, FTSE 100, CAC 40, Nikkei 225, scelti prima di guardare i dati. Valuta locale, indici di prezzo (senza dividendi).
  • Confronti: buy & hold sullo stesso indice; il rendimento medio di una finestra di sette sedute qualunque; scimmie con la stessa dieta (una finestra di sette sedute per anno, piazzata a caso dentro il suo anno); una griglia di varianti della finestra (da tre a sette sedute di dicembre, da zero a quattro di gennaio) per misurare quanto il risultato dipende dalla definizione.
  • Test: intervallo bootstrap sugli anni, scimmie con correzione di Holm sugli indici, confronto prima e dopo il 1999.
  • Costi: 1 unità di valuta a ordine, tassa del 26% con compensazione delle minusvalenze.

"""Passo 1 - Scarica da Yahoo Finance la serie giornaliera di sette indici azionari.

Salva, per ogni indice, dati/<ticker>.csv (Date, Open, High, Low, Close, Volume) e dati/metadati.json con la fonte,
la data del download, la versione di yfinance e l'impronta SHA-256 di ogni file. Gli script successivi leggono SOLO
i file salvati: i risultati restano riproducibili anche se Yahoo corregge i dati in futuro.
"""
import hashlib
import json
import time
from datetime import datetime, timezone
from zoneinfo import ZoneInfo

import pandas as pd
import yfinance as yf

import config


def scarica(ticker: str) -> pd.DataFrame:
    for tentativo in range(3):
        grezzo = yf.Ticker(ticker).history(start=config.INIZIO_DOWNLOAD, auto_adjust=False)
        if not grezzo.empty:
            return grezzo
        print(f"{ticker}: nessun dato (tentativo {tentativo + 1})")
        time.sleep(10)
    return pd.DataFrame()


def main() -> None:
    config.DATI.mkdir(parents=True, exist_ok=True)
    ora_borsa = datetime.now(ZoneInfo(config.FUSO_BORSA))
    meta = {"fonte": "Yahoo Finance", "url_fonte": config.URL_FONTE, "libreria": f"yfinance {yf.__version__}",
            "scaricato_il_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"), "indici": []}
    for ticker, nome, valuta in config.INDICI:
        riga = {"ticker": ticker, "nome": nome, "valuta": valuta}
        grezzo = scarica(ticker)
        if grezzo.empty:
            riga["stato"] = "scartato: nessun dato restituito da Yahoo Finance"
            meta["indici"].append(riga)
            continue
        df = grezzo[["Open", "High", "Low", "Close", "Volume"]].copy()
        df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
        df.index.name = "Date"
        df = df[~df.index.duplicated(keep="last")].sort_index().dropna(subset=["Close"])
        df = df[df["Close"] > 0]
        # seduta ancora in corso: barra incompleta, da scartare
        if df.index[-1].date() == ora_borsa.date() and ora_borsa.hour < 18:
            df = df.iloc[:-1]
        file = config.file_dati(ticker)
        df.to_csv(file, float_format="%.6f")
        riga.update({"stato": "ok", "file": file.name, "righe": len(df), "prima_data": str(df.index[0].date()),
                     "ultima_data": str(df.index[-1].date()), "sha256": hashlib.sha256(file.read_bytes()).hexdigest()})
        meta["indici"].append(riga)
        print(f"{ticker}: {len(df)} righe, dal {riga['prima_data']} al {riga['ultima_data']}")
    config.FILE_METADATI.write_text(json.dumps(meta, indent=2, ensure_ascii=False), encoding="utf-8")
    if sum(1 for r in meta["indici"] if r["stato"] == "ok") < 4:
        raise SystemExit("Troppo pochi indici scaricati.")


if __name__ == "__main__":
    main()

"""Passo 2 - Santa Claus rally: la finestra di sette sedute contro buy & hold, finestre generiche e scimmie.

Legge SOLO i file in dati/ e produce in risultati/:
  metriche.json       ogni numero dell'articolo, con descrizione e origine di calcolo
  finestre.csv        per ogni indice e anno: date di entrata e uscita e rendimento della finestra
  griglia.csv         per ogni indice: rendimento medio delle varianti della finestra (sedute di dicembre x sedute di gennaio)
  scimmie_medie.csv   rendimento medio delle finestre piazzate a caso, per ogni scimmia e indice
  scimmie_finali.csv  valore finale netto di ogni scimmia, per indice
  indici_usati.json   indici utilizzabili

Strategia: si compra alla chiusura della sesta-ultima seduta di dicembre e si vende alla chiusura della seconda seduta di
gennaio; il resto dell'anno si sta in cassa. Il calendario è noto in anticipo: nessun look-ahead. Scimmia: una finestra
della stessa durata per anno, piazzata a caso dentro il suo anno. Strategia e buy & hold passano dal motore di
simulazione (commissioni, tassa e minusvalenze); per le scimmie si usa la versione veloce, controllata contro il motore.
"""
import json
import platform
from datetime import datetime, timezone

import numpy as np
import pandas as pd

import config
import finestre as f
import simulazione as s

RIS: dict = {}
SORGENTE = "02_backtest_e_test.py con simulazione.py"


def reg(chiave, valore, formato, descrizione, origine=SORGENTE):
    if isinstance(valore, (np.floating, np.integer)):
        valore = valore.item()
    if isinstance(valore, float) and not np.isfinite(valore):
        raise SystemExit(f"Numero non finito per {chiave}")
    RIS[chiave] = {"valore": valore, "formato": formato, "descrizione": descrizione, "origine": origine}


def slug(ticker: str) -> str:
    return ticker.replace("^", "").replace(".", "_").lower()


def holm(p: dict) -> dict:
    ordine = sorted(p, key=p.get)
    m = len(ordine)
    out, massimo = {}, 0.0
    for k, nome in enumerate(ordine):
        massimo = max(massimo, min(1.0, (m - k) * p[nome]))
        out[nome] = massimo
    return out


def uguali(a, b, tol=1e-9):
    return abs(a - b) <= tol * max(1.0, abs(a), abs(b))


def analizza(tk: str, nome: str, valuta: str, indice_rng: int) -> dict | None:
    meta = [r for r in META["indici"] if r["ticker"] == tk][0]
    if meta["stato"] != "ok":
        return None
    t = s.carica_csv(config.file_dati(tk))
    C = t["Close"].to_numpy(float)
    n = len(C)
    date = t.index
    K_D, K_G = config.SEDUTE_DICEMBRE, config.SEDUTE_GENNAIO
    d = K_D + K_G
    ent, usc, anni = f.trova_finestre(date, K_D, K_G)
    if len(ent) < config.ANNI_MINIMI:
        print(f"{tk}: scartato, solo {len(ent)} finestre complete")
        return None
    i0 = int(ent[0])
    rend = C[usc] / C[ent] - 1
    ok_dur = bool(np.all(usc - ent == d))
    if not ok_dur:
        raise SystemExit(f"{tk}: durata delle finestre incoerente")

    # --- Motore: strategia e buy & hold -----------------------------------------------------------------------------
    stato = np.zeros(n, dtype=bool)
    for e, x in zip(ent, usc):
        stato[e:x] = True
    df = pd.DataFrame({"Open": C, "Close": C, "fx": 1.0}, index=date)
    df["segnale"] = np.r_[stato[1:], False]
    h = s.simula(df, i0, config.CAPITALE, config.COMMISSIONE, config.ALIQUOTA, config.MODO, True)
    bh = s.simula(df, i0, config.CAPITALE, config.COMMISSIONE, config.ALIQUOTA, config.MODO, False)
    anno_usc = date.year.to_numpy()[usc]
    fv, tv = f.finale_veloce(C, anno_usc, ent, usc, config.CAPITALE, config.COMMISSIONE, config.ALIQUOTA)
    if not (uguali(fv, h["valore_finale_eur"]) and uguali(tv, h["tasse_eur"])):
        raise SystemExit(f"{tk}: motore veloce e motore completo non coincidono ({fv} contro {h['valore_finale_eur']})")
    if h["n_acquisti"] != len(ent):
        raise SystemExit(f"{tk}: numero di acquisti inatteso")

    # --- Finestre reali e finestre generiche -----------------------------------------------------------------------
    rng = np.random.default_rng([config.SEME, indice_rng])
    boot = rng.choice(rend, size=(config.N_BOOTSTRAP, len(rend)), replace=True).mean(axis=1)
    alfa = (1 - config.LIVELLO_CONFIDENZA) / 2
    lo, hi = np.quantile(boot, [alfa, 1 - alfa])
    gen = C[i0 + d:] / C[i0:n - d] - 1
    media, media_gen = float(rend.mean()), float(gen.mean())

    # --- Le scimmie ------------------------------------------------------------------------------------------------
    lunghezza = int(np.median(np.diff(ent))) if len(ent) > 1 else 252
    fine_ultimo = min(n, int(ent[-1]) + lunghezza)
    pos = f.scimmie(C, ent, d, fine_ultimo, config.N_SCIMMIE, rng)
    ret_sc = C[pos + d] / C[pos] - 1
    medie_sc = ret_sc.mean(axis=1)
    quota_sc = float((ret_sc > 0).mean())
    p_media = (int((medie_sc >= media).sum()) + 1) / (config.N_SCIMMIE + 1)
    anno_calend = date.year.to_numpy()
    finali = np.empty(config.N_SCIMMIE)
    for j in range(config.N_SCIMMIE):
        finali[j], _ = f.finale_veloce(C, anno_calend[pos[j] + d], pos[j], pos[j] + d, config.CAPITALE,
                                       config.COMMISSIONE, config.ALIQUOTA)
    p_valore = (int((finali >= h["valore_finale_eur"]).sum()) + 1) / (config.N_SCIMMIE + 1)

    # --- Prima e dopo ---------------------------------------------------------------------------------------------
    prima = rend[anni < config.ANNO_DIVISIONE]
    dopo = rend[anni >= config.ANNO_DIVISIONE]
    # Test AGGIUNTO dopo aver visto i risultati principali (non è tra le ipotesi): le scimmie sulle sole finestre del periodo «dopo»
    p_dopo = None
    if len(dopo) >= 5:
        sc_dopo = ret_sc[:, anni >= config.ANNO_DIVISIONE].mean(axis=1)
        p_dopo = (int((sc_dopo >= dopo.mean()).sum()) + 1) / (config.N_SCIMMIE + 1)

    # --- Griglia di varianti --------------------------------------------------------------------------------------
    griglia = []
    for kk in config.GRIGLIA_DICEMBRE:
        for mm in config.GRIGLIA_GENNAIO:
            e2, x2, a2 = f.trova_finestre(date, kk, mm)
            sel = (a2 >= anni[0]) & (a2 <= anni[-1])
            r2 = C[x2[sel]] / C[e2[sel]] - 1
            L = kk + mm
            g2 = float((C[i0 + L:] / C[i0:n - L] - 1).mean())
            griglia.append({"indice": tk, "k": kk, "m": mm, "n": int(sel.sum()), "media": float(r2.mean()),
                            "generica": g2, "eccesso": float(r2.mean()) - g2})
    griglia = pd.DataFrame(griglia)
    standard = griglia[(griglia["k"] == K_D) & (griglia["m"] == K_G)].iloc[0]
    rango_griglia = int((griglia["eccesso"] > standard["eccesso"]).sum() + 1)

    tab = pd.DataFrame({"indice": tk, "anno": anni, "entrata": date[ent].strftime("%Y-%m-%d"),
                        "uscita": date[usc].strftime("%Y-%m-%d"), "rendimento": rend})
    return dict(tk=tk, sl=slug(tk), nome=nome, valuta=valuta, date=date, i0=i0, ent=ent, usc=usc, anni=anni, rend=rend,
                lo=lo, hi=hi, media=media, media_gen=media_gen, gen=gen, h=h, bh=bh, medie_sc=medie_sc, quota_sc=quota_sc,
                p_media=p_media, finali=finali, p_valore=p_valore, prima=prima, dopo=dopo, p_dopo=p_dopo, griglia=griglia,
                rango_griglia=rango_griglia, tab=tab)


def main() -> None:
    global META
    config.RISULTATI.mkdir(parents=True, exist_ok=True)
    META = json.loads(config.FILE_METADATI.read_text(encoding="utf-8"))
    res = []
    for k, (tk, nome, valuta) in enumerate(config.INDICI):
        r = analizza(tk, nome, valuta, k)
        if r:
            res.append(r)
            print(f"{tk}: {len(r['rend'])} finestre, media {r['media']:.4%} contro generica {r['media_gen']:.4%}, "
                  f"p scimmie {r['p_media']:.4f}, strategia {r['h']['valore_finale_eur']:.0f} contro buy & hold {r['bh']['valore_finale_eur']:.0f}")
    if len(res) < 4:
        raise SystemExit("Troppi pochi indici utilizzabili.")

    ph = holm({r["tk"]: r["p_media"] for r in res})
    ph_dopo = holm({r["tk"]: r["p_dopo"] for r in res if r["p_dopo"] is not None})
    for r in res:
        sl, h, bh, rend = r["sl"], r["h"], r["bh"], r["rend"]
        reg(f"{sl}_nome", r["nome"], "testo", "Nome dell'indice", "config.py, INDICI")
        reg(f"{sl}_data_inizio", str(r["date"][r["i0"]].date()), "data_lunga", f"Prima operazione del test su {r['nome']} (sesta-ultima seduta del primo dicembre utile)", "dati/")
        reg(f"{sl}_anni", h["anni"], "num1", f"Anni del test su {r['nome']}")
        reg(f"{sl}_n_finestre", len(rend), "int", f"Finestre complete (dicembre e gennaio seguente) per {r['nome']}", "finestre.csv")
        reg(f"{sl}_media", r["media"], "pct2", f"Rendimento medio della finestra di sette sedute, {r['nome']}", "finestre.csv")
        reg(f"{sl}_mediana", float(np.median(rend)), "pct2", f"Rendimento mediano della finestra, {r['nome']}", "finestre.csv")
        reg(f"{sl}_media_gen", r["media_gen"], "pct2", f"Rendimento medio di una finestra di sette sedute qualunque, {r['nome']}", "tutte le finestre di sette sedute, dalla prima entrata")
        reg(f"{sl}_eccesso", r["media"] - r["media_gen"], "pp2", f"Rendimento medio della finestra meno quello medio di una finestra qualunque, in punti percentuali, {r['nome']}", "finestre.csv")
        reg(f"{sl}_lo", r["lo"], "pct2", f"Estremo inferiore dell'intervallo bootstrap del rendimento medio della finestra, {r['nome']}", "bootstrap sugli anni")
        reg(f"{sl}_hi", r["hi"], "pct2", f"Estremo superiore dell'intervallo bootstrap del rendimento medio della finestra, {r['nome']}", "bootstrap sugli anni")
        reg(f"{sl}_quota_pos", float((rend > 0).mean()), "pct0", f"Quota di anni in cui la finestra ha chiuso in positivo, {r['nome']}", "finestre.csv")
        reg(f"{sl}_quota_pos_gen", float((r["gen"] > 0).mean()), "pct0", f"Quota di finestre di sette sedute qualunque chiuse in positivo, {r['nome']}", "tutte le finestre di sette sedute")
        reg(f"{sl}_n_pos", int((rend > 0).sum()), "int", f"Anni con la finestra in positivo, {r['nome']}", "finestre.csv")
        reg(f"{sl}_n_neg", int((rend <= 0).sum()), "int", f"Anni con la finestra in perdita o a zero, {r['nome']}", "finestre.csv")
        reg(f"{sl}_peggiore", float(rend.min()), "pct1", f"Peggiore rendimento della finestra in un anno, {r['nome']}", "finestre.csv")
        reg(f"{sl}_anno_peggiore", int(r["anni"][rend.argmin()]) + 1, "annonum", f"Anno (gennaio) della peggiore finestra, {r['nome']}", "finestre.csv")
        reg(f"{sl}_migliore", float(rend.max()), "pct1", f"Miglior rendimento della finestra in un anno, {r['nome']}", "finestre.csv")
        reg(f"{sl}_sc_media", float(np.mean(r["medie_sc"])), "pct2", f"Rendimento medio delle scimmie (finestre della stessa durata piazzate a caso), {r['nome']}", "scimmie_medie.csv")
        reg(f"{sl}_p_media", r["p_media"], "pval", f"p-value a una coda: quota di scimmie con rendimento medio almeno pari a quello della finestra, {r['nome']}", "scimmie_medie.csv")
        reg(f"{sl}_p_holm", ph[r["tk"]], "pval", f"p-value delle scimmie corretto con il metodo di Holm sugli indici, {r['nome']}", "scimmie_medie.csv")
        reg(f"{sl}_finale_h", h["valore_finale_eur"], "int", f"Valore finale netto della strategia (capitale iniziale e valore nella valuta dell'indice), {r['nome']}")
        reg(f"{sl}_finale_bh", bh["valore_finale_eur"], "int", f"Valore finale netto del buy & hold, {r['nome']}")
        reg(f"{sl}_cagr_h", h["cagr"], "pct2", f"Rendimento annuo composto netto della strategia, {r['nome']}")
        reg(f"{sl}_cagr_bh", bh["cagr"], "pct1", f"Rendimento annuo composto netto del buy & hold, {r['nome']}")
        reg(f"{sl}_quota_dentro", h["quota_giorni_investito"], "pct1", f"Quota di sedute con la strategia investita, {r['nome']}")
        reg(f"{sl}_n_ordini", h["n_acquisti"] + h["n_vendite"], "int", f"Ordini della strategia (acquisti e vendite), {r['nome']}")
        reg(f"{sl}_comm", h["commissioni_eur"], "int", f"Commissioni pagate dalla strategia, {r['nome']}")
        reg(f"{sl}_tasse_h", h["tasse_eur"], "int", f"Tasse pagate dalla strategia, {r['nome']}")
        reg(f"{sl}_tasse_bh", bh["tasse_eur"], "int", f"Tasse pagate dal buy & hold (a fine periodo), {r['nome']}")
        reg(f"{sl}_sc_sotto", float((r["finali"] < h["valore_finale_eur"]).mean()), "pct0", f"Quota di scimmie che chiudono sotto la strategia (valore finale), {r['nome']}", "scimmie_finali.csv")
        reg(f"{sl}_n_prima", len(r["prima"]), "int", f"Finestre iniziate a dicembre fino al {config.ANNO_DIVISIONE - 1}, {r['nome']}", "finestre.csv")
        reg(f"{sl}_n_dopo", len(r["dopo"]), "int", f"Finestre iniziate a dicembre dal {config.ANNO_DIVISIONE}, {r['nome']}", "finestre.csv")
        if len(r["prima"]):
            reg(f"{sl}_media_prima", float(r["prima"].mean()), "pct2", f"Rendimento medio della finestra fino al {config.ANNO_DIVISIONE - 1}, {r['nome']}", "finestre.csv")
        if len(r["dopo"]):
            reg(f"{sl}_media_dopo", float(r["dopo"].mean()), "pct2", f"Rendimento medio della finestra dal {config.ANNO_DIVISIONE}, {r['nome']}", "finestre.csv")
        if r["p_dopo"] is not None:
            reg(f"{sl}_p_dopo", r["p_dopo"], "pval", f"p-value a una coda delle scimmie sulle sole finestre dal {config.ANNO_DIVISIONE} (test aggiunto dopo i risultati principali), {r['nome']}", "scimmie")
            reg(f"{sl}_p_dopo_holm", ph_dopo[r["tk"]], "pval", f"p-value del test sul periodo «dopo» corretto con Holm, {r['nome']}", "scimmie")
        reg(f"{sl}_rango_griglia", r["rango_griglia"], "int", f"Posto della finestra standard tra le varianti della griglia, per eccesso sul caso (uno è la migliore), {r['nome']}", "griglia.csv")
        reg(f"{sl}_griglia_pos", int((r["griglia"]["eccesso"] > 0).sum()), "int", f"Varianti della griglia con rendimento medio sopra quello di una finestra qualunque della stessa durata, {r['nome']}", "griglia.csv")

    # --- Numeri complessivi -----------------------------------------------------------------------------------------
    n_ind = len(res)
    ecc = [r["media"] - r["media_gen"] for r in res]
    reg("n_indici", n_ind, "int", "Indici utilizzabili nel test", "config.py, INDICI e dati/")
    reg("n_scartati", len(config.INDICI) - n_ind, "int", "Indici scartati (nessun dato o storia troppo corta)", "metadati.json")
    reg("n_totale", len(config.INDICI), "int", "Indici scelti in partenza", "config.py, INDICI")

"""Funzioni della finestra del Santa Claus rally: trovare le finestre, simulare in modo veloce le scimmie.

Una finestra con k sedute di dicembre e m di gennaio: si compra alla chiusura della (k+1)-esima seduta contando dalla fine
di dicembre (così restano k sedute da vivere a dicembre) e si vende alla chiusura della m-esima seduta di gennaio
(se m = 0, all'ultima seduta di dicembre). Durata: k + m sedute da chiusura a chiusura.
"""
import numpy as np
import pandas as pd

import simulazione as s


def trova_finestre(indice: pd.DatetimeIndex, k: int, m: int):
    """Restituisce (entrata, uscita, anno_dicembre): posizioni nell'indice e anno del dicembre di ogni finestra completa."""
    anno, mese = indice.year.to_numpy(), indice.month.to_numpy()
    ent, usc, anni = [], [], []
    for a in range(int(anno.min()), int(anno.max())):
        dic = np.flatnonzero((anno == a) & (mese == 12))
        gen = np.flatnonzero((anno == a + 1) & (mese == 1))
        if len(dic) < k + 1 or len(gen) < max(m, 1):
            continue
        ent.append(int(dic[-(k + 1)]))
        usc.append(int(gen[m - 1]) if m > 0 else int(dic[-1]))
        anni.append(a)
    return np.array(ent, dtype=int), np.array(usc, dtype=int), np.array(anni, dtype=int)

                  commissione: float, aliquota: float) -> tuple[float, float]:
    """Valore finale netto e tasse di una sequenza di finestre (acquisto e vendita alla chiusura, cambio 1).
    Stesse regole di simulazione.simula: commissione su ogni ordine, tassa con compensazione delle minusvalenze."""
    zaino = s.Zaino(aliquota)
    cassa, tasse = float(capitale), 0.0
    for e, x, a in zip(ent, usc, anno_uscita):
        investito = cassa - commissione
        if investito <= 0:
            continue
        azioni = investito / C[e]
        netto = azioni * C[x] - commissione
        t = zaino.tassa(int(a), netto - cassa)
        tasse += t
        cassa = netto - t
    return cassa, tasse


def scimmie(C: np.ndarray, ini: np.ndarray, durata: int, fine_ultimo: int, n_scimmie: int, rng) -> np.ndarray:
    """Posizioni di entrata (n_scimmie x K): una finestra per anno, piazzata a caso dentro il blocco dell'anno.
    Il blocco k va dall'entrata reale k a quella k+1 (l'ultimo fino a fine_ultimo); la scimmia ha la stessa durata."""
    fine = np.r_[ini[1:], fine_ultimo]
    alto = np.maximum(fine - durata - 1, ini)
    return rng.integers(ini, alto + 1, size=(n_scimmie, len(ini)))
flusso/simulazione.pymotore: ordini, commissioni, tassa e minusvalenze
"""Motore di simulazione in euro di una regola a medie mobili su un singolo titolo.

Regole (da dichiarare nell'articolo):
  - segnale: investito se la media mobile veloce è sopra la lenta, calcolate sulle chiusure;
  - il segnale a fine seduta t decide l'operazione della seduta t+1 (nessun look-ahead);
  - esecuzione "apertura": prezzo di apertura di t+1, cambio dell'ultima chiusura nota (t);
    esecuzione "chiusura": prezzo di chiusura di t+1, cambio di t+1 (controllo di robustezza);
  - azioni frazionate, si investe tutta la cassa; commissione fissa per ogni ordine, acquisto o vendita;
  - la cassa non rende nulla; i dividendi non sono considerati;
  - tassa sulle plusvalenze calcolate in euro (cambio incluso), pagata a ogni vendita come nel regime
    amministrato; la commissione si somma al costo di acquisto e si sottrae dall'incasso; le minusvalenze
    compensano le plusvalenze successive fino al quarto anno seguente;
  - a fine periodo si vende tutto, con commissione e tassa, per confrontare cifre nette.
Non modellati: imposta di bollo, ritenuta sui dividendi, spread di cambio.
"""
from pathlib import Path

import numpy as np
import pandas as pd


class Zaino:
    """Minusvalenze compensabili con le plusvalenze successive, fino al 4° anno seguente."""

    def __init__(self, aliquota: float):
        self.aliquota = aliquota
        self.voci: list[list[float]] = []   # [ultimo anno utilizzabile, importo residuo]

    def tassa(self, anno: int, guadagno: float) -> float:
        if guadagno < 0:
            self.voci.append([anno + 4, -guadagno])
            return 0.0
        residuo = guadagno
        for voce in sorted(self.voci, key=lambda v: v[0]):      # prima le più vecchie
            if voce[0] < anno or voce[1] <= 0:
                continue
            usa = min(voce[1], residuo)
            voce[1] -= usa
            residuo -= usa
            if residuo <= 0:
                break
        return self.aliquota * residuo


def carica_csv(percorso: Path) -> pd.DataFrame:
    df = pd.read_csv(percorso, parse_dates=["Date"], index_col="Date").sort_index()
    df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
    df = df[~df.index.duplicated(keep="last")]
    if (df["Close"] <= 0).any() or df["Close"].isna().any():
        raise SystemExit(f"{percorso}: chiusure mancanti o non positive.")
    return df


def prepara(titolo: pd.DataFrame, cambio: pd.DataFrame, veloce: int, lenta: int) -> pd.DataFrame:
    df = titolo[["Open", "Close"]].astype(float).copy()
    fx = cambio["Close"].astype(float)
    df["fx"] = fx.reindex(df.index.union(fx.index)).ffill().reindex(df.index)
    df["sma_v"] = df["Close"].rolling(veloce).mean()
    df["sma_l"] = df["Close"].rolling(lenta).mean()
    df["segnale"] = (df["sma_v"] > df["sma_l"]) & df["sma_l"].notna()
    return df


def indice_data(df: pd.DataFrame, data) -> int:
    """Posizione della prima seduta con data uguale o successiva a quella indicata."""
    return int(np.searchsorted(df.index.values, np.datetime64(data)))


def simula(df: pd.DataFrame, i0: int, capitale: float, commissione: float, aliquota: float,
           modo: str, strategia: bool) -> dict:
    """strategia=True: regola a medie mobili; False: buy & hold (compra a i0, vende alla fine)."""
    O, C, FX = (df[c].to_numpy(float) for c in ("Open", "Close", "fx"))
    SEG = df["segnale"].to_numpy(bool)
    date = df.index
    n = len(df)
    cassa, azioni, base = float(capitale), 0.0, 0.0
    zaino = Zaino(aliquota)
    log: list[dict] = []
    tot = {"commissioni": 0.0, "tasse": 0.0}
    equity = np.empty(n - i0)
    in_posizione = np.zeros(n - i0, dtype=bool)

    def compra(i, px_usd, fx, etichetta):
        nonlocal cassa, azioni, base
        investito = cassa - commissione
        if investito <= 0:
            return
        px_eur = px_usd / fx
        spesa = cassa
        azioni = investito / px_eur
        base = spesa
        tot["commissioni"] += commissione
        log.append(dict(data=str(date[i].date()), lato="acquisto", motivo=etichetta, prezzo_usd=px_usd, cambio=fx,
                        prezzo_eur=px_eur, azioni=azioni, commissione=commissione, spesa_eur=spesa,
                        incasso_netto_eur=0.0, plusvalenza_eur=0.0, tassa_eur=0.0))
        cassa = 0.0

    def vendi(i, px_usd, fx, etichetta):
        nonlocal cassa, azioni, base
        px_eur = px_usd / fx
        lordo = azioni * px_eur
        netto = lordo - commissione
        guadagno = netto - base
        tassa = zaino.tassa(int(date[i].year), guadagno)
        tot["commissioni"] += commissione
        tot["tasse"] += tassa
        log.append(dict(data=str(date[i].date()), lato="vendita", motivo=etichetta, prezzo_usd=px_usd, cambio=fx,
                        prezzo_eur=px_eur, azioni=azioni, commissione=commissione, spesa_eur=0.0,
                        incasso_netto_eur=netto, plusvalenza_eur=guadagno, tassa_eur=tassa))
        cassa = netto - tassa
        azioni, base = 0.0, 0.0

    for i in range(i0, n):
        voglio = bool(SEG[i - 1]) if strategia else True
        px, fx = (O[i], FX[i - 1]) if modo == "apertura" else (C[i], FX[i])
        if voglio and azioni == 0.0:
            compra(i, px, fx, "segnale" if strategia else "acquisto iniziale")
        elif (not voglio) and azioni > 0.0:
            vendi(i, px, fx, "segnale")
        equity[i - i0] = cassa + azioni * C[i] / FX[i]
        in_posizione[i - i0] = azioni > 0.0
    if azioni > 0.0:
        vendi(n - 1, C[n - 1], FX[n - 1], "liquidazione finale")

    picco = np.maximum.accumulate(equity)
    anni = (date[-1] - date[i0]).days / 365.25
    finale = cassa
    return {
        "valore_finale_eur": finale,
        "rendimento_totale": finale / capitale - 1,
        "cagr": (finale / capitale) ** (1 / anni) - 1 if finale > 0 else float("nan"),
        "anni": anni,
        "n_acquisti": sum(1 for r in log if r["lato"] == "acquisto"),
        "n_vendite": sum(1 for r in log if r["lato"] == "vendita"),
        "commissioni_eur": tot["commissioni"],
        "tasse_eur": tot["tasse"],
        "quota_giorni_investito": float(in_posizione.mean()),
        "perdita_massima": float((equity / picco - 1).min()),
        "operazioni": log,
        "equity": equity,
        "in_posizione": in_posizione,
    }
