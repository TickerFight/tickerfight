Stop loss al 5%: protegge o taglia i guadagni?

Su una regola di trend (chiusura sopra la media a 200 sedute) uno stop loss fisso al 5% protegge dalle perdite grosse o taglia i guadagni? Condizioni di chi parte con 1.000 €: commissione per ordine in tre scenari, tassa del 26% con minusvalenze riportate, azioni frazionate.

L'articolo esce a breve sul blog⬇ Scarica il codice (.zip)

Il disegno

  • Regola di trend: investito se la chiusura sta sopra la media mobile semplice a 200 sedute; il segnale di fine seduta decide l'ordine della seduta dopo, eseguito alla chiusura (nessun look-ahead).
  • Stop loss: se la chiusura scende del 5% o più sotto il prezzo di acquisto, si vende alla chiusura della seduta successiva; si rientra solo dopo che il segnale è tornato falso e poi vero.
  • Strumenti (scelti prima di guardare i dati): un ETF azionario mondiale e un ETF sull'S&P 500 quotati a Milano in euro, più l'indice S&P 500 in dollari convertito in euro.
  • Condizioni: 1.000 €; commissione 0 €, 2 € a ordine e 0,1% dell'importo; tassa 26% con compensazione delle minusvalenze; dividendi e imposta di bollo non considerati.
  • Confronti: buy & hold; la stessa regola senza stop; tre soglie di controllo (3%, 8%, 10%); scimmie con lo stesso numero di periodi investiti e la stessa durata totale.
  • Test: scimmie (una coda) e bootstrap a blocchi della differenza di rendimento giornaliero tra regola con stop e senza, entrambi con la correzione di Holm sugli strumenti.

"""Passo 1 - Scarica da Yahoo Finance le serie giornaliere degli strumenti e del cambio euro-dollaro.

Salva dati/<ticker>.csv (Date, Open, High, Low, Close, Volume) e dati/metadati.json con la fonte, la data del download,
la versione di yfinance e l'impronta SHA-256 di ogni file. Gli script successivi leggono SOLO i file salvati: i risultati
restano riproducibili anche se Yahoo corregge i dati in futuro.
"""
import hashlib
import json
import time
from datetime import datetime, timezone
from zoneinfo import ZoneInfo

import pandas as pd
import yfinance as yf

import config


def scarica(ticker: str) -> pd.DataFrame:
    for tentativo in range(3):
        grezzo = yf.Ticker(ticker).history(start=config.INIZIO_DOWNLOAD, auto_adjust=False)
        if not grezzo.empty:
            return grezzo
        print(f"{ticker}: nessun dato (tentativo {tentativo + 1})")
        time.sleep(10)
    return pd.DataFrame()


def main() -> None:
    config.DATI.mkdir(parents=True, exist_ok=True)
    ora_borsa = datetime.now(ZoneInfo(config.FUSO_BORSA))
    meta = {"fonte": "Yahoo Finance", "url_fonte": config.URL_FONTE, "libreria": f"yfinance {yf.__version__}",
            "scaricato_il_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"), "strumenti": []}
    elenco = [(t, n, v) for t, n, v in config.STRUMENTI] + [(config.TICKER_CAMBIO, "cambio euro-dollaro", "USD")]
    for ticker, nome, valuta in elenco:
        riga = {"ticker": ticker, "nome": nome, "valuta": valuta}
        grezzo = scarica(ticker)
        if grezzo.empty:
            riga["stato"] = "scartato: nessun dato restituito da Yahoo Finance"
            meta["strumenti"].append(riga)
            continue
        df = grezzo[["Open", "High", "Low", "Close", "Volume"]].copy()
        df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
        df.index.name = "Date"
        df = df[~df.index.duplicated(keep="last")].sort_index().dropna(subset=["Close"])
        df = df[df["Close"] > 0]
        if df.index[-1].date() == ora_borsa.date() and ora_borsa.hour < 18:       # seduta in corso: barra incompleta
            df = df.iloc[:-1]
        salti = (df["Close"].pct_change().abs() > 0.25).sum()
        file = config.file_dati(ticker)
        df.to_csv(file, float_format="%.6f")
        riga.update({"stato": "ok", "file": file.name, "righe": len(df), "prima_data": str(df.index[0].date()),
                     "ultima_data": str(df.index[-1].date()), "variazioni_giornaliere_oltre_25_per_cento": int(salti),
                     "sha256": hashlib.sha256(file.read_bytes()).hexdigest()})
        meta["strumenti"].append(riga)
        print(f"{ticker}: {len(df)} righe, dal {riga['prima_data']} al {riga['ultima_data']}, salti oltre il 25%: {salti}")
    config.FILE_METADATI.write_text(json.dumps(meta, indent=2, ensure_ascii=False), encoding="utf-8")
    ok = {r["ticker"] for r in meta["strumenti"] if r["stato"] == "ok"}
    if config.TICKER_CAMBIO not in ok and any(v == "USD" for _, _, v in config.STRUMENTI):
        print("Attenzione: cambio non scaricato, gli strumenti in dollari verranno scartati.")
    if sum(1 for t, _, _ in config.STRUMENTI if t in ok) < 1:
        raise SystemExit("Nessuno strumento scaricato.")


if __name__ == "__main__":
    main()

"""Passo 2 - Regola di trend con e senza stop loss, con le condizioni di chi parte con 1.000 euro.

Legge SOLO i file in dati/ e produce in risultati/:
  metriche.json          ogni numero dell'articolo, con descrizione e origine di calcolo
  prove.csv              per strumento, regola e scenario: valore finale, rendimento annuo, ordini, costi, tasse, perdita massima
  stop_eventi.csv        per ogni stop scattato: data di vendita, prezzo, data e prezzo del rientro
  scimmie_finali.csv     valore finale netto di ogni scimmia (scenario principale), per strumento e regola
  curve.csv              valore giorno per giorno (una seduta su cinque) di buy & hold, trend e trend con stop

Regole: buy & hold (compra il primo giorno e tiene); "trend" (investito se la chiusura sta sopra la media a 200 sedute);
"s5" = trend con stop loss al 5%; "s3", "s8", "s10" = controlli con altre soglie. Quattro scenari: lordo (nessun costo,
nessuna tassa) e tre scenari di commissione, tutti con la tassa del 26%.
Scimmia: stesso numero di periodi investiti e stessa durata totale della regola, piazzati a caso senza sovrapposizioni;
gli stessi costi e le stesse tasse, passati dallo stesso contabile.
"""
import json
import platform
from datetime import datetime, timezone

import numpy as np
import pandas as pd

import config
import simulazione as s

RIS: dict = {}
SORGENTE = "02_backtest_e_test.py con simulazione.py"
SCEN = [("lordo", 0.0, 0.0, 0.0)] + [(f"c{k}", f, p, config.ALIQUOTA) for k, (_, f, p) in enumerate(config.SCENARI_COSTO)]
ID_SCEN_PRINC = f"c{config.SCENARIO_PRINCIPALE}"
NOME_SCEN = {f"c{k}": nome for k, (nome, _, _) in enumerate(config.SCENARI_COSTO)}
NOME_SCEN["lordo"] = "lordo"


def id_regola(soglia: float) -> str:
    return f"s{int(round(soglia * 100))}"


REGOLE = {"bh": None, "tr": None, id_regola(config.STOP): config.STOP}
for x in config.SOGLIE_CONTROLLO:
    REGOLE[id_regola(x)] = x
ID_STOP = id_regola(config.STOP)


def reg(chiave, valore, formato, descrizione, origine=SORGENTE):
    if isinstance(valore, (np.floating, np.integer)):
        valore = valore.item()
    if isinstance(valore, float) and not np.isfinite(valore):
        raise SystemExit(f"Numero non finito per {chiave}")
    RIS[chiave] = {"valore": valore, "formato": formato, "descrizione": descrizione, "origine": origine}


def holm(p: dict) -> dict:
    ordine = sorted(p, key=p.get)
    m = len(ordine)
    out, massimo = {}, 0.0
    for k, nome in enumerate(ordine):
        massimo = max(massimo, min(1.0, (m - k) * p[nome]))
        out[nome] = massimo
    return out


def bootstrap_blocchi(d: np.ndarray, rng) -> np.ndarray:
    """Medie di B ricampionamenti a blocchi circolari della serie d (lunghezza del blocco: config.LUNGHEZZA_BLOCCO)."""
    N, L = len(d), config.LUNGHEZZA_BLOCCO
    nb = int(np.ceil(N / L))
    medie = np.empty(config.N_BOOTSTRAP)
    passo = 250
    for a in range(0, config.N_BOOTSTRAP, passo):
        m = min(passo, config.N_BOOTSTRAP - a)
        inizi = rng.integers(0, N, size=(m, nb))
        idx = (inizi[:, :, None] + np.arange(L)[None, None, :]) % N
        medie[a:a + m] = d[idx.reshape(m, -1)[:, :N]].mean(axis=1)
    return medie


def analizza(k_str: int, tk: str, nome: str, valuta: str, meta: dict, fx_df) -> dict | None:
    riga = [r for r in meta["strumenti"] if r["ticker"] == tk][0]
    if riga["stato"] != "ok":
        print(f"{tk}: scartato, {riga['stato']}")
        return None
    t = s.carica_csv(config.file_dati(tk))
    if valuta == "USD":
        if fx_df is None:
            print(f"{tk}: scartato, manca il cambio")
            return None
        df = s.prepara(t, fx_df, config.MEDIA_MOBILE)
    else:
        df = s.prepara(t, None, config.MEDIA_MOBILE)
    if len(df) < config.MEDIA_MOBILE + 50:
        return None
    i0 = s.primo_indice(df)
    anni_test = (df.index[-1] - df.index[i0]).days / 365.25
    if anni_test < config.STORIA_MINIMA_ANNI:
        print(f"{tk}: scartato, solo {anni_test:.1f} anni di test")
        return None
    n = len(df)
    C, FX = df["Close"].to_numpy(float), df["fx"].to_numpy(float)
    anni_idx = df.index.year.to_numpy()

    # --- Tutte le simulazioni (regola x scenario) --------------------------------------------------------------------
    sim = {}
    for rid, stop in REGOLE.items():
        for sid, f, p, al in SCEN:
            sim[(rid, sid)] = s.simula(df, i0, config.CAPITALE, f, p, al, rid != "bh", stop=stop)
    prove = []
    for (rid, sid), r in sim.items():
        prove.append({"strumento": tk, "regola": rid, "scenario": sid, "finale": r["valore_finale_eur"], "cagr": r["cagr"],
                      "ordini": r["n_acquisti"] + r["n_vendite"], "n_stop": r["n_stop"], "commissioni": r["commissioni_eur"],
                      "tasse": r["tasse_eur"], "perdita_massima": r["perdita_massima"], "quota_investito": r["quota_giorni_investito"]})

    # --- Stop scattati: che cosa è successo dopo ---------------------------------------------------------------------
    eventi = []
    ops = sim[(ID_STOP, "lordo")]["operazioni"]
    for j, o in enumerate(ops):
        if o["motivo"] != "stop loss":
            continue
        succ = [x for x in ops[j + 1:] if x["lato"] == "acquisto"]
        ev = {"strumento": tk, "data_stop": o["data"], "prezzo_stop": o["prezzo"], "data_rientro": None, "prezzo_rientro": None}
        if succ:
            ev["data_rientro"], ev["prezzo_rientro"] = succ[0]["data"], succ[0]["prezzo"]
        eventi.append(ev)
    ev_df = pd.DataFrame(eventi)
    con_rientro = ev_df.dropna(subset=["prezzo_rientro"]) if len(ev_df) else ev_df
    quota_piu_caro = float((con_rientro["prezzo_rientro"] > con_rientro["prezzo_stop"]).mean()) if len(con_rientro) else float("nan")

    # --- Dove sta il livello dello stop rispetto alla media a 200 sedute, nei giorni di acquisto della regola di trend ------
    acq_tr = np.array([o["i"] for o in sim[("tr", "lordo")]["operazioni"] if o["lato"] == "acquisto"], dtype=int)
    sma = df["sma"].to_numpy(float)
    livello_stop = C[acq_tr] * (1.0 - config.STOP)
    quota_stop_sotto_media = float((livello_stop < sma[acq_tr]).mean())
    distanza_mediana = float(np.median(C[acq_tr] / sma[acq_tr] - 1.0))

    # --- Le scimmie (rispetto alla regola di trend e alla regola con lo stop del 5%) --------------------------------------
    rng_s = np.random.default_rng([config.SEME, k_str, 1])
    N = n - i0
    scim = {}
    for rid in ("tr", ID_STOP):
        a, b = s.blocchi(sim[(rid, "lordo")])
        k, D = len(a), int((b - a).sum())
        if k < 2:
            scim[rid] = None
            continue
        finali = {sid: np.empty(config.N_SCIMMIE) for sid in NOME_SCEN if sid != "lordo"}
        for j in range(config.N_SCIMMIE):
            ia, ib = s.scimmia(k, D, N, rng_s)
            for sid, f, p, al in SCEN[1:]:
                finali[sid][j] = s.valuta_blocchi(C, FX, anni_idx, ia + i0, ib + i0, config.CAPITALE, f, p, al)[0]
        scim[rid] = {"k": k, "D": D, "finali": finali}

    # --- Bootstrap a blocchi: regola con stop contro regola senza stop (rendimento giornaliero netto, scenario principale)
    e_stop = sim[(ID_STOP, ID_SCEN_PRINC)]["equity"]
    e_tr = sim[("tr", ID_SCEN_PRINC)]["equity"]
    d = (e_stop[1:] / e_stop[:-1] - 1) - (e_tr[1:] / e_tr[:-1] - 1)
    rng_b = np.random.default_rng([config.SEME, k_str, 2])
    medie = bootstrap_blocchi(d - d.mean(), rng_b)             # distribuzione sotto l'ipotesi nulla (differenza media zero)
    osservata = float(d.mean())
    p_boot = (int((np.abs(medie) >= abs(osservata)).sum()) + 1) / (config.N_BOOTSTRAP + 1)
    medie_ic = bootstrap_blocchi(d, np.random.default_rng([config.SEME, k_str, 3]))
    alfa = (1 - config.LIVELLO_CONFIDENZA) / 2
    lo, hi = np.quantile(medie_ic, [alfa, 1 - alfa])

    camp = np.arange(0, N, 5)
    curva = pd.DataFrame({"strumento": tk, "data": df.index[i0:][camp],
                          "bh": sim[("bh", ID_SCEN_PRINC)]["equity"][camp], "tr": e_tr[camp], "stop": e_stop[camp]})
    return dict(tk=tk, sl=config.slug(tk), nome=nome, valuta=valuta, df=df, i0=i0, sim=sim, prove=prove, ev=ev_df,
                n_stop=len(ev_df), n_rientri=len(con_rientro), quota_piu_caro=quota_piu_caro, scim=scim,
                quota_stop_sotto_media=quota_stop_sotto_media, distanza_mediana=distanza_mediana, p_boot=p_boot, diff_ann=osservata * 252, lo=float(lo) * 252, hi=float(hi) * 252, curva=curva, anni=anni_test)


def main() -> None:
    global META
    config.RISULTATI.mkdir(parents=True, exist_ok=True)
    META = json.loads(config.FILE_METADATI.read_text(encoding="utf-8"))
    cambio_ok = any(r["ticker"] == config.TICKER_CAMBIO and r["stato"] == "ok" for r in META["strumenti"])
    fx_df = s.carica_csv(config.file_dati(config.TICKER_CAMBIO)) if cambio_ok else None
    res = []
    for k, (tk, nome, valuta) in enumerate(config.STRUMENTI):
        r = analizza(k, tk, nome, valuta, META, fx_df)
        if r:
            res.append(r)
            sm = r["sim"]
            print(f"{tk}: {r['anni']:.1f} anni; netto ({NOME_SCEN[ID_SCEN_PRINC]}) bh {sm[('bh', ID_SCEN_PRINC)]['valore_finale_eur']:.0f}, "
                  f"trend {sm[('tr', ID_SCEN_PRINC)]['valore_finale_eur']:.0f}, con stop {sm[(ID_STOP, ID_SCEN_PRINC)]['valore_finale_eur']:.0f}; "
                  f"stop scattati {r['n_stop']}")
    if len(res) < 2:
        raise SystemExit("Troppo pochi strumenti utilizzabili.")

    ph_boot = holm({r["tk"]: r["p_boot"] for r in res})
    ph_sc = {rid: holm({r["tk"]: (int((r["scim"][rid]["finali"][ID_SCEN_PRINC] >= r["sim"][(rid, ID_SCEN_PRINC)]["valore_finale_eur"]).sum()) + 1)
                        / (config.N_SCIMMIE + 1) for r in res if r["scim"][rid]}) for rid in ("tr", ID_STOP)}
    for r in res:
        sl, nome, sm = r["sl"], r["nome"], r["sim"]
        reg(f"{sl}_nome", nome, "testo", "Nome dello strumento", "config.py, STRUMENTI")
        reg(f"{sl}_data_inizio", str(r["df"].index[r["i0"]].date()), "data_lunga", f"Prima seduta del test su {nome} (dopo le 200 sedute della media)", "dati/")
        reg(f"{sl}_anni", r["anni"], "num1", f"Anni del test su {nome}")
        reg(f"{sl}_n_stop", r["n_stop"], "int", f"Stop loss scattati con la soglia del 5%, {nome}", "stop_eventi.csv")
        reg(f"{sl}_quota_stop_sotto_media", r["quota_stop_sotto_media"], "pct0", f"Quota degli acquisti della regola di trend in cui il livello dello stop (95% del prezzo) sta sotto la media a 200 sedute del giorno, {nome}", "02_backtest_e_test.py, operazioni della regola di trend")
        reg(f"{sl}_distanza_mediana", r["distanza_mediana"], "pct1", f"Distanza mediana della chiusura dalla media a 200 sedute nei giorni di acquisto della regola di trend, {nome}", "02_backtest_e_test.py, operazioni della regola di trend")
        reg(f"{sl}_n_rientri", r["n_rientri"], "int", f"Stop seguiti da un rientro nei dati, {nome}", "stop_eventi.csv")
        if r["n_rientri"] > 0:
            reg(f"{sl}_quota_rientro_piu_caro", r["quota_piu_caro"], "pct0", f"Quota degli stop dopo i quali si è rientrati a un prezzo più alto di quello di vendita, {nome}", "stop_eventi.csv")
        reg(f"{sl}_diff_ann", r["diff_ann"], "pp1", f"Differenza media annualizzata del rendimento netto giornaliero tra regola con stop e regola senza stop, in punti percentuali, {nome}", "bootstrap a blocchi")
        reg(f"{sl}_diff_lo", r["lo"], "pp1", f"Estremo inferiore dell'intervallo bootstrap a blocchi della differenza annualizzata, {nome}", "bootstrap a blocchi")
        reg(f"{sl}_diff_hi", r["hi"], "pp1", f"Estremo superiore dell'intervallo bootstrap a blocchi della differenza annualizzata, {nome}", "bootstrap a blocchi")
        reg(f"{sl}_p_boot", r["p_boot"], "pval", f"p-value a due code: differenza media giornaliera tra regola con stop e regola senza stop, {nome}", "bootstrap a blocchi")
        reg(f"{sl}_p_boot_holm", ph_boot[r["tk"]], "pval", f"p-value del bootstrap corretto con il metodo di Holm sugli strumenti, {nome}", "bootstrap a blocchi")
        for rid in REGOLE:
            for sid in NOME_SCEN:
                x = sm[(rid, sid)]
                reg(f"{sl}_{rid}_{sid}_finale", x["valore_finale_eur"], "int", f"Valore finale di {config.CAPITALE:.0f} euro ({rid}, scenario {NOME_SCEN[sid]}), {nome}")
                reg(f"{sl}_{rid}_{sid}_cagr", x["cagr"], "pct1", f"Rendimento annuo composto ({rid}, scenario {NOME_SCEN[sid]}), {nome}")
            x = sm[(rid, ID_SCEN_PRINC)]
            reg(f"{sl}_{rid}_dd", -x["perdita_massima"], "pct0", f"Perdita massima dal picco ({rid}), {nome}")
            reg(f"{sl}_{rid}_quota", x["quota_giorni_investito"], "pct0", f"Quota di sedute investite ({rid}), {nome}")
            if rid != "bh":
                reg(f"{sl}_{rid}_ordini", x["n_acquisti"] + x["n_vendite"], "int", f"Ordini totali, acquisti e vendite ({rid}), {nome}")
                reg(f"{sl}_{rid}_n_stop", x["n_stop"], "int", f"Stop loss scattati ({rid}), {nome}")
                reg(f"{sl}_{rid}_ordini_anno", (x["n_acquisti"] + x["n_vendite"]) / r["anni"], "num1", f"Ordini all'anno ({rid}), {nome}")
            for sid in NOME_SCEN:
                if sid == "lordo":
                    continue
                reg(f"{sl}_{rid}_{sid}_costo", sm[(rid, sid)]["commissioni_eur"], "int", f"Commissioni pagate in tutto ({rid}, scenario {NOME_SCEN[sid]}), {nome}")
                reg(f"{sl}_{rid}_{sid}_tasse", sm[(rid, sid)]["tasse_eur"], "int", f"Tasse pagate in tutto ({rid}, scenario {NOME_SCEN[sid]}), {nome}")
        for rid in ("tr", ID_STOP):
            sc = r["scim"][rid]
            if not sc:
                continue
            fin = sc["finali"]
            for sid in NOME_SCEN:
                if sid == "lordo":
                    continue
                reale = sm[(rid, sid)]["valore_finale_eur"]
                reg(f"{sl}_{rid}_{sid}_sc_mediana", float(np.median(fin[sid])), "int", f"Valore finale mediano delle scimmie ({rid}, scenario {NOME_SCEN[sid]}), {nome}", "scimmie_finali.csv")

flusso/simulazione.pymotore in euro: ordini, commissioni, tassa, minusvalenze, stop loss, scimmie
"""Motore di simulazione in euro di una regola di trend su un singolo strumento, con stop loss facoltativo.

Regole (da dichiarare nell'articolo):
  - segnale: investito se la chiusura sta sopra la media mobile semplice (calcolata sulle chiusure);
  - il segnale a fine seduta t decide l'ordine della seduta t+1, eseguito alla chiusura di t+1 (nessun look-ahead),
    con il cambio della stessa seduta;
  - stop loss: se alla chiusura di t la chiusura è scesa dello `stop` o più sotto il prezzo di acquisto (nella valuta
    dello strumento) e il segnale di trend è ancora vero, si vende alla chiusura di t+1; si rientra solo dopo che il
    segnale è tornato falso e poi vero (se il segnale è già falso, l'uscita è quella del trend, senza blocco);
  - azioni frazionate, si investe tutta la cassa; commissione per ogni ordine = parte fissa + parte proporzionale
    all'importo investito o incassato;
  - la cassa non rende nulla; i dividendi non sono considerati;
  - tassa sulle plusvalenze calcolate in euro (cambio incluso), pagata a ogni vendita come nel regime amministrato;
    la commissione si somma al costo di acquisto e si sottrae dall'incasso; le minusvalenze compensano le plusvalenze
    successive fino al quarto anno seguente;
  - a fine periodo si vende tutto, con commissione e tassa, per confrontare cifre nette.
Non modellati: imposta di bollo, tassa sulle transazioni finanziarie, spread di cambio, scarto tra denaro e lettera.
"""
from pathlib import Path

import numpy as np
import pandas as pd


class Zaino:
    """Minusvalenze compensabili con le plusvalenze successive, fino al 4° anno seguente."""

    def __init__(self, aliquota: float):
        self.aliquota = aliquota
        self.voci: list[list[float]] = []   # [ultimo anno utilizzabile, importo residuo]

    def tassa(self, anno: int, guadagno: float) -> float:
        if guadagno < 0:
            self.voci.append([anno + 4, -guadagno])
            return 0.0
        residuo = guadagno
        for voce in sorted(self.voci, key=lambda v: v[0]):      # prima le più vecchie
            if voce[0] < anno or voce[1] <= 0:
                continue
            usa = min(voce[1], residuo)
            voce[1] -= usa
            residuo -= usa
            if residuo <= 0:
                break
        return self.aliquota * residuo


def carica_csv(percorso: Path) -> pd.DataFrame:
    df = pd.read_csv(percorso, parse_dates=["Date"], index_col="Date").sort_index()
    df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
    df = df[~df.index.duplicated(keep="last")]
    if (df["Close"] <= 0).any() or df["Close"].isna().any():
        raise SystemExit(f"{percorso}: chiusure mancanti o non positive.")
    return df


def prepara(titolo: pd.DataFrame, cambio: pd.DataFrame | None, media: int) -> pd.DataFrame:
    """Chiusure, cambio (dollari per euro; 1 per gli strumenti in euro), media mobile e segnale di trend."""
    df = titolo[["Close"]].astype(float).copy()
    if cambio is None:
        df["fx"] = 1.0
    else:
        fx = cambio["Close"].astype(float)
        df["fx"] = fx.reindex(df.index.union(fx.index)).ffill().reindex(df.index)
        df = df.dropna(subset=["fx"])
    df["sma"] = df["Close"].rolling(media).mean()
    df["segnale"] = (df["Close"] > df["sma"]) & df["sma"].notna()
    return df


def primo_indice(df: pd.DataFrame) -> int:
    """Prima seduta il cui segnale precedente è calcolabile con la media completa."""
    return int(np.flatnonzero(df["sma"].notna().to_numpy())[0]) + 1


def _compra_importo(cassa: float, fissa: float, perc: float) -> tuple[float, float]:
    """Dalla cassa all'importo investito e alla commissione pagata: cassa = investito + fissa + perc*investito."""
    investito = (cassa - fissa) / (1.0 + perc)
    return investito, fissa + perc * investito


def valuta_blocchi(C: np.ndarray, FX: np.ndarray, anni: np.ndarray, acquisti, vendite, capitale: float,
                   fissa: float, perc: float, aliquota: float) -> tuple[float, float, float]:
    """Valore finale, commissioni e tasse di una sequenza di blocchi (acquisto alla chiusura a[j], vendita alla chiusura
    b[j]). È la stessa contabilità di `simula`, senza il registro: serve alle scimmie."""
    cassa = capitale
    zaino = Zaino(aliquota)
    comm_tot = tasse_tot = 0.0
    for a, b in zip(acquisti, vendite):
        investito, comm = _compra_importo(cassa, fissa, perc)
        if investito <= 0:
            continue
        base = cassa
        azioni = investito / (C[a] / FX[a])
        lordo = azioni * C[b] / FX[b]
        comm_v = fissa + perc * lordo
        netto = lordo - comm_v
        tassa = zaino.tassa(int(anni[b]), netto - base)
        comm_tot += comm + comm_v
        tasse_tot += tassa
        cassa = netto - tassa
    return cassa, comm_tot, tasse_tot


def simula(df: pd.DataFrame, i0: int, capitale: float, fissa: float, perc: float, aliquota: float,
           strategia: bool, stop: float | None = None) -> dict:
    """strategia=True: regola di trend (con stop loss se `stop` è un numero); False: buy & hold (compra a i0, vende alla fine)."""
    C, FX = (df[c].to_numpy(float) for c in ("Close", "fx"))
    SEG = df["segnale"].to_numpy(bool)
    date = df.index
    n = len(df)
    cassa, azioni, base = float(capitale), 0.0, 0.0
    entrata_locale = 0.0
    bloccato = False
    zaino = Zaino(aliquota)
    log: list[dict] = []
    tot = {"commissioni": 0.0, "tasse": 0.0}
    equity = np.empty(n - i0)
    in_posizione = np.zeros(n - i0, dtype=bool)

    def compra(i, motivo):
        nonlocal cassa, azioni, base, entrata_locale
        investito, comm = _compra_importo(cassa, fissa, perc)
        if investito <= 0:
            return
        px_eur = C[i] / FX[i]
        azioni = investito / px_eur
        base = cassa
        entrata_locale = C[i]
        tot["commissioni"] += comm
        log.append(dict(i=i, data=str(date[i].date()), lato="acquisto", motivo=motivo, prezzo=C[i], cambio=FX[i],
                        prezzo_eur=px_eur, azioni=azioni, commissione=comm, spesa_eur=cassa,
                        incasso_netto_eur=0.0, plusvalenza_eur=0.0, tassa_eur=0.0))
        cassa = 0.0

    def vendi(i, motivo):
        nonlocal cassa, azioni, base
        px_eur = C[i] / FX[i]
        lordo = azioni * px_eur
        comm = fissa + perc * lordo
        netto = lordo - comm
        guadagno = netto - base
        tassa = zaino.tassa(int(date[i].year), guadagno)
        tot["commissioni"] += comm
        tot["tasse"] += tassa
        log.append(dict(i=i, data=str(date[i].date()), lato="vendita", motivo=motivo, prezzo=C[i], cambio=FX[i],
                        prezzo_eur=px_eur, azioni=azioni, commissione=comm, spesa_eur=0.0,
                        incasso_netto_eur=netto, plusvalenza_eur=guadagno, tassa_eur=tassa))
        cassa = netto - tassa
        azioni, base = 0.0, 0.0

    for i in range(i0, n):
        if strategia:
            sig = bool(SEG[i - 1])
            if bloccato and not sig:
                bloccato = False
            # uno stop che scatta nella stessa seduta in cui il trend chiede comunque di uscire è un'uscita di trend
            colpito = stop is not None and azioni > 0.0 and sig and C[i - 1] <= entrata_locale * (1.0 - stop)
            if colpito:
                vendi(i, "stop loss")
                bloccato = True
            elif sig and not bloccato and azioni == 0.0:
                compra(i, "segnale")
            elif (not sig) and azioni > 0.0:
                vendi(i, "segnale")
        elif i == i0:
            compra(i, "acquisto iniziale")
        equity[i - i0] = cassa + azioni * C[i] / FX[i]
        in_posizione[i - i0] = azioni > 0.0
    if azioni > 0.0:
        vendi(n - 1, "liquidazione finale")

    picco = np.maximum.accumulate(equity)
    anni = (date[-1] - date[i0]).days / 365.25
    finale = cassa
    return {
        "valore_finale_eur": finale,
        "rendimento_totale": finale / capitale - 1,
        "cagr": (finale / capitale) ** (1 / anni) - 1 if finale > 0 else float("nan"),
        "anni": anni,
        "n_acquisti": sum(1 for r in log if r["lato"] == "acquisto"),
        "n_vendite": sum(1 for r in log if r["lato"] == "vendita"),
        "n_stop": sum(1 for r in log if r["motivo"] == "stop loss"),
        "commissioni_eur": tot["commissioni"],
        "tasse_eur": tot["tasse"],
        "quota_giorni_investito": float(in_posizione.mean()),
        "giorni_investito": int(in_posizione.sum()),
        "perdita_massima": float((equity / picco - 1).min()),
        "operazioni": log,
        "equity": equity,
        "in_posizione": in_posizione,
    }


def blocchi(res: dict) -> tuple[np.ndarray, np.ndarray]:
    """Indici di acquisto e di vendita (nelle sedute del dataframe) dei periodi investiti di una simulazione."""
    a = np.array([r["i"] for r in res["operazioni"] if r["lato"] == "acquisto"], dtype=int)
    b = np.array([r["i"] for r in res["operazioni"] if r["lato"] == "vendita"], dtype=int)
    return a, b


def scimmia(k: int, durata_totale: int, n_sedute: int, rng) -> tuple[np.ndarray, np.ndarray]:
    """Una scimmia: k periodi investiti, la cui durata totale (sedute tra acquisto e vendita) è `durata_totale`, piazzati
    a caso nelle `n_sedute` sedute disponibili, senza sovrapposizioni. Restituisce gli indici relativi di acquisto e vendita."""
    if k == 1:
        durate = np.array([durata_totale])
    else:
        tagli = np.sort(rng.choice(durata_totale - 1, size=k - 1, replace=False)) + 1
        durate = np.diff(np.r_[0, tagli, durata_totale])               # k durate, ciascuna almeno 1
    libere = n_sedute - durata_totale - k                              # sedute rimaste fuori dai blocchi (b - a + 1 sedute l'uno)
    if libere < 0:
        raise ValueError("blocchi troppo lunghi per le sedute disponibili")
    tagli_l = np.sort(rng.choice(libere + k, size=k, replace=False)) - np.arange(k)    # k+1 vuoti non negativi
    vuoti = np.diff(np.r_[0, tagli_l, libere])
    inizi = np.cumsum(np.r_[vuoti[0], vuoti[1:k] + durate[:-1] + 1])
    return inizi, inizi + durate

# Si compra a 112 (seduta 262); a 105 (-6,25%) il prezzo sta ancora sopra la media (circa 100): lo stop deve scattare.
sega = np.r_[np.full(260, 100.0), 99.0, 112.0, 112.0, 105.0, 105.0, 105.0, 105.0, 105.0]
ds = s.prepara(pd.DataFrame({"Close": sega}, index=pd.bdate_range("2020-01-01", periods=len(sega), name="Date")), None, 200)
ns = s.simula(ds, s.primo_indice(ds), 1000, 2.0, 0.001, 0.26, True, stop=0.05)
verifica("lo stop scatta su una serie costruita a mano", ns["n_stop"] == 1, ns["n_stop"])
verifica("acquisto a 112 alla seduta 262, vendita per stop a 105 alla seduta 264",
         [(o["i"], o["lato"], o["prezzo"]) for o in ns["operazioni"]] == [(262, "acquisto", 112.0), (264, "vendita", 105.0)],
         [(o["i"], o["lato"], o["prezzo"]) for o in ns["operazioni"]])
Cs = ds["Close"].to_numpy()
n_ok = all(Cs[o["i"] - 1] <= [x for x in ns["operazioni"] if x["lato"] == "acquisto" and x["i"] < o["i"]][-1]["prezzo"] * 0.95 + 1e-9
           for o in ns["operazioni"] if o["motivo"] == "stop loss")
verifica("ogni stop è preceduto da una chiusura a -5% o peggio sul prezzo di acquisto", n_ok)
senza_s = s.simula(ds, s.primo_indice(ds), 1000, 2.0, 0.001, 0.26, True, stop=None)
verifica("senza stop non scatta nessuno stop e si resta dentro", senza_s["n_stop"] == 0 and senza_s["n_vendite"] == 1 and senza_s["operazioni"][1]["motivo"] == "liquidazione finale")

print("3. Nessun look-ahead: cambiare il futuro non cambia gli ordini già dati")
m = n - 400
corto = dfx.iloc[:m].copy()
lungo = dfx.copy()
lungo.iloc[m:, lungo.columns.get_loc("Close")] *= 0.5                # un crollo dopo la seduta m
lungo["sma"] = lungo["Close"].rolling(200).mean()
lungo["segnale"] = (lungo["Close"] > lungo["sma"]) & lungo["sma"].notna()
a = s.simula(dfx, i0, 1000, 2.0, 0.0, 0.26, True, stop=0.05)
b = s.simula(lungo, i0, 1000, 2.0, 0.0, 0.26, True, stop=0.05)
prima_a = [(o["i"], o["lato"]) for o in a["operazioni"] if o["i"] < m]
prima_b = [(o["i"], o["lato"]) for o in b["operazioni"] if o["i"] < m]
verifica(f"gli {len(prima_a)} ordini prima della seduta {m} sono identici", prima_a == prima_b)
verifica("l'equity prima della seduta m è identica", np.allclose(a["equity"][: m - i0], b["equity"][: m - i0]))

print("4. Motore completo contro motore delle scimmie, e proprietà delle scimmie")
for fissa, perc, aliq, stop in [(0.0, 0.0, 0.0, None), (2.0, 0.0, 0.26, 0.05), (0.0, 0.001, 0.26, None), (2.0, 0.001, 0.26, 0.03)]:
    rr = s.simula(dfx, i0, 1000, fissa, perc, aliq, True, stop=stop)
