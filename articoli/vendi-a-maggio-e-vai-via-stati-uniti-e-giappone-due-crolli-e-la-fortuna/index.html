«Vendi a maggio e vai via»: Stati Uniti e Giappone, due crolli e la fortuna

Tenere le azioni solo da novembre ad aprile batte il tenerle sempre, su quattro indici, e il risultato regge se si tolgono gli anni dei grandi crolli?

L'articolo esce a breve sul blog⬇ Scarica il codice (.zip)

Il disegno

  • Dati: S&P 500, Dow Jones, Nasdaq Composite e Nikkei 225.
  • Regola: azioni da novembre ad aprile, liquidità da maggio a ottobre, confrontata con il buy & hold con costi e tasse.
  • Test: bootstrap a blocchi della differenza di Sharpe, scimmie (stagioni a caso), stagioni escluse (1987 e 1998), quota di stagioni.

"""Passo 1 - Scarica da Yahoo Finance la serie giornaliera di tre indici azionari americani e del Nikkei.

Salva, per ogni indice, dati/<ticker>.csv (Date, Open, High, Low, Close, Volume) e dati/metadati.json con la fonte,
la data del download, la versione di yfinance e l'impronta SHA-256 di ogni file. Gli script successivi leggono SOLO
i file salvati: i risultati restano riproducibili anche se Yahoo corregge i dati in futuro.
"""
import hashlib
import json
import time
from datetime import datetime, timezone
from zoneinfo import ZoneInfo

import pandas as pd
import yfinance as yf

import config


def scarica(ticker: str) -> pd.DataFrame:
    for tentativo in range(3):
        grezzo = yf.Ticker(ticker).history(start=config.INIZIO_DOWNLOAD, auto_adjust=False)
        if not grezzo.empty:
            return grezzo
        print(f"{ticker}: nessun dato (tentativo {tentativo + 1})")
        time.sleep(10)
    return pd.DataFrame()


def main() -> None:
    config.DATI.mkdir(parents=True, exist_ok=True)
    ora_borsa = datetime.now(ZoneInfo(config.FUSO_BORSA))
    meta = {"fonte": "Yahoo Finance", "url_fonte": config.URL_FONTE, "libreria": f"yfinance {yf.__version__}",
            "scaricato_il_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"), "indici": []}
    for ticker, nome, valuta in config.INDICI:
        riga = {"ticker": ticker, "nome": nome, "valuta": valuta}
        grezzo = scarica(ticker)
        if grezzo.empty:
            riga["stato"] = "scartato: nessun dato restituito da Yahoo Finance"
            meta["indici"].append(riga)
            continue
        df = grezzo[["Open", "High", "Low", "Close", "Volume"]].copy()
        df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
        df.index.name = "Date"
        df = df[~df.index.duplicated(keep="last")].sort_index().dropna(subset=["Close"])
        df = df[df["Close"] > 0]
        # seduta ancora in corso: barra incompleta, da scartare
        if df.index[-1].date() == ora_borsa.date() and ora_borsa.hour < 18:
            df = df.iloc[:-1]
        file = config.file_dati(ticker)
        df.to_csv(file, float_format="%.6f")
        riga.update({"stato": "ok", "file": file.name, "righe": len(df), "prima_data": str(df.index[0].date()),
                     "ultima_data": str(df.index[-1].date()), "sha256": hashlib.sha256(file.read_bytes()).hexdigest()})
        meta["indici"].append(riga)
        print(f"{ticker}: {len(df)} righe, dal {riga['prima_data']} al {riga['ultima_data']}")
    config.FILE_METADATI.write_text(json.dumps(meta, indent=2, ensure_ascii=False), encoding="utf-8")
    if sum(1 for r in meta["indici"] if r["stato"] == "ok") < 5:
        raise SystemExit("Troppo pochi indici scaricati.")


if __name__ == "__main__":
    main()

"""Passo 2 - Sell in May and go away: strategia novembre-aprile contro buy & hold, calendari alternativi e scimmie.

Legge SOLO i file in dati/ e produce in risultati/:
  metriche.json         ogni numero dell'articolo, con descrizione e origine di calcolo
  stagioni.csv          per ogni indice e anno: rendimento di novembre-aprile, di maggio-ottobre e differenza
  calendari.csv         per ogni indice: risultato dei dodici calendari di sei mesi (quello di Halloween è "novembre")
  scimmie_finali.csv    valore finale netto di ogni scimmia, per indice
  curve.csv             valore giorno per giorno (una seduta su cinque) di strategia e buy & hold

Strategia: si compra alla chiusura dell'ultima seduta di ottobre e si vende alla chiusura dell'ultima seduta di aprile.
Il calendario è noto in anticipo: nessun look-ahead. Scimmia: stesso numero di ordini e stessa durata di ogni periodo
investito, ma ogni periodo è piazzato a caso dentro il suo anno. Tutto passa dallo stesso motore di simulazione
(commissioni, tassa e minusvalenze compresi).
"""
import json
import platform
from datetime import datetime, timezone

import numpy as np
import pandas as pd

import config
import simulazione as s

RIS: dict = {}
SORGENTE = "02_backtest_e_test.py con simulazione.py"
NOMI_MESI = ["gennaio", "febbraio", "marzo", "aprile", "maggio", "giugno", "luglio", "agosto", "settembre",
             "ottobre", "novembre", "dicembre"]


def reg(chiave, valore, formato, descrizione, origine=SORGENTE):
    if isinstance(valore, (np.floating, np.integer)):
        valore = valore.item()
    if isinstance(valore, float) and not np.isfinite(valore):
        raise SystemExit(f"Numero non finito per {chiave}")
    RIS[chiave] = {"valore": valore, "formato": formato, "descrizione": descrizione, "origine": origine}


def slug(ticker: str) -> str:
    return ticker.replace("^", "").replace(".", "_").lower()


def stato_calendario(mese: np.ndarray, mesi) -> np.ndarray:
    """True nella seduta i se si resta investiti dopo la sua chiusura: la seduta successiva cade in uno dei mesi."""
    prossimo = np.r_[mese[1:], 0]
    return np.isin(prossimo, list(mesi))


def segnale(stato: np.ndarray) -> np.ndarray:
    """Il segnale della seduta j decide l'ordine della seduta j+1 (esecuzione alla chiusura): quindi segnale[j] = stato[j+1]."""
    return np.r_[stato[1:], False]


def periodi(stato: np.ndarray) -> tuple[np.ndarray, np.ndarray]:
    x = np.r_[False, stato, False].astype(int)
    d = np.diff(x)
    return np.flatnonzero(d == 1), np.flatnonzero(d == -1)       # inizio e fine (esclusa) di ogni periodo


def holm(p: dict) -> dict:
    ordine = sorted(p, key=p.get)
    m = len(ordine)
    out, massimo = {}, 0.0
    for k, nome in enumerate(ordine):
        massimo = max(massimo, min(1.0, (m - k) * p[nome]))
        out[nome] = massimo
    return out


def scimmia(inizi: np.ndarray, durate: np.ndarray, n: int, rng) -> np.ndarray:
    """Ogni periodo investito è piazzato a caso dentro il suo anno (blocco tra un inizio e il successivo), con almeno
    una seduta in cassa prima del blocco seguente: il numero di ordini e la durata di ogni periodo restano quelli reali."""
    stato = np.zeros(n, dtype=bool)
    fini_blocco = np.r_[inizi[1:], n]
    for ini, fine, d in zip(inizi, fini_blocco, durate):
        massimo = int(fine - d - 1)
        if massimo < ini:                       # periodo che riempie il blocco (ultimo, troncato): nessuna libertà
            massimo = ini
        a = int(rng.integers(ini, massimo + 1))
        stato[a:a + d] = True
    return stato


def analizza(tk: str, nome: str, valuta: str, indice_rng: int) -> dict | None:
    sl = slug(tk)
    meta = [r for r in META["indici"] if r["ticker"] == tk][0]
    if meta["stato"] != "ok":
        return None
    t = s.carica_csv(config.file_dati(tk))
    df = pd.DataFrame({"Open": t["Close"], "Close": t["Close"], "fx": 1.0}).astype(float)
    n = len(df)
    mese = df.index.month.to_numpy()
    stato = stato_calendario(mese, config.MESI_INVESTITI)
    inizi, fini = periodi(stato)
    inizi_ok = inizi[inizi >= 1]
    if len(inizi_ok) == 0:
        return None
    i0 = int(inizi_ok[0])
    df["segnale"] = segnale(stato)
    h = s.simula(df, i0, config.CAPITALE, config.COMMISSIONE, config.ALIQUOTA, config.MODO, True)
    bh = s.simula(df, i0, config.CAPITALE, config.COMMISSIONE, config.ALIQUOTA, config.MODO, False)

    # --- Rendimenti dei due semestri, anno per anno (da chiusura a chiusura) -------------------------------------
    C = df["Close"].to_numpy()
    righe = []
    ini_p = [a for a in inizi if a >= i0]
    fini_p = [b for a, b in zip(inizi, fini) if a >= i0]
    for k in range(len(ini_p) - 1):
        if fini_p[k] >= n or ini_p[k + 1] >= n:
            break
        inv = C[fini_p[k]] / C[ini_p[k]] - 1
        est = C[ini_p[k + 1]] / C[fini_p[k]] - 1
        righe.append({"indice": tk, "anno_inizio": int(df.index[ini_p[k]].year), "inverno": inv, "estate": est, "diff": inv - est})
    st = pd.DataFrame(righe)
    if len(st) < config.STAGIONI_MINIME:
        print(f"{tk}: scartato, solo {len(st)} stagioni complete")
        return None
    d = st["diff"].to_numpy()
    rng = np.random.default_rng([config.SEME, indice_rng])
    boot = rng.choice(d, size=(config.N_BOOTSTRAP, len(d)), replace=True).mean(axis=1)
    alfa = (1 - config.LIVELLO_CONFIDENZA) / 2
    lo, hi = np.quantile(boot, [alfa, 1 - alfa])
    p_diff = (int((boot <= 0).sum()) + 1) / (config.N_BOOTSTRAP + 1)
    # Maberly e Pierce: senza le due stagioni con il crollo del 1987 e la crisi di LTCM del 1998
    escl = st[st["anno_inizio"].isin(config.STAGIONI_ESCLUSE)]
    senza = st[~st["anno_inizio"].isin(config.STAGIONI_ESCLUSE)]["diff"].to_numpy()
    rng2 = np.random.default_rng([config.SEME + 1, indice_rng])
    if len(escl) == len(config.STAGIONI_ESCLUSE) and len(senza) >= config.STAGIONI_MINIME:
        boot2 = rng2.choice(senza, size=(config.N_BOOTSTRAP, len(senza)), replace=True).mean(axis=1)
        p_senza = (int((boot2 <= 0).sum()) + 1) / (config.N_BOOTSTRAP + 1)
        senza_info = dict(media=float(senza.mean()), p=p_senza, n=len(senza), diff_escl=[float(x) for x in escl["diff"]])
    else:
        senza_info = None
    prima = st[st["anno_inizio"] <= config.DATA_PUBBLICAZIONE]["diff"]
    dopo = st[st["anno_inizio"] > config.DATA_PUBBLICAZIONE]["diff"]

    # --- I dodici calendari --------------------------------------------------------------------------------------
    cal = []
    for k in range(12):
        mesi = [((10 + k + j) % 12) + 1 for j in range(6)]        # k=0: novembre-aprile; k=1: dicembre-maggio ...
        sg = segnale(stato_calendario(mese, mesi))
        d2 = df.copy()
        d2["segnale"] = sg
        r = s.simula(d2, i0, config.CAPITALE, config.COMMISSIONE, config.ALIQUOTA, config.MODO, True)
        cal.append({"indice": tk, "k": k, "primo_mese": NOMI_MESI[mesi[0] - 1], "mese_n": mesi[0],
                    "finale": r["valore_finale_eur"], "cagr": r["cagr"]})
    cal = pd.DataFrame(cal)
    rango = int((cal["finale"] > cal.loc[0, "finale"]).sum() + 1)

    # --- Le scimmie ---------------------------------------------------------------------------------------------
    ini_i = np.array(ini_p, dtype=int)
    dur = np.array([b - a for a, b in zip(ini_p, fini_p)], dtype=int)
    dur = np.minimum(dur, n - ini_i)
    finali = np.empty(config.N_SCIMMIE)
    d3 = df.copy()
    for j in range(config.N_SCIMMIE):
        st_m = scimmia(ini_i, dur, n, rng)
        d3["segnale"] = segnale(st_m)
        r = s.simula(d3, i0, config.CAPITALE, config.COMMISSIONE, config.ALIQUOTA, config.MODO, True)
        if (r["n_acquisti"] != h["n_acquisti"] or r["n_vendite"] != h["n_vendite"]
                or abs(r["quota_giorni_investito"] - h["quota_giorni_investito"]) > 1e-9):
            raise SystemExit(f"{tk}: la scimmia {j} non ha la stessa dieta della strategia")
        finali[j] = r["valore_finale_eur"]
    p_sc = (int((finali >= h["valore_finale_eur"]).sum()) + 1) / (config.N_SCIMMIE + 1)

    camp = np.arange(0, n - i0, 5)
    curva = pd.DataFrame({"indice": tk, "data": df.index[i0:][camp], "valore_h": h["equity"][camp], "valore_bh": bh["equity"][camp]})
    return dict(tk=tk, sl=sl, nome=nome, valuta=valuta, df=df, i0=i0, h=h, bh=bh, st=st, lo=lo, hi=hi, p_diff=p_diff,
                prima=prima, dopo=dopo, senza=senza_info, cal=cal, rango=rango, finali=finali, p_sc=p_sc, curva=curva)


def main() -> None:
    global META
    config.RISULTATI.mkdir(parents=True, exist_ok=True)
    META = json.loads(config.FILE_METADATI.read_text(encoding="utf-8"))
    res = []
    for k, (tk, nome, valuta) in enumerate(config.INDICI):
        r = analizza(tk, nome, valuta, k)
        if r:
            res.append(r)
            print(f"{tk}: {len(r['st'])} stagioni, Halloween {r['h']['valore_finale_eur']:.0f} contro buy & hold {r['bh']['valore_finale_eur']:.0f}, "
                  f"rango {r['rango']}/12, p scimmie {r['p_sc']:.3f}")
    if len(res) < 3:
        raise SystemExit("Troppi pochi indici utilizzabili.")

    ph = holm({r["tk"]: r["p_sc"] for r in res})
    for r in res:
        sl, h, bh, st = r["sl"], r["h"], r["bh"], r["st"]
        reg(f"{sl}_nome", r["nome"], "testo", "Nome dell'indice", "config.py, INDICI")
        reg(f"{sl}_valuta", r["valuta"], "testo", "Valuta dell'indice", "config.py, INDICI")
        reg(f"{sl}_data_inizio", str(r["df"].index[r["i0"]].date()), "data_lunga", f"Prima operazione del test su {r['nome']} (ultima seduta di ottobre dopo l'inizio dei dati)", "dati/")
        reg(f"{sl}_anni", h["anni"], "num1", f"Anni del test su {r['nome']}")
        reg(f"{sl}_n_stagioni", len(st), "int", f"Anni completi con novembre-aprile e il maggio-ottobre seguente per {r['nome']}", "stagioni.csv")
        reg(f"{sl}_media_inv", float(st["inverno"].mean()), "pct1", f"Rendimento medio di novembre-aprile, {r['nome']}", "stagioni.csv")
        reg(f"{sl}_media_est", float(st["estate"].mean()), "pct1", f"Rendimento medio di maggio-ottobre, {r['nome']}", "stagioni.csv")
        reg(f"{sl}_diff", float(st["diff"].mean()), "pp1", f"Differenza media tra novembre-aprile e maggio-ottobre, in punti percentuali, {r['nome']}", "stagioni.csv")
        reg(f"{sl}_diff_lo", float(r["lo"]), "pp1", f"Estremo inferiore dell'intervallo bootstrap della differenza, {r['nome']}", "bootstrap sugli anni")
        reg(f"{sl}_diff_hi", float(r["hi"]), "pp1", f"Estremo superiore dell'intervallo bootstrap della differenza, {r['nome']}", "bootstrap sugli anni")
        reg(f"{sl}_p_diff", float(r["p_diff"]), "pval", f"p-value a una coda: quota di ricampionamenti con differenza media non positiva, {r['nome']}", "bootstrap sugli anni")
        reg(f"{sl}_quota_anni", float((st["diff"] > 0).mean()), "pct0", f"Quota di anni in cui novembre-aprile ha reso più del maggio-ottobre seguente, {r['nome']}", "stagioni.csv")
        reg(f"{sl}_finale_h", h["valore_finale_eur"], "int", f"Valore finale netto della strategia novembre-aprile, {r['nome']} (capitale iniziale e valore nella valuta dell'indice)")
        reg(f"{sl}_finale_bh", bh["valore_finale_eur"], "int", f"Valore finale netto del buy & hold, {r['nome']}")
        reg(f"{sl}_cagr_h", h["cagr"], "pct1", f"Rendimento annuo composto netto della strategia novembre-aprile, {r['nome']}")
        reg(f"{sl}_cagr_bh", bh["cagr"], "pct1", f"Rendimento annuo composto netto del buy & hold, {r['nome']}")
        reg(f"{sl}_dd_h", -h["perdita_massima"], "pct0", f"Perdita massima dal picco della strategia novembre-aprile, {r['nome']}")
        reg(f"{sl}_dd_bh", -bh["perdita_massima"], "pct0", f"Perdita massima dal picco del buy & hold, {r['nome']}")
        reg(f"{sl}_quota_dentro", h["quota_giorni_investito"], "pct0", f"Quota di sedute con la strategia investita, {r['nome']}")
        reg(f"{sl}_n_ordini", h["n_acquisti"] + h["n_vendite"], "int", f"Ordini della strategia novembre-aprile (acquisti e vendite), {r['nome']}")
        reg(f"{sl}_tasse_h", h["tasse_eur"], "int", f"Tasse pagate dalla strategia novembre-aprile, {r['nome']}")
        reg(f"{sl}_tasse_bh", bh["tasse_eur"], "int", f"Tasse pagate dal buy & hold (a fine periodo), {r['nome']}")
        reg(f"{sl}_rango", r["rango"], "int", f"Posto di novembre-aprile tra i dodici calendari di sei mesi (uno è il migliore), {r['nome']}", "calendari.csv")
        reg(f"{sl}_sc_mediana", float(np.median(r["finali"])), "int", f"Valore finale mediano delle scimmie, {r['nome']}", "scimmie_finali.csv")
        reg(f"{sl}_sc_sotto", float((r["finali"] < h["valore_finale_eur"]).mean()), "pct0", f"Quota di scimmie che chiudono sotto la strategia, {r['nome']}", "scimmie_finali.csv")
        reg(f"{sl}_p_sc", r["p_sc"], "pval", f"p-value a una coda: quota di scimmie che fanno almeno quanto la strategia, {r['nome']}", "scimmie_finali.csv")
        reg(f"{sl}_p_holm", ph[r["tk"]], "pval", f"p-value delle scimmie corretto con il metodo di Holm sugli indici, {r['nome']}", "scimmie_finali.csv")
        reg(f"{sl}_sc_sopra_bh", float((r["finali"] >= bh["valore_finale_eur"]).mean()), "pct1", f"Quota di scimmie che fanno almeno quanto il buy & hold, {r['nome']}", "scimmie_finali.csv")
        reg(f"{sl}_n_prima", len(r["prima"]), "int", f"Anni fino al {config.DATA_PUBBLICAZIONE} nel test, {r['nome']}", "stagioni.csv")
        reg(f"{sl}_n_dopo", len(r["dopo"]), "int", f"Anni dopo il {config.DATA_PUBBLICAZIONE} nel test, {r['nome']}", "stagioni.csv")
        if len(r["prima"]) >= 5:
            reg(f"{sl}_diff_prima", float(r["prima"].mean()), "pp1", f"Differenza media fino al {config.DATA_PUBBLICAZIONE}, {r['nome']}", "stagioni.csv")
        reg(f"{sl}_diff_dopo", float(r["dopo"].mean()) if len(r["dopo"]) else 0.0, "pp1", f"Differenza media dopo il {config.DATA_PUBBLICAZIONE}, {r['nome']}", "stagioni.csv")

    # --- Senza le due stagioni di Maberly e Pierce ---------------------------------------------------------------------
    con_senza = [r for r in res if r["senza"] is not None]
    for r in con_senza:
        sl = r["sl"]
        reg(f"{sl}_senza_diff", r["senza"]["media"], "pp1", f"Differenza media tra novembre-aprile e maggio-ottobre senza le due stagioni del 1987 e del 1998, in punti percentuali, {r['nome']}", "stagioni.csv")
        reg(f"{sl}_senza_p", r["senza"]["p"], "pval", f"p-value bootstrap della differenza media senza le due stagioni (analisi non prevista tra le ipotesi), {r['nome']}", "bootstrap sugli anni")
        reg(f"{sl}_senza_n", r["senza"]["n"], "int", f"Stagioni rimaste dopo aver tolto le due, {r['nome']}", "stagioni.csv")
        reg(f"{sl}_senza_rapporto", r["senza"]["media"] / float(r["st"]["diff"].mean()), "pct0", f"Differenza senza le due stagioni in rapporto a quella con tutte le stagioni, {r['nome']}", "stagioni.csv")
        reg(f"{sl}_escl_1987", r["senza"]["diff_escl"][0], "pp1", f"Differenza tra novembre-aprile e maggio-ottobre nella stagione del crollo di ottobre 1987, in punti percentuali, {r['nome']}", "stagioni.csv")

"""Motore di simulazione in euro di una regola a medie mobili su un singolo titolo.

Regole (da dichiarare nell'articolo):
  - segnale: investito se la media mobile veloce è sopra la lenta, calcolate sulle chiusure;
  - il segnale a fine seduta t decide l'operazione della seduta t+1 (nessun look-ahead);
  - esecuzione "apertura": prezzo di apertura di t+1, cambio dell'ultima chiusura nota (t);
    esecuzione "chiusura": prezzo di chiusura di t+1, cambio di t+1 (controllo di robustezza);
  - azioni frazionate, si investe tutta la cassa; commissione fissa per ogni ordine, acquisto o vendita;
  - la cassa non rende nulla; i dividendi non sono considerati;
  - tassa sulle plusvalenze calcolate in euro (cambio incluso), pagata a ogni vendita come nel regime
    amministrato; la commissione si somma al costo di acquisto e si sottrae dall'incasso; le minusvalenze
    compensano le plusvalenze successive fino al quarto anno seguente;
  - a fine periodo si vende tutto, con commissione e tassa, per confrontare cifre nette.
Non modellati: imposta di bollo, ritenuta sui dividendi, spread di cambio.
"""
from pathlib import Path

import numpy as np
import pandas as pd


class Zaino:
    """Minusvalenze compensabili con le plusvalenze successive, fino al 4° anno seguente."""

    def __init__(self, aliquota: float):
        self.aliquota = aliquota
        self.voci: list[list[float]] = []   # [ultimo anno utilizzabile, importo residuo]

    def tassa(self, anno: int, guadagno: float) -> float:
        if guadagno < 0:
            self.voci.append([anno + 4, -guadagno])
            return 0.0
        residuo = guadagno
        for voce in sorted(self.voci, key=lambda v: v[0]):      # prima le più vecchie
            if voce[0] < anno or voce[1] <= 0:
                continue
            usa = min(voce[1], residuo)
            voce[1] -= usa
            residuo -= usa
            if residuo <= 0:
                break
        return self.aliquota * residuo


def carica_csv(percorso: Path) -> pd.DataFrame:

    df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
    df = df[~df.index.duplicated(keep="last")]
    if (df["Close"] <= 0).any() or df["Close"].isna().any():
        raise SystemExit(f"{percorso}: chiusure mancanti o non positive.")
    return df


def prepara(titolo: pd.DataFrame, cambio: pd.DataFrame, veloce: int, lenta: int) -> pd.DataFrame:
    df = titolo[["Open", "Close"]].astype(float).copy()
    fx = cambio["Close"].astype(float)
    df["fx"] = fx.reindex(df.index.union(fx.index)).ffill().reindex(df.index)
    df["sma_v"] = df["Close"].rolling(veloce).mean()
    df["sma_l"] = df["Close"].rolling(lenta).mean()
    df["segnale"] = (df["sma_v"] > df["sma_l"]) & df["sma_l"].notna()
    return df


def indice_data(df: pd.DataFrame, data) -> int:
    """Posizione della prima seduta con data uguale o successiva a quella indicata."""
    return int(np.searchsorted(df.index.values, np.datetime64(data)))


def simula(df: pd.DataFrame, i0: int, capitale: float, commissione: float, aliquota: float,
           modo: str, strategia: bool) -> dict:
    """strategia=True: regola a medie mobili; False: buy & hold (compra a i0, vende alla fine)."""
    O, C, FX = (df[c].to_numpy(float) for c in ("Open", "Close", "fx"))
    SEG = df["segnale"].to_numpy(bool)
    date = df.index
    n = len(df)
    cassa, azioni, base = float(capitale), 0.0, 0.0
    zaino = Zaino(aliquota)
    log: list[dict] = []
    tot = {"commissioni": 0.0, "tasse": 0.0}
    equity = np.empty(n - i0)
    in_posizione = np.zeros(n - i0, dtype=bool)

    def compra(i, px_usd, fx, etichetta):
        nonlocal cassa, azioni, base
        investito = cassa - commissione
        if investito <= 0:
            return
        px_eur = px_usd / fx
        spesa = cassa
        azioni = investito / px_eur
        base = spesa
        tot["commissioni"] += commissione
        log.append(dict(data=str(date[i].date()), lato="acquisto", motivo=etichetta, prezzo_usd=px_usd, cambio=fx,
                        prezzo_eur=px_eur, azioni=azioni, commissione=commissione, spesa_eur=spesa,
                        incasso_netto_eur=0.0, plusvalenza_eur=0.0, tassa_eur=0.0))
        cassa = 0.0

    def vendi(i, px_usd, fx, etichetta):
        nonlocal cassa, azioni, base
        px_eur = px_usd / fx
        lordo = azioni * px_eur
        netto = lordo - commissione
        guadagno = netto - base
        tassa = zaino.tassa(int(date[i].year), guadagno)
        tot["commissioni"] += commissione
        tot["tasse"] += tassa
        log.append(dict(data=str(date[i].date()), lato="vendita", motivo=etichetta, prezzo_usd=px_usd, cambio=fx,
                        prezzo_eur=px_eur, azioni=azioni, commissione=commissione, spesa_eur=0.0,
                        incasso_netto_eur=netto, plusvalenza_eur=guadagno, tassa_eur=tassa))
        cassa = netto - tassa
        azioni, base = 0.0, 0.0

    for i in range(i0, n):
        voglio = bool(SEG[i - 1]) if strategia else True
        px, fx = (O[i], FX[i - 1]) if modo == "apertura" else (C[i], FX[i])
        if voglio and azioni == 0.0:
            compra(i, px, fx, "segnale" if strategia else "acquisto iniziale")
        elif (not voglio) and azioni > 0.0:
            vendi(i, px, fx, "segnale")
        equity[i - i0] = cassa + azioni * C[i] / FX[i]
        in_posizione[i - i0] = azioni > 0.0
    if azioni > 0.0:
        vendi(n - 1, C[n - 1], FX[n - 1], "liquidazione finale")

    picco = np.maximum.accumulate(equity)
    anni = (date[-1] - date[i0]).days / 365.25
    finale = cassa
    return {
        "valore_finale_eur": finale,
        "rendimento_totale": finale / capitale - 1,
        "cagr": (finale / capitale) ** (1 / anni) - 1 if finale > 0 else float("nan"),
        "anni": anni,
        "n_acquisti": sum(1 for r in log if r["lato"] == "acquisto"),
        "n_vendite": sum(1 for r in log if r["lato"] == "vendita"),
        "commissioni_eur": tot["commissioni"],
        "tasse_eur": tot["tasse"],
        "quota_giorni_investito": float(in_posizione.mean()),
        "perdita_massima": float((equity / picco - 1).min()),
        "operazioni": log,
        "equity": equity,
        "in_posizione": in_posizione,
    }

"""Controlli del motore di simulazione. Usa SOLO dati inventati: nessun numero qui è un risultato.

Si esegue prima del backtest (lo fa anche il flusso su GitHub): se un controllo fallisce, il flusso si ferma.
  1. conti fatti a mano (guadagno, perdita, cambio, compensazione delle minusvalenze);
  2. confronto con una formula indipendente;
  3. coerenza del registro delle operazioni con i prezzi grezzi.
"""
import numpy as np
import pandas as pd

import simulazione as s

ok = 0


def verifica(nome, condizione, dettaglio=""):
    global ok
    if not condizione:
        raise SystemExit(f"FALLITO: {nome} {dettaglio}")
    ok += 1
    print(f"  ok  {nome}")


def uguali(a, b, tol=1e-9):
    return abs(a - b) <= tol * max(1.0, abs(a), abs(b))


print("1. Conti a mano")
# Guadagno: compro a 100, vendo a 120, cambio 1. Azioni = 499/100 = 4,99; incasso 598,8; netto 597,8;
# plusvalenza 97,8; tassa 25,428; cassa finale 572,372.
piccolo = pd.DataFrame({"Open": [100, 100, 100, 120, 120, 120.0], "Close": [100, 100, 110, 120, 120, 120.0],
                        "fx": 1.0, "segnale": [False, True, True, False, False, False]},
                       index=pd.bdate_range("2022-03-01", periods=6))
r = s.simula(piccolo, 1, 500, 1.0, 0.26, "apertura", True)
verifica("guadagno: cassa finale 572,372", uguali(r["valore_finale_eur"], 572.372), r["valore_finale_eur"])
verifica("guadagno: tassa 25,428", uguali(r["tasse_eur"], 25.428))
verifica("guadagno: 2 commissioni", uguali(r["commissioni_eur"], 2.0))

perdita = piccolo.copy()
perdita["Open"] = [100, 100, 110, 110, 90, 90.0]
perdita["Close"] = [100, 100, 110, 100, 90, 90.0]
r = s.simula(perdita, 1, 500, 1.0, 0.26, "apertura", True)
verifica("perdita: cassa finale = 499/110*90-1", uguali(r["valore_finale_eur"], 499 / 110 * 90 - 1), r["valore_finale_eur"])
verifica("perdita: nessuna tassa", r["tasse_eur"] == 0.0)

cambio = piccolo.copy()
cambio["fx"] = 1.25
r = s.simula(cambio, 1, 500, 1.0, 0.26, "apertura", True)
verifica("cambio costante: stesso risultato in euro", uguali(r["valore_finale_eur"], 572.372))

# Con il cambio che cambia tra acquisto e vendita la plusvalenza in euro cambia: compro con cambio 1,00,
# vendo con cambio 1,25 (dollaro più debole -> meno euro): 4,99 azioni * 120 / 1,25 = 479,04; netto 478,04.
cambio_var = piccolo.copy()
cambio_var["fx"] = [1.0, 1.0, 1.0, 1.0, 1.25, 1.25]   # acquisto all'apertura di i=2 con fx[1]=1,00; vendita i=4 con fx[3]=1,00
r = s.simula(cambio_var, 1, 500, 1.0, 0.26, "apertura", True)
verifica("cambio: ordini con il cambio dell'ultima chiusura nota", uguali(r["valore_finale_eur"], 572.372), r["valore_finale_eur"])
cambio_var["fx"] = [1.0, 1.0, 1.0, 1.25, 1.25, 1.25]  # vendita i=4 con fx[3]=1,25
r = s.simula(cambio_var, 1, 500, 1.0, 0.26, "apertura", True)
netto_atteso = 4.99 * 120 / 1.25 - 1                  # 478,04 euro, sotto i 500 spesi: perdita, nessuna tassa
verifica("cambio: perdita da cambio, nessuna tassa", uguali(r["valore_finale_eur"], netto_atteso) and r["tasse_eur"] == 0.0,
         r["valore_finale_eur"])

z = s.Zaino(0.26)
verifica("zaino: perdita non tassata", z.tassa(2022, -100) == 0.0)
verifica("zaino: guadagno 150 dopo perdita 100 -> tassa su 50", uguali(z.tassa(2023, 150), 13.0))
z = s.Zaino(0.26)
z.tassa(2023, -10)
verifica("zaino: perdita scaduta (2023 -> usabile fino al 2027)", uguali(z.tassa(2028, 100), 26.0))
z = s.Zaino(0.26)
z.tassa(2023, -10)
verifica("zaino: perdita ancora valida nel 2027", uguali(z.tassa(2027, 100), 0.26 * 90))

print("2. Confronto con una formula indipendente (dati inventati, cambio 1, senza costi)")
rng = np.random.default_rng(7)
date = pd.bdate_range("2020-01-01", "2026-10-20")
n = len(date)
chiusura = 100 * np.exp(np.cumsum(rng.normal(0.0004, 0.018, n)))
apertura = np.r_[chiusura[0], chiusura[:-1]] * (1 + rng.normal(0, 0.004, n))
fx_var = 1.1 * np.exp(np.cumsum(rng.normal(0, 0.003, n)))
titolo = pd.DataFrame({"Open": apertura, "Close": chiusura}, index=pd.Index(date, name="Date"))
cambio_df = pd.DataFrame({"Close": fx_var}, index=pd.Index(date, name="Date"))
piatto = pd.DataFrame({"Close": 1.0}, index=cambio_df.index)
df = s.prepara(titolo, piatto, 50, 200)
i0 = s.indice_data(df, df.index[-1] - pd.DateOffset(years=5))
SEG = df["segnale"].to_numpy(bool)
C = df["Close"].to_numpy()
formula = 500 * np.prod([(C[i + 1] / C[i]) ** SEG[i - 1] for i in range(i0, len(df) - 1)])
r = s.simula(df, i0, 500, 0.0, 0.0, "chiusura", True)
verifica("regola, esecuzione alla chiusura, lordo = prodotto dei rendimenti", uguali(r["valore_finale_eur"], formula),
         f"{r['valore_finale_eur']} contro {formula}")
bh = s.simula(df, i0, 500, 0.0, 0.0, "chiusura", False)
verifica("buy & hold lordo = chiusura finale / chiusura iniziale", uguali(bh["valore_finale_eur"], 500 * C[-1] / C[i0]))
bh_a = s.simula(df, i0, 500, 0.0, 0.0, "apertura", False)
verifica("buy & hold, esecuzione all'apertura = ultima chiusura / prima apertura",
         uguali(bh_a["valore_finale_eur"], 500 * C[-1] / df["Open"].iloc[i0]))
con = s.simula(df, i0, 500, 1.0, 0.0, "apertura", True)
verifica("commissioni = 1 euro per ogni ordine", uguali(con["commissioni_eur"], len(con["operazioni"]) * 1.0))
netto = s.simula(df, i0, 500, 1.0, 0.26, "apertura", True)
verifica("con la tassa non si finisce mai sopra il caso senza tassa",
         netto["valore_finale_eur"] <= con["valore_finale_eur"] + 1e-9)
verifica("il registro delle operazioni quadra con la cassa finale",
         uguali(500 - sum(o["spesa_eur"] for o in netto["operazioni"]) +
                sum(o["incasso_netto_eur"] - o["tassa_eur"] for o in netto["operazioni"]), netto["valore_finale_eur"]))

print("3. Il registro delle operazioni contro i prezzi grezzi (ricerca per data, senza gli array del motore)")
dfv = s.prepara(titolo, cambio_df, 50, 200)
i0v = s.indice_data(dfv, dfv.index[-1] - pd.DateOffset(years=5))
rv = s.simula(dfv, i0v, 500, 1.0, 0.26, "apertura", True)
errori = 0
for o in rv["operazioni"]:
    d = pd.Timestamp(o["data"])
    if o["motivo"] == "liquidazione finale":
        px, fx = titolo.loc[d, "Close"], cambio_df.loc[d, "Close"]
    else:
        prec = cambio_df.index[cambio_df.index.get_loc(d) - 1]
        px, fx = titolo.loc[d, "Open"], cambio_df.loc[prec, "Close"]
    if not (uguali(o["prezzo_usd"], px) and uguali(o["cambio"], fx) and uguali(o["prezzo_eur"], px / fx)):
        errori += 1
verifica(f"{len(rv['operazioni'])} operazioni con prezzo e cambio corretti", errori == 0, f"{errori} errori")

print(f"\nTutti i {ok} controlli superati (dati inventati: nessuno di questi numeri è un risultato).")
