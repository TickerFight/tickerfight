"""Passo 1 - Scarica da Yahoo Finance la serie giornaliera dell'S&P 500.

Salva:
  dati/gspc_giornaliero.csv   la serie (Date, Open, High, Low, Close, Volume)
  dati/gspc_metadati.json     da dove arrivano i dati, quando, con quale
                              versione di yfinance, e l'impronta SHA-256 del file

Gli script successivi leggono SOLO il file salvato: i risultati sono quindi
riproducibili anche se Yahoo corregge i dati in futuro.
"""
import hashlib
import json
from datetime import datetime, timezone
from zoneinfo import ZoneInfo

import pandas as pd
import yfinance as yf

import config


def main() -> None:
    config.DATI.mkdir(parents=True, exist_ok=True)

    ticker = yf.Ticker(config.TICKER)
    grezzo = ticker.history(start=config.INIZIO_DOWNLOAD, auto_adjust=False)
    if grezzo.empty:
        raise SystemExit("Yahoo Finance non ha restituito dati.")

    df = grezzo[["Open", "High", "Low", "Close", "Volume"]].copy()
    df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
    df.index.name = "Date"
    righe_scaricate = len(df)

    # Controlli di qualità di base
    df = df[~df.index.duplicated(keep="last")].sort_index()
    df = df.dropna(subset=["Close"])
    if (df["Close"] <= 0).any():
        raise SystemExit("Trovati prezzi di chiusura non positivi.")

    # Se lo script gira durante la seduta, l'ultima barra è incompleta: la scartiamo.
    ora_borsa = datetime.now(ZoneInfo(config.FUSO_BORSA))
    barra_scartata = None
    if df.index[-1].date() == ora_borsa.date() and ora_borsa.hour < 16:
        barra_scartata = str(df.index[-1].date())
        df = df.iloc[:-1]

    df.to_csv(config.FILE_DATI, float_format="%.6f")

    sha = hashlib.sha256(config.FILE_DATI.read_bytes()).hexdigest()
    metadati = {
        "ticker": config.TICKER,
        "descrizione": "S&P 500, indice di prezzo (esclude i dividendi)",
        "fonte": "Yahoo Finance",
        "url_fonte": "https://finance.yahoo.com/quote/%5EGSPC/history/",
        "libreria": f"yfinance {yf.__version__}",
        "scaricato_il_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"),
        "righe_scaricate": righe_scaricate,
        "righe_salvate": len(df),
        "prima_data": str(df.index[0].date()),
        "ultima_data": str(df.index[-1].date()),
        "barra_incompleta_scartata": barra_scartata,
        "colonna_usata": "Close (chiusura dell'indice, non corretta: l'indice non ha dividendi)",
        "sha256_file": sha,
    }
    config.FILE_METADATI.write_text(
        json.dumps(metadati, indent=2, ensure_ascii=False), encoding="utf-8"
    )
    print(json.dumps(metadati, indent=2, ensure_ascii=False))


if __name__ == "__main__":
    main()

"""Passo 2 - La media mobile a 200 sedute sull'S&P 500, un secolo di storia.

Legge SOLO dati/gspc_giornaliero.csv; produce in risultati/: metriche.json, decadi.csv, curve.csv.
"""
import json
import platform
from datetime import datetime, timezone

import numpy as np
import pandas as pd

import config
import funzioni as fx

RIS: dict = {}
SORGENTE = "02_backtest_e_test.py con funzioni.py"
GA = config.GIORNI_ANNO


def reg(chiave, valore, formato, descrizione, origine=SORGENTE):
    if isinstance(valore, (np.floating, np.integer)):
        valore = valore.item()
    if isinstance(valore, float) and not np.isfinite(valore):
        raise SystemExit(f"Numero non finito per {chiave}")
    RIS[chiave] = {"valore": valore, "formato": formato, "descrizione": descrizione, "origine": origine}


def main() -> None:
    config.RISULTATI.mkdir(parents=True, exist_ok=True)
    df = pd.read_csv(config.FILE_DATI, parse_dates=["Date"], index_col="Date").sort_index()
    close = df["Close"].astype(float)
    if (close <= 0).any() or close.isna().any():
        raise SystemExit("Chiusure mancanti o non positive.")
    c = close.to_numpy()
    pos = fx.posizione_sma(c, config.MEDIA)
    rr, bh, p = fx.rendimenti_regola(c, pos, 0.0)
    rc, _, _ = fx.rendimenti_regola(c, pos, config.COSTO)
    date = close.index[1:][~np.isnan(pos[:-1])]
    anni = (len(rr)) / GA

    reg("prima_data", str(close.index[0].date()), "data_lunga", "Prima seduta dei dati", "dati/")
    reg("data_ultima", str(close.index[-1].date()), "data_lunga", "Ultima seduta dei dati", "dati/")
    reg("data_inizio_regola", str(date[0].date()), "data_lunga", "Primo giorno con il segnale della media", "dati/")
    reg("n_giorni", len(rr), "int", "Sedute del campione della regola", "dati/")
    reg("anni_tot", anni, "num1", "Anni di borsa del campione della regola", "dati/")
    reg("media", config.MEDIA, "int", "Lunghezza della media mobile", "config.py")
    reg("costo", config.COSTO, "pct1", "Costo a ogni cambio di posizione", "config.py")
    reg("h2_rapporto", config.H2_RAPPORTO_DD, "pct0", "Ipotesi 2: rapporto massimo tra i drawdown", "config.py")
    reg("h4_ordini", config.H4_ORDINI_ANNO, "int", "Ipotesi 4: operazioni all'anno", "config.py")

    reg("reg_cagr", fx.cagr(rr, GA), "pct2", "Rendimento annuo composto della regola", "funzioni.cagr")
    reg("bh_cagr", fx.cagr(bh, GA), "pct2", "Rendimento annuo composto del buy & hold", "funzioni.cagr")
    reg("reg_sr", fx.sharpe(rr, GA), "num2", "Sharpe della regola", "funzioni.sharpe")
    reg("bh_sr", fx.sharpe(bh, GA), "num2", "Sharpe del buy & hold", "funzioni.sharpe")
    reg("reg_dd", fx.max_drawdown(rr), "pct0", "Massimo drawdown della regola", "funzioni.max_drawdown")
    reg("bh_dd", fx.max_drawdown(bh), "pct0", "Massimo drawdown del buy & hold", "funzioni.max_drawdown")
    reg("rapporto_dd", fx.max_drawdown(rr) / fx.max_drawdown(bh), "pct0", "Drawdown della regola diviso quello del buy & hold", "rapporto")
    reg("quota", float(np.mean(p)), "pct0", "Quota di sedute in cui la regola è investita", "funzioni")
    nc = fx.n_cambi(p)
    reg("n_ordini", nc, "int", "Cambi di posizione della regola", "funzioni.n_cambi")
    reg("ordini_anno", nc / anni, "num1", "Cambi di posizione all'anno", "funzioni")
    reg("reg_sr_costo", fx.sharpe(rc, GA), "num2", "Sharpe della regola con il costo per cambio di posizione", "funzioni.sharpe")
    reg("reg_cagr_costo", fx.cagr(rc, GA), "pct2", "Rendimento annuo composto della regola con il costo", "funzioni.cagr")
    reg("costo_totale_anno", float(config.COSTO * nc / anni), "pct2", "Costo medio annuo dei cambi di posizione", "config.py e funzioni")

    righe = []
    for d0 in config.DECENNI:
        sel = (date.year >= d0) & (date.year < d0 + 10)
        if sel.sum() < 100:
            continue
        righe.append({"decennio": d0, "regola": fx.cagr(rr[sel], GA), "buy_hold": fx.cagr(bh[sel], GA), "giorni": int(sel.sum())})
    dec = pd.DataFrame(righe)
    dec.to_csv(config.RISULTATI / "decadi.csv", index=False, float_format="%.5f")
    vinti = int((dec["regola"] > dec["buy_hold"]).sum())
    reg("n_decenni", len(dec), "int", "Decenni (anche incompleti, con almeno cento sedute) nel campione", "funzioni")
    reg("decenni_vinti", vinti, "int", "Decenni in cui la regola batte il buy & hold per rendimento annuo composto", "funzioni")
    for d0 in (1930, 2000):
        r = dec[dec["decennio"] == d0].iloc[0]
        reg(f"reg_{d0}", float(r["regola"]), "pct1", f"Rendimento annuo composto della regola, decennio {d0}", "funzioni")
        reg(f"bh_{d0}", float(r["buy_hold"]), "pct1", f"Rendimento annuo composto del buy & hold, decennio {d0}", "funzioni")
    pre = date.year < 1990
    reg("anno_pre", 1990, "annonum", "Anno che divide il campione in prima e dopo", "config.py")
    reg("reg_cagr_pre", fx.cagr(rr[pre], GA), "pct1", "Rendimento annuo della regola prima del 1990", "funzioni")
    reg("bh_cagr_pre", fx.cagr(bh[pre], GA), "pct1", "Rendimento annuo del buy & hold prima del 1990", "funzioni")
    reg("reg_cagr_post", fx.cagr(rr[~pre], GA), "pct1", "Rendimento annuo della regola dal 1990", "funzioni")
    reg("bh_cagr_post", fx.cagr(bh[~pre], GA), "pct1", "Rendimento annuo del buy & hold dal 1990", "funzioni")
    pd.DataFrame({"regola": np.cumprod(1 + rr), "buy_hold": np.cumprod(1 + bh)}, index=date).resample("YE").last().to_csv(config.RISULTATI / "curve.csv", float_format="%.5f")

    h = [RIS["reg_cagr"]["valore"] < RIS["bh_cagr"]["valore"],
         RIS["rapporto_dd"]["valore"] <= config.H2_RAPPORTO_DD,
         RIS["reg_sr"]["valore"] > RIS["bh_sr"]["valore"],
         RIS["ordini_anno"]["valore"] > config.H4_ORDINI_ANNO,
         vinti < len(dec) / 2,
         RIS["reg_sr_costo"]["valore"] > RIS["bh_sr"]["valore"]]
    for i, v in enumerate(h, 1):
        reg(f"h{i}_esito", int(bool(v)), "int", f"Ipotesi {i}: 1 se confermata, 0 se smentita", "confronto con config.py")
    reg("n_conf", int(sum(bool(v) for v in h)), "int", "Ipotesi confermate (su 6)", "confronto con config.py")

    meta = {"generato": datetime.now(timezone.utc).isoformat(timespec="seconds"), "python": platform.python_version(),
            "numpy": np.__version__, "pandas": pd.__version__, "ipotesi": config.IPOTESI}
    (config.RISULTATI / "metriche.json").write_text(json.dumps({"meta": meta, "valori": RIS}, ensure_ascii=False, indent=1), encoding="utf-8")
    print(f"Scritte {len(RIS)} metriche.")


if __name__ == "__main__":
    main()

"""Parametri dell'articolo n. 122 - "Media mobile a 200 giorni: un secolo di storia".

Tutti i parametri stanno qui; gli script numerati li leggono da questo file. Le ipotesi sono scritte prima di guardare i risultati di QUESTO articolo.
Attenzione: nell'articolo 82 abbiamo già confrontato lo Sharpe di questa stessa regola con quello del buy & hold (la regola vinceva): la terza ipotesi è quindi dichiarata come già nota.

La regola: si resta investiti nell'S&P 500 (indice di prezzo, dal 1927, senza dividendi) se la chiusura è sopra la media semplice delle ultime 200 chiusure, altrimenti si sta in liquidità
con rendimento zero. Il segnale della chiusura del giorno t vale per il rendimento del giorno t+1. Si confronta con il buy & hold sugli stessi giorni per rendimento annuo composto, Sharpe,
massima perdita dal massimo (drawdown) e numero di operazioni; poi per decennio; poi con un costo per ogni cambio di posizione.

"""Funzioni di calcolo dell'articolo 122: regola a media mobile, metriche, drawdown."""
import numpy as np
import pandas as pd


def posizione_sma(close: np.ndarray, n: int) -> np.ndarray:
    """1 se la chiusura è sopra la media semplice delle ultime n chiusure (inclusa quella di oggi), 0 altrimenti; nan nei primi n-1 giorni."""
    c = pd.Series(close, dtype=float)
    m = c.rolling(n).mean()
    pos = (c > m).astype(float).to_numpy().copy()
    pos[np.isnan(m.to_numpy())] = np.nan
    return pos


def rendimenti_regola(close: np.ndarray, pos: np.ndarray, costo: float) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Rendimenti giornalieri semplici della regola e del buy & hold sugli stessi giorni, e posizioni effettive.
    La posizione decisa alla chiusura di t-1 vale per il rendimento di t; a ogni cambio di posizione si paga `costo`.
    I giorni senza segnale (prime n-1 sedute e il primo giorno dopo) sono scartati."""
    c = np.asarray(close, float)
    r = c[1:] / c[:-1] - 1.0
    p = pos[:-1]
    ok = ~np.isnan(p)
    r, p = r[ok], p[ok]
    cambi = np.abs(np.diff(np.concatenate([[p[0]], p])))
    return p * r - costo * cambi, r, p


def cagr(x: np.ndarray, ga: int) -> float:
    return float(np.exp(np.mean(np.log1p(x)) * ga) - 1.0)


def sharpe(x: np.ndarray, ga: int) -> float:
    s = np.std(x, ddof=1)
    return float(np.mean(x) / s * np.sqrt(ga)) if s > 0 else float("nan")


def max_drawdown(x: np.ndarray) -> float:
    """Massima perdita dal massimo precedente della curva dei rendimenti composti (negativa)."""
    eq = np.cumprod(1.0 + np.asarray(x, float))
    picco = np.maximum.accumulate(eq)
    return float((eq / picco - 1.0).min())


def n_cambi(p: np.ndarray) -> int:
    return int(np.sum(np.abs(np.diff(p)) > 0))

"""Controlli di funzioni.py su dati inventati. Se uno fallisce, l'Action si ferma."""
import numpy as np

import funzioni as f


def ok(nome, cond):
    assert cond, nome
    print(f"ok  {nome}")


c = np.array([10.0, 10, 10, 10, 20, 24, 24, 12, 12, 12])
pos = f.posizione_sma(c, 3)
ok("media su tre giorni: i primi due sono nan", np.isnan(pos[:2]).all())
ok("salto a 20: sopra la media di (10+10+20)/3", pos[4] == 1.0 and pos[3] == 0.0)
rr, bh, p = f.rendimenti_regola(c, pos, 0.0)
# giorni validi: dal giorno 2 (pos[2]) in poi per i rendimenti dei giorni 3..9
ok("lunghezza dei rendimenti validi", len(rr) == 7 and len(bh) == 7)
ok("il rendimento del giorno 5 usa la posizione del giorno 4", abs(rr[2] - (24 / 20 - 1)) < 1e-12)
ok("posizione 0 prima del salto", rr[0] == 0.0 and rr[1] == 0.0)
rc, _, _ = f.rendimenti_regola(c, pos, 0.01)
ok("il costo si paga sul cambio di posizione", abs((rr - rc).sum() - 0.01 * f.n_cambi(p)) < 1e-12)
ok("buy & hold sugli stessi giorni", abs(bh[2] - 0.2) < 1e-12)
ok("drawdown di +10%, -50%", abs(f.max_drawdown(np.array([0.10, -0.50])) - (-0.5)) < 1e-12)
ok("drawdown nullo se sempre su", f.max_drawdown(np.full(10, 0.01)) == 0.0)
ok("CAGR costante", abs(f.cagr(np.full(252, np.log(1.1) / 252 * 1.0), 252) - (np.exp(252 * np.log1p(np.log(1.1) / 252)) - 1)) < 1e-9)
ok("conteggio cambi", f.n_cambi(np.array([0.0, 0, 1, 1, 0])) == 2)
