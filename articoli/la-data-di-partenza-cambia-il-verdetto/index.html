La data di partenza cambia il verdetto

Il verdetto su una regola di trading, qui un filtro di tendenza a 200 sedute sull'S&P 500 contro il buy & hold, dipende dalla data in cui si comincia a guardarla?

L'articolo esce a breve sul blog⬇ Scarica il codice (.zip)

Il disegno

  • Regola: investiti se la chiusura sta sopra la media delle ultime 200 chiusure, altrimenti liquidità a rendimento zero; segnale eseguito alla chiusura successiva; 10 punti base per ogni cambio.
  • Misure: rendimento annuo composto, Sharpe e perdita massima su finestre che partono da ogni mese dal 1928.
  • Confronti: buy & hold sullo stesso indice; due decenni di riferimento (partenze a gennaio 1990 e 2000).

"""Passo 1 - Scarica da Yahoo Finance la serie giornaliera dell'S&P 500.

Salva:
  dati/gspc_giornaliero.csv   la serie (Date, Open, High, Low, Close, Volume)
  dati/gspc_metadati.json     da dove arrivano i dati, quando, con quale
                              versione di yfinance, e l'impronta SHA-256 del file

Gli script successivi leggono SOLO il file salvato: i risultati sono quindi
riproducibili anche se Yahoo corregge i dati in futuro.
"""
import hashlib
import json
from datetime import datetime, timezone
from zoneinfo import ZoneInfo

import pandas as pd
import yfinance as yf

import config


def main() -> None:
    config.DATI.mkdir(parents=True, exist_ok=True)

    ticker = yf.Ticker(config.TICKER)
    grezzo = ticker.history(start=config.INIZIO_DOWNLOAD, auto_adjust=False)
    if grezzo.empty:
        raise SystemExit("Yahoo Finance non ha restituito dati.")

    df = grezzo[["Open", "High", "Low", "Close", "Volume"]].copy()
    df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
    df.index.name = "Date"
    righe_scaricate = len(df)

    # Controlli di qualità di base
    df = df[~df.index.duplicated(keep="last")].sort_index()
    df = df.dropna(subset=["Close"])
    if (df["Close"] <= 0).any():
        raise SystemExit("Trovati prezzi di chiusura non positivi.")

    # Se lo script gira durante la seduta, l'ultima barra è incompleta: la scartiamo.
    ora_borsa = datetime.now(ZoneInfo(config.FUSO_BORSA))
    barra_scartata = None
    if df.index[-1].date() == ora_borsa.date() and ora_borsa.hour < 16:
        barra_scartata = str(df.index[-1].date())
        df = df.iloc[:-1]

    df.to_csv(config.FILE_DATI, float_format="%.6f")

    sha = hashlib.sha256(config.FILE_DATI.read_bytes()).hexdigest()
    metadati = {
        "ticker": config.TICKER,
        "descrizione": "S&P 500, indice di prezzo (esclude i dividendi)",
        "fonte": "Yahoo Finance",
        "url_fonte": "https://finance.yahoo.com/quote/%5EGSPC/history/",
        "libreria": f"yfinance {yf.__version__}",
        "scaricato_il_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"),
        "righe_scaricate": righe_scaricate,
        "righe_salvate": len(df),
        "prima_data": str(df.index[0].date()),
        "ultima_data": str(df.index[-1].date()),
        "barra_incompleta_scartata": barra_scartata,
        "colonna_usata": "Close (chiusura dell'indice, non corretta: l'indice non ha dividendi)",
        "sha256_file": sha,
    }
    config.FILE_METADATI.write_text(
        json.dumps(metadati, indent=2, ensure_ascii=False), encoding="utf-8"
    )
    print(json.dumps(metadati, indent=2, ensure_ascii=False))


if __name__ == "__main__":
    main()
flusso/02_backtest_e_test.pyregola e buy & hold da ogni mese di partenza, finestre di 5, 10 e 20 anni
"""Passo 2 - La data di partenza cambia il verdetto: filtro a 200 sedute contro buy & hold da ogni mese di partenza.

Legge SOLO dati/gspc_giornaliero.csv e produce in risultati/:
  metriche.json     ogni numero dell'articolo, con descrizione e origine
  finestre_5.csv, finestre_10.csv, finestre_20.csv   per ogni mese di partenza: rendimento annuo, Sharpe e perdita massima di regola e buy & hold
  curve.csv         valore di 1 per regola e buy & hold sull'intero periodo (una seduta su cinque)
"""
import json
from datetime import datetime, timezone

import numpy as np
import pandas as pd

import config
import funzioni as f

RIS: dict = {}
SORGENTE = "02_backtest_e_test.py con funzioni.py"
GA = config.GIORNI_ANNO


def reg(chiave, valore, formato, descrizione, origine=SORGENTE):
    if isinstance(valore, (np.floating, np.integer)):
        valore = valore.item()
    if isinstance(valore, float) and not np.isfinite(valore):
        raise SystemExit(f"Numero non finito per {chiave}")
    RIS[chiave] = {"valore": valore, "formato": formato, "descrizione": descrizione, "origine": origine}


def main() -> None:
    config.RISULTATI.mkdir(parents=True, exist_ok=True)
    df = pd.read_csv(config.FILE_DATI, parse_dates=["Date"], index_col="Date").sort_index()
    df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
    df = df[~df.index.duplicated(keep="last")]
    c = df["Close"].astype(float)
    if (c <= 0).any() or c.isna().any():
        raise SystemExit("Chiusure mancanti o non positive.")
    date, close = c.index, c.to_numpy()
    n = len(close)
    strat, bh, pos = f.rendimenti_regola(close, config.MEDIA, config.COSTO_BP / 10000.0)
    prima = config.MEDIA + 2                       # prima posizione con segnale in vigore
    anni_tot = (date[-1] - date[prima]).days / 365.25

    reg("dati_prima_data", str(date[0].date()), "data_lunga", "Prima seduta dei dati", "dati/")
    reg("dati_ultima_data", str(date[-1].date()), "data_lunga", "Ultima seduta dei dati", "dati/")
    reg("partenza_prima", str(date[prima].date()), "data_lunga", "Prima seduta con la regola in vigore", "dati/")
    reg("anni_tot", anni_tot, "num1", "Anni dalla prima seduta con la regola in vigore all'ultima")
    reg("media", config.MEDIA, "int", "Sedute della media mobile", "config.py")
    reg("costo_bp", config.COSTO_BP, "int", "Costo di ogni cambio di posizione in punti base", "config.py")
    reg("giorni_anno", GA, "int", "Sedute in un anno", "config.py")

    # intero periodo
    cs, ss, ds = f.metriche_finestra(strat[prima:], GA)
    cb, sb, db = f.metriche_finestra(bh[prima:], GA)
    reg("tot_cagr_regola", cs, "pct1", "Rendimento annuo composto della regola sull'intero periodo")
    reg("tot_cagr_bh", cb, "pct1", "Rendimento annuo composto del buy & hold sull'intero periodo")
    reg("tot_sharpe_regola", ss, "num2", "Sharpe della regola sull'intero periodo")
    reg("tot_sharpe_bh", sb, "num2", "Sharpe del buy & hold sull'intero periodo")
    reg("tot_dd_regola", ds, "pct0", "Perdita massima della regola sull'intero periodo")
    reg("tot_dd_bh", db, "pct0", "Perdita massima del buy & hold sull'intero periodo")
    reg("quota_investito", float(pos[prima:].mean()), "pct0", "Quota di sedute con la regola investita")
    reg("cambi_anno", float(np.abs(np.diff(pos[prima - 1:])).sum() / anni_tot), "num1", "Cambi di posizione all'anno")

    inizi = f.inizi_mese(date, prima)
    esiti = {}
    for h in config.ORIZZONTI:
        lung = h * GA
        righe = []
        for i in inizi:
            if i + lung >= n:
                break
            a, sa, da = f.metriche_finestra(strat[i + 1:i + 1 + lung], GA)
            b, sb_, db_ = f.metriche_finestra(bh[i + 1:i + 1 + lung], GA)
            righe.append((str(date[i].date()), a, b, sa, sb_, da, db_))
        W = pd.DataFrame(righe, columns=["data", "cagr_r", "cagr_b", "sh_r", "sh_b", "dd_r", "dd_b"])
        W.to_csv(config.RISULTATI / f"finestre_{h}.csv", index=False, float_format="%.6f")
        diff = (W["cagr_r"] - W["cagr_b"]).to_numpy()
        reg(f"h{h}_n", len(W), "int", f"Mesi di partenza con una finestra di {h} anni completa")
        reg(f"h{h}_quota_rend", float((diff > 0).mean()), "pct0", f"Quota di partenze in cui la regola batte il buy & hold sul rendimento annuo, finestre di {h} anni")
        reg(f"h{h}_quota_sharpe", float((W["sh_r"] > W["sh_b"]).mean()), "pct0", f"Quota di partenze in cui la regola ha uno Sharpe più alto, finestre di {h} anni")
        reg(f"h{h}_quota_dd", float((W["dd_r"] < W["dd_b"]).mean()), "pct0", f"Quota di partenze in cui la regola ha una perdita massima minore, finestre di {h} anni")
        reg(f"h{h}_diff_media", float(diff.mean()), "pp1", f"Differenza media di rendimento annuo (regola meno buy & hold), finestre di {h} anni")
        reg(f"h{h}_cagr_r_media", float(W["cagr_r"].mean()), "pct1", f"Rendimento annuo medio della regola, finestre di {h} anni")
        reg(f"h{h}_cagr_b_media", float(W["cagr_b"].mean()), "pct1", f"Rendimento annuo medio del buy & hold, finestre di {h} anni")
        reg(f"h{h}_diff_max", float(diff.max()), "pp1", f"Differenza massima, finestre di {h} anni")
        reg(f"h{h}_diff_min", float(diff.min()), "pp1", f"Differenza minima, finestre di {h} anni")
        reg(f"h{h}_anno_max", int(pd.Timestamp(W["data"][int(diff.argmax())]).year), "annonum", f"Anno di partenza con la differenza massima, finestre di {h} anni")
        reg(f"h{h}_anno_min", int(pd.Timestamp(W["data"][int(diff.argmin())]).year), "annonum", f"Anno di partenza con la differenza minima, finestre di {h} anni")
        reg(f"h{h}_ampiezza", float(diff.max() - diff.min()), "pp1", f"Ampiezza della differenza tra la partenza migliore e la peggiore, finestre di {h} anni")
        reg(f"h{h}_tre_su_tre", float(((diff > 0) & (W["sh_r"] > W["sh_b"]) & (W["dd_r"] < W["dd_b"])).mean()), "pct0", f"Quota di partenze in cui la regola vince su rendimento, Sharpe e perdita massima insieme, finestre di {h} anni")
        reg(f"h{h}_zero_su_tre", float(((diff <= 0) & (W["sh_r"] <= W["sh_b"]) & (W["dd_r"] >= W["dd_b"])).mean()), "pct0", f"Quota di partenze in cui la regola perde su tutte e tre le misure, finestre di {h} anni")
        esiti[h] = dict(W=W, diff=diff)

    # casi di studio: decenni che partono a gennaio di un anno
    Wp = esiti[config.ORIZZONTE_PRINCIPALE]["W"]
    for nome, anno in (("fav", config.ANNO_FAVOREVOLE), ("sfav", config.ANNO_SFAVOREVOLE)):
        r = Wp[Wp["data"].str.startswith(f"{anno}-01")].iloc[0]
        reg(f"caso_{nome}_anno", anno, "annonum", f"Anno di partenza del caso «{nome}»", "config.py")
        reg(f"caso_{nome}_regola", float(r["cagr_r"]), "pct1", f"Rendimento annuo della regola, decennio del caso «{nome}»")
        reg(f"caso_{nome}_bh", float(r["cagr_b"]), "pct1", f"Rendimento annuo del buy & hold, decennio del caso «{nome}»")
        reg(f"caso_{nome}_diff", float(r["cagr_r"] - r["cagr_b"]), "pp1", f"Differenza di rendimento annuo, decennio del caso «{nome}»")
        reg(f"caso_{nome}_dd_regola", float(r["dd_r"]), "pct0", f"Perdita massima della regola, decennio del caso «{nome}»")
        reg(f"caso_{nome}_dd_bh", float(r["dd_b"]), "pct0", f"Perdita massima del buy & hold, decennio del caso «{nome}»")

    P = config.ORIZZONTE_PRINCIPALE
    d10 = esiti[P]["diff"]
    reg("orizzonte", P, "annonum", "Anni della finestra principale", "config.py")
    reg("h1_min", config.H1_MIN, "pct0", "Ipotesi 1: quota minima", "config.py")
    reg("h1_max", config.H1_MAX, "pct0", "Ipotesi 1: quota massima", "config.py")
    reg("h2_quota", config.H2_QUOTA, "pct0", "Ipotesi 2: quota minima sulla perdita massima", "config.py")
    reg("h3_vantaggio", config.H3_VANTAGGIO, "pp1", "Ipotesi 3: vantaggio minimo nel decennio favorevole, punti", "config.py")
    reg("h3_svantaggio", config.H3_SVANTAGGIO, "pp1", "Ipotesi 3: svantaggio minimo nel decennio sfavorevole, punti", "config.py")
    reg("h5_ampiezza", config.H5_AMPIEZZA, "pp1", "Ipotesi 5: ampiezza minima, punti", "config.py")
    h = [config.H1_MIN <= RIS[f"h{P}_quota_rend"]["valore"] <= config.H1_MAX,
         RIS[f"h{P}_quota_dd"]["valore"] >= config.H2_QUOTA,
         RIS["caso_fav_diff"]["valore"] >= config.H3_VANTAGGIO and RIS["caso_sfav_diff"]["valore"] <= -config.H3_SVANTAGGIO,
         RIS[f"h{P}_cagr_r_media"]["valore"] < RIS[f"h{P}_cagr_b_media"]["valore"],
         RIS[f"h{P}_ampiezza"]["valore"] > config.H5_AMPIEZZA,
         RIS["h20_quota_rend"]["valore"] < RIS[f"h{P}_quota_rend"]["valore"]]
    for i, v in enumerate(h, 1):
        reg(f"h{i}_esito", int(v), "int", f"Ipotesi {i}: 1 se confermata, 0 se smentita", "confronto con config.py")
    reg("n_conf", int(sum(h)), "int", "Ipotesi confermate (su sei)", "confronto con config.py")
    reg("run_utc", datetime.now(timezone.utc).isoformat(timespec="seconds"), "testo", "Data di esecuzione dell'analisi")
    json.dump({"valori": RIS}, open(config.RISULTATI / "metriche.json", "w", encoding="utf-8"), indent=1, ensure_ascii=False)

    camp = np.arange(prima, n, 5)
    er, eb = np.cumprod(1.0 + strat[prima:]), np.cumprod(1.0 + bh[prima:])
    pd.DataFrame({"data": date[camp].strftime("%Y-%m-%d"), "regola": er[camp - prima], "buyhold": eb[camp - prima]}).to_csv(
        config.RISULTATI / "curve.csv", index=False, float_format="%.5f")
    print(f"Scritti {len(RIS)} numeri in metriche.json; esiti {h}")


if __name__ == "__main__":
    main()

"""Passo 3 - Grafici dell'articolo. Legge SOLO risultati/ e salva i PNG in figure/.

  fig1_valore.png     valore di 1 per regola e buy & hold, scala logaritmica
  fig2_diff_10anni.png  differenza di rendimento annuo (regola meno buy & hold) per ogni mese di partenza, finestre di dieci anni

import json

import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt  # noqa: E402
import numpy as np  # noqa: E402
import pandas as pd  # noqa: E402

import config  # noqa: E402

BLU, ARANCIO, GRIGIO = "#1f5fa8", "#c9610a", "#8a8a8a"
plt.rcParams.update({"font.size": 12, "axes.spines.top": False, "axes.spines.right": False,
                     "axes.grid": True, "grid.alpha": 0.25, "figure.dpi": 120, "savefig.bbox": "tight",
                     "savefig.facecolor": "white", "legend.frameon": False})

def main() -> None:
    config.FIGURE.mkdir(parents=True, exist_ok=True)
    V = json.loads((R / "metriche.json").read_text(encoding="utf-8"))["valori"]
    c = pd.read_csv(R / "curve.csv", parse_dates=["data"])
    fig, ax = plt.subplots(figsize=(10, 5))
    ax.plot(c["data"], c["buyhold"], color=ARANCIO, lw=1.8, label="Buy & hold")
    ax.plot(c["data"], c["regola"], color=BLU, lw=1.8, label="Filtro a 200 sedute")
    ax.set_yscale("log")
    ax.set_ylabel("Valore di 1 dollaro (scala logaritmica)")

    fig.savefig(config.FIGURE / "fig1_valore.png")
    plt.close(fig)

    diff = (W["cagr_r"] - W["cagr_b"]) * 100
    fig, ax = plt.subplots(figsize=(10, 5))
    ax.fill_between(W["data"], diff, 0, where=diff >= 0, color=BLU, alpha=0.75, label="La regola batte il buy & hold")
    ax.fill_between(W["data"], diff, 0, where=diff < 0, color=ARANCIO, alpha=0.75, label="Il buy & hold batte la regola")
    ax.axhline(0, color="#333", lw=1)
    for anno in (config.ANNO_SFAVOREVOLE, config.ANNO_FAVOREVOLE):
        ax.axvline(pd.Timestamp(f"{anno}-01-01"), color=GRIGIO, ls=":", lw=1.2)
        ax.text(pd.Timestamp(f"{anno}-01-01"), diff.max() * 0.92, str(anno), ha="center", fontsize=11,
                bbox=dict(boxstyle="round,pad=0.2", fc="white", ec="none", alpha=0.85))
    ax.set_xlabel("Mese di partenza della finestra di dieci anni")
    ax.set_ylabel("Differenza di rendimento annuo (punti)")
    ax.legend(loc="lower left", fontsize=10)
    fig.savefig(config.FIGURE / "fig2_diff_10anni.png")
    plt.close(fig)

    fig, ax = plt.subplots(figsize=(10, 5))
    x = np.arange(len(config.ORIZZONTI))
    for k, (chiave, nome, col) in enumerate((("quota_rend", "Rendimento annuo più alto", BLU), ("quota_sharpe", "Sharpe più alto", GRIGIO),
                                              ("quota_dd", "Perdita massima minore", ARANCIO))):
        ax.bar(x + (k - 1) * 0.27, [V[f"h{h}_{chiave}"]["valore"] * 100 for h in config.ORIZZONTI], width=0.27, color=col, label=nome)
    ax.axhline(50, color="#333", lw=1, ls="--")
    ax.set_xticks(x)
    ax.set_xticklabels([f"{h} anni" for h in config.ORIZZONTI])
    ax.set_ylabel("Quota di partenze in cui vince la regola (%)")
    ax.legend(loc="upper left", fontsize=10, ncol=1)
    ax.set_ylim(0, 105)
    fig.savefig(config.FIGURE / "fig3_quote.png")
    plt.close(fig)
    print("Figure salvate.")


if __name__ == "__main__":
    main()
flusso/config.pyparametri e ipotesi, scritti prima dei risultati
"""Parametri dell'articolo n. 13 - "La data di partenza cambia il verdetto".

Tutti i parametri stanno qui; gli script numerati li leggono da questo file. Le ipotesi sono scritte prima di guardare i risultati.

Regola: filtro di tendenza a 200 sedute sull'S&P 500 (indice di prezzo, senza dividendi). Si è investiti se la chiusura sta sopra la media
mobile semplice delle ultime 200 chiusure, altrimenti si sta in liquidità (rendimento zero). Il segnale si osserva a una chiusura e si
esegue alla chiusura SUCCESSIVA (nessun look-ahead); ogni cambio di posizione costa COSTO_BP punti base. Confronto: buy & hold sullo stesso indice.
La domanda non è se la regola funzioni, ma quanto il verdetto dipenda dalla data di partenza: si misura su ogni mese di partenza dal 1928,
con finestre di 5, 10 e 20 anni. Il segnale usa la storia precedente alla partenza, come farebbe chi avvia la regola in quel momento.
"""
from pathlib import Path

BASE = Path(__file__).resolve().parent.parent
DATI = BASE / "dati"
RISULTATI = BASE / "risultati"
FIGURE = BASE / "figure"

TICKER = "^GSPC"
INIZIO_DOWNLOAD = "1927-01-01"
FUSO_BORSA = "America/New_York"
FILE_DATI = DATI / "gspc_giornaliero.csv"
FILE_METADATI = DATI / "gspc_metadati.json"
DESCRIZIONE_DATI = "le chiusure giornaliere dell'indice S&P 500"
URL_FONTE = "https://finance.yahoo.com/quote/%5EGSPC/history/"
NOTA_COLONNA = "Si usano solo le chiusure; i dividendi non sono considerati"
NOMI_ESENTI = ("S&P 500",)

# --- La regola ------------------------------------------------------------------------------------------------
MEDIA = 200                      # sedute della media mobile
COSTO_BP = 10                    # punti base per ogni cambio di posizione
GIORNI_ANNO = 252
ORIZZONTI = (5, 10, 20)          # anni delle finestre
ORIZZONTE_PRINCIPALE = 10
ANNO_FAVOREVOLE = 2000           # decennio che parte a gennaio di quest'anno (caso 1)
ANNO_SFAVOREVOLE = 1990          # decennio che parte a gennaio di quest'anno (caso 2)

# --- Soglie delle ipotesi ---------------------------------------------------------------------------------------

"""Funzioni dell'articolo 13: filtro di tendenza a 200 sedute, metriche su finestre che partono da ogni mese."""
import numpy as np
import pandas as pd


def rendimenti_regola(close: np.ndarray, media: int, costo: float) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
    """Rendimenti giornalieri della regola, del buy & hold e posizione.

    Il giorno k ha il rendimento close[k]/close[k-1]-1. La posizione del giorno k è il segnale osservato alla chiusura k-2 (chiusura sopra la media
    mobile delle ultime `media` chiusure), eseguito alla chiusura k-1. Ogni cambio di posizione costa `costo` (frazione), sottratto al rendimento del giorno k.
    Gli indici 0 e 1 non hanno posizione (zero) e vanno scartati da chi usa la serie."""
    close = np.asarray(close, dtype=float)
    n = len(close)
    ret = np.zeros(n)
    ret[1:] = close[1:] / close[:-1] - 1.0
    sma = np.full(n, np.nan)
    cs = np.cumsum(np.insert(close, 0, 0.0))
    sma[media - 1:] = (cs[media:] - cs[:-media]) / media
    sig = np.zeros(n)
    ok = ~np.isnan(sma)
    sig[ok] = (close[ok] > sma[ok]).astype(float)
    pos = np.zeros(n)
    pos[2:] = sig[:-2]
    cambi = np.zeros(n)
    cambi[1:] = np.abs(pos[1:] - pos[:-1])
    strat = pos * ret - costo * cambi
    return strat, ret, pos


def metriche_finestra(r: np.ndarray, ga: int) -> tuple[float, float, float]:
    """(rendimento annuo composto, Sharpe con tasso zero, perdita massima come numero positivo) di una serie di rendimenti giornalieri."""
    r = np.asarray(r, dtype=float)
    cagr = float(np.prod(1.0 + r) ** (ga / len(r)) - 1.0)
    sd = np.std(r, ddof=1)
    sharpe = float(np.mean(r) / sd * np.sqrt(ga)) if sd > 0 else 0.0
    e = np.cumprod(1.0 + r)
    picco = np.maximum.accumulate(np.concatenate([[1.0], e]))[1:]
    return cagr, sharpe, float(-np.min(e / picco - 1.0))


def inizi_mese(date: pd.DatetimeIndex, prima: int) -> np.ndarray:
    """Posizioni della prima seduta di ogni mese, a partire dalla posizione `prima`."""
    s = pd.Series(np.arange(len(date)), index=date)
    pos = s.groupby([date.year, date.month]).first().to_numpy()
    return pos[pos >= prima]
