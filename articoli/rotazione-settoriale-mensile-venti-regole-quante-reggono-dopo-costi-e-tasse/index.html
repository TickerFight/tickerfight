Rotazione settoriale mensile: venti regole, quante reggono dopo costi e tasse?

Una rotazione mensile tra i nove ETF settoriali americani batte il buy & hold e i nove settori tenuti in parti uguali? E, tra venti regole dichiarate in anticipo, quante reggono ai costi, ai test multipli e al passare del tempo?

Leggi l'articolo sul blog →⬇ Scarica il codice (.zip)

Il disegno

  • Regole: a fine mese si scelgono i K settori con il rendimento più alto delle ultime L sedute (L = 21, 63, 126, 189, 252; K = 1, 2, 3, 4): venti regole, tutte dichiarate prima. Ordine alla chiusura della prima seduta del mese dopo.
  • Condizioni: 10.000 euro, commissione dello 0,1% dell'importo di ogni ordine, tassa del 26% con minusvalenze riportate, azioni frazionate, dollaro convertito in euro.
  • Riferimenti: buy & hold dell'ETF sull'S&P 500; i nove settori tenuti in parti uguali.
  • Test: quante regole battono i riferimenti, al lordo e al netto; classifica delle regole nelle due metà del periodo; bootstrap a blocchi della differenza giornaliera netta rispetto ai nove settori, con correzione di Holm sulle venti regole.

flusso/01_scarica_dati.py
"""Passo 1 - Scarica da Yahoo Finance le serie giornaliere dei nove ETF settoriali, dell'ETF sull'S&P 500 e del cambio euro-dollaro.

Salva dati/<ticker>.csv (Date, Open, High, Low, Close, Volume) e dati/metadati.json con la fonte, la data del download,
la versione di yfinance e l'impronta SHA-256 di ogni file. Gli script successivi leggono SOLO i file salvati: i risultati
restano riproducibili anche se Yahoo corregge i dati in futuro.
"""
import hashlib
import json
import time
from datetime import datetime, timezone
from zoneinfo import ZoneInfo

import pandas as pd
import yfinance as yf

import config


def scarica(ticker: str) -> pd.DataFrame:
    for tentativo in range(3):
        grezzo = yf.Ticker(ticker).history(start=config.INIZIO_DOWNLOAD, auto_adjust=False)
        if not grezzo.empty:
            return grezzo
        print(f"{ticker}: nessun dato (tentativo {tentativo + 1})")
        time.sleep(10)
    return pd.DataFrame()


def main() -> None:
    config.DATI.mkdir(parents=True, exist_ok=True)
    ora_borsa = datetime.now(ZoneInfo(config.FUSO_BORSA))
    meta = {"fonte": "Yahoo Finance", "url_fonte": config.URL_FONTE, "libreria": f"yfinance {yf.__version__}",
            "scaricato_il_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"), "strumenti": []}
    elenco = [(t, n, "USD") for t, n in config.SETTORI] + [(config.BENCHMARK[0], config.BENCHMARK[1], "USD"),
                                                           (config.TICKER_CAMBIO, "cambio euro-dollaro", "USD")]
    for ticker, nome, valuta in elenco:
        riga = {"ticker": ticker, "nome": nome, "valuta": valuta}
        grezzo = scarica(ticker)
        if grezzo.empty:
            riga["stato"] = "scartato: nessun dato restituito da Yahoo Finance"
            meta["strumenti"].append(riga)
            continue
        df = grezzo[["Open", "High", "Low", "Close", "Volume"]].copy()
        df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
        df.index.name = "Date"
        df = df[~df.index.duplicated(keep="last")].sort_index().dropna(subset=["Close"])
        df = df[df["Close"] > 0]
        if df.index[-1].date() == ora_borsa.date() and ora_borsa.hour < 17:       # seduta in corso: barra incompleta
            df = df.iloc[:-1]
        salti = (df["Close"].pct_change().abs() > 0.25).sum()
        file = config.file_dati(ticker)
        df.to_csv(file, float_format="%.6f")
        riga.update({"stato": "ok", "file": file.name, "righe": len(df), "prima_data": str(df.index[0].date()),
                     "ultima_data": str(df.index[-1].date()), "variazioni_giornaliere_oltre_25_per_cento": int(salti),
                     "sha256": hashlib.sha256(file.read_bytes()).hexdigest()})
        meta["strumenti"].append(riga)
        print(f"{ticker}: {len(df)} righe, dal {riga['prima_data']} al {riga['ultima_data']}, salti oltre il 25%: {salti}")
    config.FILE_METADATI.write_text(json.dumps(meta, indent=2, ensure_ascii=False), encoding="utf-8")
    ok = {r["ticker"] for r in meta["strumenti"] if r["stato"] == "ok"}
    mancanti = [t for t, _ in config.SETTORI] + [config.BENCHMARK[0], config.TICKER_CAMBIO]
    mancanti = [t for t in mancanti if t not in ok]
    if mancanti:
        raise SystemExit(f"Dati mancanti per: {', '.join(mancanti)}. Servono tutti i nove settori, il benchmark e il cambio.")


if __name__ == "__main__":
    main()
flusso/02_backtest_e_test.pyventi regole, riferimenti, metà periodo, bootstrap con correzione di Holm
"""Passo 2 - Rotazione mensile tra nove ETF settoriali: venti regole dichiarate in anticipo.

Legge SOLO i file in dati/ e produce in risultati/:
  metriche.json     ogni numero dell'articolo, con descrizione e origine di calcolo
  prove.csv         per regola e scenario: valore finale, rendimento annuo, ordini, costi, tasse, perdita massima
  meta_periodo.csv  rendimento annuo netto di ogni regola nella prima e nella seconda metà del periodo
  curve.csv         valore (una seduta su cinque) di buy & hold, nove settori e regola centrale

Regole: "bh" = buy & hold dell'ETF sull'S&P 500; "eq" = i nove settori comprati in parti uguali e tenuti;
"r<L>_<K>" = rotazione mensile verso i K settori con il rendimento più alto delle ultime L sedute.
Scenari: "lordo" (nessun costo, nessuna tassa) e "c0" (commissione proporzionale e tassa del 26%).
"""
import json
import platform
from datetime import datetime, timezone

import numpy as np
import pandas as pd

import config
import simulazione as s

RIS: dict = {}
SORGENTE = "02_backtest_e_test.py con simulazione.py"
SCEN = [("lordo", 0.0, 0.0, 0.0)] + [(f"c{k}", f, p, config.ALIQUOTA) for k, (_, f, p) in enumerate(config.SCENARI_COSTO)]
M = f"c{config.SCENARIO_PRINCIPALE}"


def rid_rot(L: int, k: int) -> str:
    return f"r{L}_{k}"


GRIGLIA = [(L, k) for k in config.N_SETTORI for L in config.LOOKBACKS]
VARIANTI = [rid_rot(L, k) for L, k in GRIGLIA]
R_CENTRALE = rid_rot(*config.CENTRALE)
NOME_REGOLA = {"bh": "buy & hold dell'S&P 500", "eq": "nove settori tenuti"}
for L, k in GRIGLIA:
    NOME_REGOLA[rid_rot(L, k)] = f"rotazione, {L} sedute, {k} {'settore' if k == 1 else 'settori'}"


def reg(chiave, valore, formato, descrizione, origine=SORGENTE):
    if isinstance(valore, (np.floating, np.integer)):
        valore = valore.item()
    if isinstance(valore, float) and not np.isfinite(valore):
        raise SystemExit(f"Numero non finito per {chiave}")
    RIS[chiave] = {"valore": valore, "formato": formato, "descrizione": descrizione, "origine": origine}


def holm(p: dict) -> dict:
    ordine = sorted(p, key=p.get)
    m = len(ordine)
    out, massimo = {}, 0.0
    for k, nome in enumerate(ordine):
        massimo = max(massimo, min(1.0, (m - k) * p[nome]))
        out[nome] = massimo
    return out


def bootstrap_blocchi(d: np.ndarray, rng) -> np.ndarray:
    N, L = len(d), config.LUNGHEZZA_BLOCCO
    nb = int(np.ceil(N / L))
    medie = np.empty(config.N_BOOTSTRAP)
    passo = 250
    for a in range(0, config.N_BOOTSTRAP, passo):
        m = min(passo, config.N_BOOTSTRAP - a)
        inizi = rng.integers(0, N, size=(m, nb))
        idx = (inizi[:, :, None] + np.arange(L)[None, None, :]) % N
        medie[a:a + m] = d[idx.reshape(m, -1)[:, :N]].mean(axis=1)
    return medie


def ranghi(x: np.ndarray) -> np.ndarray:
    return np.argsort(np.argsort(-x)) + 1          # 1 = valore più alto


def main() -> None:
    config.RISULTATI.mkdir(parents=True, exist_ok=True)
    meta = json.loads(config.FILE_METADATI.read_text(encoding="utf-8"))
    stato = {r["ticker"]: r["stato"] for r in meta["strumenti"]}
    tickers = [t for t, _ in config.SETTORI] + [config.BENCHMARK[0]]
    for t in tickers + [config.TICKER_CAMBIO]:
        if stato.get(t) != "ok":
            raise SystemExit(f"Dati mancanti per {t}: {stato.get(t)}")
    chiusure = {t: s.carica_csv(config.file_dati(t)) for t in tickers}
    cambio = s.carica_csv(config.file_dati(config.TICKER_CAMBIO))
    date, nomi, C, FX = s.prepara(chiusure, cambio)
    n = len(date)
    anni_idx = date.year.to_numpy()
    col_sett = list(range(len(config.SETTORI)))
    col_spy = nomi.index(config.BENCHMARK[0])
    t_min = max(config.LOOKBACKS)

    piani = {rid_rot(L, k): s.piano_rotazione(C, date, L, k, t_min, col_sett) for L, k in GRIGLIA}
    i0 = min(min(p) for p in piani.values())
    if any(min(p) != i0 for p in piani.values()):
        raise SystemExit("Le regole non partono tutte nella stessa seduta.")
    piani["bh"] = {i0: (col_spy,)}
    piani["eq"] = {i0: tuple(col_sett)}
    anni_test = (date[-1] - date[i0]).days / 365.25
    if anni_test < config.STORIA_MINIMA_ANNI:
        raise SystemExit(f"Solo {anni_test:.1f} anni di test: troppo pochi.")
    REGOLE = ["bh", "eq"] + VARIANTI
    print(f"{n} sedute dal {date[0].date()} al {date[-1].date()}; test dal {date[i0].date()} ({anni_test:.1f} anni)")

    sim = {}
    for rid in REGOLE:
        for sid, f, p, al in SCEN:
            sim[(rid, sid)] = s.simula(C, FX, anni_idx, piani[rid], i0, config.CAPITALE, f, p, al, serie=True)
    cg = {k: s.cagr(v["valore_finale_eur"], config.CAPITALE, date, i0) for k, v in sim.items()}
    prove = [{"regola": rid, "scenario": sid, "finale": r["valore_finale_eur"], "cagr": cg[(rid, sid)],
              "ordini": r["n_acquisti"] + r["n_vendite"], "commissioni": r["commissioni_eur"], "tasse": r["tasse_eur"],
              "perdita_massima": r["perdita_massima"]} for (rid, sid), r in sim.items()]

    # prima e seconda metà (equity netta)
    mezzo_rend = {}
    for rid in REGOLE:
        e = sim[(rid, M)]["equity"]
        m = len(e) // 2
        a1, a2 = m / 252.0, (len(e) - m) / 252.0
        mezzo_rend[rid] = ((e[m - 1] / e[0]) ** (1 / a1) - 1, (e[-1] / e[m - 1]) ** (1 / a2) - 1)
    # bootstrap a blocchi: ogni regola contro i nove settori tenuti, netto
    e_eq = sim[("eq", M)]["equity"]
    pv, boot = {}, {}
    for q, rid in enumerate(VARIANTI):
        e = sim[(rid, M)]["equity"]
        d = (e[1:] / e[:-1] - 1) - (e_eq[1:] / e_eq[:-1] - 1)
        medie = bootstrap_blocchi(d - d.mean(), np.random.default_rng([config.SEME, 2, q]))
        osserv = float(d.mean())
        pv[rid] = (int((np.abs(medie) >= abs(osserv)).sum()) + 1) / (config.N_BOOTSTRAP + 1)
        boot[rid] = osserv * 252
    ph = holm(pv)

    # --- Numeri ------------------------------------------------------------------------------------------------------------
    reg("n_sedute", n, "int", "Sedute comuni a tutti gli strumenti e al cambio", "dati/")
    reg("data_inizio", str(date[i0].date()), "data_lunga", "Prima seduta del test (primo ordine di tutte le regole)", "dati/")
    reg("data_ultima", str(date[-1].date()), "data_lunga", "Ultima seduta presente nei dati", "dati/")
    reg("anni", anni_test, "num1", "Anni del test")
    reg("n_regole", len(VARIANTI), "int", "Regole di rotazione della griglia", "config.py, LOOKBACKS e N_SETTORI")
    reg("capitale", config.CAPITALE, "int", "Capitale iniziale in euro", "config.py, CAPITALE")
    reg("commissione_perc", config.SCENARI_COSTO[config.SCENARIO_PRINCIPALE][2], "pct1", "Commissione proporzionale all'importo di ogni ordine", "config.py, SCENARI_COSTO")
    reg("aliquota", config.ALIQUOTA, "pct0", "Aliquota sulle plusvalenze", "config.py, ALIQUOTA")
    reg("n_bootstrap", config.N_BOOTSTRAP, "int", "Ricampionamenti bootstrap a blocchi", "config.py")
    reg("blocco", config.LUNGHEZZA_BLOCCO, "int", "Sedute per blocco nel bootstrap", "config.py")
    reg("seme", config.SEME, "annonum", "Seme dei sorteggi", "config.py")
    reg("livello_ic", config.LIVELLO_CONFIDENZA, "pct0", "Livello di significatività complementare", "config.py")
    for rid in REGOLE:
        for sid, *_ in SCEN:
            x = sim[(rid, sid)]
            reg(f"{rid}_{sid}_finale", x["valore_finale_eur"], "int", f"Valore finale di {config.CAPITALE:.0f} euro ({NOME_REGOLA[rid]}, scenario {sid})")
            reg(f"{rid}_{sid}_cagr", cg[(rid, sid)], "pct1", f"Rendimento annuo composto ({NOME_REGOLA[rid]}, scenario {sid})")
        x = sim[(rid, M)]
        reg(f"{rid}_dd", -x["perdita_massima"], "pct0", f"Perdita massima dal picco ({NOME_REGOLA[rid]}, netto)")
        reg(f"{rid}_ordini_anno", (x["n_acquisti"] + x["n_vendite"]) / anni_test, "num1", f"Ordini all'anno ({NOME_REGOLA[rid]})")
        reg(f"{rid}_tasse", x["tasse_eur"], "int", f"Tasse pagate in tutto ({NOME_REGOLA[rid]})")
        reg(f"{rid}_costo", x["commissioni_eur"], "int", f"Commissioni pagate in tutto ({NOME_REGOLA[rid]})")
        reg(f"{rid}_meta1", mezzo_rend[rid][0], "pct1", f"Rendimento annuo netto nella prima metà del periodo ({NOME_REGOLA[rid]})", "meta_periodo.csv")
        reg(f"{rid}_meta2", mezzo_rend[rid][1], "pct1", f"Rendimento annuo netto nella seconda metà del periodo ({NOME_REGOLA[rid]})", "meta_periodo.csv")
    for rid in VARIANTI:
        reg(f"{rid}_diff_eq", boot[rid], "pp1", f"Differenza annua netta rispetto ai nove settori tenuti, punti percentuali ({NOME_REGOLA[rid]})", "bootstrap a blocchi")
        reg(f"{rid}_p", pv[rid], "pval", f"p-value (bootstrap a blocchi) della differenza netta contro i nove settori tenuti ({NOME_REGOLA[rid]})", "bootstrap a blocchi")
        reg(f"{rid}_p_holm", ph[rid], "pval", f"p-value corretto con il metodo di Holm sulle venti regole ({NOME_REGOLA[rid]})", "bootstrap a blocchi")
    sv = sorted(VARIANTI)
    c_lordo = np.array([cg[(r, "lordo")] for r in VARIANTI])
    c_netto = np.array([cg[(r, M)] for r in VARIANTI])
    reg("n_ip1", int((c_lordo > cg[("bh", "lordo")]).sum()), "int", "Regole con rendimento annuo lordo sopra il buy & hold dell'S&P 500", "prove.csv")
    reg("n_lordo_sopra_eq", int((c_lordo > cg[("eq", "lordo")]).sum()), "int", "Regole con rendimento annuo lordo sopra i nove settori tenuti", "prove.csv")
    reg("n_netto_sopra_bh", int((c_netto > cg[("bh", M)]).sum()), "int", "Regole con rendimento annuo netto sopra il buy & hold dell'S&P 500", "prove.csv")
    reg("n_ip2", int((c_netto > cg[("eq", M)]).sum()), "int", "Regole con rendimento annuo netto sopra i nove settori tenuti", "prove.csv")
    reg("n_holm_signif", int(sum(ph[r] < 0.05 for r in VARIANTI)), "int", "Regole con p-value corretto con Holm sotto il cinque per cento", "bootstrap a blocchi")
    reg("n_p_signif", int(sum(pv[r] < 0.05 for r in VARIANTI)), "int", "Regole con p-value non corretto sotto il cinque per cento", "bootstrap a blocchi")
    m1 = np.array([mezzo_rend[r][0] for r in VARIANTI])
    m2 = np.array([mezzo_rend[r][1] for r in VARIANTI])
    r1, r2 = ranghi(m1), ranghi(m2)
    ib = int(np.argmin(r1))
    reg("migliore1_nome", NOME_REGOLA[VARIANTI[ib]], "testo", "Regola con il rendimento netto più alto nella prima metà del periodo", "meta_periodo.csv")
    reg("migliore1_cagr1", m1[ib], "pct1", "Rendimento annuo netto della migliore della prima metà, nella prima metà", "meta_periodo.csv")
    reg("migliore1_cagr2", m2[ib], "pct1", "Rendimento annuo netto della migliore della prima metà, nella seconda metà", "meta_periodo.csv")
    reg("migliore1_rango2", int(r2[ib]), "int", "Posto (1 = migliore) che la migliore della prima metà ha nella seconda metà, tra le venti regole", "meta_periodo.csv")
    ib2 = int(np.argmin(r2))
    reg("migliore2_nome", NOME_REGOLA[VARIANTI[ib2]], "testo", "Regola con il rendimento netto più alto nella seconda metà del periodo", "meta_periodo.csv")
    reg("migliore2_rango1", int(r1[ib2]), "int", "Posto che la migliore della seconda metà aveva nella prima metà", "meta_periodo.csv")
    sp = float(np.corrcoef(r1, r2)[0, 1])
    reg("spearman_meta", sp, "num2", "Correlazione tra le classifiche delle venti regole nelle due metà (Spearman)", "meta_periodo.csv")
    ig = int(np.argmax(c_netto))
    reg("migliore_nome", NOME_REGOLA[VARIANTI[ig]], "testo", "Regola con il rendimento annuo netto più alto su tutto il periodo", "prove.csv")
    reg("migliore_cagr", c_netto[ig], "pct1", "Rendimento annuo netto della migliore su tutto il periodo", "prove.csv")
    reg("migliore_p_holm", ph[VARIANTI[ig]], "pval", "p-value corretto della migliore su tutto il periodo", "bootstrap a blocchi")
    iw = int(np.argmin(c_netto))
    reg("peggiore_nome", NOME_REGOLA[VARIANTI[iw]], "testo", "Regola con il rendimento annuo netto più basso", "prove.csv")
    reg("peggiore_cagr", c_netto[iw], "pct1", "Rendimento annuo netto della peggiore su tutto il periodo", "prove.csv")
    reg("cagr_mediana_regole", float(np.median(c_netto)), "pct1", "Mediana dei rendimenti annui netti delle venti regole", "prove.csv")
    reg("attrito_medio", float(np.mean(c_lordo - c_netto)), "pp1", "Punti di rendimento annuo persi in media dal lordo al netto, sulle venti regole", "prove.csv")
    reg("ordini_anno_min", float(min(RIS[f"{r}_ordini_anno"]["valore"] for r in VARIANTI)), "num1", "Ordini all'anno della regola che ne fa meno", "prove.csv")
    reg("ordini_anno_max", float(max(RIS[f"{r}_ordini_anno"]["valore"] for r in VARIANTI)), "num1", "Ordini all'anno della regola che ne fa di più", "prove.csv")
    # per lookback: la regola migliore con k=1 e con k=4
    reg("n_lookback_breve_sopra", int(sum(cg[(rid_rot(21, k), M)] > cg[("eq", M)] for k in config.N_SETTORI)), "int", "Regole con osservazione di un mese sopra i nove settori tenuti, netto", "prove.csv")
    reg("data_scarico", meta["scaricato_il_utc"][:10], "data", "Data del download dei dati", "dati/metadati.json")
    for nome, v in config.SOGLIE.items():
        reg(f"soglia_{nome}", v, "int", f"Soglia dell'ipotesi ({nome})", "config.py, SOGLIE")
    reg("anno_mg", 1999, "annonum", "Anno dell'articolo di Moskowitz e Grinblatt", "Moskowitz e Grinblatt, Journal of Finance")

    pd.DataFrame(prove).to_csv(config.RISULTATI / "prove.csv", index=False, float_format="%.6f")
    pd.DataFrame({"regola": VARIANTI + ["bh", "eq"], "meta1": [mezzo_rend[r][0] for r in VARIANTI + ["bh", "eq"]],
                  "meta2": [mezzo_rend[r][1] for r in VARIANTI + ["bh", "eq"]]}).to_csv(config.RISULTATI / "meta_periodo.csv", index=False, float_format="%.6f")
    cur = pd.DataFrame({"data": date[i0:].astype(str), **{rid: sim[(rid, M)]["equity"] for rid in ("bh", "eq", R_CENTRALE)}}).iloc[::5]
    cur.to_csv(config.RISULTATI / "curve.csv", index=False, float_format="%.2f")
    (config.RISULTATI / "regole.json").write_text(json.dumps([{"id": rid_rot(L, k), "L": L, "k": k} for L, k in GRIGLIA]), encoding="utf-8")

flusso/simulazione.pymotore in euro: ordini, commissioni, tassa 26% con minusvalenze
"""Motore di simulazione in euro di una rotazione mensile tra più ETF (o di un buy & hold), con costi e tasse.

Regole (da dichiarare nell'articolo):
  - le chiusure sono in dollari; il valore in euro di una posizione è azioni * chiusura / cambio (dollari per euro) della stessa seduta;
  - il segnale alla chiusura dell'ultima seduta del mese decide l'ordine della prima seduta del mese dopo, eseguito alla
    chiusura di quella seduta (nessun look-ahead);
  - a ogni ribilanciamento si vendono le posizioni uscite dal gruppo scelto e con la cassa disponibile si comprano, in parti
    uguali, quelle entrate; le posizioni che restano non si toccano;
  - azioni frazionate; commissione per ogni ordine = parte fissa + parte proporzionale all'importo investito o incassato;
  - la cassa non rende nulla; i dividendi non sono considerati (si usano le chiusure non corrette);
  - tassa sulle plusvalenze calcolate in euro (cambio incluso), pagata a ogni vendita come nel regime amministrato; la
    commissione si somma al costo di acquisto e si sottrae dall'incasso; le minusvalenze compensano le plusvalenze
    successive fino al quarto anno seguente;
  - a fine periodo si vende tutto, con commissione e tassa, per confrontare cifre nette.
Non modellati: imposta di bollo, tassa sulle transazioni finanziarie, spread di cambio, scarto tra denaro e lettera.
"""
from pathlib import Path

import numpy as np
import pandas as pd


class Zaino:
    """Minusvalenze compensabili con le plusvalenze successive, fino al 4° anno seguente."""

    def __init__(self, aliquota: float):
        self.aliquota = aliquota
        self.voci: list[list[float]] = []   # [ultimo anno utilizzabile, importo residuo]

    def tassa(self, anno: int, guadagno: float) -> float:
        if guadagno < 0:
            self.voci.append([anno + 4, -guadagno])
            return 0.0
        residuo = guadagno
        for voce in sorted(self.voci, key=lambda v: v[0]):      # prima le più vecchie
            if voce[0] < anno or voce[1] <= 0:
                continue
            usa = min(voce[1], residuo)
            voce[1] -= usa
            residuo -= usa
            if residuo <= 0:
                break
        return self.aliquota * residuo


def carica_csv(percorso: Path) -> pd.DataFrame:
    df = pd.read_csv(percorso, parse_dates=["Date"], index_col="Date").sort_index()
    df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
    df = df[~df.index.duplicated(keep="last")]
    if (df["Close"] <= 0).any() or df["Close"].isna().any():
        raise SystemExit(f"{percorso}: chiusure mancanti o non positive.")
    return df


def prepara(chiusure: dict[str, pd.DataFrame], cambio: pd.DataFrame) -> tuple[pd.DatetimeIndex, list[str], np.ndarray, np.ndarray]:
    """Allinea le chiusure di tutti gli strumenti sulle sedute comuni (con cambio disponibile).
    Restituisce date, elenco dei ticker, matrice delle chiusure (sedute x strumenti) e vettore del cambio."""
    nomi = list(chiusure)
    tab = pd.concat({k: chiusure[k]["Close"].astype(float) for k in nomi}, axis=1).dropna()
    fx = cambio["Close"].astype(float)
    tab["__fx"] = fx.reindex(tab.index.union(fx.index)).ffill().reindex(tab.index)
    tab = tab.dropna()
    return tab.index, nomi, tab[nomi].to_numpy(float), tab["__fx"].to_numpy(float)


def fine_mese(date: pd.DatetimeIndex) -> np.ndarray:
    """Indici delle sedute che sono l'ultima del loro mese di calendario (l'ultima riga dei dati è esclusa)."""
    mese = date.year.to_numpy() * 12 + date.month.to_numpy()
    return np.flatnonzero(mese[1:] != mese[:-1])


def piano_rotazione(C: np.ndarray, date: pd.DatetimeIndex, lookback: int, k: int, t_min: int, colonne: list[int]) -> dict[int, tuple[int, ...]]:
    """Per ogni fine mese t >= t_min: i k strumenti (tra `colonne`) con il rendimento di prezzo più alto nelle ultime
    `lookback` sedute. Restituisce {indice della seduta di esecuzione (t+1): colonne scelte, in ordine di classifica}."""
    piano = {}
    for t in fine_mese(date):
        if t < t_min or t < lookback:
            continue
        rend = C[t, colonne] / C[t - lookback, colonne] - 1.0
        ordine = np.argsort(-rend, kind="stable")[:k]
        piano[int(t) + 1] = tuple(int(colonne[j]) for j in ordine)
    return piano


def _compra_importo(cassa: float, fissa: float, perc: float) -> tuple[float, float]:
    """Dalla cassa all'importo investito e alla commissione pagata: cassa = investito + fissa + perc*investito."""
    investito = (cassa - fissa) / (1.0 + perc)
    return investito, fissa + perc * investito


def simula(C: np.ndarray, FX: np.ndarray, anni: np.ndarray, piano: dict[int, tuple[int, ...]], i0: int, capitale: float,
           fissa: float, perc: float, aliquota: float, serie: bool = True) -> dict:
    """Esegue il piano {seduta: strumenti da tenere}. Alla prima seduta del piano si investe tutta la cassa."""
    n = C.shape[0]
    cassa = float(capitale)
    pos: dict[int, list[float]] = {}          # colonna -> [azioni, costo di acquisto in euro]
    zaino = Zaino(aliquota)
    comm_tot = tasse_tot = 0.0
    n_acq = n_ven = 0
    log: list[dict] = []
    storia: list[tuple[int, float, dict[int, float]]] = []

    def vendi(i, c, motivo):
        nonlocal cassa, comm_tot, tasse_tot, n_ven
        az, base = pos.pop(c)
        lordo = az * C[i, c] / FX[i]
        comm = fissa + perc * lordo
        netto = lordo - comm
        guadagno = netto - base
        tassa = zaino.tassa(int(anni[i]), guadagno)
        comm_tot += comm
        tasse_tot += tassa
        cassa += netto - tassa
        n_ven += 1
        if serie:
            log.append(dict(i=i, lato="vendita", col=c, motivo=motivo, azioni=az, commissione=comm,
                            incasso_netto_eur=netto, plusvalenza_eur=guadagno, tassa_eur=tassa))

    for i in sorted(piano):
        if i < i0:
            continue
        bersaglio = piano[i]
        for c in [c for c in pos if c not in bersaglio]:
            vendi(i, c, "rotazione")
        entranti = [c for c in bersaglio if c not in pos]
        if entranti and cassa > 0:
            quota = cassa / len(entranti)
            for c in entranti:
                investito, comm = _compra_importo(quota, fissa, perc)
                if investito <= 0:
                    continue
                az = investito / (C[i, c] / FX[i])
                pos[c] = [az, quota]
                comm_tot += comm
                n_acq += 1
                cassa -= quota
                if serie:
                    log.append(dict(i=i, lato="acquisto", col=c, motivo="rotazione", azioni=az, commissione=comm,
                                    spesa_eur=quota))
        if serie:
            storia.append((i, cassa, {c: v[0] for c, v in pos.items()}))
    ultimo = n - 1
    for c in list(pos):
        vendi(ultimo, c, "liquidazione finale")
    finale = cassa

    out = {"valore_finale_eur": finale, "rendimento_totale": finale / capitale - 1,
           "n_acquisti": n_acq, "n_vendite": n_ven, "commissioni_eur": comm_tot, "tasse_eur": tasse_tot}
    anni_tot = (ultimo - i0) / 252.0
    out["anni"] = anni_tot
    if not serie:
        return out
    equity = np.empty(n - i0)
    for j, (i, cs, az) in enumerate(storia):
        fine = storia[j + 1][0] if j + 1 < len(storia) else n
        val = np.full(fine - i, cs)
        for c, a in az.items():
            val = val + a * C[i:fine, c] / FX[i:fine]
        equity[i - i0:fine - i0] = val
    in_pos = np.array([bool(az) for (_, _, az) in storia])
    giorni_inv = 0
    for j, (i, cs, az) in enumerate(storia):
        fine = storia[j + 1][0] if j + 1 < len(storia) else n
        if az:
            giorni_inv += fine - i
    picco = np.maximum.accumulate(equity)
    out.update({"equity": equity, "operazioni": log, "perdita_massima": float((equity / picco - 1).min()),
                "quota_giorni_investito": giorni_inv / (n - i0)})
    return out


def cagr(finale: float, capitale: float, date: pd.DatetimeIndex, i0: int) -> float:
    anni = (date[-1] - date[i0]).days / 365.25
    return (finale / capitale) ** (1 / anni) - 1 if finale > 0 else float("nan")


def scimmia_piano(piano: dict[int, tuple[int, ...]], colonne: list[int], rng) -> dict[int, tuple[int, ...]]:
    """Una scimmia: stessi giorni di ordine e stesso numero di strumenti che entrano e escono a ogni ribilanciamento
    della regola, ma strumenti scelti a caso. Così il numero di ordini è identico a quello della regola."""
    out = {}
    prev_reale: set[int] = set()
    prev_m: list[int] = []
    for i in sorted(piano):
        T = set(piano[i])
        c = len(T - prev_reale)
        if not prev_m:
            nuovo = [int(x) for x in rng.choice(colonne, size=len(T), replace=False)]
        else:
            restano = len(prev_m) - c
            tieni = [int(x) for x in rng.choice(prev_m, size=restano, replace=False)] if restano > 0 else []
            altri = [x for x in colonne if x not in prev_m]
            entra = [int(x) for x in rng.choice(altri, size=c, replace=False)] if c > 0 else []
            nuovo = tieni + entra
        out[i] = tuple(nuovo)
        prev_m = nuovo
        prev_reale = T
    return out

r = s.simula(Cp, FXp, Ap, {1: (0,), 3: (1,)}, 1, 1000, 0.0, 0.0, 0.26)
cassa = 1000 * 0.9                       # vendita in perdita a 90: minusvalenza 100, nessuna tassa
fin_lordo = cassa * 130 / 100
tassa = 0.26 * max(0.0, (fin_lordo - cassa) - 100)
verifica("minusvalenza riportata sulla plusvalenza dopo la rotazione", uguali(r["valore_finale_eur"], fin_lordo - tassa), (r["valore_finale_eur"], fin_lordo - tassa))

# Cambio variabile: acquisto con cambio 1,00 e vendita con cambio 1,25 -> perdita da cambio, nessuna tassa.
Cc, _, Ac = mat([[100, 100]] * 3 + [[120, 100]] * 3)
FXc = np.array([1.0, 1.0, 1.0, 1.0, 1.25, 1.25])
r = s.simula(Cc, FXc, Ac, {1: (0,)}, 1, 1000, 2.0, 0.0, 0.26)

Cc2, FXc2, Ac2 = mat([[100, 100]] * 3 + [[120, 100]] * 3, fx=1.25)

verifica("cambio costante: stesso risultato in euro", uguali(r["valore_finale_eur"], 1144.744))

# Tre strumenti in parti uguali: 1000 diviso 3, commissione 2 su ogni acquisto.
C3, FX3, A3 = mat([[100, 50, 10]] * 3 + [[110, 50, 20]])

quota = 1000 / 3

# Rotazione parziale: tengo A e B, cambio C con D. Compra D con il solo ricavato di C.

print("2. Confronto con una formula indipendente (dati inventati, senza costi né tasse)")
rng = np.random.default_rng(7)
date = pd.bdate_range("2005-01-03", "2026-10-20")
n = len(date)
m = 9
ret = rng.normal(0.0003, 0.012, (n, m)) + rng.normal(0, 0.004, (n, 1))
C = 100 * np.exp(np.cumsum(ret, axis=0))
FX = 1.1 * np.exp(np.cumsum(rng.normal(0, 0.003, n)))
A = date.year.to_numpy()
colonne = list(range(m))
fm = s.fine_mese(date)
verifica("fine mese: l'ultima seduta di ogni mese e nessun'altra", all(date[t].month != date[t + 1].month for t in fm) and len(fm) == len(set(zip(date.year, date.month))) - 1)
t_min = 252
piano = s.piano_rotazione(C, date, 126, 1, t_min, colonne)
i0 = min(piano)
verifica("la prima esecuzione è il primo giorno dopo il primo fine mese con almeno 252 sedute", date[i0 - 1].month != date[i0].month and i0 - 1 >= t_min)
ok_arg = True
for i, tgt in piano.items():
    t = i - 1
    rend = C[t] / C[t - 126] - 1
    ok_arg &= tgt[0] == int(np.argmax(rend))
verifica("il settore scelto è quello con il rendimento più alto delle ultime 126 sedute alla chiusura di t", ok_arg)
piano3 = s.piano_rotazione(C, date, 126, 3, t_min, colonne)
verifica("la regola a tre settori sceglie i tre migliori", all(set(tgt) == set(np.argsort(-(C[i - 1] / C[i - 1 - 126] - 1))[:3]) for i, tgt in piano3.items()))
r = s.simula(C, FX, A, piano, i0, 1000, 0.0, 0.0, 0.0)
chiavi = sorted(piano)
formula = 1000.0
for a, b in zip(chiavi, chiavi[1:] + [n - 1]):
    c = piano[a][0]
    formula *= (C[b, c] / FX[b]) / (C[a, c] / FX[a])
verifica("rotazione a un settore, lordo = prodotto dei rendimenti in euro", uguali(r["valore_finale_eur"], formula), (r["valore_finale_eur"], formula))
r3 = s.simula(C, FX, A, piano3, i0, 1000, 0.0, 0.0, 0.0)
verifica("equity iniziale = capitale e finale coerente con la cassa", uguali(r3["equity"][0], 1000.0) and uguali(r3["equity"][-1], r3["valore_finale_eur"]))
bh = s.simula(C, FX, A, {i0: (0,)}, i0, 1000, 0.0, 0.0, 0.0)
verifica("buy & hold lordo con cambio variabile", uguali(bh["valore_finale_eur"], 1000 * (C[-1, 0] / FX[-1]) / (C[i0, 0] / FX[i0])))
eq = s.simula(C, FX, A, {i0: tuple(colonne)}, i0, 1000, 0.0, 0.0, 0.0)
atteso = sum((1000 / m) * (C[-1, c] / FX[-1]) / (C[i0, c] / FX[i0]) for c in colonne)
verifica("nove strumenti tenuti in parti uguali, lordo", uguali(eq["valore_finale_eur"], atteso))

con = s.simula(C, FX, A, piano3, i0, 1000, 2.0, 0.001, 0.0)
verifica("commissioni = 2 euro + 0,1% dell'importo, ordine per ordine",
         uguali(con["commissioni_eur"], sum(2.0 + 0.001 * (o["azioni"] * C[o["i"], o["col"]] / FX[o["i"]]) for o in con["operazioni"])))
netto = s.simula(C, FX, A, piano3, i0, 1000, 2.0, 0.001, 0.26)
verifica("con la tassa non si finisce mai sopra il caso senza tassa", netto["valore_finale_eur"] <= con["valore_finale_eur"] + 1e-9)
verifica("il registro delle operazioni quadra con la cassa finale",
         uguali(1000 - sum(o.get("spesa_eur", 0.0) for o in netto["operazioni"]) +
                sum(o.get("incasso_netto_eur", 0.0) - o.get("tassa_eur", 0.0) for o in netto["operazioni"]), netto["valore_finale_eur"]))
snella = s.simula(C, FX, A, piano3, i0, 1000, 2.0, 0.001, 0.26, serie=False)
verifica("motore senza serie = motore con serie",
         uguali(snella["valore_finale_eur"], netto["valore_finale_eur"]) and uguali(snella["commissioni_eur"], netto["commissioni_eur"])
         and uguali(snella["tasse_eur"], netto["tasse_eur"]) and snella["n_acquisti"] == netto["n_acquisti"])

print("3. Nessun look-ahead: cambiare il futuro non cambia gli ordini già dati")
cut = n - 400
C2 = C.copy()
C2[cut:] *= rng.uniform(0.3, 1.7, (n - cut, m))
p2 = s.piano_rotazione(C2, date, 126, 3, t_min, colonne)
prima_a = {i: t for i, t in piano3.items() if i < cut}
prima_b = {i: t for i, t in p2.items() if i < cut}
verifica(f"le {len(prima_a)} decisioni prima della seduta {cut} sono identiche", prima_a == prima_b)
a = s.simula(C, FX, A, piano3, i0, 1000, 2.0, 0.0, 0.26)
b = s.simula(C2, FX, A, p2, i0, 1000, 2.0, 0.0, 0.26)
verifica("l'equity prima della seduta del cambiamento è identica", np.allclose(a["equity"][: cut - i0], b["equity"][: cut - i0]))

print("4. Scimmie: stessi giorni, stesso numero di ordini")
buoni = True
rng2 = np.random.default_rng(3)
for pr in (piano, piano3):
    reale = s.simula(C, FX, A, pr, i0, 1000, 2.0, 0.0, 0.26, serie=False)
    for _ in range(100):

        r_m = s.simula(C, FX, A, sc, i0, 1000, 2.0, 0.0, 0.26, serie=False)
        buoni &= (sorted(sc) == sorted(pr) and all(len(set(sc[i])) == len(pr[i]) for i in pr)
                  and r_m["n_acquisti"] == reale["n_acquisti"] and r_m["n_vendite"] == reale["n_vendite"])
verifica("100 scimmie per regola (1 e 3 settori): stessi giorni di ordine, settori distinti, stesso numero di ordini", buoni)
vari = {tuple(s.scimmia_piano(piano, colonne, rng2)[i0]) for _ in range(200)}
verifica("le scimmie scelgono settori diversi tra loro", len(vari) > 5)
