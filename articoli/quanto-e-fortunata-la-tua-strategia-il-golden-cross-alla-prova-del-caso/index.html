"""Passo 1 - Scarica da Yahoo Finance la serie giornaliera dell'S&P 500.

Salva:
  dati/gspc_giornaliero.csv   la serie (Date, Open, High, Low, Close, Volume)
  dati/gspc_metadati.json     da dove arrivano i dati, quando, con quale
                              versione di yfinance, e l'impronta SHA-256 del file

Gli script successivi leggono SOLO il file salvato: i risultati sono quindi
riproducibili anche se Yahoo corregge i dati in futuro.
"""
import hashlib
import json
from datetime import datetime, timezone
from zoneinfo import ZoneInfo

import pandas as pd
import yfinance as yf

import config


def main() -> None:
    config.DATI.mkdir(parents=True, exist_ok=True)

    ticker = yf.Ticker(config.TICKER)
    grezzo = ticker.history(start=config.INIZIO_DOWNLOAD, auto_adjust=False)
    if grezzo.empty:
        raise SystemExit("Yahoo Finance non ha restituito dati.")

    df = grezzo[["Open", "High", "Low", "Close", "Volume"]].copy()
    df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
    df.index.name = "Date"
    righe_scaricate = len(df)

    # Controlli di qualità di base
    df = df[~df.index.duplicated(keep="last")].sort_index()
    df = df.dropna(subset=["Close"])
    if (df["Close"] <= 0).any():
        raise SystemExit("Trovati prezzi di chiusura non positivi.")

    # Se lo script gira durante la seduta, l'ultima barra è incompleta: la scartiamo.
    ora_borsa = datetime.now(ZoneInfo(config.FUSO_BORSA))
    barra_scartata = None
    if df.index[-1].date() == ora_borsa.date() and ora_borsa.hour < 16:
        barra_scartata = str(df.index[-1].date())
        df = df.iloc[:-1]

    df.to_csv(config.FILE_DATI, float_format="%.6f")

    sha = hashlib.sha256(config.FILE_DATI.read_bytes()).hexdigest()
    metadati = {
        "ticker": config.TICKER,
        "descrizione": "S&P 500, indice di prezzo (esclude i dividendi)",
        "fonte": "Yahoo Finance",
        "url_fonte": "https://finance.yahoo.com/quote/%5EGSPC/history/",
        "libreria": f"yfinance {yf.__version__}",
        "scaricato_il_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"),
        "righe_scaricate": righe_scaricate,
        "righe_salvate": len(df),
        "prima_data": str(df.index[0].date()),
        "ultima_data": str(df.index[-1].date()),
        "barra_incompleta_scartata": barra_scartata,
        "colonna_usata": "Close (chiusura dell'indice, non corretta: l'indice non ha dividendi)",
        "sha256_file": sha,
    }
    config.FILE_METADATI.write_text(
        json.dumps(metadati, indent=2, ensure_ascii=False), encoding="utf-8"
    )
    print(json.dumps(metadati, indent=2, ensure_ascii=False))


if __name__ == "__main__":
    main()

"""Passo 2 - Backtest della regola 50/200 e test sulla fortuna.

Legge SOLO dati/gspc_giornaliero.csv (scaricato dal passo 1) e produce:
  risultati/metriche.json                    ogni numero dell'articolo, con
                                             descrizione e origine di calcolo
  risultati/curve.csv                        equity curve e drawdown giornalieri
  risultati/test_casuale_periodo_intero.csv  risultati delle strategie a timing casuale
  risultati/bootstrap_diff_sharpe.csv        differenze di Sharpe ricampionate
  risultati/sottoperiodi.csv                 tabella dei sotto-periodi

Idea del test. La regola 50/200 è "investito" oppure "in cash". Per sapere se il
suo risultato è merito o caso la confrontiamo con strategie che hanno la STESSA
esposizione (stessi giorni investiti, stesso numero di cambi di posizione) ma
timing casuale: si rimescolano le durate dei periodi investiti e dei periodi in cash.
Se la regola vera batte quasi tutte le versioni a timing casuale, il suo timing
contiene informazione; altrimenti il risultato è compatibile con la fortuna.
"""
import hashlib
import json
import platform
from datetime import datetime, timezone

import numpy as np
import pandas as pd

import config

RIS: dict = {}
ORIGINE_DATI = "dati/gspc_giornaliero.csv (Yahoo Finance, ticker ^GSPC)"
RF_GIORNALIERO = config.TASSO_SENZA_RISCHIO / config.GIORNI_ANNO_BORSA
RAD = np.sqrt(config.GIORNI_ANNO_BORSA)


def mille(n: int) -> str:
    """Numero intero con il punto come separatore delle migliaia (10000 -> 10.000)."""
    return f"{n:,}".replace(",", ".")


def reg(chiave, valore, formato, descrizione, origine):
    """Registra un numero con la sua giustificazione (finisce nelle note dell'articolo)."""
    if isinstance(valore, np.floating):
        valore = float(valore)
    elif isinstance(valore, np.integer):
        valore = int(valore)
    RIS[chiave] = {
        "valore": valore,
        "formato": formato,
        "descrizione": descrizione,
        "origine": origine,
    }


# ---------------------------------------------------------------------------
# Dati e strategia
# ---------------------------------------------------------------------------
def carica_dati() -> pd.Series:
    df = pd.read_csv(config.FILE_DATI, parse_dates=["Date"]).set_index("Date").sort_index()
    if df["Close"].isna().any() or (df["Close"] <= 0).any():
        raise SystemExit("Il file dati contiene chiusure mancanti o non positive.")
    return df["Close"].astype(float)


def carica_metadati() -> dict:
    if config.FILE_METADATI.exists():
        return json.loads(config.FILE_METADATI.read_text(encoding="utf-8"))
    # Caso "file CSV scaricato a mano dal sito di Yahoo Finance": ricostruiamo il minimo.
    sha = hashlib.sha256(config.FILE_DATI.read_bytes()).hexdigest()
    return {
        "fonte": "Yahoo Finance (file CSV fornito manualmente)",
        "libreria": "download manuale dal sito, nessuna libreria",
        "scaricato_il_utc": "non registrato",
        "sha256_file": sha,
    }


def costruisci_serie(close: pd.Series) -> pd.DataFrame:
    """Segnale a fine giornata t -> posizione tenuta nella giornata t+1.

    Il segnale usa solo informazioni disponibili alla chiusura di t (media a 50 e
    a 200 giorni fino a t incluso); la posizione viene applicata al rendimento da
    t a t+1 (shift di un giorno), per non usare informazioni dal futuro.
    """
    sma_v = close.rolling(config.SMA_VELOCE).mean()
    sma_l = close.rolling(config.SMA_LENTA).mean()
    segnale = (sma_v > sma_l).astype(float).where(sma_l.notna())
    pos = segnale.shift(1)
    ret = close.pct_change()
    df = pd.DataFrame(
        {"close": close, "sma_veloce": sma_v, "sma_lenta": sma_l,
         "posizione": pos, "rend_bh": ret}
    ).dropna(subset=["posizione", "rend_bh"])
    df["rend_strategia"] = df["posizione"] * df["rend_bh"] + (1 - df["posizione"]) * RF_GIORNALIERO
    return df


# ---------------------------------------------------------------------------
# Metriche
# ---------------------------------------------------------------------------
def max_drawdown(r: np.ndarray) -> float:
    """Perdita massima dal picco, come numero positivo (0,55 = -55%)."""
    eq = np.concatenate(([1.0], np.cumprod(1.0 + r)))
    picco = np.maximum.accumulate(eq)
    return float(-(eq / picco - 1.0).min())


def metriche(r: np.ndarray, anni: float) -> dict:
    sd = r.std(ddof=1)
    return {
        "cagr": float(np.exp(np.log1p(r).sum() / anni) - 1.0),
        "vol": float(sd * RAD),
        "sharpe": float((r.mean() - RF_GIORNALIERO) / sd * RAD),
        "maxdd": max_drawdown(r),
        "molt": float(np.prod(1.0 + r)),
    }


def anni_del_periodo(indice_slice, indice_completo) -> float:
    """Anni di calendario coperti dai rendimenti: dal giorno prima del primo rendimento all'ultimo."""
    prima = indice_completo[indice_completo.get_loc(indice_slice[0]) - 1]
    return (indice_slice[-1] - prima).days / config.GIORNI_ANNO_CALENDARIO


# ---------------------------------------------------------------------------
# Test 1: strategie a timing casuale con la stessa esposizione
# ---------------------------------------------------------------------------
def blocchi(pos: np.ndarray):
    cambi = np.flatnonzero(np.diff(pos) != 0) + 1
    inizi = np.concatenate(([0], cambi))
    fini = np.concatenate((cambi, [len(pos)]))
    return pos[inizi].astype(int), (fini - inizi)


def posizione_casuale(valori, lunghezze, rng):
    """Rimescola le durate dei blocchi 'investito' tra loro e quelle 'in cash' tra loro.

    L'alternanza investito/cash resta quella originale, quindi restano identici il
    numero di giorni investiti e il numero di cambi di posizione.
    """
    lun = lunghezze.copy()
    for v in (0, 1):
        m = valori == v
        lun[m] = rng.permutation(lunghezze[m])
    return np.repeat(valori, lun)


def test_timing_casuale(pos, ret, anni, n, rng) -> pd.DataFrame:
    valori, lunghezze = blocchi(pos)
    sh = np.empty(n)
    cg = np.empty(n)
    dd = np.empty(n)
    for i in range(n):
        p = posizione_casuale(valori, lunghezze, rng)
        r = p * ret + (1 - p) * RF_GIORNALIERO
        sd = r.std(ddof=1)
        sh[i] = (r.mean() - RF_GIORNALIERO) / sd * RAD
        cg[i] = np.exp(np.log1p(r).sum() / anni) - 1.0
        dd[i] = max_drawdown(r)
    return pd.DataFrame({"sharpe": sh, "cagr": cg, "maxdd": dd})


def p_value_maggiore(nulla: np.ndarray, osservato: float) -> float:
    """Quota di strategie casuali che fanno almeno quanto quella vera (con correzione +1)."""
    return float((1 + np.sum(nulla >= osservato)) / (len(nulla) + 1))


def p_value_minore(nulla: np.ndarray, osservato: float) -> float:
    return float((1 + np.sum(nulla <= osservato)) / (len(nulla) + 1))


# ---------------------------------------------------------------------------
# Test 2: block bootstrap (circolare) delle differenze di Sharpe
# ---------------------------------------------------------------------------
def bootstrap_sharpe(rs, rb, blocco, n_boot, rng, lotto=250):
    n = len(rs)
    n_blocchi = int(np.ceil(n / blocco))
    out_s, out_b = [], []
    for _ in range(int(np.ceil(n_boot / lotto))):
        quanti = min(lotto, n_boot - len(out_s) * lotto)
        inizi = rng.integers(0, n, size=(quanti, n_blocchi))
        idx = (inizi[:, :, None] + np.arange(blocco)[None, None, :]).reshape(quanti, -1)[:, :n] % n
        s, b = rs[idx], rb[idx]
        out_s.append((s.mean(1) - RF_GIORNALIERO) / s.std(1, ddof=1) * RAD)
        out_b.append((b.mean(1) - RF_GIORNALIERO) / b.std(1, ddof=1) * RAD)
    return np.concatenate(out_s), np.concatenate(out_b)


def intervallo(x: np.ndarray):
    alfa = (1 - config.LIVELLO_CONFIDENZA) / 2
    return float(np.quantile(x, alfa)), float(np.quantile(x, 1 - alfa))

def fig_mosaico_bootstrap(curve, V):
    """Storia vera e due storie alternative del bootstrap, un quadratino per ogni mese (blocco di 21 giorni)."""
    VERDE, ROSSO = "#2f8f5b", "#d03b35"
    rb = curve["eq_bh"].pct_change().dropna().to_numpy()
    rs = curve["eq_strategia"].pct_change().dropna().to_numpy()
    pos = curve["posizione"].iloc[1:].to_numpy()
    date = curve["Date"].iloc[1:].reset_index(drop=True)
    n, L = len(rb), config.BLOCCO_BOOTSTRAP
    sharpe = lambda r: r.mean() / r.std(ddof=1) * np.sqrt(config.GIORNI_ANNO_BORSA)
    d_vera = sharpe(rs) - sharpe(rb)
    rng = np.random.default_rng(config.SEED + 5)
    cand = []
    for _ in range(200):          # 200 ricampionamenti tra cui scegliere due scenari da mostrare
        ini = rng.integers(0, n, size=int(np.ceil(n / L)))
        idx = ((ini[:, None] + np.arange(L)[None, :]).reshape(-1) % n)[:n]
        cand.append((sharpe(rs[idx]) - sharpe(rb[idx]), np.prod(1 + rs[idx]), np.prod(1 + rb[idx]), ini))
    # A: un caso tipico (differenza simile alla vera, regola avanti); B: un caso in cui vince il buy & hold
    A = min([c for c in cand if c[1] > c[2]], key=lambda c: abs(c[0] - d_vera))
    B = min([c for c in cand if c[0] < 0], key=lambda c: abs(c[0] + 0.05))
    nb = int(np.ceil(n / L)); col = 100; righe = int(np.ceil(nb / col))
    dentro = lambda starts: np.array([pos[(s + np.arange(L)) % n].mean() >= 0.5 for s in starts])
    pannelli = [("Storia vera", "i mesi in ordine cronologico", dentro(np.arange(nb) * L),
                 d_vera, np.prod(1 + rs), np.prod(1 + rb)),
                ("Scenario A", "gli stessi mesi, pescati a caso", dentro(A[3]), A[0], A[1], A[2]),
                ("Scenario B", "un altro estratto a caso", dentro(B[3]), B[0], B[1], B[2])]
    fig, assi = plt.subplots(1, 3, figsize=(12.4, 6.6))
    fig.subplots_adjust(left=0.05, right=0.99, top=0.78, bottom=0.17, wspace=0.12)
    for k, (ax, (nome, sotto, c, d, es, eb)) in enumerate(zip(assi, pannelli)):
        griglia = np.full(righe * col, np.nan); griglia[:len(c)] = c.astype(float)
        cmap = matplotlib.colors.ListedColormap([ROSSO, VERDE]); cmap.set_bad(SUP)
        ax.imshow(np.ma.masked_invalid(griglia.reshape(righe, col)), cmap=cmap, vmin=0, vmax=1,
                  aspect="auto", interpolation="nearest")
        for x in np.arange(-.5, col, 1):
            ax.axvline(x, color=SUP, lw=0.5)
        for y in np.arange(-.5, righe, 1):
            ax.axhline(y, color=SUP, lw=0.9)
        ax.grid(False)
        for s in ax.spines.values():
            s.set_visible(False)
        ax.set_xticks([])
        if k == 0:
            ax.set_yticks(range(righe))
            ax.set_yticklabels([str(date[min(r * col * L, n - 1)].year) for r in range(righe)], fontsize=8, color=MUTED)
            ax.tick_params(length=0)
        else:
            ax.set_yticks([])
        ax.set_title(nome, loc="left", fontsize=11.5, fontweight="bold", color=INK, pad=18)
        ax.text(0, 1.015, sotto, transform=ax.transAxes, fontsize=9, color=INK2, va="bottom")
        segno = "+" if d >= 0 else "−"
        ax.text(0, -0.04, f"Regola {it(es, 0)}×  ·  Buy & hold {it(eb, 0)}×\nDifferenza di Sharpe {segno}{it(abs(d))}",
                transform=ax.transAxes, fontsize=9, color=INK2, va="top")
    fig.text(0.012, 0.965, "Cosa fa il bootstrap: stessi mesi, ordine diverso", fontsize=13, fontweight="bold", color=INK, va="top")
    fig.text(0.012, 0.915, "Ogni quadratino è un mese di borsa vero (circa 21 giorni): verde se la regola era dentro il mercato, rosso se era fuori.\n"
             "Si legge a righe, da sinistra a destra; a sinistra l'anno di inizio di ogni riga. Ogni mese porta con sé il rendimento della regola e dell'indice.",
             fontsize=9.2, color=INK2, va="top")
    fig.legend(handles=[Patch(color=VERDE, label="Regola dentro il mercato"), Patch(color=ROSSO, label="Regola fuori (cash)")],
               loc="lower center", bbox_to_anchor=(0.5, 0.045), frameon=False, ncol=2, fontsize=9, labelcolor=INK2)
    fonte(fig, FONTE + " Nello scenario alcuni mesi compaiono più volte e altri mai.")
    salva(fig, "fig_mosaico_bootstrap.png")
