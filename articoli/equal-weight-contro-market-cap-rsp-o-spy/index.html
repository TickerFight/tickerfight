"""Passo 1 - Scarica da Yahoo Finance la serie giornaliera di due ETF (SPY e QQQ).

Salva, per ogni ETF, dati/<ticker>.csv (Date, Close, Adj Close, Volume) e dati/metadati.json con la fonte,
la data del download, la versione di yfinance e l'impronta SHA-256 di ogni file. Gli script successivi leggono SOLO
i file salvati: i risultati restano riproducibili anche se Yahoo corregge i dati in futuro.
"Adj Close" è il prezzo con i dividendi reinvestiti: è quello che si usa per i rendimenti.
"""
import hashlib
import json
import time
from datetime import datetime, timezone
from zoneinfo import ZoneInfo

import pandas as pd
import yfinance as yf

import config


def scarica(ticker: str) -> pd.DataFrame:
    for tentativo in range(3):
        grezzo = yf.Ticker(ticker).history(start=config.INIZIO_DOWNLOAD, auto_adjust=False)
        if not grezzo.empty:
            return grezzo
        print(f"{ticker}: nessun dato (tentativo {tentativo + 1})")
        time.sleep(10)
    return pd.DataFrame()


def main() -> None:
    config.DATI.mkdir(parents=True, exist_ok=True)
    ora_borsa = datetime.now(ZoneInfo(config.FUSO_BORSA))
    meta = {"fonte": "Yahoo Finance", "url_fonte": config.URL_FONTE, "libreria": f"yfinance {yf.__version__}",
            "scaricato_il_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"), "etf": []}
    nomi = {t: (n, a) for t, n, a in config.ETF}
    for ticker in [t for t, _, _ in config.ETF]:
        riga = {"ticker": ticker, "nome": nomi[ticker][0], "sigla": nomi[ticker][1]}
        grezzo = scarica(ticker)
        if grezzo.empty or "Adj Close" not in grezzo.columns:
            riga["stato"] = "scartato: nessun dato restituito da Yahoo Finance"
            meta["etf"].append(riga)
            continue
        df = grezzo[["Close", "Adj Close", "Volume"]].copy()
        df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
        df.index.name = "Date"
        df = df[~df.index.duplicated(keep="last")].sort_index().dropna(subset=["Adj Close"])
        df = df[df["Adj Close"] > 0]
        # seduta ancora in corso: barra incompleta, da scartare
        if df.index[-1].date() == ora_borsa.date() and ora_borsa.hour < 18:
            df = df.iloc[:-1]
        file = config.file_dati(ticker)
        df.to_csv(file, float_format="%.6f")
        riga.update({"stato": "ok", "file": file.name, "righe": len(df), "prima_data": str(df.index[0].date()),
                     "ultima_data": str(df.index[-1].date()), "sha256": hashlib.sha256(file.read_bytes()).hexdigest()})
        meta["etf"].append(riga)
        print(f"{ticker}: {len(df)} righe, dal {riga['prima_data']} al {riga['ultima_data']}")
    config.FILE_METADATI.write_text(json.dumps(meta, indent=2, ensure_ascii=False), encoding="utf-8")
    principali = {t for t, _, _ in config.ETF}
    ok = {r["ticker"] for r in meta["etf"] if r["stato"] == "ok"}
    if not principali <= ok:
        raise SystemExit(f"Mancano ETF principali: {sorted(principali - ok)}")


if __name__ == "__main__":
    main()

"""Passo 2 - Equal weight contro market cap, RSP contro SPY: rendimento, rischio, finestre mobili, fasi di crisi, persistenza e scimmie.

Legge SOLO i file in dati/ e produce in risultati/:
  metriche.json        ogni numero dell'articolo, con descrizione e origine di calcolo
  curve.csv            valore di 1 dollaro nei due ETF (una seduta su cinque)
  finestre_5anni.csv   rendimento annuo delle finestre mobili di cinque anni a fine mese, per ETF
  bootstrap.csv        differenze di Sharpe ricampionate (RSP - SPY)
  scimmie.csv          valore finale di 1 dollaro per ogni scimmia
Scimmia: ogni mese sceglie a caso uno dei due ETF e prende il rendimento di quel mese (senza costi).
"""
import json
from datetime import datetime, timezone

import numpy as np
import pandas as pd

import config
import motore as m

RIS: dict = {}
SORGENTE = "02_backtest_e_test.py con motore.py"
GA = config.GIORNI_ANNO
MESI_IT = ["gennaio", "febbraio", "marzo", "aprile", "maggio", "giugno", "luglio", "agosto", "settembre", "ottobre",
           "novembre", "dicembre"]


def reg(chiave, valore, formato, descrizione, origine=SORGENTE):
    if isinstance(valore, (np.floating, np.integer)):
        valore = valore.item()
    if isinstance(valore, float) and not np.isfinite(valore):
        raise SystemExit(f"Numero non finito per {chiave}")
    RIS[chiave] = {"valore": valore, "formato": formato, "descrizione": descrizione, "origine": origine}


def carica(ticker: str) -> pd.Series:
    df = pd.read_csv(config.file_dati(ticker), parse_dates=["Date"], index_col="Date").sort_index()
    df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
    df = df[~df.index.duplicated(keep="last")]
    c = df[config.COLONNA_PREZZO].astype(float)
    if (c <= 0).any() or c.isna().any():
        raise SystemExit(f"{ticker}: prezzi mancanti o non positivi.")
    return c


def sotto_acqua(equity: np.ndarray, date: pd.DatetimeIndex) -> dict:
    """Massima perdita dal picco: date di picco, minimo e (se avvenuto) recupero."""
    picco = np.maximum.accumulate(equity)
    dd = equity / picco - 1
    t = int(np.argmin(dd))
    p = int(np.argmax(equity[: t + 1]))
    rec = np.flatnonzero(equity[t:] >= equity[p])
    out = {"dd": float(dd[t]), "picco": date[p], "minimo": date[t], "recupero": date[t + int(rec[0])] if len(rec) else None}
    return out


def main() -> None:
    config.RISULTATI.mkdir(parents=True, exist_ok=True)
    sigle = [s for _, _, s in config.ETF]
    prezzi = pd.concat({s: carica(t) for t, _, s in config.ETF}, axis=1).dropna()
    anni = (prezzi.index[-1] - prezzi.index[0]).days / 365.25
    if anni < config.ANNI_MINIMI:
        raise SystemExit(f"Storia comune di {anni:.1f} anni: troppo corta.")
    R = prezzi.pct_change().dropna()
    cum = np.cumprod(1.0 + R.to_numpy(), axis=0)
    fini = m.fine_mese(R.index)
    rng = np.random.default_rng(config.SEME)
    ci = {s: i for i, s in enumerate(sigle)}
    A, B = "RSP", "SPY"

    reg("data_inizio", str(prezzi.index[0].date()), "data_lunga", "Prima seduta in comune dei due ETF", "dati/")
    reg("data_ultima", str(prezzi.index[-1].date()), "data_lunga", "Ultima seduta nei dati", "dati/")
    reg("anni", anni, "num1", "Anni di storia comune", "dati/")
    reg("n_sedute", len(R), "int", "Rendimenti giornalieri in comune", "dati/")
    reg("capitale", config.CAPITALE, "int", "Capitale iniziale per tradurre i rendimenti in dollari", "config.py, CAPITALE")
    reg("h4_min", config.H4_MIN, "pct0", "Ipotesi 4: quota minima di finestre di cinque anni con RSP davanti", "config.py")
    reg("h4_max", config.H4_MAX, "pct0", "Ipotesi 4: quota massima di finestre di cinque anni con RSP davanti", "config.py")
    reg("n_scimmie", config.N_SCIMMIE, "int", "Scimmie simulate", "config.py, N_SCIMMIE")
    reg("n_perm", config.N_PERMUTAZIONI, "int", "Permutazioni del test di persistenza", "config.py, N_PERMUTAZIONI")
    reg("n_boot", config.N_BOOTSTRAP, "int", "Ricampionamenti bootstrap", "config.py, N_BOOTSTRAP")
    reg("blocco", config.BLOCCO_BOOTSTRAP, "int", "Giorni per blocco del bootstrap", "config.py, BLOCCO_BOOTSTRAP")
    reg("livello_ic", config.LIVELLO_CONFIDENZA, "pct0", "Livello degli intervalli di confidenza", "config.py")
    reg("soglia", 0.05, "pct0", "Livello di significatività dei test", "convenzione")

    # --- Periodo comune -----------------------------------------------------------------------------------------
    stat = {}
    for s in sigle:
        r = R[s].to_numpy()
        e = m.equity(r)
        stat[s] = dict(cagr=m.cagr(r, GA), vol=m.vol(r, GA), sharpe=m.sharpe(r, GA), dd=m.max_dd(r),
                       finale=config.CAPITALE * float(e[-1]), tot=float(e[-1] - 1), ua=sotto_acqua(np.r_[1.0, e], R.index.insert(0, prezzi.index[0])))
        k = s.lower()
        reg(f"{k}_cagr", stat[s]["cagr"], "pct1", f"Rendimento annuo composto di {s} (dividendi reinvestiti)", "motore.cagr")
        reg(f"{k}_vol", stat[s]["vol"], "pct1", f"Volatilità annua di {s}", "motore.vol")
        reg(f"{k}_sharpe", stat[s]["sharpe"], "num2", f"Rapporto di Sharpe di {s} (tasso privo di rischio zero)", "motore.sharpe")
        reg(f"{k}_dd", -stat[s]["dd"], "pct0", f"Perdita massima dal picco di {s}", "motore.max_dd")
        reg(f"{k}_finale", stat[s]["finale"], "int", f"Valore finale del capitale iniziale in {s}", "motore.equity")
        reg(f"{k}_tot", stat[s]["tot"], "pct0", f"Rendimento totale di {s} nel periodo", "motore.equity")
        ua = stat[s]["ua"]
        reg(f"{k}_dd_picco", f"{MESI_IT[ua['picco'].month - 1]} {ua['picco'].year}", "testo", f"Mese del picco prima della perdita massima di {s}", "dati/")
        reg(f"{k}_dd_minimo", f"{MESI_IT[ua['minimo'].month - 1]} {ua['minimo'].year}", "testo", f"Mese del minimo della perdita massima di {s}", "dati/")
        if ua["recupero"] is not None:
            reg(f"{k}_dd_recupero_anni", (ua["recupero"] - ua["picco"]).days / 365.25, "num1", f"Anni dal picco al pieno recupero della perdita massima di {s}", "dati/")
            reg(f"{k}_dd_recupero", f"{MESI_IT[ua['recupero'].month - 1]} {ua['recupero'].year}", "testo", f"Mese del pieno recupero della perdita massima di {s}", "dati/")
        else:
            reg(f"{k}_dd_non_recuperato", 1, "int", f"1 se {s} non ha ancora recuperato la perdita massima", "dati/")
    reg("diff_cagr", stat[A]["cagr"] - stat[B]["cagr"], "pp1", "Rendimento annuo composto di RSP meno quello di SPY, in punti percentuali", "motore.cagr")
    reg("diff_dd", stat[B]["dd"] - stat[A]["dd"], "pp1", "Perdita massima di RSP meno quella di SPY, in punti percentuali", "motore.max_dd")
    reg("diff_vol", stat[A]["vol"] - stat[B]["vol"], "pp1", "Volatilità di RSP meno quella di SPY, in punti percentuali", "motore.vol")
    reg("rapporto_finale", stat[A]["finale"] / stat[B]["finale"], "num2", "Valore finale di RSP diviso quello di SPY", "motore.equity")
    reg("corr", float(np.corrcoef(R[A], R[B])[0, 1]), "num2", "Correlazione dei rendimenti giornalieri tra RSP e SPY", "numpy.corrcoef")

    # --- Bootstrap della differenza di Sharpe ---------------------------------------------------------------------
    boot = m.bootstrap_diff_sharpe(R[A].to_numpy(), R[B].to_numpy(), config.N_BOOTSTRAP, config.BLOCCO_BOOTSTRAP, GA, rng)
    lo, hi = np.quantile(boot, [(1 - config.LIVELLO_CONFIDENZA) / 2, 1 - (1 - config.LIVELLO_CONFIDENZA) / 2])
    d0 = stat[A]["sharpe"] - stat[B]["sharpe"]
    reg("diff_sharpe", d0, "num2", "Sharpe di RSP meno Sharpe di SPY", "motore.sharpe")
    reg("diff_sharpe_lo", float(lo), "num2", "Estremo inferiore dell'intervallo bootstrap a blocchi della differenza di Sharpe", "bootstrap a blocchi")
    reg("diff_sharpe_hi", float(hi), "num2", "Estremo superiore dell'intervallo bootstrap a blocchi della differenza di Sharpe", "bootstrap a blocchi")
    pd.DataFrame({"diff_sharpe": boot}).to_csv(config.RISULTATI / "bootstrap.csv", index=False, float_format="%.5f")

    # --- Finestre mobili --------------------------------------------------------------------------------------------
    for anni_fin in config.FINESTRE_ANNI:
        pos, rend = m.rendimento_finestre(cum, fini, anni_fin * GA)
        quote = m.quote_vincitrici(rend)
        ann = (1.0 + rend) ** (1.0 / anni_fin) - 1.0
        reg(f"fin{anni_fin}_n", len(pos), "int", f"Finestre mobili di {anni_fin} anni (a fine mese)", "motore.rendimento_finestre")
        reg(f"fin{anni_fin}_quota_rsp", float(quote[ci[A]]), "pct0", f"Quota di finestre di {anni_fin} anni in cui RSP rende più di SPY", "motore.quote_vincitrici")
        reg(f"fin{anni_fin}_quota_spy", float(quote[ci[B]]), "pct0", f"Quota di finestre di {anni_fin} anni in cui SPY rende più di RSP", "motore.quote_vincitrici")
        reg(f"fin{anni_fin}_peggio_rsp", float(ann[:, ci[A]].min()), "pct0", f"Peggior rendimento annuo di RSP su finestre di {anni_fin} anni", "motore.rendimento_finestre")
        reg(f"fin{anni_fin}_miglio_rsp", float(ann[:, ci[A]].max()), "pct0", f"Miglior rendimento annuo di RSP su finestre di {anni_fin} anni", "motore.rendimento_finestre")
        reg(f"fin{anni_fin}_peggio_spy", float(ann[:, ci[B]].min()), "pct0", f"Peggior rendimento annuo di SPY su finestre di {anni_fin} anni", "motore.rendimento_finestre")
        reg(f"fin{anni_fin}_miglio_spy", float(ann[:, ci[B]].max()), "pct0", f"Miglior rendimento annuo di SPY su finestre di {anni_fin} anni", "motore.rendimento_finestre")
        if anni_fin == config.FINESTRA_RIF:
            out = pd.DataFrame(ann, columns=sigle)
            out.insert(0, "data", R.index[pos].strftime("%Y-%m-%d"))
            out.to_csv(config.RISULTATI / "finestre_5anni.csv", index=False, float_format="%.5f")
            ultima_rsp = bool(ann[-1, ci[A]] > ann[-1, ci[B]])
            reg("fin_rif_ultima_rsp", int(ultima_rsp), "int", "1 se nell'ultima finestra di riferimento RSP rende più di SPY", "motore.rendimento_finestre")
    reg("fin_rif_anni", config.FINESTRA_RIF, "annonum", "Anni della finestra di riferimento", "config.py, FINESTRA_RIF")

    # --- Fasi di crisi -----------------------------------------------------------------------------------------------
    n_peggio = 0
    for k, (nome, a, b) in enumerate(config.CRISI):
        sub = prezzi.loc[a:b]
        if len(sub) < 20:
            continue
        ra = float(sub[A].iloc[-1] / sub[A].iloc[0] - 1)
        rb = float(sub[B].iloc[-1] / sub[B].iloc[0] - 1)
        reg(f"crisi{k}_nome", nome, "testo", "Nome della fase di crisi", "config.py, CRISI")
        reg(f"crisi{k}_rsp", ra, "pct0", f"Rendimento di RSP nella fase «{nome}»", "dati/")
        reg(f"crisi{k}_spy", rb, "pct0", f"Rendimento di SPY nella fase «{nome}»", "dati/")
        reg(f"crisi{k}_rsp_peggio", int(ra < rb), "int", f"1 se RSP perde più di SPY nella fase «{nome}»", "dati/")
        n_peggio += int(ra < rb)
    reg("n_crisi_rsp_peggio", n_peggio, "int", "Fasi di crisi in cui RSP rende meno di SPY", "dati/")
    reg("n_crisi", len(config.CRISI), "int", "Fasi di crisi fissate in anticipo", "config.py, CRISI")

    # --- Persistenza ----------------------------------------------------------------------------------------------
    passato, futuro = m.persistenza(cum, fini, config.FINESTRA_RIF * GA)
    oss, p_perm = m.p_permutazione(passato, futuro, config.N_PERMUTAZIONI, rng)
    reg("pers_n", len(passato), "int", "Mesi usati nel test di persistenza", "motore.persistenza")
    reg("pers_quota", oss, "pct0", "Quota di mesi in cui il vincitore dei cinque anni precedenti vince anche nei cinque successivi", "motore.p_permutazione")
    reg("pers_p", p_perm, "pval", "P-value del test di permutazione per la persistenza del vincitore", "motore.p_permutazione")
    reg("pers_atteso", 0.5, "pct0", "Quota attesa per puro caso con due ETF", "calcolo")
    iq = sigle.index("RSP")
    pq, fq = float(np.mean(passato == iq)), float(np.mean(futuro == iq))
    reg("pers_passato_rsp", pq, "pct0", "Quota di mesi in cui RSP è il vincitore dei cinque anni precedenti", "motore.persistenza")
    reg("pers_futuro_rsp", fq, "pct0", "Quota di mesi in cui RSP è il vincitore dei cinque anni successivi", "motore.persistenza")
    reg("pers_atteso_marg", pq * fq + (1 - pq) * (1 - fq), "pct0", "Quota di coincidenze attesa se passato e futuro fossero indipendenti, date le frequenze di RSP", "calcolo")

    # --- Scimmie ---------------------------------------------------------------------------------------------------
    mensili = m.rendimenti_mensili(R.to_numpy(), fini)
    fin = m.scimmie_valore_finale(mensili, config.N_SCIMMIE, rng)
    bh = np.prod(1.0 + mensili, axis=0)
    pd.DataFrame({"finale": fin}).to_csv(config.RISULTATI / "scimmie.csv", index=False, float_format="%.5f")
    reg("sc_mediana", config.CAPITALE * float(np.median(fin)), "int", "Valore finale mediano delle scimmie (scelta casuale tra i due ETF ogni mese)", "scimmie.csv")
    reg("sc_p5", config.CAPITALE * float(np.percentile(fin, 5)), "int", "Quinto percentile del valore finale delle scimmie", "scimmie.csv")
    reg("sc_p95", config.CAPITALE * float(np.percentile(fin, 95)), "int", "Novantacinquesimo percentile del valore finale delle scimmie", "scimmie.csv")
    reg("sc_batte_spy", float((fin > bh[ci[B]]).mean()), "pct0", "Quota di scimmie che chiudono sopra SPY", "scimmie.csv")
    reg("sc_batte_rsp", float((fin > bh[ci[A]]).mean()), "pct0", "Quota di scimmie che chiudono sopra RSP", "scimmie.csv")
    reg("sc_tra", int(min(stat[A]["finale"], stat[B]["finale"]) <= config.CAPITALE * float(np.median(fin)) <= max(stat[A]["finale"], stat[B]["finale"])), "int", "1 se il valore finale mediano delle scimmie sta tra quello di SPY e quello di RSP", "scimmie.csv")

    camp = np.arange(0, len(R), 5)
    pd.DataFrame({"data": R.index[camp].strftime("%Y-%m-%d"), "spy": cum[camp, ci[B]], "rsp": cum[camp, ci[A]]}).to_csv(
        config.RISULTATI / "curve.csv", index=False, float_format="%.5f")

    # --- Verdetti sulle ipotesi -----------------------------------------------------------------------------------
    quota5 = float(m.quote_vincitrici(m.rendimento_finestre(cum, fini, config.FINESTRA_RIF * GA)[1])[ci[A]])
    h = [stat[B]["cagr"] > stat[A]["cagr"],
         stat[A]["vol"] > stat[B]["vol"] and stat[A]["dd"] < stat[B]["dd"],
         bool(lo <= 0 <= hi),
         config.H4_MIN <= quota5 <= config.H4_MAX,
         oss < 0.5,
         RIS["crisi0_rsp_peggio"]["valore"] == 1 and RIS["crisi1_rsp_peggio"]["valore"] == 1 and RIS["crisi2_rsp_peggio"]["valore"] == 0]
    for i, v in enumerate(h, 1):
        reg(f"h{i}_esito", int(bool(v)), "int", f"Ipotesi {i}: 1 se confermata, 0 se smentita", "confronto con config.py")
    reg("n_conf", int(sum(bool(v) for v in h)), "int", "Ipotesi confermate (su sei)", "confronto con config.py")

    out = {"generato_il_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"), "pandas": pd.__version__,
           "numpy": np.__version__, "ipotesi": config.IPOTESI, "valori": RIS}
    (config.RISULTATI / "metriche.json").write_text(json.dumps(out, indent=1, ensure_ascii=False), encoding="utf-8")
    print(f"Registrati {len(RIS)} numeri")


if __name__ == "__main__":
    main()

"""Passo 3 - Grafici dell'articolo. Legge SOLO risultati/ e salva i PNG in figure/.

  fig1_valore_dei_due_etf.png   valore di un dollaro in SPY e RSP (scala logaritmica)
  fig2_finestre_5anni.png       rendimento annuo delle finestre mobili di cinque anni, per ETF
  fig3_scimmie.png              valore finale delle scimmie contro SPY e RSP
"""
import json

import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt  # noqa: E402
import pandas as pd  # noqa: E402

import config  # noqa: E402

BLU, ARANCIO, GRIGIO = "#1f5fa8", "#c9610a", "#8a8a8a"
plt.rcParams.update({"font.size": 12, "axes.spines.top": False, "axes.spines.right": False,
                     "axes.grid": True, "grid.alpha": 0.25, "figure.dpi": 120, "savefig.bbox": "tight",
                     "savefig.facecolor": "white", "legend.frameon": False})
R = config.RISULTATI


def main() -> None:

"""Funzioni di calcolo dell'articolo 202: metriche, finestre mobili, persistenza, bootstrap, scimmie.

Nessuna strategia di trading: sono confronti tra buy & hold. Le funzioni sono pure, così verifica_motore.py
le controlla su dati inventati con risultati calcolabili a mano.
"""
import numpy as np
import pandas as pd


def equity(r) -> np.ndarray:
    return np.cumprod(1.0 + np.asarray(r, dtype=float))


def cagr(r, ga: int) -> float:
    r = np.asarray(r, dtype=float)
    return float(np.prod(1.0 + r) ** (ga / len(r)) - 1.0)


def vol(r, ga: int) -> float:
    return float(np.std(np.asarray(r, dtype=float), ddof=1) * np.sqrt(ga))


def sharpe(r, ga: int) -> float:
    r = np.asarray(r, dtype=float)
    return float(np.mean(r) / np.std(r, ddof=1) * np.sqrt(ga))


def max_dd(r) -> float:
    e = equity(r)
    picco = np.maximum.accumulate(np.concatenate([[1.0], e]))[1:]
    return float(np.min(e / picco - 1.0))


def fine_mese(indice: pd.DatetimeIndex) -> np.ndarray:
    """Posizioni dell'ultima seduta di ogni mese (escluso l'ultimo mese se incompleto non è controllato: si usa quanto c'è)."""
    s = pd.Series(np.arange(len(indice)), index=indice)
    return s.groupby([indice.year, indice.month]).last().to_numpy()


def rendimento_finestre(cum: np.ndarray, fini: np.ndarray, lung: int) -> tuple[np.ndarray, np.ndarray]:
    """Rendimento totale delle ultime `lung` sedute alla fine di ogni mese utile. cum: (T, k). Restituisce (posizioni, matrice)."""
    ok = fini[fini >= lung]
    base = np.concatenate([np.ones((1, cum.shape[1])), cum], axis=0)   # base[i] = valore prima della seduta i
    return ok, cum[ok] / base[ok - lung + 1] - 1.0


def quote_vincitrici(rend: np.ndarray) -> np.ndarray:
    """Quota di finestre in cui ciascuna colonna ha il rendimento più alto."""
    v = np.argmax(rend, axis=1)
    return np.bincount(v, minlength=rend.shape[1]) / len(v)


def persistenza(cum: np.ndarray, fini: np.ndarray, lung: int) -> tuple[np.ndarray, np.ndarray]:
    """Per ogni mese t con `lung` sedute prima e dopo: area vincitrice nel passato (usa solo dati fino a t) e nel futuro."""
    base = np.concatenate([np.ones((1, cum.shape[1])), cum], axis=0)
    passato, futuro = [], []
    for e in fini:
        if e - lung + 1 < 0 or e + lung >= len(cum):
            continue
        passato.append(int(np.argmax(cum[e] / base[e - lung + 1])))
        futuro.append(int(np.argmax(cum[e + lung] / cum[e])))
    return np.array(passato), np.array(futuro)


def p_permutazione(passato: np.ndarray, futuro: np.ndarray, n: int, rng) -> tuple[float, float]:
    """Quota di coincidenze osservata e p-value: frazione di permutazioni del futuro con almeno altrettante coincidenze."""
    oss = float(np.mean(passato == futuro))
    sup = 0
    for _ in range(n):
        if np.mean(passato == rng.permutation(futuro)) >= oss - 1e-12:
            sup += 1
    return oss, (sup + 1) / (n + 1)


def volatilita_mobile(r: pd.Series, finestra: int, ga: int) -> pd.Series:
    return r.rolling(finestra).std(ddof=1) * np.sqrt(ga)


def correlazione_mobile(a: pd.Series, b: pd.Series, finestra: int) -> pd.Series:
    return a.rolling(finestra).corr(b)


def bootstrap_diff_sharpe(ra, rb, n: int, blocco: int, ga: int, rng) -> np.ndarray:
    """Differenza di Sharpe (a - b) ricampionando a blocchi le stesse date per entrambe le serie."""
    ra, rb = np.asarray(ra, dtype=float), np.asarray(rb, dtype=float)
    T = len(ra)
    nb = int(np.ceil(T / blocco))
    out = np.empty(n)
    for i in range(n):
        inizi = rng.integers(0, T - blocco + 1, size=nb)
        idx = (inizi[:, None] + np.arange(blocco)[None, :]).ravel()[:T]
        out[i] = sharpe(ra[idx], ga) - sharpe(rb[idx], ga)
    return out


def rendimenti_mensili(r: np.ndarray, fini: np.ndarray) -> np.ndarray:
    """Rendimenti composti per mese: r (T, k); i mesi sono i tratti che finiscono nelle posizioni `fini`."""
    e = np.vstack([np.ones((1, r.shape[1])), np.cumprod(1.0 + r, axis=0)])
    estremi = np.concatenate([[0], fini + 1])
    if estremi[-1] != len(r):            # coda: ultimo mese incompleto, si include
        estremi = np.concatenate([estremi, [len(r)]])
    return e[estremi[1:]] / e[estremi[:-1]] - 1.0


def scimmie_valore_finale(mensili: np.ndarray, n: int, rng) -> np.ndarray:
    """Valore finale di 1 per n scimmie che scelgono a caso l'area (colonna) ogni mese."""
    M, k = mensili.shape
    scelta = rng.integers(0, k, size=(n, M))
    return np.prod(1.0 + mensili[np.arange(M)[None, :], scelta], axis=1)

"""Controlli di motore.py su dati inventati, con risultati calcolabili a mano. Se uno fallisce, il flusso si ferma."""
import numpy as np
import pandas as pd

import motore as m

rng = np.random.default_rng(1)


def uguale(a, b, tol=1e-9, nome=""):
    assert abs(a - b) < tol, f"{nome}: {a} != {b}"
    print(f"ok  {nome}")


# 1. rendimento composto: +10% e -10% in due sedute
r = np.array([0.10, -0.10])
uguale(m.equity(r)[-1], 0.99, nome="equity")
uguale(m.cagr(np.full(252, 0.0), 252), 0.0, nome="cagr nullo")
uguale(m.cagr(np.full(252, 0.001), 252), 1.001 ** 252 - 1, nome="cagr costante")
# 2. perdita massima: sale a 1,2 poi scende a 0,6 -> -50%
uguale(m.max_dd(np.array([0.2, -0.5])), -0.5, nome="perdita massima")
# 3. volatilità e Sharpe di una serie alternata +1% / -1%
alt = np.tile([0.01, -0.01], 126)
uguale(m.vol(alt, 252), np.std(alt, ddof=1) * np.sqrt(252), nome="volatilità")
uguale(m.sharpe(alt + 0.0, 252), 0.0, tol=1e-9, nome="Sharpe nullo")
# 4. fine mese: tre mesi finti
idx = pd.bdate_range("2020-01-01", periods=70)
fini = m.fine_mese(idx)
assert idx[fini[0]].month == 1 and idx[fini[0] + 1].month == 2, "fine mese"
print("ok  fine mese")
# 5. finestre: A cresce 1% al giorno, B ferma, C scende 1%: A vince sempre, rendimento a 10 sedute noto
T = 600
R = np.column_stack([np.full(T, 0.01), np.zeros(T), np.full(T, -0.01)])
cum = np.cumprod(1.0 + R, axis=0)
pos, rend = m.rendimento_finestre(cum, np.array([50, 100, 599]), 10)
uguale(rend[0, 0], 1.01 ** 10 - 1, nome="rendimento finestra A")
uguale(rend[0, 1], 0.0, nome="rendimento finestra B")
uguale(rend[0, 2], 0.99 ** 10 - 1, nome="rendimento finestra C")
q = m.quote_vincitrici(rend)
assert q.tolist() == [1.0, 0.0, 0.0], "quote vincitrici"
print("ok  quote vincitrici")
# 6. persistenza: se A vince sempre, passato = futuro sempre
fm = m.fine_mese(pd.bdate_range("2000-01-03", periods=T))
pa, fu = m.persistenza(cum, fm, 60)
assert len(pa) > 5 and (pa == fu).all() and (pa == 0).all(), "persistenza"
print("ok  persistenza")
# 7. permutazione: su dati indipendenti il p-value è alto; su dati identici con due classi è basso
a = rng.integers(0, 3, 300)
b = rng.integers(0, 3, 300)
_, p = m.p_permutazione(a, b, 500, rng)
assert p > 0.05, f"permutazione su dati indipendenti: p={p}"
_, p2 = m.p_permutazione(a, a, 500, rng)
assert p2 < 0.01, "permutazione su dati identici"
print("ok  permutazione")
# 8. correlazione mobile: serie identiche -> 1; opposte -> -1
x = pd.Series(rng.normal(size=300))
uguale(m.correlazione_mobile(x, x, 50).dropna().min(), 1.0, tol=1e-9, nome="correlazione 1")
uguale(m.correlazione_mobile(x, -x, 50).dropna().max(), -1.0, tol=1e-9, nome="correlazione -1")
# 9. bootstrap: due serie identiche -> differenza di Sharpe zero
d = m.bootstrap_diff_sharpe(x.to_numpy() * 0.01, x.to_numpy() * 0.01, 50, 21, 252, rng)
assert np.abs(d).max() < 1e-9, "bootstrap identico"
print("ok  bootstrap")
# 10. mensili e scimmie: tre aree con rendimento mensile costante 1%, 0, -1% -> la scimmia ha valore tra i tre estremi
Rm = np.column_stack([np.full(T, 0.0005), np.zeros(T), np.full(T, -0.0005)])
mens = m.rendimenti_mensili(Rm, fm)
uguale(np.prod(1 + mens[:, 0]), np.prod(1 + Rm[:, 0]), tol=1e-9, nome="mesi ricompongono il totale")
sc = m.scimmie_valore_finale(mens, 2000, rng)
assert (sc <= np.prod(1 + mens[:, 0]) + 1e-9).all() and (sc >= np.prod(1 + mens[:, 2]) - 1e-9).all(), "scimmie nei limiti"
print("ok  scimmie")
