"""Passo 1 - Scarica da Yahoo Finance la serie giornaliera dell'S&P 500.

Salva:
  dati/gspc_giornaliero.csv   la serie (Date, Open, High, Low, Close, Volume)
  dati/gspc_metadati.json     da dove arrivano i dati, quando, con quale
                              versione di yfinance, e l'impronta SHA-256 del file

Gli script successivi leggono SOLO il file salvato: i risultati sono quindi
riproducibili anche se Yahoo corregge i dati in futuro.
"""
import hashlib
import json
from datetime import datetime, timezone
from zoneinfo import ZoneInfo

import pandas as pd
import yfinance as yf

import config


def main() -> None:
    config.DATI.mkdir(parents=True, exist_ok=True)

    ticker = yf.Ticker(config.TICKER)
    grezzo = ticker.history(start=config.INIZIO_DOWNLOAD, auto_adjust=False)
    if grezzo.empty:
        raise SystemExit("Yahoo Finance non ha restituito dati.")

    df = grezzo[["Open", "High", "Low", "Close", "Volume"]].copy()
    df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
    df.index.name = "Date"
    righe_scaricate = len(df)

    # Controlli di qualità di base
    df = df[~df.index.duplicated(keep="last")].sort_index()
    df = df.dropna(subset=["Close"])
    if (df["Close"] <= 0).any():
        raise SystemExit("Trovati prezzi di chiusura non positivi.")

    # Se lo script gira durante la seduta, l'ultima barra è incompleta: la scartiamo.
    ora_borsa = datetime.now(ZoneInfo(config.FUSO_BORSA))
    barra_scartata = None
    if df.index[-1].date() == ora_borsa.date() and ora_borsa.hour < 16:
        barra_scartata = str(df.index[-1].date())
        df = df.iloc[:-1]

    df.to_csv(config.FILE_DATI, float_format="%.6f")

    sha = hashlib.sha256(config.FILE_DATI.read_bytes()).hexdigest()
    metadati = {
        "ticker": config.TICKER,
        "descrizione": "S&P 500, indice di prezzo (esclude i dividendi)",
        "fonte": "Yahoo Finance",
        "url_fonte": "https://finance.yahoo.com/quote/%5EGSPC/history/",
        "libreria": f"yfinance {yf.__version__}",
        "scaricato_il_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"),
        "righe_scaricate": righe_scaricate,
        "righe_salvate": len(df),
        "prima_data": str(df.index[0].date()),
        "ultima_data": str(df.index[-1].date()),
        "barra_incompleta_scartata": barra_scartata,
        "colonna_usata": "Close (chiusura dell'indice, non corretta: l'indice non ha dividendi)",
        "sha256_file": sha,
    }
    config.FILE_METADATI.write_text(
        json.dumps(metadati, indent=2, ensure_ascii=False), encoding="utf-8"
    )
    print(json.dumps(metadati, indent=2, ensure_ascii=False))


if __name__ == "__main__":
    main()

"""Passo 2 - Comprare dopo un calo rapido del 10% sull'S&P 500.

Legge SOLO dati/gspc_giornaliero.csv; produce in risultati/: metriche.json, eventi.csv.
"""
import json
import platform
from datetime import datetime, timezone

import numpy as np
import pandas as pd

import config
import funzioni as fx

RIS: dict = {}
SORGENTE = "02_backtest_e_test.py con funzioni.py"
NOMI = {21: "1m", 63: "3m", 252: "12m"}


def reg(chiave, valore, formato, descrizione, origine=SORGENTE):
    if isinstance(valore, (np.floating, np.integer)):
        valore = valore.item()
    if isinstance(valore, float) and not np.isfinite(valore):
        raise SystemExit(f"Numero non finito per {chiave}")
    RIS[chiave] = {"valore": valore, "formato": formato, "descrizione": descrizione, "origine": origine}


def main() -> None:
    config.RISULTATI.mkdir(parents=True, exist_ok=True)
    rng = np.random.default_rng(config.SEME)
    df = pd.read_csv(config.FILE_DATI, parse_dates=["Date"], index_col="Date").sort_index()
    close = df["Close"].astype(float)
    if (close <= 0).any() or close.isna().any():
        raise SystemExit("Chiusure mancanti o non positive.")
    c = close.to_numpy()
    date = close.index
    anni = date.year.to_numpy()
    ev = fx.eventi_calo(c, config.FINESTRA, config.SOGLIA_CALO, config.PAUSA)

    reg("prima_data", str(date[0].date()), "data_lunga", "Prima seduta dei dati", "dati/")
    reg("data_ultima", str(date[-1].date()), "data_lunga", "Ultima seduta dei dati", "dati/")
    reg("n_rend", len(c) - 1, "int", "Rendimenti giornalieri", "dati/")
    reg("anni_tot", (len(c) - 1) / 252, "num1", "Anni di borsa nei dati", "dati/")
    reg("finestra", config.FINESTRA, "int", "Sedute della finestra del massimo di riferimento", "config.py")
    reg("soglia_calo", config.SOGLIA_CALO, "pct0", "Calo minimo dal massimo", "config.py")
    reg("pausa", config.PAUSA, "int", "Sedute di pausa dopo un evento", "config.py")
    reg("h1_ecc", config.H1_ECCESSO, "pct0", "Ipotesi 1: eccesso minimo a dodici mesi", "config.py")
    reg("h3_quota", config.H3_QUOTA, "pct0", "Ipotesi 3: quota minima di eventi negativi a tre mesi", "config.py")
    reg("h4_calo", config.H4_CALO, "pct0", "Ipotesi 4: ulteriore calo minimo", "config.py")
    reg("h4_quota", config.H4_QUOTA, "pct0", "Ipotesi 4: quota di eventi", "config.py")
    reg("anno_escluso", config.ANNO_ESCLUSO_FINO, "annonum", "Ultimo anno degli eventi tolti nell'ipotesi 5", "config.py")
    reg("soglia_p", config.SOGLIA_P, "num2", "Soglia di significatività", "config.py")
    reg("n_casuali", config.N_CASUALI, "int", "Estrazioni casuali", "config.py")
    reg("n_eventi", len(ev), "int", "Eventi di calo rapido", "funzioni.eventi_calo")
    reg("primo_evento", str(date[ev[0]].date()), "data_lunga", "Data del primo evento", "funzioni")
    reg("ultimo_evento", str(date[ev[-1]].date()), "data_lunga", "Data dell'ultimo evento", "funzioni")

    righe = []
    for h in config.ORIZZONTI:
        av = fx.rend_avanti(c, h)
        ok_ev = ev[~np.isnan(av[ev])]
        tutti = av[~np.isnan(av)]
        n = NOMI[h]
        reg(f"n_{n}", len(ok_ev), "int", f"Eventi con {n} di dati successivi", "funzioni")
        reg(f"m_{n}", float(av[ok_ev].mean()), "pct1", f"Rendimento medio dopo l'evento, orizzonte {n}", "funzioni")
        reg(f"t_{n}", float(tutti.mean()), "pct1", f"Rendimento medio di un giorno qualunque, orizzonte {n}", "funzioni")
        reg(f"ecc_{n}", float(av[ok_ev].mean() - tutti.mean()), "pp2", f"Eccesso, orizzonte {n}, punti percentuali", "funzioni")
        reg(f"neg_{n}", float((av[ok_ev] < 0).mean()), "pct0", f"Quota di eventi con rendimento negativo, orizzonte {n}", "funzioni")
        reg(f"neg_tutti_{n}", float((tutti < 0).mean()), "pct0", f"Quota di giorni qualunque con rendimento negativo, orizzonte {n}", "funzioni")
        if h == 252:
            p = fx.pval_casuali(av, float(av[ok_ev].mean()), len(ok_ev), config.N_CASUALI, rng)
            reg("pval", p, "pval", "Quota di estrazioni casuali con rendimento medio a dodici mesi almeno pari a quello degli eventi", "funzioni.pval_casuali")
            s = ok_ev[anni[ok_ev] > config.ANNO_ESCLUSO_FINO]
            tt = av[(anni > config.ANNO_ESCLUSO_FINO) & ~np.isnan(av)]
            reg("n_senza", len(s), "int", "Eventi a dodici mesi dopo il 1939", "funzioni")
            reg("ecc_senza", float(av[s].mean() - tt.mean()), "pp2", "Eccesso a dodici mesi senza gli eventi fino al 1939, punti percentuali", "funzioni")
            reg("m_senza", float(av[s].mean()), "pct1", "Rendimento medio a dodici mesi dopo gli eventi dopo il 1939", "funzioni")
            reg("peggiore_12m", float(av[ok_ev].min()), "pct0", "Il peggior rendimento a dodici mesi dopo un evento", "funzioni")
            reg("migliore_12m", float(av[ok_ev].max()), "pct0", "Il miglior rendimento a dodici mesi dopo un evento", "funzioni")
            for d_i in ok_ev:
                righe.append({"data": str(date[d_i].date()), "r12": av[d_i], "r3": fx.rend_avanti(c, 63)[d_i]})
    mn = fx.minimo_avanti(c, 63)
    ok_mn = ev[~np.isnan(mn[ev])]
    reg("quota_calo_ulteriore", float((mn[ok_mn] <= -config.H4_CALO).mean()), "pct0", "Quota di eventi con un ulteriore calo di almeno il 5% nei 63 giorni successivi", "funzioni.minimo_avanti")
    reg("calo_medio", float(mn[ok_mn].mean()), "pct1", "Ulteriore calo medio massimo nei 63 giorni successivi", "funzioni.minimo_avanti")
    pd.DataFrame(righe).to_csv(config.RISULTATI / "eventi.csv", index=False, float_format="%.5f")

    h = [RIS["ecc_12m"]["valore"] >= config.H1_ECCESSO,
         RIS["m_1m"]["valore"] < RIS["t_1m"]["valore"],
         RIS["neg_3m"]["valore"] >= config.H3_QUOTA,
         RIS["quota_calo_ulteriore"]["valore"] > config.H4_QUOTA,
         RIS["ecc_senza"]["valore"] > 0,
         RIS["pval"]["valore"] < config.SOGLIA_P]
    for i, v in enumerate(h, 1):
        reg(f"h{i}_esito", int(bool(v)), "int", f"Ipotesi {i}: 1 se confermata, 0 se smentita", "confronto con config.py")
    reg("n_conf", int(sum(bool(v) for v in h)), "int", "Ipotesi confermate (su 6)", "confronto con config.py")

    meta = {"generato": datetime.now(timezone.utc).isoformat(timespec="seconds"), "python": platform.python_version(),
            "numpy": np.__version__, "pandas": pd.__version__, "seme": config.SEME, "ipotesi": config.IPOTESI}
    (config.RISULTATI / "metriche.json").write_text(json.dumps({"meta": meta, "valori": RIS}, ensure_ascii=False, indent=1), encoding="utf-8")
    print(f"Scritte {len(RIS)} metriche.")


if __name__ == "__main__":
    main()

"""Parametri dell'articolo n. 151 - "Comprare dopo un -10%".

Tutti i parametri stanno qui; gli script numerati li leggono da questo file. Le ipotesi sono scritte prima di guardare i risultati.

«Quando il mercato crolla del dieci per cento, compra». Sull'S&P 500 (indice di prezzo, dal 1927, senza dividendi) un «calo rapido» è il giorno in cui la chiusura è almeno il
10% sotto il massimo delle 20 chiusure precedenti. Per rendere gli eventi quasi indipendenti, dopo un evento se ne ignorano altri per un anno di borsa (252 sedute). Si misura
il rendimento a 1, 3 e 12 mesi (21, 63 e 252 sedute) dalla chiusura dell'evento e lo si confronta con quello di un giorno qualunque. Si guarda anche quanto l'indice scende ancora
dopo l'evento, e se i risultati dipendono dagli anni della Grande Depressione.
"""

FIGURE = BASE / "figure"

FILE_METADATI = DATI / "gspc_metadati.json"
DESCRIZIONE_DATI = "le chiusure giornaliere dell'indice S&P 500"
URL_FONTE = "https://finance.yahoo.com/quote/%5EGSPC/history/"

    "2) A un mese dall'evento il rendimento medio è inferiore a quello medio a un mese di un giorno qualunque: il calo, nel breve, continua. "

"""Funzioni di calcolo dell'articolo 151: eventi di calo rapido e rendimenti successivi."""
import numpy as np


def eventi_calo(close: np.ndarray, finestra: int, soglia: float, pausa: int) -> np.ndarray:
    """Indici dei giorni in cui la chiusura è almeno `soglia` sotto il massimo delle `finestra` chiusure precedenti.
    Dopo un evento se ne ignorano altri per `pausa` sedute."""
    c = np.asarray(close, float)
    ev = []
    ultimo = -10**9
    for t in range(finestra, len(c)):
        if t - ultimo < pausa:
            continue
        if c[t] <= (1 - soglia) * c[t - finestra:t].max():
            ev.append(t)
            ultimo = t
    return np.array(ev, int)


def rend_avanti(close: np.ndarray, h: int) -> np.ndarray:
    """Rendimento semplice da t a t+h; nan dove non esiste."""
    c = np.asarray(close, float)
    out = np.full(len(c), np.nan)
    out[: len(c) - h] = c[h:] / c[:-h] - 1.0
    return out


def minimo_avanti(close: np.ndarray, h: int) -> np.ndarray:
    """Variazione minima della chiusura nei `h` giorni successivi rispetto alla chiusura di t (<= 0 se scende; nan se mancano giorni)."""
    c = np.asarray(close, float)
    out = np.full(len(c), np.nan)
    for t in range(len(c) - h):
        out[t] = c[t + 1:t + h + 1].min() / c[t] - 1.0
    return out


def pval_casuali(valori: np.ndarray, osservato: float, n_eventi: int, n: int, rng: np.random.Generator) -> float:
    """Quota di estrazioni casuali di `n_eventi` giorni (senza pausa) con rendimento medio >= osservato (con +1)."""
    v = valori[~np.isnan(valori)]
    cont = 0
    for _ in range(n):
        if v[rng.integers(0, len(v), n_eventi)].mean() >= osservato:
            cont += 1
    return float((1 + cont) / (n + 1))
flusso/verifica_motore.py
"""Controlli di funzioni.py su dati inventati. Se uno fallisce, l'Action si ferma."""
import numpy as np

import funzioni as f


def ok(nome, cond):
    assert cond, nome
    print(f"ok  {nome}")


c = np.concatenate([np.full(30, 100.0), [95.0, 89.0, 88.0, 100.0], np.full(30, 100.0), [85.0], np.full(5, 85.0)])
ev = f.eventi_calo(c, 20, 0.10, 20)
ok("il primo evento è la chiusura a 89 (11% sotto il massimo)", ev[0] == 31 and c[ev[0]] == 89.0)
ok("la pausa evita il secondo giorno consecutivo", 32 not in ev)
ok("un secondo evento dopo la pausa", len(ev) == 2 and c[ev[1]] == 85.0)
ok("rendimento avanti", abs(f.rend_avanti(np.array([100.0, 110, 121]), 2)[0] - 0.21) < 1e-12 and np.isnan(f.rend_avanti(np.array([100.0, 110, 121]), 2)[1]))
m = f.minimo_avanti(np.array([100.0, 90, 95, 80, 100, 100]), 3)
ok("minimo nei tre giorni successivi", abs(m[0] - (-0.20)) < 1e-12 and abs(m[1] - (80 / 90 - 1)) < 1e-12 and np.isnan(m[3]))
rng = np.random.default_rng(0)
v = rng.normal(0, 1, 2000)
ok("p-value vicino a mezzo per un'osservazione tipica", 0.2 < f.pval_casuali(v, 0.0, 20, 500, rng) < 0.8)
ok("p-value piccolo per un valore enorme", f.pval_casuali(v, 3.0, 20, 500, rng) < 0.01)
