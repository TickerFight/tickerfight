Il filtro sul rendimento passato funziona davvero? Prova fuori campione su 150 titoli

Il filtro trovato (regola 50/200 solo sui titoli con rendimento passato basso, buy & hold sugli altri) funziona anche fuori campione, cioè in un periodo precedente e su titoli mai usati prima?

Leggi l'articolo sul blog →⬇ Scarica il codice (.zip)Vedi su GitHub

Il disegno

  • Fuori campione due volte: periodo diverso (i cinque anni fino al 20 ottobre 2021, mentre il filtro è stato trovato sul 2021-2026) e titoli diversi (150 mai usati negli articoli precedenti).
  • Strategia: il primo giorno si ordinano i titoli per rendimento dei dodici mesi precedenti. Sul terzo più basso si applica la regola 50/200, sugli altri si compra e si tiene. Controllo: la metà più bassa.
  • Confronti: buy & hold su tutti, regola su tutti, e una scelta casuale dello stesso numero di titoli su cui applicare la regola (10.000 sorteggi).
  • Riferimento: lo stesso filtro sul periodo 2021-2026, dove però è stato trovato: serve solo come confronto, non come prova.

"""Passo 1 - Scarica da Yahoo Finance i titoli nuovi per il test fuori campione.

Si ripete il sorteggio degli articoli precedenti (stesso elenco, stesso seme, stesso ordine) saltando i titoli
già usati, e si prendono i primi N_TITOLI con dati completi. I prezzi sono tagliati a FINE_TEST.

Salva:
  dati/titoli/<TICKER>.csv   Date, Open, High, Low, Close, Volume
  dati/eurusd_giornaliero.csv   cambio, in dollari per un euro, dal 2015
  dati/selezione.csv         ordine del sorteggio, esito per ogni titolo provato
  dati/metadati.json         fonte, data del download, versione di yfinance, impronte SHA-256
"""
import hashlib
import json
import time
from datetime import datetime, timezone

import numpy as np
import pandas as pd
import yfinance as yf

import config


def scarica(ticker: str, tentativi: int = 3) -> pd.DataFrame | None:
    for t in range(tentativi):
        try:
            grezzo = yf.Ticker(ticker).history(start=config.INIZIO_DOWNLOAD, end="2021-11-10", auto_adjust=False)
        except Exception as e:  # noqa: BLE001 - errori di rete: si riprova
            print(f"  {ticker}: errore {type(e).__name__}, tentativo {t + 1}")
            grezzo = None
        if grezzo is not None and not grezzo.empty:
            break
        time.sleep(3)
    else:
        return None
    df = grezzo[["Open", "High", "Low", "Close", "Volume"]].copy()
    df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
    df.index.name = "Date"
    df = df[~df.index.duplicated(keep="last")].sort_index()
    df = df[df.index <= pd.Timestamp(config.FINE_TEST)].dropna(subset=["Open", "Close"])
    if df.empty or (df[["Open", "Close"]] <= 0).any().any():
        return None
    return df


def main() -> None:
    config.DATI_TITOLI.mkdir(parents=True, exist_ok=True)
    comp = pd.read_csv(config.FILE_COMPOSIZIONE)
    usati = set()
    for f in (config.FILE_SELEZ_SVIL, config.FILE_SELEZ_VAL):
        s = pd.read_csv(f)
        usati |= set(s.loc[s["esito"] == "preso", "ticker"])
    cambio = scarica(config.TICKER_CAMBIO)
    if cambio is None:
        raise SystemExit("Yahoo Finance non ha restituito il cambio euro/dollaro.")
    cambio.to_csv(config.FILE_CAMBIO, float_format="%.6f")
    ordine = np.random.default_rng(config.SEME).permutation(len(comp))
    prese, righe = [], []
    for pos, k in enumerate(ordine, 1):
        if len(prese) >= config.N_TITOLI:
            break
        r = comp.iloc[k]
        t = r["ticker_yahoo"]
        if t in usati:
            continue
        df = scarica(t)
        esito = "preso"
        if df is None:
            esito = "scartato: dati non disponibili"
        else:
            fine = df.index[-1]
            i0 = int(df.index.searchsorted(fine - pd.DateOffset(years=config.ANNI_TEST)))
            if (pd.Timestamp(config.FINE_TEST) - fine).days > 5:
                esito = "scartato: nessun dato vicino alla fine del test"
            elif i0 < config.SEDUTE_MINIME_PRIMA:
                esito = f"scartato: storia troppo corta ({i0} sedute prima dell'inizio del test)"
        if esito == "preso":
            df.to_csv(config.DATI_TITOLI / f"{t}.csv", float_format="%.6f")
            prese.append(t)
        righe.append({"ordine_sorteggio": pos, "ticker": t, "nome": r["nome"], "settore": r["settore"], "esito": esito})
        print(f"{pos:3d} {t:8s} {esito}")
    if len(prese) < config.N_TITOLI:
        raise SystemExit(f"Solo {len(prese)} titoli validi su {config.N_TITOLI} richiesti.")
    pd.DataFrame(righe).to_csv(config.FILE_SELEZIONE, index=False)

    def impronta(p):
        return hashlib.sha256(p.read_bytes()).hexdigest()

    metadati = {
        "descrizione": config.DESCRIZIONE_DATI,
        "fonte": "Yahoo Finance",
        "libreria": f"yfinance {yf.__version__}",
        "scaricato_il_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"),
        "seme": config.SEME,
        "titoli_gia_usati": len(usati),
        "titoli_provati": len(righe),
        "titoli_scartati": len(righe) - len(prese),
        "titoli": prese,
        "fine_test": config.FINE_TEST,
        "colonne_usate": config.NOTA_COLONNA,
        "sha256_titoli": {t: impronta(config.DATI_TITOLI / f"{t}.csv") for t in prese},
        "sha256_cambio": impronta(config.FILE_CAMBIO),
    }
    config.FILE_METADATI.write_text(json.dumps(metadati, indent=2, ensure_ascii=False), encoding="utf-8")
    print(f"\nScaricati {len(prese)} titoli, {len(righe) - len(prese)} scartati.")


if __name__ == "__main__":
    main()
flusso/02_backtest_e_test.pysimulazione dei portafogli, test, scelta casuale, confronto con il 2021-2026
"""Passo 2 - Il filtro sul rendimento passato, provato fuori campione.

Legge SOLO i file in dati/ e produce in risultati/:
  metriche.json              ogni numero dell'articolo, con descrizione e origine di calcolo
  titoli.csv                 una riga per titolo: rendimento passato, gruppo, risultati di regola, buy & hold e filtro
  curva_portafoglio.csv      valore giorno per giorno dei portafogli: buy & hold, regola, filtro, filtro sulla metà
  simulazioni_casuali.csv    valore finale del portafoglio quando la regola è assegnata a caso a un terzo dei titoli

Regole del metodo:
  - ipotesi, strategia e parametri sono fissati in config.py prima di guardare i risultati;
  - il rendimento passato è misurato sui dodici mesi PRIMA del primo ordine e l'ordinamento dei titoli si fa
    il primo giorno: nessuna informazione dal futuro;
  - ogni titolo parte con 500 euro, 1 euro a ordine e 26% di tassa in euro: gli stessi conti degli articoli precedenti;
  - ogni numero dell'articolo passa da reg(): niente cifre scritte a mano nel testo.
"""
import json
import platform
from datetime import datetime, timezone

import numpy as np
import pandas as pd

import config
import simulazione as s

RIS: dict = {}


def reg(chiave, valore, formato, descrizione, origine):
    """Registra un numero con la sua giustificazione (finisce nelle note dell'articolo)."""
    if isinstance(valore, (np.floating, np.integer)):
        valore = valore.item()
    if isinstance(valore, float) and not np.isfinite(valore):
        raise SystemExit(f"Numero non finito per {chiave}")
    RIS[chiave] = {"valore": valore, "formato": formato, "descrizione": descrizione, "origine": origine}


def spearman(x, y) -> float:
    return float(np.corrcoef(pd.Series(x).rank().to_numpy(), pd.Series(y).rank().to_numpy())[0, 1])


def test_correlazione(x, y, rng):
    """Spearman, p-value di permutazione a una coda (ipotesi: correlazione negativa), IC bootstrap al 95%."""
    n = len(x)
    rho = spearman(x, y)
    rx = pd.Series(x).rank().to_numpy()
    ry = pd.Series(y).rank().to_numpy()
    rxc = (rx - rx.mean()) / np.linalg.norm(rx - rx.mean())
    ryc = ry - ry.mean()
    idx = rng.random((config.N_PERMUTAZIONI, n)).argsort(axis=1)
    rho_perm = (ryc[idx] * rxc).sum(axis=1) / np.linalg.norm(ryc)
    p = (int((rho_perm <= rho).sum()) + 1) / (config.N_PERMUTAZIONI + 1)
    ib = rng.integers(0, n, (config.N_BOOTSTRAP, n))
    a = pd.DataFrame(np.asarray(x)[ib]).rank(axis=1).to_numpy()
    b = pd.DataFrame(np.asarray(y)[ib]).rank(axis=1).to_numpy()
    a, b = a - a.mean(axis=1, keepdims=True), b - b.mean(axis=1, keepdims=True)
    boot = (a * b).sum(axis=1) / np.sqrt((a ** 2).sum(axis=1) * (b ** 2).sum(axis=1))
    lo, hi = np.percentile(boot, [2.5, 97.5])
    return rho, p, float(lo), float(hi)


def maxdd(v) -> float:
    v = np.asarray(v)
    return float(-(v / np.maximum.accumulate(v) - 1).min())


def main() -> None:
    config.RISULTATI.mkdir(parents=True, exist_ok=True)
    meta = json.loads(config.FILE_METADATI.read_text(encoding="utf-8"))
    comp = pd.read_csv(config.FILE_COMPOSIZIONE).set_index("ticker_yahoo")
    cambio = s.carica_csv(config.FILE_CAMBIO)
    rng = np.random.default_rng(config.SEME_STAT)

    righe, eq_mm, eq_bh, fine_set, inizio_set = [], {}, {}, set(), set()
    for t in meta["titoli"]:
        df = s.prepara(s.carica_csv(config.DATI_TITOLI / f"{t}.csv"), cambio, config.SMA_VELOCE, config.SMA_LENTA)
        df = df[df.index <= pd.Timestamp(config.FINE_TEST)]
        i0 = s.indice_data(df, df.index[-1] - pd.DateOffset(years=config.ANNI_TEST))
        if i0 < config.SEDUTE_MINIME_PRIMA:
            raise SystemExit(f"{t}: storia troppo corta.")
        if df["fx"].iloc[i0 - 1:].isna().any():
            raise SystemExit(f"{t}: cambio mancante nel periodo del test.")
        fine_set.add(df.index[-1])
        inizio_set.add(df.index[i0])
        C = df["Close"].to_numpy()
        ret = float(C[i0 - 1] / C[i0 - 1 - config.FINESTRA_RENDIMENTO] - 1)
        mm = s.simula(df, i0, config.CAPITALE, config.COMMISSIONE, config.ALIQUOTA, config.MODO_PRINCIPALE, True)
        bh = s.simula(df, i0, config.CAPITALE, config.COMMISSIONE, config.ALIQUOTA, config.MODO_PRINCIPALE, False)
        righe.append({"ticker": t, "nome": comp.loc[t, "nome"], "settore": comp.loc[t, "settore"], "ret": ret,
                      "finale_mm": mm["valore_finale_eur"], "finale_bh": bh["valore_finale_eur"],
                      "differenza": mm["valore_finale_eur"] - bh["valore_finale_eur"],
                      "rend_bh": bh["rendimento_totale"], "rend_mm": mm["rendimento_totale"]})
        d = df.index[i0:]
        eq_mm[t] = pd.Series(mm["equity"], index=d)
        eq_bh[t] = pd.Series(bh["equity"], index=d)
    if max(fine_set) - min(fine_set) > pd.Timedelta(days=5) or max(inizio_set) - min(inizio_set) > pd.Timedelta(days=5):
        raise SystemExit("I titoli non coprono lo stesso periodo di test.")

    tab = pd.DataFrame(righe)
    n = len(tab)
    ordine = tab["ret"].rank(method="first").astype(int)       # 1 = rendimento passato più basso
    nt, nm = int(n * config.QUOTA_FILTRO), int(n * config.QUOTA_CONTROLLO)
    tab["gruppo"] = np.where(ordine <= nt, "basso", np.where(ordine > n - nt, "alto", "medio"))
    tab["filtro"] = ordine <= nt
    tab["filtro_meta"] = ordine <= nm
    tab["finale_filtro"] = np.where(tab["filtro"], tab["finale_mm"], tab["finale_bh"])
    tab["finale_filtro_meta"] = np.where(tab["filtro_meta"], tab["finale_mm"], tab["finale_bh"])
    tab["vince"] = tab["differenza"] > 0
    tab.to_csv(config.RISULTATI / "titoli.csv", index=False, float_format="%.6f")

    ok = pd.DataFrame(eq_mm).notna().all(axis=1) & pd.DataFrame(eq_bh).notna().all(axis=1)
    EM, EB = pd.DataFrame(eq_mm)[ok], pd.DataFrame(eq_bh)[ok]
    f_set = list(tab.loc[tab["filtro"], "ticker"])
    m_set = list(tab.loc[tab["filtro_meta"], "ticker"])
    curva = pd.DataFrame({
        "data": EM.index,
        "valore_bh": EB.sum(axis=1).to_numpy(), "valore_regola": EM.sum(axis=1).to_numpy(),
        "valore_filtro": (EM[f_set].sum(axis=1) + EB.drop(columns=f_set).sum(axis=1)).to_numpy(),
        "valore_filtro_meta": (EM[m_set].sum(axis=1) + EB.drop(columns=m_set).sum(axis=1)).to_numpy(),
    })
    curva.to_csv(config.RISULTATI / "curva_portafoglio.csv", index=False, float_format="%.6f")

    cap = config.CAPITALE * n
    F = {"bh": tab["finale_bh"].sum(), "mm": tab["finale_mm"].sum(), "fl": tab["finale_filtro"].sum(),
         "fm": tab["finale_filtro_meta"].sum()}
    tl = "titoli.csv"
    origine_dati = f"dati/ ({meta['fonte']}, scaricati il {meta['scaricato_il_utc'][:10]})"

    # --- Parametri e campione -------------------------------------------------------------------------------
    reg("capitale", config.CAPITALE, "int", "Capitale iniziale per ogni titolo, in euro", "config.py, CAPITALE")
    reg("capitale_totale", cap, "int", "Capitale iniziale complessivo, in euro", "config.py: CAPITALE per numero di titoli")
    reg("commissione", config.COMMISSIONE, "num2", "Commissione fissa per ogni ordine, in euro", "config.py, COMMISSIONE")
    reg("aliquota", config.ALIQUOTA, "pct0", "Aliquota sulle plusvalenze", "config.py, ALIQUOTA")
    reg("sma_veloce", config.SMA_VELOCE, "annonum", "Giorni della media veloce", "config.py, SMA_VELOCE")
    reg("sma_lenta", config.SMA_LENTA, "annonum", "Giorni della media lenta", "config.py, SMA_LENTA")
    reg("n_permutazioni", config.N_PERMUTAZIONI, "int", "Permutazioni per i p-value", "config.py, N_PERMUTAZIONI")
    reg("n_bootstrap", config.N_BOOTSTRAP, "int", "Ricampionamenti bootstrap", "config.py, N_BOOTSTRAP")
    reg("livello_ic", 0.95, "pct0", "Livello degli intervalli di confidenza e percentile alto usato per la distribuzione casuale", "02_backtest_e_test.py, percentili 2,5 e 97,5 del bootstrap; 95° percentile")
    reg("seme", config.SEME, "annonum", "Seme del sorteggio dei titoli", "config.py, SEME")
    reg("data_inizio", str(min(inizio_set).date()), "data_lunga", "Prima seduta del test", origine_dati)
    reg("data_fine", str(max(fine_set).date()), "data_lunga", "Ultima seduta del test", origine_dati)
    reg("data_scarico", meta["scaricato_il_utc"][:10], "data", "Data del download dei dati", "dati/metadati.json")
    reg("n_titoli", n, "int", "Titoli del campione fuori campione", f"{tl}")
    reg("n_gia_usati", meta["titoli_gia_usati"], "int", "Titoli già usati negli articoli precedenti, esclusi dal sorteggio", "dati/metadati.json")
    reg("n_provati", meta["titoli_provati"], "int", "Titoli provati per ottenerne il numero richiesto", "dati/metadati.json")
    reg("n_scartati", meta["titoli_scartati"], "int", "Titoli scartati per dati assenti o storia troppo corta", "dati/metadati.json")
    reg("n_filtro", nt, "int", "Titoli nel gruppo con il rendimento passato più basso (a cui si applica la regola)", f"{tl}: un terzo di n")
    reg("n_filtro_meta", nm, "int", "Titoli nel gruppo di controllo (metà più bassa)", f"{tl}: metà di n")
    reg("soglia_ret", float(tab.loc[tab["filtro"], "ret"].max()), "pct0",
        "Rendimento passato più alto tra i titoli del terzo più basso (soglia del filtro)", f"{tl}")

    # --- 1) La relazione si ripete? ---------------------------------------------------------------------------
    rho, p, lo, hi = test_correlazione(tab["ret"].to_numpy(), tab["differenza"].to_numpy(), rng)
    reg("rho", rho, "num2", "Correlazione di rango tra rendimento passato e differenza regola meno buy & hold, fuori campione", f"{tl}: Spearman")
    reg("rho_lo", lo, "num2", "Estremo inferiore dell'intervallo di confidenza al 95%", "bootstrap sui titoli")
    reg("rho_hi", hi, "num2", "Estremo superiore dell'intervallo di confidenza al 95%", "bootstrap sui titoli")
    reg("rho_p", p, "pval", "p-value di permutazione a una coda (ipotesi: correlazione negativa)", "permutazione")
    reg("quota_vince", float(tab["vince"].mean()), "pct0", "Quota di titoli su cui la regola batte il buy & hold, fuori campione", f"{tl}: differenza > 0")
    reg("n_vince", int(tab["vince"].sum()), "int", "Titoli su cui la regola batte il buy & hold", f"{tl}: differenza > 0")
    reg("mediana_diff", float(tab["differenza"].median()), "num2", "Differenza mediana in euro, regola meno buy & hold", f"{tl}")
    for g in ("basso", "medio", "alto"):
        d = tab[tab["gruppo"] == g]
        reg(f"g_{g}_vince", float(d["vince"].mean()), "pct0", f"Quota di titoli su cui vince la regola nel gruppo con rendimento passato {g}", f"{tl}")
        reg(f"g_{g}_ret", float(d["ret"].median()), "pct0", f"Rendimento passato mediano del gruppo {g}", f"{tl}")
        reg(f"g_{g}_rbh", float(d["rend_bh"].median()), "pct0", f"Rendimento mediano del buy & hold nei cinque anni, gruppo {g}", f"{tl}")
        reg(f"g_{g}_diff", float(d["differenza"].median()), "num2", f"Differenza mediana in euro, gruppo {g}", f"{tl}")
    reg("rho_diff_rbh", spearman(tab["rend_bh"].to_numpy(), tab["differenza"].to_numpy()), "num2",
        "Correlazione di rango tra rendimento del buy & hold e differenza", f"{tl}: Spearman")
    reg("rho_ret_rbh", spearman(tab["ret"].to_numpy(), tab["rend_bh"].to_numpy()), "num2",
        "Correlazione di rango tra rendimento passato e rendimento del buy & hold nei cinque anni", f"{tl}: Spearman")

    # --- 2) e 3) Portafogli ------------------------------------------------------------------------------------
    nomi = {"bh": "buy & hold su tutti i titoli", "mm": "regola su tutti i titoli",
            "fl": "regola sul terzo con rendimento passato più basso, buy & hold sugli altri",
            "fm": "regola sulla metà con rendimento passato più basso, buy & hold sugli altri"}
    for k, v in F.items():
        reg(f"finale_{k}", float(v), "num2", f"Valore finale netto in euro del portafoglio: {nomi[k]}", f"{tl}: somma dei valori finali")
        reg(f"rend_{k}", float(v / cap - 1), "pct0", f"Rendimento totale netto del portafoglio: {nomi[k]}", "finale / capitale_totale - 1")
    for k, col in (("bh", "valore_bh"), ("mm", "valore_regola"), ("fl", "valore_filtro"), ("fm", "valore_filtro_meta")):
        reg(f"maxdd_{k}", maxdd(curva[col]), "pct0", f"Perdita massima dal picco del portafoglio: {nomi[k]}", "curva_portafoglio.csv")
    reg("diff_fl_bh", float(F["fl"] - F["bh"]), "num2", "Filtro meno buy & hold, in euro (segno: + vantaggio del filtro)", "finale_fl - finale_bh")
    reg("diff_fl_bh_abs", float(abs(F["fl"] - F["bh"])), "num2", "Distanza in euro tra filtro e buy & hold", "|finale_fl - finale_bh|")
    reg("diff_fl_bh_pct_abs", float(abs(F["fl"] / F["bh"] - 1)), "pct1", "Distanza tra filtro e buy & hold in percentuale del valore finale del buy & hold", "|finale_fl / finale_bh - 1|")
    reg("diff_fl_mm", float(F["fl"] - F["mm"]), "num2", "Filtro meno regola su tutti i titoli, in euro", "finale_fl - finale_mm")
    reg("diff_mm_bh_abs", float(abs(F["mm"] - F["bh"])), "num2", "Distanza in euro tra regola su tutti i titoli e buy & hold", "|finale_mm - finale_bh|")
    reg("diff_fm_bh", float(F["fm"] - F["bh"]), "num2", "Controllo: filtro sulla metà meno buy & hold, in euro", "finale_fm - finale_bh")
    d_stock = (tab["finale_filtro"] - tab["finale_bh"]).to_numpy()
    ib = rng.integers(0, n, (config.N_BOOTSTRAP, n))
    medie = d_stock[ib].mean(axis=1)
    lo_b, hi_b = np.percentile(medie, [2.5, 97.5])

flusso/simulazione.pymotore: segnale, ordini, commissioni, cambio, tassa e minusvalenze
"""Motore di simulazione in euro di una regola a medie mobili su un singolo titolo.

Regole (da dichiarare nell'articolo):
  - segnale: investito se la media mobile veloce è sopra la lenta, calcolate sulle chiusure;
  - il segnale a fine seduta t decide l'operazione della seduta t+1 (nessun look-ahead);
  - esecuzione "apertura": prezzo di apertura di t+1, cambio dell'ultima chiusura nota (t);
    esecuzione "chiusura": prezzo di chiusura di t+1, cambio di t+1 (controllo di robustezza);
  - azioni frazionate, si investe tutta la cassa; commissione fissa per ogni ordine, acquisto o vendita;
  - la cassa non rende nulla; i dividendi non sono considerati;
  - tassa sulle plusvalenze calcolate in euro (cambio incluso), pagata a ogni vendita come nel regime
    amministrato; la commissione si somma al costo di acquisto e si sottrae dall'incasso; le minusvalenze
    compensano le plusvalenze successive fino al quarto anno seguente;
  - a fine periodo si vende tutto, con commissione e tassa, per confrontare cifre nette.
Non modellati: imposta di bollo, ritenuta sui dividendi, spread di cambio.
"""
from pathlib import Path

import numpy as np
import pandas as pd


class Zaino:
    """Minusvalenze compensabili con le plusvalenze successive, fino al 4° anno seguente."""

    def __init__(self, aliquota: float):
        self.aliquota = aliquota
        self.voci: list[list[float]] = []   # [ultimo anno utilizzabile, importo residuo]

    def tassa(self, anno: int, guadagno: float) -> float:
        if guadagno < 0:
            self.voci.append([anno + 4, -guadagno])
            return 0.0
        residuo = guadagno
        for voce in sorted(self.voci, key=lambda v: v[0]):      # prima le più vecchie
            if voce[0] < anno or voce[1] <= 0:
                continue
            usa = min(voce[1], residuo)
            voce[1] -= usa
            residuo -= usa
            if residuo <= 0:
                break
        return self.aliquota * residuo


def carica_csv(percorso: Path) -> pd.DataFrame:
    df = pd.read_csv(percorso, parse_dates=["Date"], index_col="Date").sort_index()
    df.index = pd.to_datetime(df.index).tz_localize(None).normalize()
    df = df[~df.index.duplicated(keep="last")]
    if (df["Close"] <= 0).any() or df["Close"].isna().any():
        raise SystemExit(f"{percorso}: chiusure mancanti o non positive.")
    return df


def prepara(titolo: pd.DataFrame, cambio: pd.DataFrame, veloce: int, lenta: int) -> pd.DataFrame:
    df = titolo[["Open", "Close"]].astype(float).copy()
    fx = cambio["Close"].astype(float)
    df["fx"] = fx.reindex(df.index.union(fx.index)).ffill().reindex(df.index)
    df["sma_v"] = df["Close"].rolling(veloce).mean()
    df["sma_l"] = df["Close"].rolling(lenta).mean()
    df["segnale"] = (df["sma_v"] > df["sma_l"]) & df["sma_l"].notna()
    return df


def indice_data(df: pd.DataFrame, data) -> int:
    """Posizione della prima seduta con data uguale o successiva a quella indicata."""
    return int(np.searchsorted(df.index.values, np.datetime64(data)))


def simula(df: pd.DataFrame, i0: int, capitale: float, commissione: float, aliquota: float,
           modo: str, strategia: bool) -> dict:
    """strategia=True: regola a medie mobili; False: buy & hold (compra a i0, vende alla fine)."""
    O, C, FX = (df[c].to_numpy(float) for c in ("Open", "Close", "fx"))
    SEG = df["segnale"].to_numpy(bool)
    date = df.index
    n = len(df)
    cassa, azioni, base = float(capitale), 0.0, 0.0
    zaino = Zaino(aliquota)
    log: list[dict] = []
    tot = {"commissioni": 0.0, "tasse": 0.0}
    equity = np.empty(n - i0)
    in_posizione = np.zeros(n - i0, dtype=bool)

    def compra(i, px_usd, fx, etichetta):
        nonlocal cassa, azioni, base
        investito = cassa - commissione
        if investito <= 0:
            return
        px_eur = px_usd / fx
        spesa = cassa
        azioni = investito / px_eur
        base = spesa
        tot["commissioni"] += commissione
        log.append(dict(data=str(date[i].date()), lato="acquisto", motivo=etichetta, prezzo_usd=px_usd, cambio=fx,
                        prezzo_eur=px_eur, azioni=azioni, commissione=commissione, spesa_eur=spesa,
                        incasso_netto_eur=0.0, plusvalenza_eur=0.0, tassa_eur=0.0))
        cassa = 0.0

    def vendi(i, px_usd, fx, etichetta):
        nonlocal cassa, azioni, base
        px_eur = px_usd / fx
        lordo = azioni * px_eur
        netto = lordo - commissione
        guadagno = netto - base
        tassa = zaino.tassa(int(date[i].year), guadagno)
        tot["commissioni"] += commissione
        tot["tasse"] += tassa
        log.append(dict(data=str(date[i].date()), lato="vendita", motivo=etichetta, prezzo_usd=px_usd, cambio=fx,
                        prezzo_eur=px_eur, azioni=azioni, commissione=commissione, spesa_eur=0.0,
                        incasso_netto_eur=netto, plusvalenza_eur=guadagno, tassa_eur=tassa))
        cassa = netto - tassa
        azioni, base = 0.0, 0.0

    for i in range(i0, n):
        voglio = bool(SEG[i - 1]) if strategia else True
        px, fx = (O[i], FX[i - 1]) if modo == "apertura" else (C[i], FX[i])
        if voglio and azioni == 0.0:
            compra(i, px, fx, "segnale" if strategia else "acquisto iniziale")
        elif (not voglio) and azioni > 0.0:
            vendi(i, px, fx, "segnale")
        equity[i - i0] = cassa + azioni * C[i] / FX[i]
        in_posizione[i - i0] = azioni > 0.0
    if azioni > 0.0:
        vendi(n - 1, C[n - 1], FX[n - 1], "liquidazione finale")

    picco = np.maximum.accumulate(equity)
    anni = (date[-1] - date[i0]).days / 365.25
    finale = cassa
    return {
        "valore_finale_eur": finale,
        "rendimento_totale": finale / capitale - 1,
        "cagr": (finale / capitale) ** (1 / anni) - 1 if finale > 0 else float("nan"),
        "anni": anni,
        "n_acquisti": sum(1 for r in log if r["lato"] == "acquisto"),
        "n_vendite": sum(1 for r in log if r["lato"] == "vendita"),
        "commissioni_eur": tot["commissioni"],
        "tasse_eur": tot["tasse"],
        "quota_giorni_investito": float(in_posizione.mean()),
        "perdita_massima": float((equity / picco - 1).min()),
        "operazioni": log,
        "equity": equity,

"""Controlli del motore di simulazione. Usa SOLO dati inventati: nessun numero qui è un risultato.

Si esegue prima del backtest (lo fa anche il flusso su GitHub): se un controllo fallisce, il flusso si ferma.
  1. conti fatti a mano (guadagno, perdita, cambio, compensazione delle minusvalenze);
  2. confronto con una formula indipendente;
  3. coerenza del registro delle operazioni con i prezzi grezzi.
"""
import numpy as np
import pandas as pd

import simulazione as s

ok = 0


def verifica(nome, condizione, dettaglio=""):
    global ok
    if not condizione:
        raise SystemExit(f"FALLITO: {nome} {dettaglio}")
    ok += 1
    print(f"  ok  {nome}")


def uguali(a, b, tol=1e-9):
    return abs(a - b) <= tol * max(1.0, abs(a), abs(b))


print("1. Conti a mano")
# Guadagno: compro a 100, vendo a 120, cambio 1. Azioni = 499/100 = 4,99; incasso 598,8; netto 597,8;
# plusvalenza 97,8; tassa 25,428; cassa finale 572,372.
piccolo = pd.DataFrame({"Open": [100, 100, 100, 120, 120, 120.0], "Close": [100, 100, 110, 120, 120, 120.0],
                        "fx": 1.0, "segnale": [False, True, True, False, False, False]},
                       index=pd.bdate_range("2022-03-01", periods=6))
r = s.simula(piccolo, 1, 500, 1.0, 0.26, "apertura", True)
verifica("guadagno: cassa finale 572,372", uguali(r["valore_finale_eur"], 572.372), r["valore_finale_eur"])
verifica("guadagno: tassa 25,428", uguali(r["tasse_eur"], 25.428))
verifica("guadagno: 2 commissioni", uguali(r["commissioni_eur"], 2.0))

perdita = piccolo.copy()
perdita["Open"] = [100, 100, 110, 110, 90, 90.0]
perdita["Close"] = [100, 100, 110, 100, 90, 90.0]
r = s.simula(perdita, 1, 500, 1.0, 0.26, "apertura", True)
verifica("perdita: cassa finale = 499/110*90-1", uguali(r["valore_finale_eur"], 499 / 110 * 90 - 1), r["valore_finale_eur"])
verifica("perdita: nessuna tassa", r["tasse_eur"] == 0.0)

cambio = piccolo.copy()
cambio["fx"] = 1.25
r = s.simula(cambio, 1, 500, 1.0, 0.26, "apertura", True)
verifica("cambio costante: stesso risultato in euro", uguali(r["valore_finale_eur"], 572.372))

# Con il cambio che cambia tra acquisto e vendita la plusvalenza in euro cambia: compro con cambio 1,00,
# vendo con cambio 1,25 (dollaro più debole -> meno euro): 4,99 azioni * 120 / 1,25 = 479,04; netto 478,04.
cambio_var = piccolo.copy()
cambio_var["fx"] = [1.0, 1.0, 1.0, 1.0, 1.25, 1.25]   # acquisto all'apertura di i=2 con fx[1]=1,00; vendita i=4 con fx[3]=1,00
r = s.simula(cambio_var, 1, 500, 1.0, 0.26, "apertura", True)
verifica("cambio: ordini con il cambio dell'ultima chiusura nota", uguali(r["valore_finale_eur"], 572.372), r["valore_finale_eur"])
cambio_var["fx"] = [1.0, 1.0, 1.0, 1.25, 1.25, 1.25]  # vendita i=4 con fx[3]=1,25
r = s.simula(cambio_var, 1, 500, 1.0, 0.26, "apertura", True)
netto_atteso = 4.99 * 120 / 1.25 - 1                  # 478,04 euro, sotto i 500 spesi: perdita, nessuna tassa
verifica("cambio: perdita da cambio, nessuna tassa", uguali(r["valore_finale_eur"], netto_atteso) and r["tasse_eur"] == 0.0,
         r["valore_finale_eur"])

z = s.Zaino(0.26)
verifica("zaino: perdita non tassata", z.tassa(2022, -100) == 0.0)
verifica("zaino: guadagno 150 dopo perdita 100 -> tassa su 50", uguali(z.tassa(2023, 150), 13.0))
z = s.Zaino(0.26)
z.tassa(2023, -10)
verifica("zaino: perdita scaduta (2023 -> usabile fino al 2027)", uguali(z.tassa(2028, 100), 26.0))
z = s.Zaino(0.26)
z.tassa(2023, -10)
verifica("zaino: perdita ancora valida nel 2027", uguali(z.tassa(2027, 100), 0.26 * 90))

print("2. Confronto con una formula indipendente (dati inventati, cambio 1, senza costi)")
rng = np.random.default_rng(7)
date = pd.bdate_range("2020-01-01", "2026-10-20")
n = len(date)
chiusura = 100 * np.exp(np.cumsum(rng.normal(0.0004, 0.018, n)))
apertura = np.r_[chiusura[0], chiusura[:-1]] * (1 + rng.normal(0, 0.004, n))
fx_var = 1.1 * np.exp(np.cumsum(rng.normal(0, 0.003, n)))
titolo = pd.DataFrame({"Open": apertura, "Close": chiusura}, index=pd.Index(date, name="Date"))
cambio_df = pd.DataFrame({"Close": fx_var}, index=pd.Index(date, name="Date"))
piatto = pd.DataFrame({"Close": 1.0}, index=cambio_df.index)
df = s.prepara(titolo, piatto, 50, 200)
i0 = s.indice_data(df, df.index[-1] - pd.DateOffset(years=5))
SEG = df["segnale"].to_numpy(bool)
C = df["Close"].to_numpy()
formula = 500 * np.prod([(C[i + 1] / C[i]) ** SEG[i - 1] for i in range(i0, len(df) - 1)])
r = s.simula(df, i0, 500, 0.0, 0.0, "chiusura", True)
verifica("regola, esecuzione alla chiusura, lordo = prodotto dei rendimenti", uguali(r["valore_finale_eur"], formula),
         f"{r['valore_finale_eur']} contro {formula}")
bh = s.simula(df, i0, 500, 0.0, 0.0, "chiusura", False)
verifica("buy & hold lordo = chiusura finale / chiusura iniziale", uguali(bh["valore_finale_eur"], 500 * C[-1] / C[i0]))
bh_a = s.simula(df, i0, 500, 0.0, 0.0, "apertura", False)
verifica("buy & hold, esecuzione all'apertura = ultima chiusura / prima apertura",
         uguali(bh_a["valore_finale_eur"], 500 * C[-1] / df["Open"].iloc[i0]))
con = s.simula(df, i0, 500, 1.0, 0.0, "apertura", True)
verifica("commissioni = 1 euro per ogni ordine", uguali(con["commissioni_eur"], len(con["operazioni"]) * 1.0))
netto = s.simula(df, i0, 500, 1.0, 0.26, "apertura", True)
verifica("con la tassa non si finisce mai sopra il caso senza tassa",
         netto["valore_finale_eur"] <= con["valore_finale_eur"] + 1e-9)
verifica("il registro delle operazioni quadra con la cassa finale",
         uguali(500 - sum(o["spesa_eur"] for o in netto["operazioni"]) +
                sum(o["incasso_netto_eur"] - o["tassa_eur"] for o in netto["operazioni"]), netto["valore_finale_eur"]))

print("3. Il registro delle operazioni contro i prezzi grezzi (ricerca per data, senza gli array del motore)")
dfv = s.prepara(titolo, cambio_df, 50, 200)
i0v = s.indice_data(dfv, dfv.index[-1] - pd.DateOffset(years=5))
rv = s.simula(dfv, i0v, 500, 1.0, 0.26, "apertura", True)
errori = 0
for o in rv["operazioni"]:
    d = pd.Timestamp(o["data"])
    if o["motivo"] == "liquidazione finale":
        px, fx = titolo.loc[d, "Close"], cambio_df.loc[d, "Close"]
    else:
        prec = cambio_df.index[cambio_df.index.get_loc(d) - 1]
        px, fx = titolo.loc[d, "Open"], cambio_df.loc[prec, "Close"]
    if not (uguali(o["prezzo_usd"], px) and uguali(o["cambio"], fx) and uguali(o["prezzo_eur"], px / fx)):
        errori += 1
verifica(f"{len(rv['operazioni'])} operazioni con prezzo e cambio corretti", errori == 0, f"{errori} errori")

print(f"\nTutti i {ok} controlli superati (dati inventati: nessuno di questi numeri è un risultato).")
